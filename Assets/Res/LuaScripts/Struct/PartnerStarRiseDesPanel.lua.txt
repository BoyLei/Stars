-- 设置升星信息
function PartnerStarRiseDesPanel:SetRiseStarData()

	-- 设置属性
	local attrNum = self.StarExpendCfg.Value[i];
	for i = 1, #self.StarExpendCfg.Value do
		attrNum = self.StarExpendCfg.Value[i];
		break;
	end
	if attrNum == nil then
		attrNum = 100;
	end
	self.AttrDes.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_AttrDes"), {"<color=#E1AF8C> ".. (attrNum / 100) .." </color>"});

	self.SkillDes.text = self:GetParSkillDes() .. "<color=#E1AF8C>1%</color>";

	-- 资质
	self.QualificationDes.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_QualificationDes"), {"<color=#E1AF8C>".. self.StarExpendCfg.Star .."</color>",});
end

-- 获得技能描述
function PartnerStarRiseDesPanel:GetParSkillDes()

	for _k,_v in pairs(ConfigDataAccessor.ParSkill) do
		if _v.Par == self.CurPartnerCfg.Id and _v.Star == self.StarExpendCfg.Star then
			return _v.Desc;
		end
	end

	return "";

end

-- 获得品质-颜色图标
function PartnerStarRiseDesPanel:GetQualityName(quality)
	if quality== 1 then
		return "Partner_quality_n"
	elseif quality== 2 then
		return "Partner_quality_r"
	elseif quality== 3 then
		return "Partner_quality_sr"
	elseif quality== 4 then
		return "Partner_quality_ssr"
	end
end

-- 获得当前升星的配置
function PartnerStarRiseDesPanel:GetStarExpendCfg(index)
	local cfg = self.CurQualityStarExpendCfg[index];
	return cfg;
end

-- 点击翻页
function PartnerStarRiseDesPanel:OnBtnPagingClick(num)

	local index = self.Index + num;
	if index < 1 or index > #self.CurQualityStarExpendCfg then
		Log("PartnerStarRiseDesPanel:OnBtnPagingClick index=" .. index .. "count=" .. #self.CurQualityStarExpendCfg);
		return;
	end

	self.Index = index;
	local cfg = self:GetStarExpendCfg(self.Index);
	if cfg == nil then
		Log("PartnerStarRiseDesPanel:OnBtnPagingClick cfg==nil");
		return;
	end

	self.StarExpendCfg = cfg;
	self:SetBaseData();
	self:SetRiseStarData();

end

function PartnerStarRiseDesPanel:ClosePanel()
	self.Prefab.gameObject:SetActive(false);
	--if self.RiseStarItem ~= nil then
	--self.RiseStarItem.prefab.gameObject:SetActive(false);
	--end
end

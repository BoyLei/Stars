function PartnerStarRiseFinishPanel:Init(CurSelectPartnerData, CurPartnerCfg, StarExpendCfg)
    if CurSelectPartnerData == nil or CurPartnerCfg == nil or StarExpendCfg == nil then
        Log(
            "PartnerStarRiseFinishPanel:Init CurSelectPartnerData==nil or CurPartnerCfg==nil or StarExpendCfg==nil err!!!"
        )
        self:ClosePanel()
        return
    end

    self.CurSelectPartnerData = CurSelectPartnerData
    self.CurPartnerCfg = CurPartnerCfg
    self.StarExpendCfg = StarExpendCfg

    self:SetBaseData()
    self:SetRiseStarData()

    self.Prefab.gameObject:SetActive(true)
end



-- 设置基础信息

                self.m_RoleIcon.gameObject:SetActive(true)
            end)
 
        end
    end

    -- 目前先只读配置表里面的战斗力
    --local PartnerNature = ConfigDataAccessor.PartnerNature[self.CurPartnerCfg.Nature][self.CurSelectPartnerData.Level]
    --if PartnerNature ~= nil then
    --self.Power.text = tostring(PartnerNature.Power);
    --end
end

-- 设置升星信息
function PartnerStarRiseFinishPanel:SetRiseStarData()
    -- 设置属性
    for i = 1, #self.StarExpendCfg.Nature do
        local attrIndex = self.StarExpendCfg.Nature[i]
        local attrNum = self.StarExpendCfg.Value[i]
        if attrNum == nil then
            attrNum = 100
        end
        local attrNumStr = tostring(attrNum / 100) .. "%"
        if attrIndex > 0 then

            local config1=  ConfigDataAccessor.GetAttrByID(attrIndex)
            local attrName1 = "";
            if config1 ~= nil then
                attrName1 = config1.Name;
            end


            if attrName1 ~= "" then
                local attrItem = self:GetAttrItemByIndex(i)
                if attrItem ~= nil then
                    attrItem:Init(attrName1, attrNumStr)
                end
            else
                Log("PartnerStarRiseFinishPanel:SetRiseStarData err!!! AoiAttr=nil,attrIndex=" .. attrIndex)
            end
        end
    end

    -- 技能
    self.SkillDes.text = self:GetParSkillDes()

    -- 资质
    self.QualificationDes.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_QualificationDes"), {self.StarExpendCfg.Star});
end

-- 根据下标获得属性Item
function PartnerStarRiseFinishPanel:GetAttrItemByIndex(index)
    local attrItem = self.AttrItemList[index]

    if attrItem == nil then
        local _go = newObject(self.AttrItem)
        _go.transform:SetParent(self.AttrRoot)
        _go.transform.localPosition = Vector3.zero
        _go.transform.localScale = Vector3.one
        attrItem = RiseStarAttrItem.new(_go)
        self.AttrItemList[index] = attrItem
    end

    return attrItem
end

-- 获得技能描述
function PartnerStarRiseFinishPanel:GetParSkillDes()
    for _k, _v in pairs(ConfigDataAccessor.ParSkill) do
        if _v.Par == self.CurPartnerCfg.Id and _v.Star == self.CurSelectPartnerData.CurStar then
            return _v.Desc
        end
    end

    return ""
end

-- 获得品质-颜色图标
function PartnerStarRiseFinishPanel:GetQualityName(quality)
    if quality == 1 then
        return "Partner_quality_n"
    elseif quality == 2 then
        return "Partner_quality_r"
    elseif quality == 3 then
        return "Partner_quality_sr"
    elseif quality == 4 then
        return "Partner_quality_ssr"
    end
end

-- 关闭属性Item
function PartnerStarRiseFinishPanel:CloseAttrItemList()
    for i = 1, #self.AttrItemList do
        local item = self.AttrItemList[i]
        if item ~= nil and item.prefab ~= nil then
            item.prefab.gameObject:SetActive(false)
        end
    end
end

function PartnerStarRiseFinishPanel:ClosePanel()
    self.Prefab.gameObject:SetActive(false)
    self:CloseAttrItemList()
end

return PartnerStarRiseFinishPanel

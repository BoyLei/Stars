PurchaseFilterItem=class("PurchaseFilterItem",UIPanel)
PurchaseFilterItem.m_selected=nil;
PurchaseFilterItem.m_selectedTex=nil;
PurchaseFilterItem.m_noselected=nil;
PurchaseFilterItem.m_noselectedTex=nil;
PurchaseFilterItem.m_Item=nil;
PurchaseFilterItem.Type=nil;
local ItemFunc=nil;


function PurchaseFilterItem:ctor(go, atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_selected=self:Find("selected").gameObject;
	self.m_noselected=self:Find("noselected").gameObject;
	self.m_selectedTex=self:Find("selected/tex"):GetComponent(typeof(Text));
	self.m_noselectedTex=self:Find("noselected/tex"):GetComponent(typeof(Text));
	self.m_Item=self.prefab:GetComponent(typeof(JButton));
	ItemFunc=function() self:OnClickItem(); end;
	self.m_Item.OnClick= ItemFunc;
end


function PurchaseFilterItem:OnClickItem()
	if self.OnSelectCallBack~=nil then
		self.OnSelectCallBack(self.Type);
	end
end

function PurchaseFilterItem:Select(select)
	self.m_selected:SetActive(select)
	self.m_noselected:SetActive(not select)
end

function PurchaseFilterItem:UpdateContent(arg,cb)
	self.OnSelectCallBack=cb;
	self.Type=arg.Type
	self.m_selectedTex.text=arg.Content
	self.m_noselectedTex.text=arg.Content
end



return  PurchaseFilterItem

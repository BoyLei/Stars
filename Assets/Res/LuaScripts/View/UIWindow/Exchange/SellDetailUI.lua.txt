function SellDetailUI:SellTansBtn1()

function SellDetailUI:OnSellBtn()
	if self.SelectID==0 then
		return
	end
	local total=self.Count*self.Price
	local cost=math.floor(total*SystemConstConfigs.TradePutawayTax/100)
	--查看背包银币数量
	
	local money= BusinessManager.Instance:GetCurrencyNum(1);
	if cost>money then
		Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_81"));
		return 
	end
	
	LuaModuleDispatcher:SendMessage(ModuleDef.ExchangeModule,"OnBankAddGoodsReq",{self.SelectData.ID,self.Count,self.Price})
end

function  SellDetailUI:OnItemRefesh(tr,idx)
	if self.ItemList==nil then
		return;
	end
	for i=1,#self.ItemList do
		if self.ItemList[i].transform == tr then
			self.ItemList[i]:UpdateContent(self.ItemDatas[idx+1],self.SelectID,function (data)
					self:OnShowItem(data);
				end);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
			return
		end
	end

	local bi = SellDetailItem.new(tr.gameObject,self.AtlasPath);
	bi.prefab:SetActive(true);
	bi:UpdateContent(self.ItemDatas[idx+1],self.SelectID,function (data)

			self:OnShowItem(data);
		end);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
	table.insert(self.ItemList, bi);
end

function SellDetailUI:OnShowItem(data)
	
	self.SelectID=data.CIndex
	self.SelectData=data.data
	for k, v in pairs(self.ItemList) do
		v:Select(v.Data.CIndex==self.SelectID)
	end
	ItemID=data.data.BaseID;
	local config=ConfigDataAccessor.Item[data.data.BaseID]
	if config~=nil then
		if not IsEquip(config.ItemType) then
			self.itemTipsItem:Refesh(config,data.data.ID)
			self.m_Item:SetActive(true)
			self.m_Equip:SetActive(false)
		else
			--self.itemTipsEquip:Refesh(config,data.data.ID)
			self.itemTipsEquip:RefeshEquipByData(data.data);
			self.m_Item:SetActive(false)
			self.m_Equip:SetActive(true)
		end
	else
		self.m_Item:SetActive(false)
		self.m_Equip:SetActive(false)
	end

	self.SellRecord:OnOpen(data.data.BaseID)

	self.SellCount:OnOpen(data.data,function (count)
			self:OnCountChange(count)
		end);
	
	self.SellPrice:OnOpen(data.data,function (price)
			self:OnPriceChange(price)
		end);
end

function SellDetailUI:OnCountChange(count)
	self.Count=count;
	self:OnFreshPirce();
end

function SellDetailUI:OnPriceChange(price)
	self.Price=price;
	self:OnFreshPirce();
end


function SellDetailUI:OnFreshPirce()
	local total=self.Count*self.Price

	-- 税费 买方交易金币税率（/100）
	local rate=math.floor(total*SystemConstConfigs.TradeSuccessTax/100)
	self.m_RatePrice.text=rate

	
	---上架费 买方上架银币税率（/100）
	local cost=math.floor(total*SystemConstConfigs.TradePutawayTax/100)
	self.m_Price.text=cost
	
	self.m_Gain.text=math.floor(total-rate)
end

function SellDetailUI:OnOpen(arg)
	self.IsActive=true;
	self.CallBack=arg;
	self.SelectData=nil;
	self:InitGoods();
end



function SellDetailUI:InitGoods()
	self.BagDatas={}

	local items=BusinessManager.Instance:GetItemsByItemSpaceId(0);
	local CIndex=1;
	for k, v in pairs(items) do
		local id= v.BaseID
		local config=ConfigDataAccessor.Item[id]
		if config~=nil then
			if config.IsTradable and not(v.IsBind) then
				table.insert(self.BagDatas,{data=v,CIndex=CIndex})
				CIndex=CIndex+1;
			end
			--local key=GetSubItemID(tonumber(config.ItemType),tonumber(config.SubType))
			--local itemSubtype= ConfigDataAccessor.ItemType[key]
			--if itemSubtype~=nil then
		
			--end
		end
	end

	local empytCnt=pageCount;
	local cnt=#self.BagDatas;
	if cnt> 0 then
		empytCnt=empytCnt+cnt%pageCount;
		self.m_Has:SetActive(true);
		self.m_None:SetActive(false);
	else
		self.m_Has:SetActive(false);
		self.m_None:SetActive(true);
	end

	self.ItemDatas={}
	for k, v in pairs(self.BagDatas) do
		if self.SelectID==0 then
			self.SelectID=v.CIndex
		end

		table.insert(self.ItemDatas,v);
	end

	for i = 1, empytCnt do
		table.insert(self.ItemDatas,nil);
	end

	self.m_ScrollView:ContinueScroll(#self.ItemDatas);
end

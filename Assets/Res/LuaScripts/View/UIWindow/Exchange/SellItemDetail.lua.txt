    self.m_SortBtn = self:Find("SortBtn"):GetComponent(typeof(JButton));

function SellItemDetail:OnClickSortBtnBtn()
    self.OrderPass = not self.OrderPass;
    self.m_SortUp:SetActive(self.OrderPass)
    self.m_SortDown:SetActive(not self.OrderPass)
    self:SortGoods();
    self:InitGrids();

    -- LuaModuleDispatcher:SendMessage(ModuleDef.ExchangeModule,"OnSetOrderRule",{true,PurchaseType,self:GetSortIndex(),self.CurrentDropIndex})
end

function SellItemDetail:GetSortIndex()
    if self.OrderPass then
        return 1
    end
    return 2
end

function SellItemDetail:SortGoods()
    local order = self:GetSortIndex();
    if self.CurrentDropIndex == 0 then
        -- "持续时间"
        if order == 1 then
            table.sort(self.Goods, function(good1, good2)
                return good1.Etime < good2.Etime
            end)
            -- 从小到大
        else
            -- 从大到小
            table.sort(self.Goods, function(good1, good2)
                return good1.Etime > good2.Etime
            end)
        end

    elseif self.CurrentDropIndex == 1 then
        -- "可售出总价"
        if order == 1 then
            table.sort(self.Goods, function(good1, good2)
                return good1.Price < good2.Price
            end)
            -- 从小到大
        else
            -- 从大到小
            table.sort(self.Goods, function(good1, good2)
                return good1.Price > good2.Price
            end)
        end

    elseif self.CurrentDropIndex == 2 then
        -- "品阶"
        if order == 1 then
            table.sort(self.Goods, function(good1, good2)
                return good1.Quality < good2.Quality
            end)
            -- 从小到大
        else
            -- 从大到小
            table.sort(self.Goods, function(good1, good2)
                return good1.Quality > good2.Quality
            end)
        end

    end
end

function SellItemDetail:InitGrids()
    self.SellGoods = {}
    local Index = 1;
    for k, v in pairs(self.Goods) do
        if self.SelectGoods == 0 then
            self.SelectGoods = Index
        end
        table.insert(self.SellGoods, {
            Index = Index,
            data = v
        })
        Index = Index + 1
    end
    self.m_ScrollView:ContinueScroll(#self.SellGoods);
end
function SellItemDetail:OnDropDownHandler(index)
    self.CurrentDropIndex = index;
    -- 重新排序
    self:SortGoods();
    self:InitGrids();
end

function SellItemDetail:OnDestroy()

    self.m_SellBtn.OnClick = nil;

    self.m_Dropdown.onValueChanged:RemoveListener(OnDropDownFunc);

    CS.SGF.Network.MsgRetManager.Instance:OffMessageEnum(OnBankOffShelfGoodsRetFunc, CS.MsgIDEnum.BankOffShelfGoodsReqID);

end

function SellItemDetail:OnSellGridItemRefesh(tr, idx)
    if self.SellItemList == nil then
        return;
    end
    for i = 1, #self.SellItemList do
        if self.SellItemList[i].transform == tr then
            self.SellItemList[i]:UpdateContent(self.SellGoods[idx + 1], self.SelectGoods, function(data, index)
                self:OnSelectCallBack(data, index)
            end); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end

    local bi = SellGoodsDetailItem.new(tr.gameObject, self.AtlasPath);
    bi.prefab:SetActive(true);
    bi:UpdateContent(self.SellGoods[idx + 1], self.SelectGoods, function(data, index)
        self:OnSelectCallBack(data, index)
    end); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.SellItemList, bi);
end

function SellItemDetail:OnSelectCallBack(data, index)
    for i = 1, #self.SellItemList do
        self.SellItemList[i]:Select(false)
    end
    self.GoodsID = data.GoodsID
    self.SelectGoods = index
    self.m_SellCount.text = data.ItemNum;
    self.m_SellPrice.text = data.Price;
    local rate = math.floor(data.TotalPrice * SystemConstConfigs.TradeSuccessTax / 100)
    self.m_Gain.text = (data.TotalPrice - rate);
    local config = ConfigDataAccessor.Item[data.ItemID]
    if config ~= nil then
        self.m_ItemName.text = config.Name
        -- local quality=config.Quality;
        -- ResourceHelper:LoadIconByAtlasAsync(
        -- self.m_BgRare,
        -- ResourceHelper.AtlasPathCommonItem,
        -- "Bag_Grid_sq"..quality,
        -- false
        -- )
        ResourceHelper:LoadIconAsync(self.m_ItemIcon, "", config.Icon, true)
    end
    self.m_Time.text = SecToStringColon((data.Etime - TimeUtils.GetServerTimeNow() / 1000))
end

function SellItemDetail:OnOpen(arg)
    self.IsActive = true;
    self.SellItemList = {}
    self.CallBack = arg.Back;
    self.Goods = arg.Goods;
    self.SelectGoods = 0;
    self:InitDropdown();
    self:SortGoods();
    self:InitGrids();
end

function SellItemDetail:InitDropdown()
    self.m_Dropdown:ClearOptions();
    self.CurrentDropIndex = 0;
    local list = List_String()
    for i = 1, #OptionsItems do
        list:Add(OptionsItems[i])
    end
    self.m_Dropdown:AddOptions(list);
    -- self:OnFresh();
end

---@class SellRecordItem
SellRecordItem=class("SellRecordItem",UIPanel)
SellRecordItem.m_price=nil;
SellRecordItem.m_num=nil;

function SellRecordItem:ctor(go, atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_price=self:Find("price"):GetComponent(typeof(Text));
	self.m_num=self:Find("num"):GetComponent(typeof(Text));
end

function SellRecordItem:UpdateContent(arg)
	self.m_price.text=arg.TotalPrice
	self.m_num.text=arg.ItemNum
end



return  SellRecordItem

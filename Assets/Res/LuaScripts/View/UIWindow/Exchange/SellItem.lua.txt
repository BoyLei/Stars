SellItem=class("SellItem",UIPanel)
SellItem.m_BgRare=nil;
SellItem.m_Icon=nil;
SellItem.m_Name=nil;
SellItem.m_Time=nil;
SellItem.m_Price=nil;
SellItem.ItemID=0;


function SellItem:ctor(go, atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_BgRare=self:Find("BgRare"):GetComponent(typeof(Image));
	self.m_Icon=self:Find("Icon"):GetComponent(typeof(Image));
	self.m_Name=self:Find("Name"):GetComponent(typeof(Text));
	self.m_Time=self:Find("Time"):GetComponent(typeof(Text));
	self.m_Price=self:Find("Price"):GetComponent(typeof(Text));
end



function SellItem:UpdateContent(arg)
	--BankLogBase
	self.ItemID=arg.ItemID;
	local config=ConfigDataAccessor.Item[arg.ItemID]
	if config~=nil then
		self.m_Name.text=config.Name
		local quality=config.Quality;
		ResourceHelper:LoadIconByAtlasAsync(
			self.m_BgRare,
			ResourceHelper.AtlasPathCommonItem,
			"Bag_Grid_sq"..quality,
			false
		)
		ResourceHelper:LoadIconAsync(self.m_Icon, "", config.Icon, false)
	end

	self.m_Price.text=arg.TotalPrice
	self.m_Time.text=TimeToFullString2(TimeUtils.Stamp2DataTime(arg.TranTime) )
end



return  SellItem

local DefAmuletEquipInfoEmblemItem = {}

DefAmuletEquipInfoEmblemItem.__index = DefAmuletEquipInfoEmblemItem

function DefAmuletEquipInfoEmblemItem:New(_go)
    local self = {}
    setmetatable(self, DefAmuletEquipInfoEmblemItem)

    self.gameObject = _go
    self.transform = _go.transform

    self.go_mach = self.transform:Find("mach").gameObject

    self.IconBg = self.transform:Find("iconbg"):GetComponent(typeof(Image))
    self.Icon = self.transform:Find("icon"):GetComponent(typeof(Image))

    self.Name = self.transform:Find("name"):GetComponent(typeof(Text)) --名字

    --点击事件
    self.btnClick = self.transform:GetComponent(typeof(JButton))
    if self.btnClick ~= nil then
        self.btnClick.OnClick = (function()
            if self.OnItemClick ~= nil then
                self.OnItemClick(self)
            end
        end)
    end

    return self --返回自身
end

function DefAmuletEquipInfoEmblemItem:InitItem(idx, onItemClick)
    self.Idx = idx
    self.OnItemClick = onItemClick
end

function DefAmuletEquipInfoEmblemItem:SetInfo(gemSlot)
    local slotID = gemSlot.SlotID
    local gemID = gemSlot.GemID
    --self.IconBg.color=GetEmblemBgColorRGB(slotID);

    self.SlotID = slotID

    if slotID > 9 then
        --self.IconBg.sprite = AtlasManager.Instance:GetSprite("ui/amulet/atlas/rune", "Rune_slot_" .. slotID)
        ResourceHelper:LoadIconByAtlasAsync(
            self.IconBg,
            ResourceHelper.AtlasPathRune,
            "Rune_slot_" .. slotID,
            false
        )
    else
        --self.IconBg.sprite = AtlasManager.Instance:GetSprite("ui/amulet/atlas/rune", "Rune_slot_0" .. slotID)
        ResourceHelper:LoadIconByAtlasAsync(
            self.IconBg,
            ResourceHelper.AtlasPathRune,
            "Rune_slot_0" .. slotID,
            false
        )
    end

    --刷新宝石槽颜色

    if gemID ~= nil and gemID ~= 0 then
        self.Icon.gameObject:SetActive(true)

        local itemData = BusinessManager.Instance:GetItemByItemEntityId(gemID)

        self.EntityID = itemData.EntityID
        self.BaseID = itemData.BaseID

        local cfg = ConfigDataAccessor.Item[self.BaseID]
        --LoadIcon(self.Icon,cfg.Icon);
        ResourceHelper:LoadIconAsync(self.Icon, "", cfg.Icon, false)

        --宝石的共鸣
        local heraldryEquipCfg = ConfigDataAccessor.HeraldryEquip[self.BaseID]

        --判断是是否触发了共鸣
        if slotID == heraldryEquipCfg.ResonanceSlot then
            --self.Met.text="[已共鸣]";
            self.go_mach:SetActive(true)

            local BaseAttrId = heraldryEquipCfg.BaseAttrId[1]
            local BaseAttrVal = heraldryEquipCfg.BaseAttrVal[1]

            local ResonanceAttrID = heraldryEquipCfg.ResonanceAttrID[1]
            local ResonanceAttrVal = heraldryEquipCfg.ResonanceAttrVal[1]

            local config1=  ConfigDataAccessor.GetAttrByID(BaseAttrId)
            local attrName1 = "";
            if config1 ~= nil then
                attrName1 = config1.Name;
            end
        
            local config2=  ConfigDataAccessor.GetAttrByID(ResonanceAttrID)
            local attrName2 = "";
            if config2 ~= nil then
                attrName2 = config2.Name;
            end

            local name =
            attrName1 ..
                "+" .. (BaseAttrVal + ResonanceAttrVal) .. "[" .. BaseAttrVal .. "+" .. ResonanceAttrVal .. "]"

            self.Name.text = cfg.Name .. ":" .. name
        else
            --self.Met.text="[未共鸣]";
            self.go_mach:SetActive(false)

            local BaseAttrId = heraldryEquipCfg.BaseAttrId[1]
            local BaseAttrVal = heraldryEquipCfg.BaseAttrVal[1]

            local config1=  ConfigDataAccessor.GetAttrByID(BaseAttrId)
            local attrName1 = "";
            if config1 ~= nil then
                attrName1 = config1.Name;
            end
        
            local config2=  ConfigDataAccessor.GetAttrByID(ResonanceAttrID)
            local attrName2 = "";
            if config2 ~= nil then
                attrName2 = config2.Name;
            end

            local name = attrName1 .. "+" .. BaseAttrVal

            self.Name.text = cfg.Name .. ":" .. name
        end
    else
        self.Icon.gameObject:SetActive(false)
        self.go_mach:SetActive(false)
        self.Name.text = LanguageHelp:GetLanguageByKey("Local_Str_ItemTipsEmblemItem_Name2")
    end
end

function DefAmuletEquipInfoEmblemItem:RefeshBoard(slotID)
    if slotID > 9 then
        --self.IconBg.sprite = AtlasManager.Instance:GetSprite("ui/amulet/atlas/rune", "Rune_slot_" .. slotID)
        ResourceHelper:LoadIconByAtlasAsync(self.IconBg, ResourceHelper.AtlasPathRune, "Rune_slot_" .. slotID, false)
    else
        --self.IconBg.sprite = AtlasManager.Instance:GetSprite("ui/amulet/atlas/rune", "Rune_slot_0" .. slotID)
        ResourceHelper:LoadIconByAtlasAsync(self.IconBg, ResourceHelper.AtlasPathRune, "Rune_slot_0" .. slotID, false)
    end

    --刷新宝石槽颜色
    self.Icon.gameObject:SetActive(false)
    self.Name.text = ""
    self.go_mach:SetActive(false)
end

return DefAmuletEquipInfoEmblemItem

local PolishAmuletSkillPoint = {}

PolishAmuletSkillPoint.__index = PolishAmuletSkillPoint

function PolishAmuletSkillPoint:New(_go)
    local self = {}
    setmetatable(self, PolishAmuletSkillPoint)

    self.gameObject = _go
    self.transform = _go.transform

    self.points = {}
    self.pointIcons = {}
    self.pointIconsAnim = {}
    for i = 1, 10 do
        local pt = self.transform:Find("points/point" .. i).gameObject
        local ptIc = self.transform:Find("points/point" .. i .. "/Image").gameObject
        local ptIcAmin = self.transform:Find("points/point" .. i .. "/anim").gameObject

        table.insert(self.points, pt)
        table.insert(self.pointIcons, ptIc)
        table.insert(self.pointIconsAnim, ptIcAmin)
    end

    self.skillIcon = self.transform:Find("iconbg/icon"):GetComponent(typeof(Image))

    self.title = self.transform:Find("title"):GetComponent(typeof(Text))
    self.nextLv = self.transform:Find("title2"):GetComponent(typeof(Text))
    self.add = self.transform:Find("add"):GetComponent(typeof(Text))

    --详细信息
    local btnInfo = self.transform:Find("iconbg"):GetComponent(typeof(JButton))
    btnInfo.OnClick = (function()
        LuaModuleDispatcher:SendMessage(ModuleDef.AmuletModule, "OnShowEmblemSkillInfo", {self.SkillID})
    end)

    return self --返回自身
end

function PolishAmuletSkillPoint:Refesh(maxnum, cs, allList)
    for i = 1, #self.points do
        self.points[i].gameObject:SetActive(false)
        self.pointIcons[i].gameObject:SetActive(false)
        self.pointIconsAnim[i].gameObject:SetActive(false)
    end

    if cs ~= nil then
        self.SkillID = cs.ID --技能id
        --显示图标
        local ID = cs.ID --技能id

        --点的数量
        local num = 0

        for i = 0, allList.Count - 1 do
            local cs2 = allList[i]
            if self.SkillID == cs2.ID then
                local Points2 = cs2.Points --点数分布，二进制
                local pList2 = self:_byte2bin(Points2)

                for i = 1, maxnum do
                    if pList2[33 - i] > 0 then
                        num = num + 1
                    end
                end
            end
        end

        local num2 = 0

        local Points = cs.Points --点数分布，二进制
        local pList = self:_byte2bin(Points)
        for i = 1, maxnum do
            self.points[i].gameObject:SetActive(true)
            if pList[33 - i] > 0 then
                self.pointIcons[i].gameObject:SetActive(true)
                num2 = num2 + 1
            end
        end

        local maxLv = ConfigDataAccessor.GetPassiveDescMaxLv(ID)

        --每一级的点数
        local perLevelPoint = 5

        local isMaxLv = false
        if num >= maxLv * perLevelPoint then
            isMaxLv = true
        end

        if isMaxLv then
            local pd = ConfigDataAccessor.GetPassiveDesc(ID, maxLv)
            ResourceHelper:LoadIconByAtlasAsync(self.skillIcon, ResourceHelper.AtlasPathEtchSkill, pd.IconPath, false)

            self.title.text = pd.SkillName
            --self.add.text = "精通+" .. num2
            self.add.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_master"), {num2})

            self.nextLv.text = LanguageHelp:GetLanguageByKey("Local_Str_MaxLevel2")
        else
            --计算到下一个等级需要的点数
            local leftNum = 1

            --计算技能等级

            local lvNext = 1
            for i = 1, (maxLv - 1) do
                if lvNext * perLevelPoint > num then
                    leftNum = lvNext * perLevelPoint - num
                    break
                else
                    lvNext = lvNext + 1
                end
            end

            local lvNow = lvNext - 1 --当前等级

            local pd = ConfigDataAccessor.GetPassiveDesc(ID, lvNext)
            ResourceHelper:LoadIconByAtlasAsync(self.skillIcon, ResourceHelper.AtlasPathEtchSkill, pd.IconPath, false)

            self.title.text = pd.SkillName
            --self.add.text = "精通+" .. num2
            self.add.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_master"), {num2})

            --self.nextLv.text = "距离下一级提升还差<color=#FFAC5A>" .. leftNum .. "</color>点"
            self.nextLv.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_nextLv"), {leftNum})
        end
    end
end

function PolishAmuletSkillPoint:_byte2bin(n)
    --print("_byte2bin"..n);

    local t = {}
    for i = 32, 0, -1 do
        t[#t + 1] = math.floor(n / 2 ^ i)
        n = n % 2 ^ i
    end

    --print(table.concat(t));

    return t
    --return table.concat(t)
end

function PolishAmuletSkillPoint:ShowPolishPs(idx)
    if self.pointIcons[idx].gameObject.activeSelf then
        self.pointIconsAnim[idx].gameObject:SetActive(true)
    end
end

return PolishAmuletSkillPoint

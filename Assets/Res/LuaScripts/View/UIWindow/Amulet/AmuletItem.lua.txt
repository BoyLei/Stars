local AmuletItem = {}

AmuletItem.__index = AmuletItem

function AmuletItem:New(_go)
    local self = {}
    setmetatable(self, AmuletItem)

    self.gameObject = _go
    self.transform = _go.transform

    self.transform:GetComponent(typeof(JButton)).OnClick = (function()
        if self.OnSelected ~= nil then
            self.OnSelected(self)
        end
    end)

    self.go_selected = self.transform:Find("Selected").gameObject

    self.ps1 = self.transform:Find("center_Fx_Yuan"):GetComponent(typeof(Image))
    self.ps2 = self.transform:Find("center_Fx"):GetComponent(typeof(Image))

    self.go_have = self.transform:Find("EquipItem/Have").gameObject
    self.go_lock = self.transform:Find("EquipItem/Lock").gameObject
    self.tex_lock = self.transform:Find("EquipItem/Lock/title"):GetComponent(typeof(Text))

    self.go_null = self.transform:Find("EquipItem/Null").gameObject

    self.Center = self.transform:Find("center"):GetComponent(typeof(Image))
    self.Icon = self.transform:Find("EquipItem/Have/Icon"):GetComponent(typeof(Image))
    self.Lv = self.transform:Find("EquipItem/Have/LvBg/name"):GetComponent(typeof(Text))

    self.go_LvBg = self.transform:Find("EquipItem/Have/LvBg").gameObject
    self.go_LvBg:SetActive(false)

    self.EmblemSlotBg = {}
    self.EmblemSlotIcon = {}

    for i = 1, 4 do
        local bg = self.transform:Find("EquipItem/Have/RuneItem" .. i .. "/iconbg"):GetComponent(typeof(Image))
        local icon = self.transform:Find("EquipItem/Have/RuneItem" .. i .. "/icon"):GetComponent(typeof(Image))
        table.insert(self.EmblemSlotBg, bg)
        table.insert(self.EmblemSlotIcon, icon)
    end

    self.LockTitle = self.transform:Find("EquipItem/Lock/title"):GetComponent(typeof(Text))

    self.Name = self.transform:Find("Name"):GetComponent(typeof(Text))

    return self --返回自身
end

function AmuletItem:InitItem(idx, onSelected)
    self.OnSelected = onSelected
    self.Idx = idx
end

function AmuletItem:MakeSelect(b)
    self.go_selected:SetActive(b)
    if b then
        if self.Type == 1 then
            self.ps1.gameObject:SetActive(true)
            self.ps2.gameObject:SetActive(true)
        else
            self.ps1.gameObject:SetActive(false)
            self.ps2.gameObject:SetActive(false)
        end
    else
        self.ps1.gameObject:SetActive(false)
        self.ps2.gameObject:SetActive(false)
    end
end

function AmuletItem:SetInfo(amuletSlot)
    if amuletSlot.IsEquiped then
        local data = BusinessManager.Instance:GetItemByItemEntityId(amuletSlot.EntityID)

        local baseID = data.BaseID
        local combatSkills = data.CombatSkills
        local gemSlots = data.GemSlots

        --查看宝石
        for i = 0, gemSlots.List.Count - 1 do
            self.EmblemSlotIcon[i + 1].gameObject:SetActive(false)
            --刷新宝石槽位颜色
            local slotID = gemSlots.List[i].SlotID
            local gemID = gemSlots.List[i].GemID

            if gemID ~= nil and gemID ~= 0 then
                self.EmblemSlotIcon[i + 1].gameObject:SetActive(true)
                local itemData = BusinessManager.Instance:GetItemByItemEntityId(gemID)
                local cfg = ConfigDataAccessor.Item[itemData.BaseID]
                --LoadIcon(self.EmblemSlotIcon[i+1],cfg.Icon);
                ResourceHelper:LoadIconAsync(self.EmblemSlotIcon[i + 1], "", cfg.Icon, false)
            end

            --self.EmblemSlotBg[i+1].color=GetEmblemBgColorRGB(slotID);
            if slotID > 9 then
                ResourceHelper:LoadIconByAtlasAsync(
                    self.EmblemSlotBg[i + 1],
                    ResourceHelper.AtlasPathRune,
                    "Rune_slot_s_" .. slotID,
                    false
                )
            else
                ResourceHelper:LoadIconByAtlasAsync(
                    self.EmblemSlotBg[i + 1],
                    ResourceHelper.AtlasPathRune,
                    "Rune_slot_s_0" .. slotID,
                    false
                )
            end
        end

        --图标
        local cfg2 = ConfigDataAccessor.Item[baseID]
        --LoadIcon(self.Icon,cfg2.Icon);
        ResourceHelper:LoadIconAsync(self.Icon, "", cfg2.Icon, false)

        --特效颜色值
        self.ps1.color = self:GetPsColor1(cfg2.Quality)
        self.ps2.color = self:GetPsColor2(cfg2.Quality)

        ResourceHelper:LoadIconByAtlasAsync(
            self.Center,
            ResourceHelper.AtlasPathRune,
            "Rune_quality_0" .. (6 - cfg2.Quality),
            false
        )
        self.Name.text = cfg2.Name
    else
        ResourceHelper:LoadIconByAtlasAsync(self.Center, ResourceHelper.AtlasPathRune, "Rune_quality_05", false)
        --道具类型
        if self.Type==3 then
            self.Name.text = ""
        else
            if self.Idx == 1 then
                self.Name.text = LanguageHelp:GetLanguageByKey("Local_Str_Amulet_Atk")
            elseif self.Idx == 2 then
                self.Name.text = LanguageHelp:GetLanguageByKey("Local_Str_Amulet_Def")
            end
        end
    end
end

function AmuletItem:GetPsColor1(quality)
    if quality == 1 then
        return Color(210 / 255, 220 / 255, 215 / 255, 10 / 255)
    elseif quality == 2 then
        return Color(126 / 255, 255 / 255, 198 / 255, 10 / 255)
    elseif quality == 3 then
        return Color(85 / 255, 140 / 255, 255 / 255, 10 / 255)
    elseif quality == 4 then
        return Color(173 / 255, 47 / 255, 220 / 255, 10 / 255)
    elseif quality == 5 then
        return Color(255 / 255, 167 / 255, 47 / 255, 10 / 255)
    end
end

function AmuletItem:GetPsColor2(quality)
    if quality == 1 then
        return Color(210 / 255, 220 / 255, 215 / 255)
    elseif quality == 2 then
        return Color(126 / 255, 255 / 255, 198 / 255)
    elseif quality == 3 then
        return Color(85 / 255, 140 / 255, 255 / 255)
    elseif quality == 4 then
        return Color(173 / 255, 47 / 255, 220 / 255)
    elseif quality == 5 then
        return Color(255 / 255, 167 / 255, 47 / 255)
    end
end

function AmuletItem:SetType(Type,tex)
    self.Type = Type
    self.IsNull = false
    if Type == 1 then
        self.IsLock=false;
        --装备中
        self.go_have:SetActive(true)
        self.go_lock:SetActive(false)
        self.go_null:SetActive(false)
    elseif Type == 2 then
        --未装备
        self.IsLock=false;
        self.go_have:SetActive(false)
        self.go_lock:SetActive(false)
        self.go_null:SetActive(true)
        self.ps1.gameObject:SetActive(false)
        self.ps2.gameObject:SetActive(false)
        self.IsNull = true
    elseif Type == 3 then
        --未解锁
        self.IsLock=true;
        self.go_have:SetActive(false)
        self.go_lock:SetActive(true)
        self.go_null:SetActive(false)
        self.ps1.gameObject:SetActive(false)
        self.ps2.gameObject:SetActive(false)
        self.tex_lock.text=tex;
        self.Name.text = ""
    end
end

return AmuletItem

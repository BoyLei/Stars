local EmblemItemFull = {}

EmblemItemFull.__index = EmblemItemFull

function EmblemItemFull:New(_go) 
    local self = {};    
    setmetatable(self, EmblemItemFull); 


    self.gameObject=_go;
    self.transform=_go.transform;


    self.Num=self.transform:Find("Have/Num"):GetComponent(typeof(Text));--数量
    self.Name=self.transform:Find("Have/Name"):GetComponent(typeof(Text));--数量
    self.Attr1=self.transform:Find("Have/attr1"):GetComponent(typeof(Text));--数量
    self.Attr2=self.transform:Find("Have/attr2"):GetComponent(typeof(Text));--数量


    self.go_gongming=self.transform:Find("Have/gongming").gameObject;
    self.go_notgongming=self.transform:Find("Have/notgongming").gameObject;


    self.Icon=self.transform:Find("Have/Icon"):GetComponent(typeof(Image));

    self.go_Selected=self.transform:Find("Have/Selected").gameObject;
    self.go_Selected:SetActive(false);


    self.go_Have=self.transform:Find("Have").gameObject;
    self.go_Have:SetActive(true);

    self.go_Menu=self.transform:Find("Menu").gameObject;
    self.go_Menu:SetActive(false);

    self.IsUsed=false;



    --点击事件
	self.btnClick=self.transform:GetComponent(typeof(JButton));
	self.btnClick.OnClick=(function() 
        if self.OnItemClick ~= nil then
            if self.IsUsed then
		        self.OnItemClick(self); 
            else
                self.OnItemClick(nil); 
            end
        end
	end);





    --点击事件
	self.btnClick=self.transform:GetComponent(typeof(JButton));
	self.btnClick.OnClick=(function() 
        if self.OnItemClick ~= nil then
            if self.IsUsed then
		        self.OnItemClick(self); 
            else
                self.OnItemClick(nil); 
            end
        end
	end);
	


    return self;    --返回自身
end


--初始化Item
function EmblemItemFull:InitItem(onItemClick,isEquiped)
    self.IsEquiped=isEquiped;

    self.OnItemClick=onItemClick;
    self:SetUse(false);
end



function EmblemItemFull:SetUse(b)
    self.IsUsed=b;

    --如果没有有底板就隐藏
    self.gameObject:SetActive(b);
  
end



--设置Item
function EmblemItemFull:SetItem(item,slotIdx)


    --背包特判
    if item.BaseID== 0 then
        self:SetUse(false);
        self:MakeSelected(false);
    else
        self:SetUse(true);


        self.Item=item;

        self.EntityID=item.EntityID;
        self.BaseID=item.BaseID;
        


        local cfg=ConfigDataAccessor.Item[self.BaseID];

        local heraldryEquipCfg=ConfigDataAccessor.HeraldryEquip[self.BaseID];

        local isGongming=false;
        --判断是是否触发了共鸣
        if slotIdx== heraldryEquipCfg.ResonanceSlot then

            local BaseAttrId=heraldryEquipCfg.BaseAttrId[1];
            local BaseAttrVal=heraldryEquipCfg.BaseAttrVal[1];

            local ResonanceAttrID=heraldryEquipCfg.ResonanceAttrID[1];
            local ResonanceAttrVal=heraldryEquipCfg.ResonanceAttrVal[1];

            local config1=  ConfigDataAccessor.GetAttrByID(BaseAttrId)
            local attrName1 = "";
            if config1 ~= nil then
                attrName1 = config1.Name;
            end
        
            local config2=  ConfigDataAccessor.GetAttrByID(ResonanceAttrID)
            local attrName2 = "";
            if config2 ~= nil then
                attrName2 = config2.Name;
            end
    
            --self.Attr1.text="基础：<color=#8AACD1>"..attrcfg1.desc.."+"..BaseAttrVal.."</color>";

            self.Attr1.text = LanguageHelp:GetLanguageByKey("Local_Str_amuletattrs1") .. "：<color=#8AACD1>"..attrName1.."+"..BaseAttrVal.."</color>";
            self.Attr2.text="<color=#8AACD1>"..attrName2.."+"..ResonanceAttrVal.."</color>";

            self.go_gongming:SetActive(true);
            self.go_notgongming:SetActive(false);

            isGongming=true;
        else

            local BaseAttrId=heraldryEquipCfg.BaseAttrId[1];
            local BaseAttrVal=heraldryEquipCfg.BaseAttrVal[1];

            local config1=  ConfigDataAccessor.GetAttrByID(BaseAttrId)
            local attrName1 = "";
            if config1 ~= nil then
                attrName1 = config1.Name;
            end
        
            local config2=  ConfigDataAccessor.GetAttrByID(ResonanceAttrID)
            local attrName2 = "";
            if config2 ~= nil then
                attrName2 = config2.Name;
            end
    
            --self.Attr1.text="基础：<color=#8AACD1>"..attrcfg1.desc.."+"..BaseAttrVal.."</color>";
            self.Attr1.text = LanguageHelp:GetLanguageByKey("Local_Str_amuletattrs1") .. "：<color=#8AACD1>"..attrName1.."+"..BaseAttrVal.."</color>";


            --self.Attr2.text="<color=#B6C9D5>未共鸣</color>";
            self.Attr2.text="";
            self.go_gongming:SetActive(false);
            self.go_notgongming:SetActive(true);
        end

		--LoadIcon(self.Icon,cfg.Icon);
        ResourceHelper:LoadIconAsync(self.Icon, "", cfg.Icon, false)

        --战力
		local equipPower=FightPowerManager.Instance:GetShowHeraldryPowerByGemID(self.BaseID,isGongming);
		local power=LanguageHelp:GetLanguageByKey("Local_Str_Tips_Condition27_Power")..(equipPower);

        if isGongming then
            local str="<color=#94D18A>"..power.."</color>"
           -- self.Name.text = cfg.Name.. str;
           self.Name.text = cfg.Name.." <size=24>"..str.."</size>";
        else
            local str=""..power;
            self.Name.text = cfg.Name.." <size=24>"..str.."</size>";
        end

        self.Num.text=item.Num;


        --判断是否能升级(配置表没有下一级)
        local hCfg2=ConfigDataAccessor.GetHeraldryEquip(heraldryEquipCfg.ResonanceSlot, heraldryEquipCfg.HeraldryLevel+1);
        if hCfg2== nil then
            self.IsMaxLv=true;
        else
            --判断是否能升级
            local maxLv=ConfigDataAccessor.GetHeraldryGrowMaxLv();
            if heraldryEquipCfg.HeraldryLevel == maxLv then
                self.IsMaxLv=true;
            else
                self.IsMaxLv=false;
            end
        end


        if item.IsSelected~=nil then
            self:MakeSelected(item.IsSelected)
        end

    end


end

--是否被选中
function EmblemItemFull:MakeSelected(b)
	
    if self.Item~=nil and self.Item.IsSelected~=nil then
        self.Item.IsSelected=b
    end
    self.go_Selected:SetActive(b)
    --self.go_Menu:SetActive(b);
end


return EmblemItemFull;

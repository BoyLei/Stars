local ExtractAmuletEquip = {}

ExtractAmuletEquip.__index = ExtractAmuletEquip

local EquipedAmuletItem=require("View/Uiwindow/Amulet/EquipedAmuletItem")
local SelectBtn= require("View/Item/SelectBtn")
function ExtractAmuletEquip:New(_go,extractAmulet) 
    local self = {};    
    setmetatable(self, ExtractAmuletEquip); 
    
    self.ExtractAmulet=extractAmulet;


    self.gameObject=_go;
    self.transform=_go.transform;


    
    --道具列表
    self.EquipedAmuletItemList={};
	--无限列表
	self.go_EquipedAmuletItem = self.transform:Find("Items/EquipedAmuletItem").gameObject;

	self.rect_grid=self.transform:Find("Items/Scroll View/Mask/List"):GetComponent(typeof(RectTransform));

	self.tr_ItemRoot=self.transform:Find("Items/Scroll View/Mask/List");
	self.go_EquipedAmuletItem:SetActive(false);
	self.go_EquipedAmuletItem.transform:SetParent(self.transform:Find("Items"));
	self.EquipedAmuletScorllView=self.transform:Find("Items/Scroll View"):GetComponent(typeof(SuperScrollViewController));
	self.EquipedAmuletScorllView:NewScroll(self.go_EquipedAmuletItem, 0, function (tr,idx)
		self:OnEquipedAmuletItemRefesh(tr,idx);
	end);

	--装备菜单
	self.EquipMenuBtns={};
	for i=1,2 do 
		local btn=SelectBtn:New(self.transform:Find("Items/EquipMenu/Scroll View/Mask/List/EquipTagItem"..i).gameObject);
		btn:InitItem(i,function (idx)
			self:OnEquipMenuBtnClick(idx);
		end);
		table.insert( self.EquipMenuBtns,btn);
	end
    self.EquipMenuIdx=1;



    self:Show(false);

    return self;    --返回自身
end

function ExtractAmuletEquip:Show(b)
    self.gameObject:SetActive(b);
    if b then
    	self:OnEquipMenuBtnClick(1);
        --self:Refesh();
    end
end

function ExtractAmuletEquip:Refesh()

	--拿到所有的
	self.allitems=BusinessManager.Instance:GetItemsByItemType(5,0);
	self:ClearItemData();
		

	if  self.EquipMenuIdx==1 then
		--装备槽位1
		self:GetDataByEquipType(1);
	elseif self.EquipMenuIdx==2 then
		--装备槽位2
		self:GetDataByEquipType(2);
	end

	self:RefeshItems();
	--关闭道具Tips
	if GameManager.Instance.IsOpenItemTips then
		LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnCloseTips",{  });
	end


end

------------------------------无限列表相关



--刷新装备列表
function ExtractAmuletEquip:RefeshItems()
	


	self:SortItem(self.ItemDataList);

	self.EquipedAmuletScorllView:ContinueScroll(#self.ItemDataList,false);



end



function ExtractAmuletEquip:GetDataByEquipType(equipType)


	local spaceID=13;

    --装备中
    if equipType==1 then
        spaceID=13;
    else
    --背包里
		spaceID=14;
    end


	for i=0,self.allitems.Count-1 do
		local data =self.allitems[i];
		if data~=nil and data.SpaceId==spaceID then

			local cfg=ConfigDataAccessor.Item[data.BaseID];--首选的

			--剔除掉没有洗练满的
			local maxNum=ConfigDataAccessor.GetMaxPolishNum(cfg.Quality);

			local combatSkills =data.CombatSkills;
			local PolishNum=combatSkills.CurPolish;--已经铭刻的次数

			local cfgAmulet=ConfigDataAccessor.AmuletEquip[data.BaseID];--首选的
			--一阶的不能萃
			if PolishNum == maxNum and cfgAmulet.EquipLevel>1 then

				local info={}
				info.Data=data;
				info.IsSelected=false;
				info.Quality=cfg.Quality;


				table.insert( self.ItemDataList,info);
			end

			
		end
	end


end

function ExtractAmuletEquip:OnEquipMenuBtnClick(idx,noResetPos)
	if noResetPos==nil then
		self.rect_grid.anchoredPosition=Vector2(self.rect_grid.anchoredPosition.x,0);
	end

	self:CleanAllSelected();

	for i=1,#self.EquipMenuBtns do
		self.EquipMenuBtns[i]:MakeSelected(false);
	end


    if idx==nil then
        idx =self.EquipMenuIdx;
    end
	self.EquipMenuIdx=idx;
	self.EquipMenuBtns[idx]:MakeSelected(true);


	--清除
	self.ExtractAmulet:SetSelectedItemData(nil);
	--刷新面板
	self.ExtractAmulet:Refesh();

	self:Refesh();

	if #self.EquipedAmuletItemList>0 then
		for i=1,#self.EquipedAmuletItemList do
			if self.EquipedAmuletItemList[i].gameObject.activeSelf then
				self:OnItemClick(self.EquipedAmuletItemList[i]);
				self.EquipedAmuletItemList[i]:MakeSelected(true);
				break;
			end
			
		end
	end
end


function ExtractAmuletEquip:CleanAllSelected()
	--无限列表
	for i=1,#self.EquipedAmuletItemList do
		self.EquipedAmuletItemList[i]:MakeSelected(false);
	end
end

function ExtractAmuletEquip:OnEquipedAmuletItemRefesh(tr,idx)


	for i=1,#self.EquipedAmuletItemList do
		if self.EquipedAmuletItemList[i].transform == tr then
            self.EquipedAmuletItemList[i]:SetItem(self.ItemDataList[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
		end

	end

	--tr.gameObject:SetActive(true);
    local bi = EquipedAmuletItem:New(tr.gameObject);
	bi:InitItem(function (ritem)
		self:OnItemClick(ritem);
	end)
	bi.gameObject:SetActive(true);
	bi:SetItem(self.ItemDataList[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1

    table.insert(self.EquipedAmuletItemList, bi);

end

function ExtractAmuletEquip:OnItemClick(bitem)

	self:CleanAllSelected();
	bitem:MakeSelected(true);
	

    --刷新
    self.ExtractAmulet:SetSelectedItemData(bitem.Data);

end

function ExtractAmuletEquip:ClearItemData()

	if self.ItemDataList==nil then
		self.ItemDataList ={}
	else
		for k,_ in pairs(self.ItemDataList) do self.ItemDataList[k] = nil end
		--self.ItemDataList ={}
	end
end


function ExtractAmuletEquip:SortItem(tbl)

	if #tbl > 1 then

		--[[
		table.sort(tbl, function(a, b)
			if a.Heroid < b.Heroid then
				-- 按是否装备降序
				return true
			elseif a.Heroid == b.Heroid then
				-- 如果是否装备相等，按Type升序
				if a.Type < b.Type then
					return true
				elseif a.Type == b.Type then
					-- 如果Type相等，按稀有度降序
					return a.Quality > b.Quality
				end
			end
			return false
		end)

		]]
		table.sort(tbl, function(a, b)
			if a.Data.Heroid > b.Data.Heroid then
				-- 按是否装备降序
				return true
			elseif a.Data.Heroid == b.Data.Heroid then
				-- 如果是否装备相等，按Quality升序
				return a.Quality > b.Quality
			end
			return false
		end)

	end

end

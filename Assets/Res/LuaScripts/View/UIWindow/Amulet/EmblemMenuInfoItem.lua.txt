local EmblemMenuInfoItem = {}

EmblemMenuInfoItem.__index = EmblemMenuInfoItem


function EmblemMenuInfoItem:New(_go) 
    local self = {};    
    setmetatable(self, EmblemMenuInfoItem); 
    

    self.gameObject=_go;
    self.transform=_go.transform;


    self.transform:GetComponent(typeof(JButton)).OnClick=(function()
        self:OnItemClick()
	end);

    self.go_have=self.transform:Find("have").gameObject;
    self.go_null=self.transform:Find("null").gameObject;

    self.IconBg=self.transform:Find("iconbg"):GetComponent(typeof(Image));
    self.IconBg2=self.transform:Find("iconbg2"):GetComponent(typeof(Image));

    self.Icon=self.transform:Find("have/icon"):GetComponent(typeof(Image));

    self.Attr2=self.transform:Find("have/attr2"):GetComponent(typeof(Text));--属性，【共鸣额外】
    self.Attr=self.transform:Find("have/attr"):GetComponent(typeof(Text));--属性，【共鸣额外】
    self.Met=self.transform:Find("have/met"):GetComponent(typeof(Text));--共鸣标记


    self.MetEff=self.transform:Find("have/metEff").gameObject;


    self.Name=self.transform:Find("have/name"):GetComponent(typeof(Text));--名字

    self.go_up=self.transform:Find("have/up").gameObject;--是否可以合成
    self.go_up:SetActive(false);

    self.go_selected=self.transform:Find("select").gameObject;--选中
    self.go_selected:SetActive(false);

    return self;    --返回自身
end


function EmblemMenuInfoItem:OnItemClick()
    if self.OnSelected~= nil then
        self.OnSelected(self.SlotID,self.Idx);
    end
end


function EmblemMenuInfoItem:InitItem(idx,onSelected)
    self.OnSelected=onSelected;
    self.Idx=idx;
end

function EmblemMenuInfoItem:MakeSelect(b)
    self.go_selected:SetActive(b);
end

function EmblemMenuInfoItem:SetInfo(gemSlot)



    local slotID=gemSlot.SlotID;
    local gemID=gemSlot.GemID;
    --self.IconBg.color=GetEmblemBgColorRGB(slotID);

    self.SlotID=slotID;

    --刷新宝石槽颜色
    if slotID > 9 then
        --self.IconBg.sprite = AtlasManager.Instance:GetSprite("ui/amulet/atlas/rune", "Rune_slot_" .. slotID)
        ResourceHelper:LoadIconByAtlasAsync(
            self.IconBg,
            ResourceHelper.AtlasPathRune,
            "Rune_slot_" .. slotID,
            false
        )
    else
        --self.IconBg.sprite = AtlasManager.Instance:GetSprite("ui/amulet/atlas/rune", "Rune_slot_0" .. slotID)
        ResourceHelper:LoadIconByAtlasAsync(
            self.IconBg,
            ResourceHelper.AtlasPathRune,
            "Rune_slot_0" .. slotID,
            false
        )
    end



    if gemID~= nil and gemID~=0 then

        self.go_have:SetActive(true);
        self.go_null:SetActive(false);


        local itemData=BusinessManager.Instance:GetItemByItemEntityId(gemID);

        self.EntityID=itemData.EntityID;
        self.BaseID=itemData.BaseID;
        
        local cfg=ConfigDataAccessor.Item[self.BaseID];
		--LoadIcon(self.Icon,cfg.Icon);
        ResourceHelper:LoadIconAsync(self.Icon, "", cfg.Icon, false)


        --宝石的共鸣
        local heraldryEquipCfg=ConfigDataAccessor.HeraldryEquip[self.BaseID];



        --判断是是否触发了共鸣
        if slotID== heraldryEquipCfg.ResonanceSlot then
            self.Met.text=LanguageHelp:GetLanguageByKey("Local_Str_Resonated");
            self.MetEff:SetActive(true);

            local BaseAttrId=heraldryEquipCfg.BaseAttrId[1];
            local BaseAttrVal=heraldryEquipCfg.BaseAttrVal[1];

            local ResonanceAttrID=heraldryEquipCfg.ResonanceAttrID[1];
            local ResonanceAttrVal=heraldryEquipCfg.ResonanceAttrVal[1];


            local config1=  ConfigDataAccessor.GetAttrByID(BaseAttrId)
            local attrName1 = "";
            if config1 ~= nil then
                attrName1 = config1.Name;
            end
        
            local config2=  ConfigDataAccessor.GetAttrByID(ResonanceAttrID)
            local attrName2 = "";
            if config2 ~= nil then
                attrName2 = config2.Name;
            end
        

            local name=attrName1.."<color=#B9AEF1>+"..(BaseAttrVal+ResonanceAttrVal).."</color>";
            self.Attr.text=name;

            self.Attr2.text="["..BaseAttrVal.."+"..ResonanceAttrVal.."]";
        else
            self.Met.text=LanguageHelp:GetLanguageByKey("Local_Str_unresonance");
            self.MetEff:SetActive(false);

            local BaseAttrId=heraldryEquipCfg.BaseAttrId[1];
            local BaseAttrVal=heraldryEquipCfg.BaseAttrVal[1];

            local config1=  ConfigDataAccessor.GetAttrByID(BaseAttrId)
            local attrName1 = "";
            if config1 ~= nil then
                attrName1 = config1.Name;
            end
        
            local config2=  ConfigDataAccessor.GetAttrByID(ResonanceAttrID)
            local attrName2 = "";
            if config2 ~= nil then
                attrName2 = config2.Name;
            end
        

            local name=attrName1.."<color=#B9AEF1>+"..BaseAttrVal.."</color>";
            self.Attr.text=name;

            self.Attr2.text="";
        end


        self.Name.text=cfg.Name;
    

    else
        self.go_have:SetActive(false);
        self.go_null:SetActive(true);
        

    end

end





return EmblemMenuInfoItem;

local ExtractAmuletSelectMenu = {}

ExtractAmuletSelectMenu.__index = ExtractAmuletSelectMenu

local ExtractAmuletItem = require("View/Uiwindow/Amulet/ExtractAmuletItem")

local ExtractSkillSort = require("View/Uiwindow/Amulet/ExtractSkillSort")

local SortList = require("View/Item/SortList")

function ExtractAmuletSelectMenu:New(_go, extractAmulet)
    local self = {}
    setmetatable(self, ExtractAmuletSelectMenu)

    self.ExtractAmulet = extractAmulet

    self.gameObject = _go
    self.transform = _go.transform

    --道具列表
    self.BagItemList = {}
    --无限列表
    self.go_BagItem = self.transform:Find("ExtractAmuletItem").gameObject

    self.rect_grid = self.transform:Find("Scroll View/Mask/List"):GetComponent(typeof(RectTransform))

    self.tr_ItemRoot = self.transform:Find("Scroll View/Mask/List")
    self.go_BagItem:SetActive(false)
    self.go_BagItem.transform:SetParent(self.transform)
    self.BagScorllView = self.transform:Find("Scroll View"):GetComponent(typeof(SuperScrollViewController))
    self.BagScorllView:NewScroll(
        self.go_BagItem,
        0,
        function(tr, idx)
            self:OnBagItemRefesh(tr, idx)
        end
    )

    self.SelectedExtractAmuletItem = nil

    --点击关闭
    self.transform:Find("BtnClose"):GetComponent(typeof(JButton)).OnClick = (function()
        self.ExtractAmulet:ShowExtractAmuletSelectMenu(false)
    end)

    --点击选择
    self.transform:Find("BtnConfirm"):GetComponent(typeof(JButton)).OnClick = (function()
        if self.SelectedExtractAmuletItem ~= nil then
            self.ExtractAmulet:SetSelectedToExtractItemData(self.SelectedExtractAmuletItem.Data)
        end
        self.ExtractAmulet:ShowExtractAmuletSelectMenu(false)
    end)

    --技能筛选
    self.transform:Find("ExtractSortMenu/MultSelectBtn"):GetComponent(typeof(JButton)).OnClick = (function()
        self.ExtractSkillSort:Show(true, self.Type)
    end)
    self.MultSelectWarSkillId = 0 --当前筛选选择的技能 0表示全部

    self.MultSelect = self.transform:Find("ExtractSortMenu/MultSelectBtn/tex"):GetComponent(typeof(Text))

    self.ExtractSkillSort = ExtractSkillSort:New(self.transform:Find("ExtractSortMenu/SkillSort").gameObject, self)

    --等级筛选
    self.sortLvs = {}
    for k, v in pairs(ConfigDataAccessor.AmuletEquip) do
        if not (self:CheckIsInLvs(v.EquipLevel)) and v.EquipLevel ~= 1 then
            table.insert(self.sortLvs, v.EquipLevel)
        end
    end
    table.sort(
        self.sortLvs,
        function(d1, d2)
            if d1 ~= d2 then
                return d1 > d2
            end
            return false
        end
    )

    local temp = {}
    table.insert(temp, LanguageHelp:GetLanguageByKey("Local_Str_Quality_All"))
    for i = 1, #self.sortLvs do
        --table.insert( temp, ""..self.sortLvs[i].."级" )
        table.insert(temp, StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LvLimit3"), {self.sortLvs[i]}))
    end

    self.SortListLv = SortList:New(self.transform:Find("ExtractSortMenu/AmuletSortList").gameObject)
    self.SortListLv:Init(
        temp,
        function(idx)
            self:OnSelectSortListLv(idx)
        end
    )
    self.SortListLvTex = self.transform:Find("ExtractSortMenu/LvBtn/tex"):GetComponent(typeof(Text))
    self.SortListLvIdx = 1
    self.transform:Find("ExtractSortMenu/LvBtn"):GetComponent(typeof(JButton)).OnClick = (function()
        self.SortListLv:Show(true, self.SortListLvIdx)
    end)

    self:Show(false)

    return self --返回自身
end

function ExtractAmuletSelectMenu:CheckIsInLvs(lv)
    for i = 1, #self.sortLvs do
        if self.sortLvs[i] == lv then
            return true
        end
    end
    return false
end

function ExtractAmuletSelectMenu:OnSelectSortListLv(idx)
    self.SortListLvIdx = idx
    self.SortListLvTex.text = self.SortListLv.SortNames[idx]
    self:RefeshItems()
end

function ExtractAmuletSelectMenu:Show(b)
    self.gameObject:SetActive(b)
    if b then
        self.MultSelectStr = LanguageHelp:GetLanguageByKey("Local_Str_AmuletSortBtnTex1") --重置
        self.MultSelectWarSkillId = 0 --重置
        self:Refesh()
    end
end

function ExtractAmuletSelectMenu:Refesh()
    self:RefeshItems()
end

------------------------------
------------------------------无限列表相关

--刷新装备列表
function ExtractAmuletSelectMenu:RefeshItems()
    if not (self.gameObject.activeSelf) then
        do
            return
        end
    end

    if self.ExtractAmulet.selectedData == nil then
        do
            return
        end
    end

    --关闭道具Tips
    if GameManager.Instance.IsOpenItemTips then
        LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnCloseTips", {})
    end

    --拿到所有的
    self.allitems = BusinessManager.Instance:GetItemsByItemType(5, 0)
    self:ClearItemData()

    local cfg = ConfigDataAccessor.Item[self.ExtractAmulet.selectedData.BaseID] --首选的
    self.Type = cfg.SubType

    --设置下面筛选文字
    self.MultSelect.text = self.MultSelectStr

    --设置筛选等级
    local selectLv = 0
    if self.SortListLvIdx > 1 then
        selectLv = self.sortLvs[self.SortListLvIdx - 1]
    end
	self.SortListLvTex.text = self.SortListLv.SortNames[self.SortListLvIdx]

    --要剔除掉已装备的，和自己一样的
    for i = 0, self.allitems.Count - 1 do
        if self.allitems[i].Heroid == 0 and self.allitems[i].EntityID ~= self.ExtractAmulet.selectedData.EntityID then
            local cfg1 = ConfigDataAccessor.Item[self.ExtractAmulet.selectedData.BaseID] --首选的
            local cfg2 = ConfigDataAccessor.Item[self.allitems[i].BaseID] --需要消失的

            local amuletCfg1 = ConfigDataAccessor.AmuletEquip[self.ExtractAmulet.selectedData.BaseID] --首选的
            local amuletCfg2 = ConfigDataAccessor.AmuletEquip[self.allitems[i].BaseID] --需要消失的

            --剔除掉稀有度不同的
            --剔除掉类型不同的
            --剔除掉等级更低的
            if
                cfg1.Quality == cfg2.Quality and cfg1.SubType == cfg2.SubType and amuletCfg1.EquipLevel <= amuletCfg2.EquipLevel and
                    (selectLv == 0 or (selectLv > 0 and selectLv == amuletCfg2.EquipLevel))
             then
                local maxNum = ConfigDataAccessor.GetMaxPolishNum(cfg2.Quality) --铭刻次数的上限

                local combatSkills = self.allitems[i].CombatSkills
                local PolishNum = combatSkills.CurPolish --已经铭刻的次数

                if PolishNum == maxNum then
                    --寻找选中的技能
                    if self.MultSelectWarSkillId == 0 then
                        --全选
                        table.insert(self.ItemDataList, self.allitems[i])
                    else
                        --找到只有相等技能的
                        --满足筛选条件
                        local list = combatSkills.List
                        for j = 0, list.Count - 1 do
                            local ID = list[j].ID
                            if ID == self.MultSelectWarSkillId then
                                table.insert(self.ItemDataList, self.allitems[i])
                                break
                            end
                        end
                    end
                end
            end
        end
    end

    --self:SortItem(self.ItemDataList);

    self.BagScorllView:ContinueScroll(#self.ItemDataList)
end

function ExtractAmuletSelectMenu:CleanAllSelected()
    --无限列表
    for i = 1, #self.BagItemList do
        self.BagItemList[i]:MakeSelected(false)
    end
end

function ExtractAmuletSelectMenu:OnBagItemRefesh(tr, idx)
    for i = 1, #self.BagItemList do
        if self.BagItemList[i].transform == tr then
            self.BagItemList[i]:SetItem(self.ItemDataList[idx + 1], self.MultSelectWarSkillId) --idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end

    --tr.gameObject:SetActive(true);
    local bi = ExtractAmuletItem:New(tr.gameObject)
    bi:InitItem(
        function(ritem)
            self:OnItemClick(ritem)
        end
    )
    bi.gameObject:SetActive(true)
    bi:SetItem(self.ItemDataList[idx + 1], self.MultSelectWarSkillId) --idx是从0开始，如果ItemDataList是lua的table需要idx+1

    table.insert(self.BagItemList, bi)
end

function ExtractAmuletSelectMenu:OnItemClick(bitem)
    self:CleanAllSelected()
    bitem:MakeSelected(true)

    --刷新右侧
    self.SelectedExtractAmuletItem = bitem
end

function ExtractAmuletSelectMenu:ClearItemData()
    if self.ItemDataList == nil then
        self.ItemDataList = {}
    else
        --self.ItemDataList ={}
        for k, _ in pairs(self.ItemDataList) do
            self.ItemDataList[k] = nil
        end
    end
end

function ExtractAmuletSelectMenu:SortItem(tbl)
    if #tbl > 1 then
        table.sort(
            tbl,
            function(a, b)
                if a.SortProp < b.SortProp then
                    -- 按SortProp升序
                    return true
                elseif a.SortProp == b.SortProp then
                    -- 如果SortProp相等，按Type升序
                    if a.ItemType < b.ItemType then
                        return true
                    elseif a.ItemType == b.ItemType then
                        -- 如果Type相等，按Id升序
                        return a.BaseID < b.BaseID
                    end
                end
                return false
            end
        )
    end
end

------------------------------无限列表相关

return ExtractAmuletSelectMenu

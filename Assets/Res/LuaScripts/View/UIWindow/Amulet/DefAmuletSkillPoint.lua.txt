local DefAmuletSkillPoint = {}

DefAmuletSkillPoint.__index = DefAmuletSkillPoint

function DefAmuletSkillPoint:New(_go)
    local self = {}
    setmetatable(self, DefAmuletSkillPoint)

    self.gameObject = _go
    self.transform = _go.transform

    self.points = {}
    self.pointIcons = {}
    for i = 1, 10 do
        local pt = self.transform:Find("points/point (" .. i .. ")").gameObject
        local ptIc = self.transform:Find("points/point (" .. i .. ")/Image").gameObject

        table.insert(self.points, pt)
        table.insert(self.pointIcons, ptIc)
    end

    self.skillIcon = self.transform:Find("iconbg/icon"):GetComponent(typeof(Image))

    self.title = self.transform:Find("title"):GetComponent(typeof(Text))
    self.add = self.transform:Find("add"):GetComponent(typeof(Text))
    --self.nextLv=self.transform:Find("title2"):GetComponent(typeof(Text));

    --详细信息
    local btnInfo = self.transform:Find("iconbg"):GetComponent(typeof(JButton))
    btnInfo.OnClick = (function()
        LuaModuleDispatcher:SendMessage(ModuleDef.AmuletModule, "OnShowEmblemSkillInfo", {self.SkillID})
    end)

    return self --返回自身
end

function DefAmuletSkillPoint:Refesh(maxnum, cs)
    for i = 1, #self.points do
        self.points[i].gameObject:SetActive(false)
        self.pointIcons[i].gameObject:SetActive(false)
    end

    if cs ~= nil then
        self.SkillID = cs.ID --技能id

        --显示图标
        local ID = cs.ID --技能id
        local Points = cs.Points --点数分布，二进制

        local pList = self:_byte2bin(Points)

        --点的数量
        local num = 0

        for i = 1, maxnum do
            self.points[i].gameObject:SetActive(true)

            if pList[33 - i] > 0 then
                self.pointIcons[i].gameObject:SetActive(true)
                num = num + 1
            end
        end

        local maxLv = ConfigDataAccessor.GetPassiveDescMaxLv(ID)

        --每一级的点数
        local perLevelPoint = 5

        local isMaxLv = false
        if num >= maxLv * perLevelPoint then
            isMaxLv = true
        end

        if isMaxLv then
            --self.nextLv.text="已达到最大等级";
            local pd = ConfigDataAccessor.GetPassiveDesc(ID, maxLv)
            ResourceHelper:LoadIconByAtlasAsync(self.skillIcon, ResourceHelper.AtlasPathEtchSkill, pd.IconPath, false)

            self.title.text = pd.SkillName
            --self.add.text="精通+"..num;
            self.add.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_master"), {num})
        else
            --self.nextLv.text="距离下一级提升还差"..leftNum.."点"
            --计算到下一个等级需要的点数
            local leftNum = 1

            --计算技能等级

            local lvNext = 1
            for i = 1, (maxLv - 1) do
                if lvNext * perLevelPoint > num then
                    leftNum = lvNext * perLevelPoint - num
                    break
                else
                    lvNext = lvNext + 1
                end
            end

            local lvNow = lvNext - 1 --当前等级

            local pd = ConfigDataAccessor.GetPassiveDesc(ID, lvNext)
            ResourceHelper:LoadIconByAtlasAsync(self.skillIcon, ResourceHelper.AtlasPathEtchSkill, pd.IconPath, false)

            --self.title.text="【"..pd.SkillName.."】精通+"..num;
            --self.add.text="精通+"..num;
            self.add.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_master"), {num})

            self.title.text = pd.SkillName
        end
    end
end

function DefAmuletSkillPoint:_byte2bin(n)
    --print("_byte2bin"..n);

    local t = {}
    for i = 32, 0, -1 do
        t[#t + 1] = math.floor(n / 2 ^ i)
        n = n % 2 ^ i
    end

    --print(table.concat(t));

    return t
    --return table.concat(t)
end

return DefAmuletSkillPoint

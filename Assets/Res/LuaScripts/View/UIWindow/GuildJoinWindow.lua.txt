GuildJoinWindow=class("GuildJoinWindow",UIPanel)
GuildJoinWindow.IsOpen=false;


local SelectBtn= require("View/Item/SelectBtn")

local GuildJoin=require("View/Uiwindow/Guild/GuildJoin")
local GuildCreate=require("View/Uiwindow/Guild/GuildCreate")
local CurrencyBar =require("View/Item/CurrencyBar")

function GuildJoinWindow:Awake()

end

function GuildJoinWindow:BindModule(m)
	self.module=m;
end

function GuildJoinWindow:InitItems()
	self.transform =self.prefab.transform;
	self.gameObject=self.transform.gameObject;

	self.idxNow=-1;

	self.currencyBar=CurrencyBar:New(nil,self.transform:Find("Cut4Cam90/AssemblyRatio2/CurrencyAhchor"));


	--关闭信息
	local btnCloseInfo=self.transform:Find("Block"):GetComponent(typeof(Button));
	btnCloseInfo.onClick:AddListener(function()
		self:OnCloseInfoClick();
	end);


	--背包菜单按钮
	self.MenuBtns={};

	local btn1=SelectBtn:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/RightMenu/BtnJoin").gameObject);
	btn1:InitItem(1,function (idx)
		self:OnMenuBtnClick(idx);
	end);
	table.insert(self.MenuBtns,btn1);

	local btn2=SelectBtn:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/RightMenu/BtnCreate").gameObject);
	btn2:InitItem(2,function (idx)
		self:OnMenuBtnClick(idx);
	end);
	table.insert( self.MenuBtns,btn2);



	self.GuildJoin= GuildJoin:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/Join").gameObject,self);
	self.GuildCreate=GuildCreate:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/Create").gameObject,self);


	self.m_CloseBtn=self:Find("Cut4Cam90/AssemblyRatio2/RightMenu/BtnClose"):GetComponent(typeof(Button));
	self.m_CloseBtn.onClick:AddListener(function() self:OnClickCloseBtn() end);

end

function GuildJoinWindow:Refesh()
	--刷新货币
	local coins={4,3,2,1}
	self.currencyBar:RefeshBar(coins);
	self.currencyBar:PlayAnim(true);

	self.idxNow=-1;
	self:OnMenuBtnClick(1);
end



function GuildJoinWindow:OnMenuBtnClick(idx)

	if self.idxNow==idx then
		return;
	end

	if idx==1 then
		self.PlayAnimation("GuildJoinWindow_Join", function ()
		
		end)
		self.GuildJoin:Show(true);
		self.GuildCreate:Show(false);

	elseif idx==2 then
		self.PlayAnimation("GuildJoinWindow_Create", function ()
		
		end)
		self.GuildJoin:Show(false);
		self.GuildCreate:Show(true);

	end


	self.idxNow=idx;

	for i=1,#self.MenuBtns do
		self.MenuBtns[i]:MakeSelected(false);
	end
	self.MenuBtns[idx]:MakeSelected(true);


	--关闭道具Tips
	if GameManager.Instance.IsOpenItemTips then
		self.module:CloseItemTips();
	end
end


function GuildJoinWindow:Start()
	print("GuildJoinWindow:Start")

end

function GuildJoinWindow:OnEnable()
	
end

function GuildJoinWindow:OnDisable()
	
end

function GuildJoinWindow:OnDestroy()

end

function GuildJoinWindow:OnOpen(arg)
	self.IsOpen=true;

	self.PlayAnimation("GuildJoinWindow_Open", function ()
		
	end)
end

function GuildJoinWindow:OnClose(arg)
	--self:CloseWindow();

	self.GuildJoin:Show(false);
	self.GuildCreate:Show(false);

	self.module.guildJoinWindowView = nil
end

function GuildJoinWindow:OnCloseInfoClick()
	--关闭道具Tips
	if self.module.IsOpenTips then
		self.module.IsOpenTips=false;
		self.module:CloseItemTips();
	else
		--UIManager.Instance:CloseWindow("Item/GuildJoinWindow");
	end
	self.module.guildJoinWindowView = nil
end


function GuildJoinWindow:OnClickCloseBtn()

	if self.module.IsOpenTips then
		self.module.IsOpenTips=false;
		self.module:CloseItemTips();

	end
	self.idxNow=-1;
	self.IsOpen=false;
	self.module:OnCloseBox();

	self.Close();

	self.module.guildJoinWindowView = nil


end



return  GuildJoinWindow

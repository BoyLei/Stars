local EquipUpgradeAttrSimpleSuccess = {}

EquipUpgradeAttrSimpleSuccess.__index = EquipUpgradeAttrSimpleSuccess


function EquipUpgradeAttrSimpleSuccess:New(_go,onClose) 
    local self = {};    
    setmetatable(self, EquipUpgradeAttrSimpleSuccess); 
    
	self.canClose=false;

    self.gameObject=_go;
    self.transform=_go.transform;
	self.OnClose=onClose;



    self:Show(false);

    return self;    --返回自身
end



function EquipUpgradeAttrSimpleSuccess:Show(b,entity_id,base_id,oldAttr,newAttr)
    self.gameObject:SetActive(b);
    if b then

		if self.tw~= nil then
            self.tw:Kill();
        end
        --延迟一段时间后调用
        self.tw=DOVirtual.DelayedCall(0.6,function ()
    
            if self.OnClose~=nil then
                self.OnClose();
            end
            self:Show(false);
        end)

	else

		

    end
end




return EquipUpgradeAttrSimpleSuccess;

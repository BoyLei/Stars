local EquipRecastEventLog = {}

EquipRecastEventLog.__index = EquipRecastEventLog

local EquipRecastEventLogItem=require("View/Uiwindow/EquipUpgrade/EquipRecastEventLogItem")

function EquipRecastEventLog:New(_go) 
    local self = {};    
    setmetatable(self, EquipRecastEventLog); 
    

    self.gameObject=_go;
    self.transform=_go.transform;

    --道具列表
    self.EventLogItemList={};
	--无限列表
	self.go_EventLogItem = self.transform:Find("Scroll View/Mask/List/GuildEventLogItem").gameObject;

	self.tr_ItemRoot=self.transform:Find("Scroll View/Mask/List");
	self.go_EventLogItem:SetActive(false);
	self.go_EventLogItem.transform:SetParent(self.transform);
	self.EventLogScorllView=self.transform:Find("Scroll View"):GetComponent(typeof(SuperScrollViewController));
	self.EventLogScorllView:NewScroll(self.go_EventLogItem, 0, function (tr,idx)
		self:OnEventLogItemRefesh(tr,idx);
	end);

	self.go_null=self.transform:Find("Null").gameObject;

    --关闭按钮
    self.transform:Find("BtnClose"):GetComponent(typeof(JButton)).OnClick=(function() 
        self:Show(false);
    end);


    self:Show(false);

    return self;    --返回自身
end



function EquipRecastEventLog:OnEventLogItemRefesh(tr,idx)

	for i=1,#self.EventLogItemList do
		if self.EventLogItemList[i].transform == tr then
            self.EventLogItemList[i]:SetItem(self.ItemDataList[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
		end

	end

    local bi = EquipRecastEventLogItem:New(tr.gameObject);

	bi.gameObject:SetActive(true);
	bi:SetItem(self.ItemDataList[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1

    table.insert(self.EventLogItemList, bi);

end


function EquipRecastEventLog:Show(b)
    self.gameObject:SetActive(b);
    if b then


        self:Refesh();

    end
end



function EquipRecastEventLog:Refesh()


    if self.ItemDataList==nil then
        self.ItemDataList ={}
    else
        for k,_ in pairs(self.ItemDataList) do self.ItemDataList[k] = nil end
    end

	--判断是否还有没选择的
	local eqRecastMD=FixMessageManager.Instance:GetMDMgr(FixUpdateDef.EqRecasts):GetEqRecastMD();

	if eqRecastMD~= nil then
		local eventInfos=eqRecastMD.RehisEvs.EventInfos;

		
		--刷新数据 EventInfo
		for i=0,eventInfos.Count-1 do
			local data={};

			data.ItemInfos=eventInfos[i].ItemInfos;--List
			data.EventTime=eventInfos[i].EventTime;--ulong
			data.EventID=eventInfos[i].EventID;--int
			
				
			table.insert( self.ItemDataList, data );
		end



	end


	self.go_null:SetActive((#self.ItemDataList)<=0);


	self:SortItem(self.ItemDataList);
	self.EventLogScorllView:ContinueScroll(#self.ItemDataList);

end


function EquipRecastEventLog:SortItem(tbl)


	if #tbl > 1 then

		table.sort(tbl, function(a, b)

			-- 如果Type相等，按Id升序
			return a.EventTime > b.EventTime

		end)

	end

end


return EquipRecastEventLog;

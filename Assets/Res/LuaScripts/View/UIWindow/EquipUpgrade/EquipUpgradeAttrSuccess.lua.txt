local EquipUpgradeAttrSuccess = {}

EquipUpgradeAttrSuccess.__index = EquipUpgradeAttrSuccess
local SelectBtn= require("View/Item/SelectBtn")


local EquipUpgradeSlotAttr=require("View/Uiwindow/EquipUpgrade/EquipUpgradeSlotAttr")

function EquipUpgradeAttrSuccess:New(_go,onClose,equipUpgradeAttrUpgrade) 
    local self = {};    
    setmetatable(self, EquipUpgradeAttrSuccess); 
    
	self.canClose=false;

    self.gameObject=_go;
    self.transform=_go.transform;

	self.EquipUpgradeAttrUpgrade=equipUpgradeAttrUpgrade;
	--关闭信息
	local btnClose=self.transform:Find("BtnClose"):GetComponent(typeof(JButton));
	btnClose.OnClick=(function()
		if self.canClose then
			self.EquipUpgradeAttrUpgrade:PlayAttrSuccessCloseAnim(function ()
				self:Show(false);
				if onClose~=nil then
					onClose();
				end
			end);
		end
	end);

    --属性
    self.attrs={}
    for i=1,5 do
        local eai = EquipUpgradeSlotAttr:New(self.transform:Find("AttrPanel/Attr"..i).gameObject);
        table.insert( self.attrs,eai);
    end

    self.equipImage=self.transform:Find("EquipItem/Have/Icon"):GetComponent(typeof(Image));
	self.equipBg=self.transform:Find("EquipItem/Have/BgRare"):GetComponent(typeof(Image));
	self.equipAdd=self.transform:Find("EquipItem/Have/Num"):GetComponent(typeof(Text));

    --成功页签
    self.successTag={}
    for i=1,5 do
        local btn=SelectBtn:New(self.transform:Find("tab/Btn"..i).gameObject);
        btn:InitItem(i,function (idx)
            self:OnSelectBtnClick(idx);
        end);
        table.insert(self.successTag,btn);
    end


    self:Show(false);

    return self;    --返回自身
end

function EquipUpgradeAttrSuccess:OnSelectBtnClick(idx)
	
	for i=1,#self.successTag do
		self.successTag[i]:MakeSelected(false);
	end

	self.successTag[idx]:MakeSelected(true);
	
	
	--主属性


	local isOld=true;

	local PrevProps=nil;
	if idx>1 then
		isOld=false;
		PrevProps=self.NewAttrs[idx-2].MainPropList;
	end

	local MainProps=self.NewAttrs[idx-1].MainPropList;

	for i=1,#self.attrs do
		self.attrs[i].gameObject:SetActive(false);
	end

	for i=0,MainProps.Count-1 do
		self.attrs[i+1].gameObject:SetActive(true);

		local maxVal=self.eCfg.NumMax[i+1];

		local oldVal=self.OldAttr[i+1];
		if not(isOld) then
			oldVal=PrevProps[i].Value;
		end

		local newVal=MainProps[i].Value;

		self.attrs[i+1]:RefeshAttrProb(MainProps[i].ID,oldVal,newVal);

	end
	
end


function EquipUpgradeAttrSuccess:Show(b,entity_id,base_id,oldAttr,newAttr)
    self.gameObject:SetActive(b);
    if b then
        self.canClose=false;

        local cfg=ConfigDataAccessor.Item[base_id];

        self.equipAdd.text="";
        local quality=cfg.Quality; 
        --self.equipBg.sprite= AtlasManager.Instance:GetSprite("ui/common/atlas/commonitem","Bag_Grid_sq"..quality);
		ResourceHelper:LoadIconByAtlasAsync(
			self.equipBg,
			ResourceHelper.AtlasPathCommonItem,
			"Bag_Grid_sq" .. quality,
			false
		)
        --LoadIcon(self.equipImage,cfg.Icon);
		ResourceHelper:LoadIconAsync(self.equipImage, "", cfg.Icon, false)


		self.eCfg=ConfigDataAccessor.Equip[base_id];

		
		
		for i=1,#self.successTag do
			self.successTag[i].gameObject:SetActive(false);
		end
		
		
		--显示页签
		for i=0,newAttr.Count-1 do
			self.successTag[i+1].gameObject:SetActive(true);
		end


		self.NewAttrs=newAttr;
		self.OldAttr={};
		for i=1,#oldAttr do
			table.insert( self.OldAttr, oldAttr[i] );
		end

		self:OnSelectBtnClick(self.NewAttrs.Count);
	else

		self.canClose=true;

    end
end




return EquipUpgradeAttrSuccess;

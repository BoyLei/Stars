function TransJobWindow:OnOpen(arg)
    self.IsOpen = true

    -- UIManager.Instance:SetRawCamrea(true);
    local entityBase = GameManager.Instance.M_MainPlayerCtrlBase
    if entityBase ~= nil then
        self.m_JobID = entityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Uint,
            CS.StarProjectDef.AOIAttrDefine.Job)
	else
		self.m_JobID = arg;
    end
    ---- 初始化职业icon
    self:InitTransJobIcon();
    ---- 初始化转职任务
    self:InitTransJobTaskData();
    ---- 初始化主角职业转职后技能信息
    self:InitTransJobSkillData();

    self:InitData(arg)
end

-- 初始化界面信息
function TransJobWindow:InitData(arg)

    self:CheckTaskProgress(0);

-- 刷新任务进度
function TransJobWindow:CheckTaskProgress(finishTaskID)
    for i = 1, #self.m_ItemTransJobTaskTb, 1 do
        local item = self.m_ItemTransJobTaskTb[i];
        local isGet = item.TaskCfg.ID == finishTaskID and finishTaskID ~= 0;
        item:CheckTaskProgress(isGet);
    end
    self:CheckTransJobBtn();
end

function TransJobWindow:CheckTransJobBtn()
    local HeroMD = GameManager.Instance.HeroMD;
    local TransJobRewardList = HeroMD.TransJobRewardList;
    local getTaskCount = TransJobRewardList.List.Count;
    local isFinish = getTaskCount >= self.m_InitTaskCount;
    self.JBtnTransJob.UnClick = not isFinish;
    self.JBtnTransJobRedPoint:SetActive(isFinish);
end

---设置转职信息
function TransJobWindow:SetTransJobData()
    if self.m_TransferJobCfg == nil then
        return;
    end
    self.JobName.text = self.m_TransferJobCfg.Name;
    local index = 1;
    -- 先把大招加在最前面
    -- 技能点
    local gobPointUltimateJobSkills = newObject(self.TogglePointItem);
    gobPointUltimateJobSkills.transform:SetParent(self.TogglePointContent);
    gobPointUltimateJobSkills:SetActive(true);
    gobPointUltimateJobSkills.transform.localPosition = Vector3.zero;
    gobPointUltimateJobSkills.transform.localScale = Vector3.one;
    table.insert(self.m_ItemTogglePointTb, gobPointUltimateJobSkills.transform:Find("Check").gameObject);
    -- 技能图标
    local gobSkillIconItem = newObject(self.SkillIconItem);
    gobSkillIconItem.transform:SetParent(self.SkillIconContent);
    gobSkillIconItem.transform.localPosition = Vector3.zero;
    gobSkillIconItem.transform.localScale = Vector3.one;
    local gobSkillIconSpItem = gobSkillIconItem.transform:GetComponent(typeof(Image))
    table.insert(self.m_ItemSkillIconTb, gobSkillIconItem.gameObject);
    self:SetSkillIcon(gobSkillIconSpItem, self.m_TransferJobCfg.UltimateJobSkills, index);
    -- 技能
    local gobBornJobSkills = newObject(self.ToggleSkillItem);
    gobBornJobSkills.transform:SetParent(self.SkillScrollView);
    gobBornJobSkills:SetActive(true);
    gobBornJobSkills.transform.localPosition = Vector3.zero;
    gobBornJobSkills.transform.localScale = Vector3.one;
    local itemBornJobSkills = ItemTransJobSkill:New(gobBornJobSkills, function(skillCfg, index)
        self:ClickSkillItemCB(skillCfg, index)
    end);
    table.insert(self.m_ItemSkillTb, itemBornJobSkills);
    itemBornJobSkills:OnUpdateContent(self.m_TransferJobCfg.UltimateJobSkills, index);
    index = index + 1;
    for key, value in pairs(self.m_TransferJobCfg.TransferJobSkills) do
        -- 技能点
        local gob = newObject(self.TogglePointItem);
        gob.transform:SetParent(self.TogglePointContent);
        gob:SetActive(true);
        gob.transform.localPosition = Vector3.zero;
        gob.transform.localScale = Vector3.one;
        table.insert(self.m_ItemTogglePointTb, gob.transform:Find("Check").gameObject);
        -- 技能图标
        local gob1 = newObject(self.SkillIconItem);
        gob1.transform:SetParent(self.SkillIconContent);
        gob1.transform.localPosition = Vector3.zero;
        gob1.transform.localScale = Vector3.one;
        local gobSkillIconSpItem = gob1.transform:GetComponent(typeof(Image))
        table.insert(self.m_ItemSkillIconTb, gob1.gameObject);
        self:SetSkillIcon(gobSkillIconSpItem, value, index);
        -- 技能
        local gob2 = newObject(self.ToggleSkillItem);
        gob2.transform:SetParent(self.SkillScrollView);
        gob2:SetActive(true);
        gob2.transform.localPosition = Vector3.zero;
        gob2.transform.localScale = Vector3.one;
        local item = ItemTransJobSkill:New(gob2, function(skillCfg, index)
            self:ClickSkillItemCB(skillCfg, index)
        end);
        table.insert(self.m_ItemSkillTb, item);
        item:OnUpdateContent(value, index);
        index = index + 1;
    end
    -- 默认选中第一个
    local defaultSkillItem = self.m_ItemSkillTb[1];
    if defaultSkillItem ~= nil then
        defaultSkillItem:SetCheckState(true);
    end
end

function TransJobWindow:SetSkillIcon(skillIconGob, skillId, index)
    local skillIconPath = "";
    local talentID = 0;
    local talentCfgs = GetJobSkillTalents(skillId);
    -- 天赋 ID 采用 默认的天赋ID
    if talentCfgs ~= nil and #talentCfgs > 0 then
        talentID = talentCfgs[1].ID;
    end
    if talentID == 0 then
        return;
    end
    local cfg = GetJobSkillLevelDescCfg(talentID, 1);
    if cfg == nil then
        return;
    end
	skillIconPath = IconPath .. "TransJob_SkillIcon_" .. cfg.SkillID;
    ResourceHelper:LoadIconAsync(skillIconGob, "", skillIconPath, false)
end

---技能点击回调
---@param skillCfg 技能配置
---@param index 点击的下标
function TransJobWindow:ClickSkillItemCB(skillCfg, index)
    self:SetSkillPointShow(index);
    self.SkillName.text = skillCfg.SkillName;
    for i = 1, #self.m_ItemSkillTb, 1 do
        local child = self.m_ItemSkillTb[i];
        local isCheck = i == index;
        child:SetCheckState(isCheck);
    end
    -- 播放技能适配，等待资源
end

---设置技能点显示
---@param index 下标
function TransJobWindow:SetSkillPointShow(index)
    for i = 1, #self.m_ItemTogglePointTb, 1 do
        local item = self.m_ItemTogglePointTb[i];
        local isShow = i == index;
        item:SetActive(isShow);
        local skillItem = self.m_ItemSkillIconTb[i];
        skillItem:SetActive(isShow);
    end
end

    msg.ConfigID = taskID

function TransJobWindow:RegisterMessage()
    -- 领取转职条件奖励 返回
    self.TransJobCondRewardRetCmd = function(data)
        self:OnTransJobCondRewardRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.TransJobCondRewardRetID, self.TransJobCondRewardRetCmd, nil)
end

function TransJobWindow:UnRegisterMessage()
    -- 领取转职条件奖励 返回
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.TransJobCondRewardRetID, self.TransJobCondRewardRetCmd, nil)
end

function TransJobWindow:OnTransJobCondRewardRet(data)
    local res = data.data
    -- 转职条件表ID
    self:CheckTaskProgress(res.ConfigID);
end

---- 初始化职业icon
function TransJobWindow:InitTransJobIcon()
    local dipath = IconPath .. "TransJob_di_" .. self.m_JobID;
    ResourceHelper:LoadIconAsync(self.DiIcon, "", dipath, false)
    local framepath = IconPath .. "TransJob_frame_" .. self.m_JobID;
    ResourceHelper:LoadIconAsync(self.FrameIcon, "", framepath, false)
    -- local bgpath = IconPath .. "TransJob_bg_" .. self.m_JobID;
    -- ResourceHelper:LoadIconAsync(self.bgcolor, "", bgpath, false)
    local jobshi1 = math.floor(self.m_JobID / 10)
    local JobIconPath = ResourceHelper.HeadAtlasPath .. "Login_LogoChara" .. jobshi1;
    ResourceHelper:LoadIconAsync(self.JobIcon, "", JobIconPath, false)

    local JobPortraitPath = IconPath .. "TransJob_role_" .. self.m_JobID;
    ResourceHelper:LoadIconAsync(self.JobPortrait, "", JobPortraitPath, false)
end

---- 初始化转职任务
function TransJobWindow:InitTransJobTaskData()
    for key, value in pairs(ConfigDataAccessor.TransferCondition) do
        local gob = newObject(self.TaskItem);
        gob.transform:SetParent(self.TaskScrollView);
        gob:SetActive(true);
        gob.transform.localPosition = Vector3.zero;
        gob.transform.localScale = Vector3.one;
        local item = ItemTransJobTask:New(gob, function(taskID)
            self:ClickTaskItemCB(taskID);
        end);
        table.insert(self.m_ItemTransJobTaskTb, item);
        item:OnUpdateContent(value);
        self.m_InitTaskCount = self.m_InitTaskCount + 1
    end
end

---- 初始化主角职业转职后技能信息
function TransJobWindow:InitTransJobSkillData()
    if self.m_JobID == -1 then
        return;
    end
    local jobCfg = ConfigDataAccessor.Job[self.m_JobID];
    self.m_TransferJobCfg = nil;
    if jobCfg ~= nil then
        for key, value in pairs(jobCfg.TransferJob) do
            self.m_TransferJobCfg = ConfigDataAccessor.Job[value];
        end
        self:SetTransJobData();
    end
end

-- 点击【转职】按钮
function TransJobWindow:OnClickJBtnTransJob()
    local HeroMD = GameManager.Instance.HeroMD;
    local TransJobRewardList = HeroMD.TransJobRewardList;
    local getTaskCount = TransJobRewardList.List.Count;
    local isFinish = getTaskCount >= self.m_InitTaskCount;
    if isFinish == false then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_227"));
        return;
    end
    -- 寻路去目标点
    local jobCfg = ConfigDataAccessor.Job[self.m_JobID];
    local config = TaskHelper.GetTaskConfig(jobCfg.TransferTask)
    if config ~= nil then
        local TaskInfo = TaskHelper.GetRunningTask(jobCfg.TransferTask)
        if TaskInfo == nil then
            return;
        end

        if TaskInfo.TaskState == CS.ProtoMsg.TaskStateEnum.TaskStateNoGetReward then
            -- 任务未提交or未接取
        elseif TaskInfo.TaskState == CS.ProtoMsg.TaskStateEnum.TaskStageRunning then
            -- 任务进行中
            local taskTarget = TaskHelper.GetTaskTarget(jobCfg.TransferTask)
            if taskTarget ~= nil then
                CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(jobCfg.TransferTask)
                UIManager.Instance:CloseAllLoadedWindowNWidget();
                TaskHelper.FindPathByTask(taskTarget.MapID, taskTarget.FindPath,taskTarget.TargetID, function(result)
                    CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(0)
                    if result and
                        (taskTarget.FindPath.FindPathType == CS.Task.E_FindPath.NPC or taskTarget.FindPath.FindPathType ==
                            CS.Task.E_FindPath.InterAction) then
                        local EventID = TaskHelper.GetTaskRunningEventID(jobCfg.TransferTask)
                        self:OnFinishPathFinding(taskTarget.TargetID, jobCfg.TransferTask, EventID, taskTarget.taskType)
                    end
                end)
            end
        elseif TaskInfo.TaskState == CS.ProtoMsg.TaskStateEnum.TaskStageFinished then
            -- 任务已完成（未领奖）
            CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(jobCfg.TransferTask)
            UIManager.Instance:CloseAllLoadedWindowNWidget();
            -- 任务已完成未提交
            TaskHelper.FindPathByNpc(config.TaskFinishInfo.FinishTaskMapID, config.TaskFinishInfo.FinishTaskNPC, 1,
                function(result)
                    CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(0)
                    if result then
                        self:OnFinishPathFinding(config.TaskFinishInfo.FinishTaskNPC, jobCfg.TransferTask, 0,
                            CS.Task.TaskType.Dialogue)
                    end
                end)
        end
    end
end

function TransJobWindow:OnFinishPathFinding(targetid, SourceID, SourceSubID, taskType)
    if taskType == CS.Task.TaskType.Dialogue then
        self:OnOpenDialogue(targetid, SourceID, SourceSubID)
    elseif taskType == CS.Task.TaskType.InterAction then
        self:OnInterAction(targetid, SourceID, SourceSubID)
    end
end

function TransJobWindow:OnInterAction(objectID, SourceID, SourceSubID)
    local uid = GameManager.Instance:GetObjectEntityIDByConfig(objectID)
    BusinessManager.Instance:TurnToEntity(uid)
    LuaModuleDispatcher:SendMessage(ModuleDef.InterActionModule, "QueryInter", uid, true, SourceID, SourceSubID)
end

function TransJobWindow:OnOpenDialogue(npcID, SourceID, SourceSubID)
    local lobbyService = CS.ProtoMsg.LobbyServiceReq()
    lobbyService.MapID = GameManager.Instance:GetCurMapId()
    lobbyService.UID = GameManager.Instance:GetNPCEntityIDByConfig(npcID)
    lobbyService.ObjID = npcID
    lobbyService.SourceType = CS.ProtoMsg.MapServiceSourceTypeEnum.Task
    lobbyService.SourceID = SourceID
    lobbyService.SourceSubID = SourceSubID
    lobbyService.ObjType = CS.ProtoMsg.ServiceObjEnum.ObjNpc
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby,
        lobbyService, false)
end

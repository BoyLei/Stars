    self.IsOpen = true;

    self:CloseWindow()

-- 初始化界面信息
function TransJobFinishWindow:InitData(arg)
    local job = arg;
    self.m_TransferJobCfg = ConfigDataAccessor.Job[job];
    if self.m_TransferJobCfg == nil then
        return;
    end
    self.JobName.text = self.m_TransferJobCfg.Name;
    self.JobDes.text = self.m_TransferJobCfg.TransferDesc;
    self:SetAttrData();
    self:SetTransJobData();

    self.PlayAnimation("TransJobFinishWindow_Go_Anim", nil);
end

-- 设置属性信息
function TransJobFinishWindow:SetAttrData()
    for i = 1, #self.m_TransferJobCfg.AttrID, 1 do
        local attrID = self.m_TransferJobCfg.AttrID[i];
        local cfg = ConfigDataAccessor.GetAttrByID(attrID);
        local attrName = "";
        if cfg ~= nil then
            attrName = cfg.Name;
        end
        local attrVal = self.m_TransferJobCfg.AttrValue[i];
        local gob = newObject(self.AttrItem);
        gob.transform:SetParent(self.AttrLayout);
        gob:SetActive(true);
        gob.transform.localPosition = Vector3.zero;
        gob.transform.localScale = Vector3.one;
        local ItemAttr = ItemTransJobAttr:New(gob);
        ItemAttr:SetAttr(attrName, attrVal);
    end
end

---设置转职信息
function TransJobFinishWindow:SetTransJobData()
    local index = 1;
    -- 先把大招加在最前面
    -- 技能
    local gobBornJobSkills = newObject(self.ToggleSkillItem);
    gobBornJobSkills.transform:SetParent(self.SkillScrollView);
    gobBornJobSkills:SetActive(true);
    gobBornJobSkills.transform.localPosition = Vector3.zero;
    gobBornJobSkills.transform.localScale = Vector3.one;
    local itemBornJobSkills = ItemTransJobSkill:New(gobBornJobSkills, nil);
    itemBornJobSkills:OnUpdateContent(self.m_TransferJobCfg.UltimateJobSkills, index);
    index = index + 1;
    for key, value in pairs(self.m_TransferJobCfg.TransferJobSkills) do
        -- 技能
        local gob2 = newObject(self.ToggleSkillItem);
        gob2.transform:SetParent(self.SkillScrollView);
        gob2:SetActive(true);
        gob2.transform.localPosition = Vector3.zero;
        gob2.transform.localScale = Vector3.one;
        local item = ItemTransJobSkill:New(gob2, nil);
        item:OnUpdateContent(value, index);
        index = index + 1;
    end
    
    self.UnlockGob.transform:SetParent(self.SkillScrollView);
    self.UnlockGob:SetActive(true);
end

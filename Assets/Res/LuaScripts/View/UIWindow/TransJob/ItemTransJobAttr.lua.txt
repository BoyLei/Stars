local ItemTransJobAttr = {}
ItemTransJobAttr.__index = ItemTransJobAttr

function ItemTransJobAttr:New(_go)
    local self = {}
    setmetatable(self, ItemTransJobAttr)

    self.Transform = _go.transform
    -- 任务描述
    self.AttrName = self.Transform:Find("AttrName"):GetComponent(typeof(Text));
    -- 任务进度
    self.Val = self.Transform:Find("Val"):GetComponent(typeof(Text));


    return self
end

---comment
---@param attrName 属性名
---@param attrVal 属性值
function ItemTransJobAttr:SetAttr(attrName, attrVal)
    self.AttrName.text = attrName;
    self.Val.text = "+" .. attrVal;
end

return ItemTransJobAttr

--- 代拍仓库 ui 界面的初始化接口, 代拍仓库先不采用 new prefab 的方式.通过 UnionWarehouse:New 接口将节点跟脚本绑定
---@param _go "传过来的 代拍仓库 根节点"
function UnionWarehouse:New(_go)
    local self = {}
    setmetatable(self, UnionWarehouse)

    self.gameObject = _go
    self.transform = _go.transform

    self.ItemDescContent = self.transform:Find("ItemDescContent")
    self.ItemName = self.ItemDescContent:Find("ItemName"):GetComponent(typeof(Text))
    self.ItemType = self.ItemDescContent:Find("ItemType"):GetComponent(typeof(Text))
    self.ItemCount = self.ItemDescContent:Find("ItemCount"):GetComponent(typeof(Text))
    self.ItemPreDesc = self.ItemDescContent:Find("ItemPreDesc"):GetComponent(typeof(Text))
    self.ItemDesc = self.ItemDescContent:Find("ItemDesc"):GetComponent(typeof(Text))
    self.ItemIconGrid = self.ItemDescContent:Find("ItemIconGrid").gameObject
    self.DescItemGrid = ItemGrid:New(self.ItemIconGrid)
    self.DescItemGrid:InitItem(true, nil)

    self.ItemContent = self.transform:Find("ItemContent")
    self.ItemGrid = self.ItemContent:Find("ItemGrid").gameObject
    self.ItemGrid:SetActive(false)

    self.tr_ItemRoot = self.ItemContent:Find("Scroll View/Mask/List")

    self.BagScorllView = self.ItemContent:Find("Scroll View"):GetComponent(typeof(SuperScrollViewController))
    self.BagScorllView:NewScroll(
        self.ItemGrid,
        0,
        function(tr, idx)
            self:OnItemRefesh(tr, idx)
        end
    )

    self.NotOpen = self.transform:Find("NotOpen").gameObject

    self.ItemList = {}

    return self -- 返回自身
end

function UnionWarehouse:MarkOpen(isOpen)
    self.NotOpen:SetActive(not isOpen)
    self.ItemDescContent.gameObject:SetActive(isOpen)
    self.ItemContent.gameObject:SetActive(isOpen)
end

-- 初始化格子 ， 先初始化 7*7 49个格子
function UnionWarehouse:OnShow(itemDatas)
    self:MarkOpen(true)
    -- 格子数据 list
    self.SlotList = {}

    -- item道具数据 list
    self.ItemDataList = {}

    -- 默认选中的 格子idx
    self.SelectIdx = 0

    -- 根据 itemDatas 生成 itemDataList
    -- 数据结构如下: data {EntityID, BaseID, Num}
    for i = 1, #itemDatas.goodsInfo do
        local itemData = {}
        itemData.ItemInfo = itemDatas.goodsInfo[i].ItemInfo
        itemData.EntityID = itemDatas.goodsInfo[i].UniID
        itemData.BaseID = itemDatas.goodsInfo[i].ItemID
        -- 数量先默认为 1
        itemData.Num = itemDatas.goodsInfo[i].count

        table.insert(self.ItemDataList, itemData)
    end

    local count = #self.ItemDataList

    for i = 1, count do
        -- 创造背包格子(带道具的格子)

        local data = self.ItemDataList[i]

        -- 处理堆叠问题
        local cfg = ConfigDataAccessor.Item[data.BaseID]
        local canStack = cfg.CanOverlay
        if canStack then
            local maxStack = 999 -- 配置表没有，暂时是临时的

            local tempNum = data.Num
            while tempNum > maxStack do
                tempNum = tempNum - maxStack

                data.Num = maxStack
                table.insert(self.SlotList, data)
            end

            data.Num = tempNum
            table.insert(self.SlotList, data)
        else
            table.insert(self.SlotList, data)
        end
    end

    -- 格子行数 默认最小 7 行
    local row = 7

    -- 向上取整 计算数据 的最小行数
    local dataRow = math.ceil(count / 7)

    -- 如果数据的 行数 大于默认的行数 , 那就用 数据的行数
    if dataRow > row then
        row = dataRow
    end

    -- 生成 空格子
    for i = count + 1, row * 7 do
        local it = {}
        it.BaseID = 0
        table.insert(self.SlotList, it)
    end

    self.BagScorllView:ContinueScroll(#self.SlotList, false)

    if count == 0 then
        self.SelectIdx = -1
        self.ItemDescContent.gameObject:SetActive(false)
    else
        -- TODO  选中第一个
        self.SelectIdx = 0
        self.ItemDescContent.gameObject:SetActive(true)
    end
end

function UnionWarehouse:CleanAllSelected()
    -- 一开始还没有数据
    if self.ItemList == nil then
        return
    end
    -- 无限列表
    for i = 1, #self.ItemList do
        self.ItemList[i]:MakeSelected(false)
    end
end

function UnionWarehouse:OnItemRefesh(tr, idx)
    for i = 1, #self.ItemList do
        if self.ItemList[i].transform == tr then
            -- self.ItemList[i]:SetItem(self.SlotList[idx + 1]); -- idx是从0开始，如果BagSlotList是lua的table需要idx+1
            self.ItemList[i]:SetItemNum(self.SlotList[idx + 1].BaseID, self.SlotList[idx + 1].Num)

            -- 刷新的就是默认选中的 道具icon
            if idx == self.SelectIdx then
                self:OnItemClick(self.ItemList[i])
            end
            return
        end
    end

    -- tr.gameObject:SetActive(true);
    local bi = ItemGrid:New(tr.gameObject, self.tr_ItemRoot)
    bi:InitItem(
        true,
        function(ritem)
            self.SelectIdx = idx
            self:OnItemClick(ritem)
        end
    )
    bi.gameObject:SetActive(true)
    -- bi:SetItem(self.SlotList[idx + 1]); -- idx是从0开始，如果BagSlotList是lua的table需要idx+1
    bi:SetItemNum(self.SlotList[idx + 1].BaseID, self.SlotList[idx + 1].Num)

    -- 刷新的就是默认选中的 道具icon
    if idx == self.SelectIdx then
        self:OnItemClick(bi)
    end
    table.insert(self.ItemList, bi)
end

function UnionWarehouse:OnItemClick(bitem)
    -- 格子状态为开启的时候
    if bitem ~= nil and bitem.BaseID > 0 then
        self:CleanAllSelected()

        bitem:MakeSelected(true)

        -- TODO: 刷新道具描述显示
        self:RefreshItemDesc(bitem)
    end
end

function UnionWarehouse:RefreshItemDesc(bitem)
    if bitem == nil or bitem.BaseID == 0 then
        self.ItemDescContent.gameObject:SetActive(false)
        return
    end
    self.ItemDescContent.gameObject:SetActive(true)

    local baseID = bitem.BaseID
    local cfg = ConfigDataAccessor.Item[baseID]
    local itemMaintType = ConfigDataAccessor.ItemMainType[cfg.ItemType]

    self.ItemName.text = cfg.Name

    if itemMaintType ~= nil then
        -- self.ItemType.text = "道具类型：" .. itemMaintType.Name;
        self.ItemType.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_UnionWarehouse_ItemType"), {itemMaintType.Name})
    else
        -- self.ItemType.text = "道具类型：" .. cfg.ItemType;
        self.ItemType.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_UnionWarehouse_ItemType"), {cfg.ItemType})
    end

    local bindInfo = CheckItemIsBind(bitem.Item)
    if bindInfo.IsBind then
        -- 如果道具已经绑定, 那就采用默认值 -1
        self.ItemCount.text = LanguageHelp:GetLanguageByKey("Local_Str_Bound")
    else
        -- self.ItemCount.text = "可交易" .. bindInfo.LeftTimes .. "次";
        self.ItemCount.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_leftTimes"), {bindInfo.LeftTimes})
    end

    self.ItemDesc.text = cfg.Desc

    -- icon的显示
    self.DescItemGrid:SetItemNum(bitem.BaseID, 0)
    self.DescItemGrid.Num.text = bitem.Num.text

    -- 拍卖道具的战力评分
    local score = FightPowerManager.Instance:GetBagItemShowPower(bitem)
    self.ItemPreDesc.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Equip_Level"), {score})
end

return UnionWarehouse

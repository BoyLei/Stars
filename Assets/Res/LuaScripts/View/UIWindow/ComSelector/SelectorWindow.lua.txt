function SelectorWindow:OnSubTypeClick(idx)

		self.selectedSubType[idx] = 0;

	else
		self.selectedSubType[idx] = 1;
		self.go_SelectedSubType[idx]:SetActive(true);
	end
end

function SelectorWindow:Refesh()
	-- for i = 1, 3 do
	-- 	self.commonSettingsSelect[i]:SetActive(false);
	-- 	self.commonSel[i] = 0;
	-- end

	self.SelectedSortLv = 0;
	if self.EquipLevel ~= nil then
		for i = 0, #self.EquipLevel do
			if self.EquipLevel[i] == self.SelectedSortLv then
				self.SortListIdx = i + 1;
				self.SortListTex.text = self.SortList.SortNames[i + 1];
			end
		end
	end

	self.QualitySelectedSortLv = 0;
	if self.QualityLevel ~= nil then
		for k, v in pairs(self.QualityLevel) do
			if self.QualityLevel[k] == self.QualitySelectedSortLv then
				self.QualitySortListIdx = k + 1;
				self.QualitySortListTex.text = self.QualitySortList.SortNames[k + 1];
			end
		end
	end

	for k, v in pairs(self.selectedSubType) do
		self.go_SelectedSubType[k]:SetActive(false);
		self.selectedSubType[k] = 0;
	end
	-- 词条
	-- self:RefeshItems();

	self.selectorData.commonSelector = self.commonSel;
	self.selectorData.levelRange = self.SelectedSortLv;
	self.selectorData.selectQuality = self.selectedQuality;
	self.selectorData.equipType = self.selectedSubType;
	if self.selectorData.wordsData == nil then
		self.selectorData.wordsData = {};
		local tb = {};
		for i = 1, 5 do
			tb[i] = 0;
		end
		self.selectorData.wordsData.quality = tb;
		self.selectorData.wordsData.wordsNum = -1;
		self.selectorData.wordsData.words = {};
	end
	self.selectorWords:OnBtnResetClick();
end

function SelectorWindow:RefeshItems()

	self:ClearItemData();

	for k, v in pairs(self.selectedSubAttr) do
		table.insert(self.ItemDataList, v);
	end

	for i = #self.ItemDataList + 1, 7 do
		table.insert(self.ItemDataList, 0);
	end

	self:SortItem(self.ItemDataList);

	self.EquipedAmuletScorllView:ContinueScroll(#self.ItemDataList);
end

function SelectorWindow:OnItemClick(bitem)
	self.selectorWords:Show(true);
end

function SelectorWindow:ClearItemData()

	if self.ItemDataList == nil then
		self.ItemDataList = {}
	else
		for k, _ in pairs(self.ItemDataList) do
			self.ItemDataList[k] = nil
		end
	end
end

function SelectorWindow:SortItem(tbl)
	if #tbl > 1 then
		table.sort(tbl, function(a, b)
				if a.Data.Heroid > b.Data.Heroid then
					return true
				elseif a.Data.Heroid == b.Data.Heroid then
					return a.Quality > b.Quality
				end
				return false
			end)
	end
end

function SelectorWindow:SetType(t, it)
	if t == nil then
		return;
	end

	if self.ItemType == it then
		return;
	end

	for k, v in pairs(self.filterAllTypes) do
		v:SetActive(false);
	end

	self.wordsType = {};

	for _, v in pairs(t) do
		local itemfilter = ConfigDataAccessor.Itemfilter[v];
		local pt = itemfilter.FilterType;
		if self.filterAllTypes[pt] ~= nil then
			self.filterAllTypes[pt]:SetActive(true);
			local op = itemfilter.OperaOption;
			if pt == 1 then
				self:SetCommonSettings(op);
			elseif pt == 2 then
				self:SetLevel(op);
			elseif pt == 3 then
				self:SetQualityLevel(op);
			elseif pt == 4 then
				self:SetEquip(op);
			end
		else
			table.insert(self.wordsType, v);
		end
	end

	self.ItemType = it;
	self:Refesh();
end

function SelectorWindow:SetCommonSettings(t)
	for i = 1, #self.commonSettingsSelect do
		self.commonSettingsSelect[i].transform.parent.gameObject:SetActive(false);
	end

	for k, v in pairs(t) do
		self.commonSettingsSelect[v].transform.parent.gameObject:SetActive(true);
	end
end

function SelectorWindow:SetLevel(t)
	self.EquipLevel = {}
	for k,v in pairs(t) do
		self.EquipLevel[k] = v;
	end

	local tb = {};
	table.insert(tb, LanguageHelp:GetLanguageByKey("Local_Str_SelectorWindow_LV"));
	for k, v in pairs(t) do
		--table.insert(tb, "等级" .. v .. "以上");
		table.insert(tb, StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SelectorWindow_LV2"), {v}));
	end
	
	self.EquipLevel[0] = 0;
	self.SortList:Init(tb, function(idx)
			self:OnSelectSortList(idx);
		end)
	self.SortListIdx = 1;
	self.SortListTex.text = self.SortList.SortNames[self.SortListIdx];
end

function SelectorWindow:GetQualityName(t)
	if t == 1 then
		return LanguageHelp:GetLanguageByKey("Local_Str_QualityName1");
	elseif t == 2 then
		return LanguageHelp:GetLanguageByKey("Local_Str_QualityName2");
	elseif t == 3 then
		return LanguageHelp:GetLanguageByKey("Local_Str_QualityName3");
	elseif t == 4 then
		return LanguageHelp:GetLanguageByKey("Local_Str_QualityName4");
	elseif t == 5 then
		return LanguageHelp:GetLanguageByKey("Local_Str_QualityName5");
	end
	return "";
end

function SelectorWindow:SetQualityLevel(t)
	self.QualityLevel = t;

	local tb = {};
	table.insert(tb, LanguageHelp:GetLanguageByKey("Local_Str_SelectorWindow_Quality"));
	for i = 1, #t do
		--table.insert(tb, "品质" .. self:GetQualityName(t[i]) .. "以上");
		table.insert(tb, StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SelectorWindow_Quality2"), {self:GetQualityName(t[i])}));
	end
	self.QualitySortList:Init(tb, function(idx)
			self:OnSelectQualitySortList(idx);
		end)

	self.QualityLevel[0] = 0;
	self.QualitySortListIdx = 1;
	self.QualitySortListTex.text = self.QualitySortList.SortNames[self.QualitySortListIdx];
end

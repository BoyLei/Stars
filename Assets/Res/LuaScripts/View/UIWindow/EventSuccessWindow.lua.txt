EventSuccessWindow=class("EventSuccessWindow",UIPanel)

local EventSuccessItem=require("View/Uiwindow/Event/EventSuccessItem")

function EventSuccessWindow:Awake()
	self.transform =self.prefab.transform:Find("Cut4Cam90/AssemblyRatio2/Success");
	self.gameObject=self.transform.gameObject;


	self.go_List=self.transform:Find("Center/Scroll View").gameObject;

	self.scoreTx=self.transform:Find("Top/score"):GetComponent(typeof(Text));

	
    --道具列表
    self.EventSuccessItemList={};
	--无限列表
	self.go_EventSuccessItem = self.transform:Find("Center/Scroll View/Mask/List/EventSuccessItem").gameObject;

	self.tr_ItemRoot=self.transform:Find("Center/Scroll View/Mask/List");
	self.go_EventSuccessItem:SetActive(false);
	self.go_EventSuccessItem.transform:SetParent(self.transform);
	self.EventSuccessScorllView=self.transform:Find("Center/Scroll View"):GetComponent(typeof(SuperScrollViewController));
	self.EventSuccessScorllView:NewScroll(self.go_EventSuccessItem, 0, function (tr,idx)
		self:OnEventSuccessItemRefesh(tr,idx);
	end);


	self.m_CloseBtn=self.transform:Find("Bottom/BtnConfirm"):GetComponent(typeof(JButton));
	self.m_CloseBtn.OnClick=(function() self:OnClickCloseBtn() end);




end



function EventSuccessWindow:OnEventSuccessItemRefesh(tr,idx)

	for i=1,#self.EventSuccessItemList do
		if self.EventSuccessItemList[i].transform == tr then
            self.EventSuccessItemList[i]:SetItem(self.ItemDataList[idx+1],idx+1);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
		end

	end

    local bi = EventSuccessItem:New(tr.gameObject);

	bi.gameObject:SetActive(true);
	bi:SetItem(self.ItemDataList[idx+1],idx+1);--idx是从0开始，如果ItemDataList是lua的table需要idx+1

    table.insert(self.EventSuccessItemList, bi);

end



function EventSuccessWindow:Refesh()
	self.scoreTx.text=self.Score;



    if self.ItemDataList==nil then
        self.ItemDataList ={}
    else
        for k,_ in pairs(self.ItemDataList) do self.ItemDataList[k] = nil end
    end


    local cfgs=ConfigDataAccessor.GVEGuildReward;

    local maxScore=0;

    for k,v in pairs(cfgs) do

        if v.PlayMode==self.playMode and self.effectLv == v.EffectLevel then;
            table.insert( self.ItemDataList, v );
            if v.ScoOneP > maxScore then
                maxScore=v.ScoOneP;
            end
        end

    end
    

	self:SortItem(self.ItemDataList);

	self.EventSuccessScorllView:ContinueScroll(#self.ItemDataList);


end

function EventSuccessWindow:SortItem(tbl)

	if #tbl > 1 then

		table.sort(tbl, function(a, b)
			--升序
			if a.ScoOneP < b.ScoOneP then
				return true
			end
			return false
		end)

	end

end

function EventSuccessWindow:Start()

end

function EventSuccessWindow:OnEnable()
	
end

function EventSuccessWindow:OnDisable()
	
end

function EventSuccessWindow:OnDestroy()

end

function EventSuccessWindow:OnOpen(arg)

	self.Score=arg.score;
	self.playMode=arg.playID;
	self.effectLv=arg.effectLv;


	self:Refesh();

end



function EventSuccessWindow:OnClose(arg)
	--self:CloseWindow();
end



function EventSuccessWindow:OnClickCloseBtn()



	self.Close();



end



return  EventSuccessWindow

function PlayerAttrWindow:SetAttrInfo(arg)
    if arg == nil then
        self:CloseWindow();
        return;
    end

    if type(arg) == "table" then
        self.m_EntityBase = arg[1];
        self.m_SyncBaseInfo = arg[2];
    else
        self.m_EntityBase = arg[0];
        self.m_SyncBaseInfo = arg[1];
    end

    if self.m_EntityBase == nil or self.m_SyncBaseInfo == nil then
        self:CloseWindow();
        return;
    end

    self:Init();

    if not self.transform.gameObject.activeSelf then
        self.transform.gameObject:SetActive(true);
    end

    LuaModuleDispatcher:SendMessage(ModuleDef.EquipModule, "OnOpenEquip", {1});
end

function PlayerAttrWindow:CloseWindow()
    if self.m_EntityBase ~= nil then
        self:ReleaseInfo();
    end
    -- 关闭道具Tips
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnCloseTips", {});
    LuaModuleDispatcher:SendMessage(ModuleDef.EquipModule, "OnCloseEquip", {});
end

function PlayerAttrWindow:GetSyncBaseInfoByIndex(index)
    if self.m_SyncBaseInfo == nil then
        return 0;
    end

    for i = 0, self.m_SyncBaseInfo.Count - 1 do
        local value = self.m_SyncBaseInfo[i];
        if value.Index == index then
            local type = LocalDataManager.Instance:GetAttrPropByAttrIdx(value.Index);
            return GetValueByPropType(value, type.type);
        end
    end

    return 0;
end

function PlayerAttrWindow:Init()
    local name = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.String,
        CS.StarProjectDef.AOIAttrDefine.Name);
    self.m_Name.text = name;
    -- local uid = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Ulong,CS.StarProjectDef.AOIAttrDefine.UserID);
    -- self.m_Uid.text = "UID："..tostring(uid);
    local jobId = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Uint,
        CS.StarProjectDef.AOIAttrDefine.Job);
    local jobName = "";
    local jobCfg = ConfigDataAccessor.Job[jobId];
    if jobCfg then
        jobName = jobCfg.Name;
    end
    self.m_Job.text = jobName;
    local level = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,
        CS.StarProjectDef.AOIAttrDefine.PlayerLevel);
    -- local maxLevel = self:GetMaxLevel();
    self.m_Level.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_PlayerAttrWindow_Level"), {level});

    local expCur = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,
        CS.StarProjectDef.AOIAttrDefine.PlayerExp);
    local expSum = self:GetCurLevelExp(level);
    self.m_Exp.text = tostring(expCur) .. "/" .. tostring(expSum);

    local ratio = 0;
    if expSum > 0 then
        ratio = expCur / expSum;
        if ratio > 1 then
            ratio = 1;
        end
    end
    self.m_ExpRectBar:SetWidth(self.expWidthCfg * ratio);

    local guidName = BusinessManager.Instance:GetGuildName();
    if guidName == "" then
        guidName = LanguageHelp:GetLanguageByKey("Local_Str_guidName");
    end
    self.m_Guild.text = guidName;

    self:SetModel(self.m_EntityBase.avatarDataCell);

    -- 根据OwnModule分类
    local list = ConfigDataAccessor.Attribute;
    self.BaseAttrConfList = {}; -- 一维数组
    self.HighAttrConfList = {}; -- 二维数组
    for k, v in pairs(list) do
        local value = self:GetSyncBaseInfoByIndex(v.AttrId);
        -- local value = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(v.AttrId);
        v.Value = value;

        v.ExtraVal = 0;
        local com = v.ComputationalFormula;
        if com ~= nil and #com >= 3 then
            local attr1ID = tonumber(com[1]);
            local attr2ID = tonumber(com[2]);
            local del = tonumber(com[3]);
            local RealityVal = self:GetSyncBaseInfoByIndex(attr1ID);
            local ExtraVal = self:GetSyncBaseInfoByIndex(attr2ID);
            v.ExtraVal = roundToDecimal((RealityVal - ExtraVal - del) / 100, 1);
        end

        if v.OwnModule == 0 then
            table.insert(self.BaseAttrConfList, v);
        elseif v.OwnModule ~= -1 then
            if self.HighAttrConfList[v.OwnModule] == nil then
                self.HighAttrConfList[v.OwnModule] = {};
            end
            table.insert(self.HighAttrConfList[v.OwnModule], v);
        end
    end
    -- 创建基础属性
    self.m_BaseAttrScroll:ContinueScroll(#self.BaseAttrConfList);
    -- 创建高级属性
    for i = 1, #self.ItemHighAttrfList do
        GameObject.Destroy(self.ItemHighAttrfList[i].Transform.gameObject);
    end
    self.ItemHighAttrfList = {};
    for k, v in pairs(self.HighAttrConfList) do
        local go = newObject(self.m_ItemHighAttr);
        go.transform:SetParent(self.m_HighAttrScroll.content);
        go.transform.localPosition = Vector3.zero;
        go.transform.localScale = Vector3.one;
        local item = ItemHighAttr:New(go);
        item:SetItem(k, self.HighAttrConfList[k], self.m_ItemHighBaseAttr, self.ItemHighAttrClickFun);
        table.insert(self.ItemHighAttrfList, item);
    end
    -- CS.UnityEngine.UI.LayoutRebuilder:ForceRebuildLayoutImmediate(self.m_HighAttrScroll.content:GetComponent(typeof(RectTransform)));
    -- self.m_HighAttrScroll.content:GetComponent(typeof(VerticalLayoutGroup)):CalculateLayoutInputVertical();
    -- self.m_HighAttrScroll.content:GetComponent(typeof(ContentSizeFitter)):SetLayoutVertical();

    ---------------- [战力] [start]------------

    self.PowerLevel.text = GetPlayerTotalPowerRankLevel(false)
    local RankLevelCfg = GetPlayerCurRankLevelCfg(false)
    if RankLevelCfg ~= nil then
        local level = RankLevelCfg.ParentRank;
        local iconName = "Icon_PowerLevel_" .. level;
        ResourceHelper:LoadIconByAtlasAsync(self.PowerIcon, ResourceHelper.AtlasPathCommonctrl, iconName, false)
    end

    ---------------- [战力] [END]------------

    ---------------- [虚拟列表] [start]------------

    ---------------- [虚拟列表] [END]------------

    UIManager.Instance:SetRawCamrea(true);

    -- 默认选中基础属性
    self:OnAttrMenuJBtnClick(1);
    -- 默认清空基础属性的选中
    self:CleanBaseAttrAllSelected();
end

function PlayerAttrWindow:OnItemRefesh(tr, idx)
    if self.BaseAttrItemList == nil then
        return;
    end

    local data = self.BaseAttrConfList[idx + 1];
    for i = 1, #self.BaseAttrItemList do
        if self.BaseAttrItemList[i].transform == tr then
            self.BaseAttrItemList[i]:UpdateContent(data);
            if self.CurCheckAttrData ~= nil and self.BaseAttrItemList[i].AttrData == self.CurCheckAttrData then
                self.BaseAttrItemList[i]:OnSelect(true);
            end
            return;
        end
    end
    -- 找不到，就创建一个新的
    self:CreateItemBaseAttr(tr.gameObject, data);
end

function PlayerAttrWindow:CreateItemBaseAttr(gameObject, data)
    local item = ItemBaseAttr:New(gameObject, self.ItemBaseAttrClickFun);
    item:UpdateContent(data);
    if self.CurCheckAttrData ~= nil and data == self.CurCheckAttrData then
        item:OnSelect(true);
    end
    item.prefab.gameObject:SetActive(true);

    table.insert(self.BaseAttrItemList, item);
end

-- 清空所有基础属性的选中
function PlayerAttrWindow:CleanBaseAttrAllSelected()
    for i = 1, #self.BaseAttrItemList do
        self.BaseAttrItemList[i]:OnSelect(false);
    end
end

-- 基础属性点击事件
function PlayerAttrWindow:OnItemBaseAttrClick(data)
    self:CleanBaseAttrAllSelected();
    if data == nil then
        return;
    end
    self:OpenAttrDeclareWidget(data);
end

function PlayerAttrWindow:OpenAttrDeclareWidget(data)
    -- 显示属性详细界面
    self.CurCheckAttrData = data;
    if self.AttrDeclareWidget == nil then
        local list = {};
        table.insert(list, self.m_EntityBase);
        table.insert(list, self.BaseAttrConfList);
        table.insert(list, self.HighAttrConfList);
        table.insert(list, self.CurCheckAttrData);
        local widget = UIManager.Instance:OpenWidget(self.AttrDeclareWidgetPath, false, list, self.Root,
            CS.StarProjectDef.MainPageCommond.HideNone, false, false);
        self.AttrDeclareWidget = widget:GetLuaPanel().ScriptTable;
        self.AttrDeclareWidget.RootParent = self.Root;
    else
        self.AttrDeclareWidget:SetCheckAttr(self.CurCheckAttrData);
    end
    -- 关闭道具Tips
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnCloseTips", {});
    -- LuaModuleDispatcher:SendMessage(ModuleDef.EquipModule, "OnCloseEquip", { });
end

function PlayerAttrWindow:CleanHighAttrAllSelected()
    for i = 1, #self.ItemHighAttrfList do
        local item = self.ItemHighAttrfList[i];
        item:CleanBaseAttrAllSelected();
    end
end

function PlayerAttrWindow:OnAttrMenuJBtnClick(idx)
    for i = 1, #self.AttrMenuJBtns do
        self.AttrMenuJBtns[i]:MakeSelected(false);
    end

    self.AttrMenuJBtns[idx]:MakeSelected(true);

    self.m_BaseAttrScroll.gameObject:SetActive(idx == 1);
    self.m_HighAttrScroll.gameObject:SetActive(idx == 2);
    if idx == 1 then
        -- self:UpdateData(self.BaseAttrConfList);
    elseif idx == 2 then
        if self.CurCheckAttrData ~= nil then
            for i = 1, #self.ItemHighAttrfList do
                local item = self.ItemHighAttrfList[i];
                item:CleanBaseAttrAllSelected();
                item:CheckItemByAttrId(self.CurCheckAttrData.AttrId);
            end
        end
    end
end

function PlayerAttrWindow:GetCurLevelExp(level)
    --return ConfigDataAccessor.LevelExp[level].Exp;
    if level == 0 then
		return 0;
	end
	local cfg = ConfigDataAccessor.LevelExp[level];
	if cfg ~= nil then
		return cfg.Exp;
	end
	return 0;
end

function PlayerAttrWindow:GetMaxLevel()
    local count = #ConfigDataAccessor.LevelExp;
    return ConfigDataAccessor.LevelExp[count].Id;
end

function PlayerAttrWindow:SetModel(avatarDataCell)
    if self.m_avatarDataCell ~= avatarDataCell then
        self:PushDisplayGob();
        self.m_avatarDataCell = avatarDataCell;

        ------- 加载头像
        local headItem = ConfigDataAccessor.ModelHead[tonumber(self.m_avatarDataCell.HeadID)];
        if headItem ~= nil then
            self:GetSpriteByAtlas(headItem.HeadAtlasName, headItem.MinHead, function(sp)
                self.m_Portrait.sprite = sp
            end);
        end
        ------- 加载头像

        ------- 设置模型
        ResourceManager.Instance:PopGameObject("Roles/Template/Display_Model", function(go)
            if go == nil then
                return;
            end

            if self.m_avatarDataCell == nil then
                ResourceManager.Instance:PushGameObject("Roles/Template/Display_Model", go);
                return;
            end

            local pos = Vector3.zero;
            pos.x = -0.5;
            UIManager.Instance:SetModel2UI2(go, self.m_avatarDataCell.AvatarID, CS.StarProjectDef.ThreeDModelPos.Raw,
                Vector3.one, Vector3.zero, Vector3.zero, pos, "idle_01");
            go:SetActive(true);
            self.m_ModelViewDisplay = go:GetComponent(typeof(RoleViewDisplay));
            self.m_DisplayRawing:SetModelView(self.m_ModelViewDisplay);

            if self.m_ModelViewDisplay ~= nil then
                local rotation = Quaternion.Euler(0, 0, 0);
                self.m_ModelViewDisplay:SetRotation(rotation);
                self.m_DisplayRawing:SetState(true);
            end

            ------- 设置模型
            self.renderTexture.texture = ResourceManager.Instance.RenderTexture;
        end);
    end
    if self.m_ModelViewDisplay ~= nil then
        local rotation = Quaternion.Euler(0, 0, 0);
        self.m_ModelViewDisplay:SetRotation(rotation);
        self.m_DisplayRawing:SetState(true);
    end
end

function PlayerAttrWindow:PushDisplayGob()
    if self.m_ModelViewDisplay ~= nil then
        ResourceManager.Instance:PushGameObject("Roles/Template/Display_Model", self.m_ModelViewDisplay.gameObject);
        -- GameObject.Destroy(self.m_ModelViewDisplay.gameObject);
        self.m_ModelViewDisplay = nil;
        self.m_DisplayRawing:SetModelView(self.m_ModelViewDisplay);
    end
end

function PlayerAttrWindow:CloseAttrDeclareWidget()
    if self.AttrDeclareWidget ~= nil then
        self.AttrDeclareWidget:CloseWidget();
        self:CleanBaseAttrAllSelected();
        self:CleanHighAttrAllSelected();
    end
end

function PlayerAttrWindow:ReleaseInfo()
    self:PushDisplayGob();
    UIManager.Instance:SetRawCamrea(false);
    self.m_avatarDataCell = nil;
    self.m_EntityBase = nil;
    if self.m_DisplayRawing ~= nil then
        self.m_DisplayRawing:SetState(false);
    end
    self.BaseAttrConfList = {};
    self.HighAttrConfList = {};

    self:CloseAttrDeclareWidget();
    -- self.m_HighAttrScroll.content:DestroyChildren();
end

return PlayerAttrWindow

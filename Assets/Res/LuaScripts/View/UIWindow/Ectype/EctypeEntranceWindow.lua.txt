-- 默认选中可挑战的最高等级关卡
function EctypeEntranceWindow:SetDefaultSelData()
    if self.curSel ~= nil then return end

    local dailyCopyData = nil
    local defaultDailyCopyData = nil
    local lv = GameManager.Instance:GetPlayerLevel();
    local dailyCopyDifficulty = ConfigDataAccessor.DailyCopyDifficultyList;
    for _, v in ipairs(dailyCopyDifficulty) do
        if defaultDailyCopyData == nil then
            defaultDailyCopyData = v
			self.curDif = v.DifficultyType
        end
        -- 找到当前玩家等级最高可挑战的关卡
        if  GameManager.Instance:IsConditionMete(v.UnlockConditions) then -- v.UnlockConditions >= lockedLevel and lv >= v.UnlockConditions then
            dailyCopyData = v
			self.curDif = v.DifficultyType
        end
    end

    -- 还未解锁任意关卡默认选第一个
    if dailyCopyData == nil then
        dailyCopyData = defaultDailyCopyData
    end

    if dailyCopyData ~= nil then
        self.curSel = ConfigDataAccessor.DailyCopyList[dailyCopyData.CopyID];
        self:SetCurMode(ConfigDataAccessor.DailyCopyList[dailyCopyData.CopyID], dailyCopyData.ID);
    end
end

function EctypeEntranceWindow:GetUnlockCondition()
    local lvlimit = 0;
    local dailyCopyDifficulty = ConfigDataAccessor.DailyCopyDifficultyList;
    for _, v in pairs(dailyCopyDifficulty) do
        if self.curSel.ID == v.CopyID and self.curDif == v.DifficultyType then
            lvlimit = v.UnlockConditions;
            break
        end
    end
    return lvlimit
end

function EctypeEntranceWindow:SetCurMode(data, id)
    for k, v in pairs(self.Items) do
        v.sel:SetActive(k == id);
        --v:SetSelect(k == data.ID);
    end

    self.nameTxt.text = data.Name;
    self.descTxt.text = data.Decs;
    self.curSel = data;
    self:SetCount();

    local dailyCopyDifficulty = ConfigDataAccessor.DailyCopyDifficultyList;
    local dailyCopyList = ConfigDataAccessor.DailyCopyList;
    ResourceHelper:LoadIconAsync(self.bgImg, "", dailyCopyList[dailyCopyDifficulty[id].CopyID].BG, false)
end

function EctypeEntranceWindow:SetDifficult()
    for k, v in pairs(self.difObjs) do
        v:SetActive(k == self.curDif);
    end

    for _, v in pairs(self.firstPassItems) do
        GameObject.Destroy(v);
    end
    for _, v in pairs(self.normalRewardItems) do
        GameObject.Destroy(v);
    end
    self.firstPassItems = {}
    self.normalRewardItems = {}

    local firstSuc = false;
    local key = (self.curSel.ID << 48) | (self.curDif << 16 >> 16);
    if self.module.allDailyDatas ~= nil then
        for k, v in pairs(self.module.allDailyDatas) do
            if key == k then
                firstSuc = true;
                break
            end
        end
    end
    local dailyCopyDifficulty = ConfigDataAccessor.DailyCopyDifficultyList;
    for k, v in pairs(dailyCopyDifficulty) do
        if v.CopyID == self.curSel.ID and v.DifficultyType == self.curDif then
            local frewards = v.FirstReward;
            for index, reward in pairs(frewards) do
                local gob = newObject(self.firstPassRewardItem);
                gob:SetActive(true);
                gob.transform:SetParent(self.firstPassRewardItem.transform.parent);
                gob.transform.localPosition = Vector3.zero;
                gob.transform.localScale = Vector3.one;

                local it = RewardItem:New(gob, reward, v.FirstRewardNum[index], firstSuc);
				table.insert(self.firstPassItems, gob);
                --self.firstPassItems[reward] = gob;
            end

            local nrewards = v.BaseAwardsShow;
            for _, reward in pairs(nrewards) do
                local gob = newObject(self.normalRewardItem);
                gob:SetActive(true);
                gob.transform:SetParent(self.normalRewardItem.transform.parent);
                gob.transform.localPosition = Vector3.zero;
                gob.transform.localScale = Vector3.one;
                local it = RewardItem:New(gob, reward, 0);
                self.normalRewardItems[reward] = gob;
            end

            --for _, item in pairs(self.Items) do
                --for _, itemdata in pairs(dailyCopyDifficulty) do
                    --if item.data.ID == itemdata.CopyID and itemdata.DifficultyType == self.curDif then
                        --item:SetLevel(itemdata.UnlockConditions);
                        --break
                    --end
                --end
            --end
        end
    end

    self:SetCount();
end

function EctypeEntranceWindow:SetCount()

    local info = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.ComCount):GetData(8);

    if info == nil then
        return;
    end

    self.timeInfo = info;

    local firstSuc = false;
    local key = (self.curSel.ID << 48) | (self.curDif << 16 >> 16);
    if self.module.allDailyDatas ~= nil then
        for k, v in pairs(self.module.allDailyDatas) do
            if key == k then
                firstSuc = true;
                break
            end
        end
    end

    local lvlimit = self:GetUnlockCondition();

    if not GameManager.Instance:IsConditionMete(lvlimit) then 
        local groups = GetCMconditionByGroupID(lvlimit)
        self.leftTimes.text = groups[1].Desc
    else
        if not firstSuc then
            --self.leftTimes.text = "<color=#D3D698>(首次挑战不消耗次数)</color>";
            self.leftTimes.text = LanguageHelp:GetLanguageByKey("Local_Str_EctypeEntranceWindow_leftTimes");

            -- "今日剩余次数：<color=#4CC9F8>" .. info.RemainCount .. "/" .. info.AddCount ..
            --                           "</color><color=#D3D698>(首次挑战不消耗次数)</color>";
        else
            --self.leftTimes.text = "今日剩余次数：<color=#4CC9F8>" .. info.RemainCount .. "/" .. info.AddCount .. "</color>";
            self.leftTimes.text = LanguageHelp:GetLanguageByKey("Local_Str_EctypeEntranceWindow_leftTimes2") .. "：<color=#4CC9F8>" .. info.RemainCount .. "/" .. info.AddCount .. "</color>";
        end
    end
end

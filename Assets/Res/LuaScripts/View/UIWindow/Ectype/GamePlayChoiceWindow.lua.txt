--时序残响 单人本
function GamePlayChoiceWindow:OpenDailyCopy()
    -- 检测等级是否满足最小等级

    local config = ConfigDataAccessor.PlayOpenConditions[6]
    if not (CheckSysOpen(6, true)) then
        return
    end

    local cb = function()
        local mapID = config.MapID
        local npcID = config.NPCID
        local mete = GameManager.Instance:IsConditionMete(config.IndexChangeCondition)
        if mete then
            mapID = config.MapID2
            npcID = config.NPCID2
        end

        TaskHelper.FindPathByNpc(
            mapID,
            npcID,
            1,
            function(result)
                if result then
                    LuaModuleDispatcher:SendMessage(ModuleDef.EctypeEntranceModule, "OnOpenEctypeEntrance", {})
                --LuaModuleDispatcher:SendMessage(ModuleDef.EctypePopModule, "OnOpenEctypePop", {});
                --LuaModuleDispatcher:SendMessage(ModuleDef.RingTaskModule, "OnCkeckAndOpenRingTaskWindow",{});
                end
            end
        )
        if GamePlayChoiceWindow.IsOpen then
            self.starAnima:Play("Selected")
        end
    end

    local closeCb = function()
        if GamePlayChoiceWindow.IsOpen then
            self.starAnima:Play("Selected")
        end
    end
    self.starAnima:Play(
        "UnSelected",
        function()
            LuaModuleDispatcher:SendMessage(ModuleDef.DailyTeamModule, "OpenDailyTeamOpenTipsWidget", {6, cb, "", closeCb})
        end
    )

    --LuaModuleDispatcher:SendMessage(ModuleDef.EctypePopModule, "OnOpenEctypePop", {});
end

--研究手册  环任务
function GamePlayChoiceWindow:OpenRingTask()
    local allRingTasks = BusinessManager.Instance:GetAllRingTaskIDs()
    -- 领取了第一个环任务
    if allRingTasks ~= nil and allRingTasks.Count > 0 then
        local isContain = TaskHelper.ContainTask(allRingTasks[0])
        local isFinish = TaskHelper.IsTaskFinsh(allRingTasks[0])

        if (TaskHelper.ContainTask(allRingTasks[0]) or TaskHelper.IsTaskFinsh(allRingTasks[0])) then
            local closeCb = function()
                self.starAnima_1:Play("Selected")
            end
            self.starAnima_1:Play(
                "UnSelected",
                function()
                    -- 打开UI
                    UIManager.Instance:CloseAllLoadedWindowNWidget()
                    --UIManager.Instance:OpenWindowAsync(RingTaskWindowPath, nil, nil)
                    LuaModuleDispatcher:SendMessage(ModuleDef.RingTaskModule, "OnCkeckAndOpenRingTaskWindow", {})
                end
            )
        else
            --LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "OpenSecretAreaOpenTipsWidget", {});
            local config = ConfigDataAccessor.PlayOpenConditions[5]
            if not (CheckSysOpen(5, true)) then
                return
            end

            local cb = function()
                local mapID = config.MapID
                local npcID = config.NPCID
                local mete = GameManager.Instance:IsConditionMete(config.IndexChangeCondition)
                if mete then
                    mapID = config.MapID2
                    npcID = config.NPCID2
                end

                TaskHelper.FindPathByNpc(
                    mapID,
                    npcID,
                    1,
                    function(result)
                        if result then
                            LuaModuleDispatcher:SendMessage(ModuleDef.RingTaskModule, "OnCkeckAndOpenRingTaskWindow", {})
                        end
                    end
                )
                if GamePlayChoiceWindow.IsOpen then
                    self.starAnima_1:Play("Selected")
                end
            end

            local closeCb = function()
                self.starAnima_1:Play("Selected")
            end
            self.starAnima_1:Play(
                "UnSelected",
                function()
                    LuaModuleDispatcher:SendMessage(ModuleDef.DailyTeamModule, "OpenDailyTeamOpenTipsWidget", {5, cb, "", closeCb})
                end
            )
        end
    else
        local config = ConfigDataAccessor.PlayOpenConditions[5]
        if not (CheckSysOpen(5, true)) then
            return
        end
    end
end

--侵蚀裂隙 通缉任务

    if not (CheckSysOpen(3, true)) then
        return
    end

    local str = LanguageHelp:GetLanguageByKey("Local_Str_Go")
    local isReceiveTask = false
    local info = GameManager.Instance:GetMainPlayerWantTaskMD()
    if info ~= nil then
        isReceiveTask = info.IsAcc
    end
    if isReceiveTask then
        str = LanguageHelp:GetLanguageByKey("Local_Str_examine")
    end

    local cb = function()
        if isReceiveTask then
            -- 打开通缉界面
            LuaModuleDispatcher:SendMessage(ModuleDef.WantedModule, "OnOpenWantedWindow", {})
        else
            local mapID = config.MapID
            local npcID = config.NPCID
            local mete = GameManager.Instance:IsConditionMete(config.IndexChangeCondition)
            if mete then
                mapID = config.MapID2
                npcID = config.NPCID2
            end

            TaskHelper.FindPathByNpc(
                mapID,
                npcID,
                1,
                function(result)
                    if result then
                    -- 打开通缉界面
                    --LuaModuleDispatcher:SendMessage(ModuleDef.WantedModule, "OnOpenWantedWindow", {})
                    end
                end
            )
        end
        if GamePlayChoiceWindow.IsOpen then
            self.starAnima_4:Play("Selected")
        end
    end

    local closeCb = function()
        self.starAnima_4:Play("Selected")
    end
    self.starAnima_4:Play(
        "UnSelected",
        function()
            --LuaModuleDispatcher:SendMessage(ModuleDef.WantedModule, "OnOpenWantedTipsWidget", {});
            LuaModuleDispatcher:SendMessage(ModuleDef.DailyTeamModule, "OpenDailyTeamOpenTipsWidget", {3, cb, str, closeCb})

function GamePlayChoiceWindow:OnPageChange()
    if self.PageIndex < 1 then
        self.PageIndex = 1
    end

    if self.PageIndex > MaxPage then
        self.PageIndex = MaxPage
    end

    for i = 1, #self.Pages do
        local page = self.Pages[i]
        if page ~= nil then
            page:SetActive(i == self.PageIndex)
        end
    end

    if self.leftBtn ~= nil then
        self.leftBtn.gameObject:SetActive(self.PageIndex > 1)
    end

    if self.rightBtn ~= nil then
        self.rightBtn.gameObject:SetActive(self.PageIndex < MaxPage)
    end

    if self.PageIndex == 1 then
        self.PlayAnimation(
            "GamePlayChoiceWindow_QieHuan_Anim01",
            function()
                --self.PlayAnimation("GamePlayChoiceWindow_Loop_Anim")
            end
        )
    elseif self.PageIndex == 2 then
        self.PlayAnimation(
            "GamePlayChoiceWindow_QieHuan_Anim",
            function()
                --self.PlayAnimation("GamePlayChoiceWindow_Loop02_Anim")
            end
        )
    end
end

function GamePlayChoiceWindow:FreshDailyCopyleftTime()
    -- self.PersonSecretleftTimes
    local config = ConfigDataAccessor.PlayOpenConditions[6]

    if not (CheckSysOpen(6)) then
        self.m_DailyCopybg.color = LockColor
        self.leftTimes.text = LanguageHelp:GetLanguageByKey("Local_Str_Tips3")
        --self.leftTimes.transform.parent.gameObject:SetActive(false)
        return
    end

    --self.leftTimes.transform.parent.gameObject:SetActive(true)

    local info = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.ComCount):GetData(8)
    if info == nil then
        return
    end
    --self.leftTimes.text = "今日剩余次数：" .. info.RemainCount .. "/" .. info.AddCount;

    local isHaveLeft = (info.RemainCount > 0)
    if isHaveLeft then
        self.go_finish_DailyCopy:SetActive(false)
        self.m_DailyCopybg.color = WhiteColor
        self.leftTimes.text = LanguageHelp:GetLanguageByKey("Local_Str_EctypeEntranceWindow_leftTimes2") .. "：" .. info.RemainCount .. "/" .. info.AddCount
    else
        self.go_finish_DailyCopy:SetActive(true)
        self.m_DailyCopybg.color = LockColor
        self.leftTimes.text = LanguageHelp:GetLanguageByKey("RingTaskWindow_12")
    end
end

function GamePlayChoiceWindow:SetRingTaskInfo()
    local config = ConfigDataAccessor.PlayOpenConditions[5]
    if not (CheckSysOpen(5)) then
        self.m_WeekCopybg.color = LockColor
        self.leftTimes2.text = LanguageHelp:GetLanguageByKey("Local_Str_Tips3")
        return
    end

    local allRingTasks = BusinessManager.Instance:GetAllRingTaskIDs()
    -- 领取了第一个环任务
    if allRingTasks ~= nil and allRingTasks.Count > 0 then
        local finishNum = 0
        local allTaskNum = allRingTasks.Count - 1
        for i = 0, allRingTasks.Count - 2 do
            local isfinish = self:CheckRingTaskFinish(allRingTasks[i])
            if isfinish then
                finishNum = finishNum + 1
            end
        end

        --self.leftTimes2.text = "已完成：" .. finishNum .. "/" .. allTaskNum;

        if finishNum >= allTaskNum then
            self.go_finish_WeekCopy:SetActive(true)
            self.m_WeekCopybg.color = LockColor
            self.leftTimes2.text = LanguageHelp:GetLanguageByKey("RingTaskWindow_12")
        else
            --self.leftTimes2.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_leftTimes2"), {finishNum, allTaskNum})
            self.go_finish_WeekCopy:SetActive(false)
            self.m_WeekCopybg.color = WhiteColor

            self.leftTimes2.text = LanguageHelp:GetLanguageByKey("Local_Str_EctypeEntranceWindow_leftTimes2") .. " " .. (allTaskNum - finishNum) .. "/" .. allTaskNum
        end
    else
        self.m_WeekCopybg.color = LockColor
        self.leftTimes2.text = LanguageHelp:GetLanguageByKey("Local_Str_Tips3")
    end
end

function GamePlayChoiceWindow:CheckRingTaskFinish(id)
    local isFinish = false

    if id ~= nil and id > 0 then
        local idNow = id
        local isLast = false
        local configNow = TaskHelper.GetTaskConfig(idNow)
        local baseNow = nil
        if configNow ~= nil then
            baseNow = configNow.Base
        else
            isLast = true
        end

        while not (isLast) do
            -- 任务状态
            if TaskHelper.ContainTask(idNow) then
                -- 正在进行
                isLast = true
            elseif TaskHelper.IsTaskFinsh(idNow) then
                if baseNow.NextTaskID == 0 then
                    -- 已经完成
                    isLast = true
                    isFinish = true
                else
                    -- 继续往下查
                    configNow = TaskHelper.GetTaskConfig(baseNow.NextTaskID)
                    if configNow ~= nil then
                        idNow = baseNow.NextTaskID
                        baseNow = configNow.Base
                    else
                        -- 结束
                        baseNow = nil
                        isLast = true
                        print("任务ID：" .. baseNow.NextTaskID .. "不存在")
                    end
                end
            else
                isLast = true
            end
        end
    end

    return isFinish
end

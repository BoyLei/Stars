PlotImageWindow=class("PlotImageWindow",UIPanel)
PlotImageWindow.m_RawImage=nil;
PlotImageWindow.m_AutoAlpha=nil;
PlotImageWindow.EffectID=0;

function PlotImageWindow:Awake()
	self.m_RawImage=self:Find("RawImage"):GetComponent(typeof(RawImage));
	self.m_AutoAlpha=self:Find("RawImage"):GetComponent(typeof(AutoAlpha));

end

function PlotImageWindow:Start()

end

function PlotImageWindow:OnEnable()
	
end

function PlotImageWindow:OnDisable()
	
end

function PlotImageWindow:OnDestroy()

end

function PlotImageWindow:OnOpen(arg)
	local config=arg[1]
	local EffectID=arg[2]
	self.EffectID=EffectID;
	if config==nil then
		self:OnEnd();
	else
		ResourceHelper:LoadRawIconAsync(self.m_RawImage, config.Path, true, function ()
			self.m_AutoAlpha.Play(function ()
				self:OnEnd();
			end)
		end)

		-- 		self.m_AutoAlpha.Play(function ()
		-- 			self:OnEnd();
		-- 		end)
		-- end)
		
	end
end

function PlotImageWindow:OnEnd()
	CS.StarProject.GlobalEvent.OnEffectEnd:Invoke(self.EffectID)
	self:Close();
end

end

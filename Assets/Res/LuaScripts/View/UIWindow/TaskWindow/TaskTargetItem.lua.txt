TaskTargetItem=class("TaskTargetItem",UIPanel)
TaskTargetItem.Content=nil;
TaskTargetItem.Image=nil;

function TaskTargetItem:ctor(go,atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.Content=self:Find("Text"):GetComponent(typeof(Text));
	self.Image=self:Find("Image"):GetComponent(typeof(Image));
end


function TaskTargetItem:OnShow(data)
	self.Content.text=data.TaskDesc
	local color=data:GetStateColor()
	self.Image.color=color;
	self.prefab:SetActive(true)
end

function TaskTargetItem:OnSpecShow(desc,color)
	self.Content.text=desc
	self.Image.color=color;
	self.prefab:SetActive(true)
end

function TaskTargetItem:OnHide()
	self.prefab:SetActive(false)
end
return  TaskTargetItem

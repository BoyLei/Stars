---@class TaskChapter
TaskChapter = class("TaskChapter",UIPanel)
local TaskScrolIItem=require('View/UIWindow/TaskWindow/TaskScrolIItem')
TaskChapter.m_ChapterBg=nil;
TaskChapter.m_Tasks=nil;
TaskChapter.m_TaskTemplate=nil;
TaskChapter.m_TaskGroups={};

function TaskChapter:ctor(go,atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_TaskTemplate=self:Find("Tasks/Task").gameObject
	self.m_TaskGroups={}
end

function TaskChapter:OnSelect()
	if self.m_TaskGroups[0]~=nil then
		self.m_TaskGroups[0]:OnSelectHandler();
	end
end

function TaskChapter:GetIconName(taskType)
	if taskType==1 then
		return "Task_title_01"
	else
		return "Task_title_02"
	end
end

--数据更新
function TaskChapter:OnShow(Chapter,data)

	if data==nil or data.Count<1 then
		self:OnHide();
		return
	end
	local type=1;
	for k, v in pairs(data) do
		if self.m_TaskGroups[k]==nil then
			local go=GameObject.Instantiate(self.m_TaskTemplate)
			go.transform:SetParent(self.m_TaskTemplate.transform.parent)
			go.transform.localPosition=Vector3(0,0,0)
			go.transform.localScale=Vector3(1,1,1)
			self.m_TaskGroups[k]=TaskScrolIItem.new(go,self.AtlasPath)
		end
		type=v.TaskType
		self.m_TaskGroups[k]:OnShow(v)
	end
	self.prefab:SetActive(true)
end

function TaskChapter:OnHide()
	if self.m_TaskGroups~=nil then
		for k, v in pairs(self.m_TaskGroups) do
			v:OnHide()
		end
	end
	self.prefab:SetActive(false)
end

return  TaskChapter;

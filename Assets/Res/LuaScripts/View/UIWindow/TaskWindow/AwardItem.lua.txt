AwardItem=class("AwardItem",UIPanel)
AwardItem.bg=nil;
AwardItem.icon=nil;
AwardItem.num=nil;
AwardItem.Btn=nil;
AwardItem.Cansell=nil;
local data=nil;

function AwardItem:Awake()
	self.bg=self:Find("bg"):GetComponent(typeof(Image));
	self.icon=self:Find("Icon"):GetComponent(typeof(Image));
	self.num=self:Find("Image/num"):GetComponent(typeof(Text));
	self.Btn=self:Find("bg"):GetComponent(typeof(Button));
	self.Cansell=self:Find("cansell"):GetComponent(typeof(RectTransform));
	self.Btn.onClick:AddListener(function ()
			self:OnClickHandler()
		end);
end

function  AwardItem:OnClickHandler()
	--CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(self.TaskInfo.TaskID)
	Log("on click")
end


function AwardItem:OnUpdateContent(data)
	self.data=data

	local config=ConfigDataAccessor.Item[self.data.ItemID]
	if config~=nil then
		--self.icon=
		--self.bg=
	end

	self.num.text=data.ItemNum
	self.Cansell.gameObject:SetActive(self.data.CanSell)
end

return  AwardItem

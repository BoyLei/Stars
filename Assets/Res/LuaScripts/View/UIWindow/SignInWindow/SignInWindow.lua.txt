function SignInWindow:RefreshBeginnerRedPoint()

    local BeginnerTargetData

function SignInWindow:InitViewByTabIndex(systemtype)

    end

    if self.TabIndex == 3 and not self:IsPreviewTabCanShow() then
        self:CalCurrentOpen()

        self:CalCurrentOpen()
    end

    local issignopen = CS.StarProject.Service.SystemOpen.SystemOpenManager.Instance:SystemIsOpen(SystemOpenType.Activ_SignIn)
    if self.TabIndex == 2 and not issignopen then
        self:CalCurrentOpen()

	if self.TabIndex == 1 and  sevenDayGoalMD == nil or self:CalSeventDayLastTime(sevenDayGoalMD.ActiveDateTime) <= 0 or not sevenDayGoalMD.IsActive then

    end

        self.daySign:SetActive(false);
        self.m_PreviewRoot:SetActive(false);
        self.m_PartnerTaskPanel:Show(false);

        self.daySignSel:SetActive(false);
        self.m_PreviewSel:SetActive(false)
        self.m_PartnerTaskSel:SetActive(false)

        end

        self.daySign:SetActive(true);
        self.m_PreviewRoot:SetActive(false);
        self.m_PartnerTaskPanel:Show(false);

function SignInWindow:SetLeftTime(st)
    if self.module.signInWindowView == nil then
        return;
    end
    local t = math.floor((st + 3600*24*30) - TimeUtils.DateTimeToStampMilli(TimeUtils.GetServerTime())/1000);
    if t < 0 then
        t = 0;
    end
    if self.leftTimeText == nil then
        self.leftTimeText = self.transform:Find("Cut4Cam90/AssemblyRatio2/Root/Right/container/ActivityTime/LeftTime"):GetComponent(typeof(Text));
    end
    if self.leftTimeText ~= nil then
        self.leftTimeText.text = self:FormatShowTime(t);
    end
end

function SignInWindow:FormatShowTime(timeInSec)
    local day=0;
    local hour=0;
    local min=0;
    if (timeInSec / 86400) > 0 then
        day = math.floor(timeInSec / 86400);
        --_text = showTime.ToString() + "天";
    end

    timeInSec = timeInSec - 86400*day;
    if (timeInSec / 3600) > 0 then
        hour = math.floor(timeInSec / 3600);
        --_text = showTime.ToString() + "小时";
    end

    timeInSec = timeInSec - 3600*hour;
    
    if (timeInSec / 60) > 0 then
        min = math.ceil(timeInSec / 60);
        --_text = showTime.ToString() + "分钟";
    end

    --return string.format("%d天%d小时%d分钟", day, hour, min);
    return StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Time"), {day, hour, min});
end

function SignInWindow:InitDayRewards(seasonId, checkinData, rewardData, st)
	local signInData = {};
    local signSeasonList = ConfigDataAccessor.SignSeasonList;
    for k,v in pairs(signSeasonList) do
       -- logError("check in1 seasonid == ",v.SeasonID,"compare == ",seasonId)
       --当前只配置了1季度
        if v.SeasonID == 1 then --seasonId then
            signInData[v.SignNum] = v;
        end
    end
    local allinitcount = 0
    local initednum = 0
    local initcb = function ()
        initednum = initednum + 1
        if initednum == allinitcount then 
            for k,v in pairs(self.allDailyItems) do
                self.allDailyItems[k].gameObject:SetActive(true);
            end
        end
    end
    for k,v in pairs(signInData) do
        allinitcount = allinitcount + 1
     --   self.allDailyItems[k].gameObject:SetActive(true);
        self.allDailyItems[k]:Init(v, checkinData, rewardData, st,initcb);
    end
end

function SignInWindow:InitAccmulateRewards(seasonId, checkInCount, countList)
	local accmulateData = {};
    local signAccrueList = ConfigDataAccessor.SignAccrueList;
    for k,v in pairs(signAccrueList) do
      --  logError("check in seasonid == ",v.SeasonID,"compare == ",seasonId)
        if v.SeasonID == 1 then --seasonId then
            accmulateData[v.ID] = v;
        end
    end

    for k,v in pairs(accmulateData) do
        self.allAccmulateItems[k].gameObject:SetActive(true);
        self.allAccmulateItems[k]:Init(v, checkInCount, countList);
    end
end

function SignInWindow:IsShowBeginnerTarget()
    if BusinessManager.Instance:SystemIsOpen(CS.StarProjectDef.SystemOpenType.BeginnerTarget) then
        local FriendModule = ModuleManager.Instance:GetModule(ModuleDef.BeginnerTargetModule);
        if FriendModule then
            local luatable = FriendModule:GetLuaTable()
            local BeginnerTargetData = luatable:GetChapterData();
            if BeginnerTargetData and BeginnerTargetData.LastTime then
                if BeginnerTargetData.LastTime == 0 or not BeginnerTargetData.IsEnd then
                    return true
                else
                    local dt = GameManager.Instance:GetServerTimeStamp()/1000 - BeginnerTargetData.LastTime;
                    local t = math.floor((3600*24) - dt);
                    if t > 0 then
                        return true
                    end
                end
            end
        end
    end
    return false
end

	self:CloseWindow();
    self.module.signInWindowView=nil;
    self.m_PartnerTaskPanel:Show(false);
    MonoHelper.RemoveSecTimeUpdateListener(self.TimeTickFunc);
    MonoHelper.RemoveUpdateListener(self.updatefunc);
    if self.sevenDayGoalPanel ~= nil then
        self.sevenDayGoalPanel:OnClose();
    end
end

function SignInWindow:CloseWindow()
	self.IsOpen = false;
    self.module.signInWindowView=nil;
end

function SignInWindow:CalSininday(RewardData)
    local signindays = 0
    for i=1, 30 do
        if (1<<i-1 & RewardData)>0 then 
            signindays = signindays + 1
        end
    end
    return signindays
end

return SignInWindow

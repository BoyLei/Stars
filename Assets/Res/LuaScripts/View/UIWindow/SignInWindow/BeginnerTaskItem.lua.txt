---@class BeginnerTaskItem
local TaskStatus = {
	Rewarded = 0,
	Running = 1,
	Finish = 2,
}
BeginnerTaskItem = class("BeginnerTaskItem", UIPanel)

BeginnerTaskItem.m_IsLock = true;
local ItemGrid = require("View/Item/ItemGrid")
function BeginnerTaskItem:ctor(go, parent, atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.TxtDes = self:Find("textbox/TxtDes"):GetComponent(TText);
	self.TxtProcess = self:Find("num/TxtProcess"):GetComponent(TText);
	self.BtnGet = self:Find("BtnGet"):GetComponent(TButton);
	self.BtnJump = self:Find("BtnJump"):GetComponent(TButton);
	self.ObjFinish = self:Find("ObjFinish").gameObject
	self.items = {}
	for i = 1, 2 do
		local it = ItemGrid:New(self:Find("ObjRewardRoot/ItemGrid" .. i).gameObject)
		it:InitItem(
				false,
				function(ritem)
					self:OnItemClick(ritem)
				end
		)
		table.insert(self.items, it)
	end
	self.BtnGet.onClick:AddListener(function ()
		self:OnGetReward()
	end)
	self.BtnJump.onClick:AddListener(function ()
		self:OnJumpTo()
	end)
end
function BeginnerTaskItem:OnItemClick(ritem)
	LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {ritem.BaseID, false, 0, ritem.transform, 0})
end
function BeginnerTaskItem:OnGetReward()
	if self.TaskData then
		local message = CS.ProtoMsg.SubmitTaskReq();
		local battleSocket = NetworkManager.Instance.gameSocket;
		message.ID =  self.TaskData.TaskID;
		battleSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, message, false);
	end
end
function BeginnerTaskItem:OnJumpTo()
	if self.TaskData then
		local cfg=ConfigDataAccessor.SystemJump[self.TaskData.JumpID];
		if cfg~=nil then
			local fromModule=ModuleDef.OnShowSignInWindow;
			local formArgs={}
			JumpBySystemJumpEnum(cfg.WayString,fromModule,formArgs);
		end
	end
end
--数据更新
function BeginnerTaskItem:UpdateContent(taskData)
	self.TaskData = taskData;
	local configNow = TaskHelper.GetTaskConfig(self.TaskData.TaskID)
	if configNow then
		self.TxtDes.text = configNow.Base.Desc
		local taskTarget = TaskHelper.GetTaskTarget(self.TaskData.TaskID)
		self.BtnGet.gameObject:SetActive(false)
		self.BtnJump.gameObject:SetActive(true)
		self.ObjFinish:SetActive(false)
		local str
		if taskTarget ~= nil then
			if self.TaskData.State == 0 then
				str = taskTarget.MaxCount .. "/" .. taskTarget.MaxCount
			else
				str = taskTarget.CurProgress .. "/" .. taskTarget.MaxCount
			end
		else
			local process  = 0
			for i = 1, configNow.TaskFinishInfo.Targets.Count do
				local target = configNow.TaskFinishInfo.Targets[i-1]
				process = process + target.Target.MaxNum
			end
			str = process .. "/" .. process
		end
		self.TxtProcess.text = str
		if self.TaskData.State == TaskStatus.Rewarded then
			self.BtnGet.gameObject:SetActive(false)
			self.BtnJump.gameObject:SetActive(false)
			self.ObjFinish:SetActive(true)
		elseif self.TaskData.State == TaskStatus.Running then
			self.BtnGet.gameObject:SetActive(false)
			self.BtnJump.gameObject:SetActive(true)
			self.ObjFinish:SetActive(false)
		elseif self.TaskData.State == TaskStatus.Finish then
			self.BtnGet.gameObject:SetActive(true)
			self.BtnJump.gameObject:SetActive(false)
			self.ObjFinish:SetActive(false)
		end
		local Awards=ConfigDataAccessor.GetTaskReward(self.TaskData.TaskID);
		local idx=0;
		if self.items then
			for i = 1, #self.items do
				self.items[i].gameObject:SetActive(false)
			end
			for k,v in pairs(Awards) do
				idx = idx + 1
				if idx <= #self.items then
					self.items[idx].gameObject:SetActive(true)
					self.items[idx]:SetItemNum(k, v);
				end
			end
		end
	end
end

return BeginnerTaskItem;

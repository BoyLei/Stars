---@class TreasureSolt
TreasureSolt = class("TreasureSolt",UIPanel)
TreasureSolt.m_Icon=nil;
TreasureSolt.m_Num=nil;
TreasureSolt.m_Select=nil;

function TreasureSolt:ctor(go,atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_Num=self:Find("Num"):GetComponent(typeof(Text))
	self.m_Icon=self:Find("icon"):GetComponent(typeof(Image))
	self.m_Select=self:Find("Sle").gameObject;
end

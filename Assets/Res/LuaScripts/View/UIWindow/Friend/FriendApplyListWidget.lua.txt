function FriendApplyListWidget:OnItemRefesh(tr, idx)
	if self.ItemFriendApplyList == nil then
		return;
	end
	local data = self.m_ApplyFriends[idx + 1]; --idx是从0开始，如果ItemDataList是lua的table需要idx+1
	local isNew = self:GetIsNewData(data.FromUserPID)
	for i = 1, #self.ItemFriendApplyList do
		if self.ItemFriendApplyList[i].prefab.transform == tr then
			self.ItemFriendApplyList[i]:UpdateContent(data, isNew);
			return;
		end
	end
	local sp = ItemFriendApply.new(tr.gameObject, self.ScrollViewContent, function(pidList)
		self:OnRespondFriendApplyReq(pidList)
	end, function(UserInfo, tr)
		self:OnHeadSelect(UserInfo, tr);
	end);
	sp:UpdateContent(data, isNew);
	table.insert(self.ItemFriendApplyList, sp);
end

-- 获取是否是新数据
function FriendApplyListWidget:GetIsNewData(pid)
	for key, value in pairs(self.m_ApplyFriendsRead) do
		if value ~= nil and value.FromUserPID == pid then
			return false;
		end
	end
	return true;
end

-- 删除申请列表
function FriendApplyListWidget:OnRespondFriendApplyReq(pidList)
	if pidList == nil then
		return;
	end
	local count = #pidList;
	if count > 0 then
		for i = 1, #pidList do
			local pid = pidList[i];
			local index = self:GetIsHasLocalFriendListHasByPID(pid);
			if index ~= -1 then
				table.remove(self.m_ApplyFriends, index);
				-- 删除本地缓存的申请列表
				LuaModuleDispatcher:SendMessage(ModuleDef.FriendModule, "OnDelApplyListByPID", { pid });
			end
		end
		local count = #self.m_ApplyFriends;
		--self.ApplyNum.text = "申请数：" .. count;
		self.ApplyNum.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_FriendApplyListWidget_ApplyNum"), {count});

		self.EmptyTips:SetActive(count <= 0);
		self.ScrollView:ContinueScroll(count, false);

		-- 红点通知
		RedPointManager.Instance:RefreshRedPointCount(RedPointType.FriendRequest, count);
	end
end

-- 获取item
function FriendApplyListWidget:GetItemByPID(PID)
	if self.ItemFriendApplyList ~= nil or self.ItemFriendApplyList ~= {} or #self.ItemFriendApplyList > 0 then
		for i = 1, #self.ItemFriendApplyList do
			local item = self.ItemFriendApplyList[i];
			if item ~= nil and item.FriendInfo ~= nil and item.FriendInfo.FromUserPID == PID then
				return item;
			end
		end
	end

	return nil;
end

-- 全部同意
function FriendApplyListWidget:OnClickAllAgreeJBtn(res)
	local pids = {};
	for j = 1, #self.m_ApplyFriends do
		local friendInfo = self.m_ApplyFriends[j];
		if friendInfo ~= nil then
			table.insert(pids, friendInfo.FromUserPID);
		end
	end
	local key = "FriendApplyListWidget";
	local cb = self.OnRespondFriendApplyRespFun;
	if res == -1 then
		cb = self.OnRespondFriendApplyResp2Fun;
	end
	LuaModuleDispatcher:SendMessage(ModuleDef.FriendModule, "OnSendRespondFriendApply", { pids, res, key, cb });
end

-- 对好友申请响应（同意或拒绝） 回调
function FriendApplyListWidget:OnRespondFriendApplyResp(data, res)
	if data == nil then
		return;
	end

	-- 服务器返回的是申请失败的PID
	-- 过滤成-》只有成功的pid
	local pids = {};
	for i = 1, #self.m_ApplyFriends do
		local friendInfo = self.m_ApplyFriends[i];
		if friendInfo ~= nil then
			local isHave = false;
			for j = 0, data.FailToUserPID.Count - 1 do
				local pid = data.FailToUserPID[j];
				if friendInfo.FromUserPID == pid then
					isHave = true;
				end
			end
			if isHave == false then
				table.insert(pids, friendInfo.FromUserPID);
			end
		end
	end

	if pids == nil or pids == {} or #pids <= 0 then
		return;
	end

	-- 同意的发送系统消息
	if res == 1 then
		for i = 1, #pids do
			local pid = pids[i];
			local userName = "";
			local friendInfo = self:GetFriendListHasByPID(pid);
			if friendInfo ~= nil then
				userName = friendInfo.FromUserName;
				-- 先添加好友缓存
				LuaModuleDispatcher:SendMessage(ModuleDef.FriendModule, "OnAddFriendListByPID", {
					friendInfo.FromUserPID,
					friendInfo.FromUserName,
					friendInfo.FromModelID,
					friendInfo.FromLevel,
					friendInfo.FromGuildName,
					friendInfo.FromJobID,
				});
			end
			-- 发送私聊信息
			local chatMsg = LanguageHelp:GetLanguageByKey("Local_Str_add_friends_tips");
			local chatChannel = CS.ProtoMsg.ChatChannel.ChannelPrivate;
			local extrainfos = nil;
			LuaModuleDispatcher:SendMessage(ModuleDef.ChatModule, "OnSendChatMsg",
				{ chatMsg, chatChannel, extrainfos, pid });
			-- 模拟接收系统消息
			local msgList = {};
			local chatMsgNotice = CS.ProtoMsg.ChatMsgNotice();
			chatMsgNotice.EID = 0;
			chatMsgNotice.PID = 0;
			--chatMsgNotice.Data = "你与<color=#b23738><size=21>" .. userName .. "</size></color>成为好友";
			chatMsgNotice.Data = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_add_friends_tips2"), {userName});
			chatMsgNotice.Channel = CS.ProtoMsg.ChatChannel.ChannelSystem;
			chatMsgNotice.ChatTime = TimeUtils.ServerNowStampMilli;
			chatMsgNotice.ChatLevel = 1;
			chatMsgNotice.JobID = GameManager.Instance:GetPlayerJob();

			table.insert(msgList, chatMsgNotice);
			LuaModuleDispatcher:SendMessage(ModuleDef.ChatModule, "OnAnalogReceptionChatMsg", { msgList });
		end
	end

	self:OnRespondFriendApplyReq(pids);
end

-- 本地缓存的数据是否有包含pid
function FriendApplyListWidget:GetIsHasLocalFriendListHasByPID(pid)
	if self.m_ApplyFriends == nil then
		return -1;
	end
	for i = 1, #self.m_ApplyFriends do
		local friendInfo = self.m_ApplyFriends[i];
		if friendInfo ~= nil and friendInfo.FromUserPID == pid then
			return i;
		end
	end
	return -1;
end

-- 本地缓存的数据是否有包含pid
function FriendApplyListWidget:GetFriendListHasByPID(pid)
	if self.m_ApplyFriends == nil then
		return nil;
	end
	for i = 1, #self.m_ApplyFriends do
		local friendInfo = self.m_ApplyFriends[i];
		if friendInfo ~= nil and friendInfo.FromUserPID == pid then
			return friendInfo;
		end
	end
	return nil;
end

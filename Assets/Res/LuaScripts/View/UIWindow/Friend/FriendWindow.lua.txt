	self.module = module;

end

function FriendWindow:InitTabJbtn()
	self.AttrMenuJBtns = {};

	self.TabBtnClickFun = function(idx)
		self:OnAttrMenuJBtnClick(idx, false);
	end

	local FrameTransForm = self.transform:Find("Cut4Cam90/AssemblyRatio2/bg").transform;
	-- 页签[系统]
	local btn1 = SelectBtn:New(FrameTransForm:Find("MenuBg/MenuContent/TabJBtnFriend").gameObject);
	btn1:InitItem(0, self.TabBtnClickFun);
	table.insert(self.AttrMenuJBtns, btn1);
	-- 页签[当前]
	local btn6 = SelectBtn:New(FrameTransForm:Find("MenuBg/MenuContent/TabJBtnAdd").gameObject);
	btn6:InitItem(1, self.TabBtnClickFun);
	table.insert(self.AttrMenuJBtns, btn6);
	-- 页签[世界]
	local btn1 = SelectBtn:New(FrameTransForm:Find("MenuBg/MenuContent/TabJBtnBlackList").gameObject);
	btn1:InitItem(2, self.TabBtnClickFun);
	table.insert(self.AttrMenuJBtns, btn1);
end

function FriendWindow:InitContent()
	self.PanelList = {};

	local FrameTransForm = self.transform:Find("Cut4Cam90/AssemblyRatio2/bg").transform;
	-- 内容[好友]
	FirendPanelSp = FirendPanel:New(FrameTransForm:Find("PanelRoot/FirendPanel"), 0);
	table.insert(self.PanelList, FirendPanelSp);
	FirendPanelSp:BindParent(self);
	
	-- 内容[添加]
	AddfriendsPanelSp = AddfriendsPanel:New(FrameTransForm:Find("PanelRoot/AddfriendsPanel"), 1);
	table.insert(self.PanelList, AddfriendsPanelSp);
	AddfriendsPanelSp:BindParent(self);
	-- 内容[黑名单]
	BlackListPanelSp = BlackListPanel:New(FrameTransForm:Find("PanelRoot/BlackListPanel"), 2);
	table.insert(self.PanelList, BlackListPanelSp);
	BlackListPanelSp:BindParent(self);
end

-- 判断与该玩家PID的关系
------ 判断是否好友关系和在不在双方黑名单中
-------- 0没有任何关系 1只有好友关系 2在我的黑名单中
function FriendWindow:GetFriendRelation(pid)
	if self.module == nil then
		return 0;
	end
	local res = self.module:GetFriendRelation(pid);
	return res;
end

-- 获取缓存列表有无pid
function FriendWindow:GetIsHasCacheApplyFinishPIDListByPID(pid)
	if self.module == nil then
		return false;
	end
	local res = self.module:GetIsHasCacheApplyFinishPIDListByPID(pid);
	return res;
end

	--todo 切换到最近聊天，加入缓存，显示消息列表

-- 页签点击
function FriendWindow:OnAttrMenuJBtnClick(idx, forceRecent)
	self.CurrentTabIdx = idx;
	for i = 1, #self.AttrMenuJBtns do
		local child = self.AttrMenuJBtns[i];
		local child2 = self.PanelList[i];
		if child ~= nil and child2 ~= nil then
			local isShow = child2.Idx == self.CurrentTabIdx;
			child:MakeSelected(isShow);
			child2:MakeSelected(isShow);
			if isShow then
				self.CurrentContent = child2;
				-- 只有好友列表才会主动刷新
				if forceRecent then
					child2:OnAttrMenuJBtnClick(0);
				end
			end
		end
	end
end

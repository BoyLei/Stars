	--计算装备每个部位的战力
	--背包格子列表
	if self.EquipPower == nil then
		self.EquipPower = {}
	else
		for k, _ in pairs(self.EquipPower) do
			self.EquipPower[k] = nil
		end
	end
	for i = 1, 8 do
		--获取每个部位的当前战力
		local itemData = BusinessManager.Instance:GetHeroEquipedEquip(i)

			table.insert(self.EquipPower, power)
		end
	end

	--默认点个1
	self:OnMenuBtnClick(self.BagMenuIdx, true)
end

--创建一个背包格子
function BagWindow:CreateBagSlot(baseID, entityID, num, state, maxStack)
	--创建格子
	local bagSlot = SlotManager.Instance:CreatSlotToGroup(E_SlotModule.BagModule, 0, 1, state, maxStack) --新建一个格子(正在使用)

	--监听事件
	bagSlot:RegOnUpdateCallBacke(SlotUpdateActionType.SelectAction):AddListener(
		function(id, args)
			self:CleanAllSelected()
		end
	)

	--放入数据
	bagSlot:PutIn(baseID, entityID, num)

	--监听事件
	--[[
	bagSlot:RegUpdateAction(SlotUpdateActionType.SelectAction,function (id,args)
	self:CleanAllSelected();
	end);

	bagSlot.UpdateActionDic[SlotUpdateActionType.SelectAction]=(function (id,args)
	self:CleanAllSelected();
	end);
	]]
	table.insert(self.BagSlotList, bagSlot)
end

function BagWindow:RefeshItems()
	self:SortItem(self.ItemDataList) --排序

	--背包格子列表
	if self.BagSlotList == nil then
		self.BagSlotList = {}
	else
		for k, _ in pairs(self.BagSlotList) do
			self.BagSlotList[k] = nil
		end
	end

	--背包数量限制
	local nowNum = 0

	for i = 1, #self.ItemDataList do
		--创造背包格子(带道具的格子)

		local data = self.ItemDataList[i]

		--处理堆叠问题
		local cfg = ConfigDataAccessor.Item[data.BaseID]
		local canStack = cfg.CanOverlay
		if canStack then
			local maxStack = 999 --配置表没有，暂时是临时的

			local tempNum = data.Num
			while tempNum > maxStack do
				tempNum = tempNum - maxStack

				--self:CreateBagSlot(data.BaseID,data.EntityID,maxStack,1,maxStack)
				local d = {}
				d.BaseID = data.BaseID
				d.EntityID = data.EntityID
				d.Num = maxStack

				table.insert(self.BagSlotList, d)

				--nowNum=nowNum+1;--如果占最大格子数要打开
			end

			--self:CreateBagSlot(data.BaseID,data.EntityID,tempNum,1,maxStack)

			local d2 = {}
			d2.BaseID = data.BaseID
			d2.EntityID = data.EntityID
			d2.Num = tempNum
			table.insert(self.BagSlotList, d2)

			nowNum = nowNum + 1 --暂时是堆叠拆分不占格子数
		else
			nowNum = nowNum + 1 --格子数量

			--self:CreateBagSlot(data.BaseID,data.EntityID,data.Num,1,0)
			local d2 = {}
			d2.BaseID = data.BaseID
			d2.EntityID = data.EntityID
			d2.Num = data.Num
			table.insert(self.BagSlotList, d2)
		end
	end

	--追加到100个空格子
	--for i=#self.ItemDataList+1,self.maxNum do
	for i = #self.ItemDataList + 1, 100 do
		--local it={};
		--it.BaseID=0;
		--table.insert( self.ItemDataList,it);

		--创造背包格子（空格子）
		--self:CreateBagSlot(0,0,0,0,0)

		local d2 = {}
		d2.BaseID = 0
		d2.EntityID = 0
		d2.Num = 0
		table.insert(self.BagSlotList, d2)
	end

	self.BagScorllView:ContinueScroll(#self.BagSlotList, false)

	local str = "" .. nowNum
	if nowNum < 100 and nowNum > 10 then
		str = "0" .. nowNum
	elseif nowNum < 10 then
		str = "00" .. nowNum
	end

	--self.BagLimit.text="<color=#E38830>"..str.."</color>/"..self.maxNum;
	self.BagLimit.text = str .. "/" .. self.maxNum

	--local coins={3,2,1}
	--self.currencyBar:RefeshBar(coins,true);
	--self.currencyBar:PlayAnim(true);

function BagWindow:OnMenuBtnClick(idx, noResetPos)
	if noResetPos == nil then
		self.rect_grid.anchoredPosition = Vector2(self.rect_grid.anchoredPosition.x, 0)
	end

	for i = 1, #self.MenuBtns do
		self.MenuBtns[i]:MakeSelected(false)
	end

	self.MenuBtns[idx]:MakeSelected(true)

	self:ClearItemData()

	self.BagMenuIdx = idx

	--计算最大背包数量
	local numEquip = ConfigDataAccessor.ItemSpace[2].BagNum
	local numCommon = ConfigDataAccessor.ItemSpace[11].BagNum
	local numFriend = ConfigDataAccessor.ItemSpace[11].BagNum
	local numOther = ConfigDataAccessor.ItemSpace[13].BagNum
	local numRuin = ConfigDataAccessor.ItemSpace[14].BagNum --符文
	--local numGem=ConfigDataAccessor.ItemSpace[21].BagNum;--星次石
	local numGem = 0

	--无限列表
	if idx == 1 then
		self.maxNum = numEquip + numCommon + numFriend + numOther + numRuin + numGem

		--全部
		self:GetDataBySpaceId(0)
	elseif idx == 2 then
		self.maxNum = numCommon

		--道具
		self:GetDataBySpaceId(11)
	elseif idx == 3 then
		self.maxNum = numEquip

		--装备
		self:GetDataBySpaceId2(2, 15)
	elseif idx == 4 then
		self.maxNum = numOther

		--伙伴
		self:GetDataBySpaceId3(19, 20, 17)
	elseif idx == 5 then
		self.maxNum = numRuin + numGem

		--星次石，纹章 14
		self:GetDataBySpaceId2(14, 16)
	elseif idx == 6 then
		self.maxNum = numOther

		--其他
		self:GetDataBySpaceId(24)
	end

	self:RefeshItems()
end

function BagWindow:GetBoxType(cfg)
	if cfg.ItemType == 22 then
		if cfg.SubType == 1 then
			return 1
		elseif cfg.SubType == 2 then
			return 2
		end
	end
	return 0
end

function BagWindow:GetCantEquipLv(EntityID, cfg)
	local CantEquipLv = 0

	if cfg.ItemType == 2 then
		local cantEquip = false

		--等级不足无法穿戴
		local lvLimit = ConfigDataAccessor.Equip[cfg.Id].Level
		local playerLv = GameManager.Instance:GetPlayerLevel()
		if lvLimit > playerLv then
			CantEquipLv = 1
		end

		--职业不同无法装备
		if not (cantEquip) then
			local JobType = ConfigDataAccessor.Equip[cfg.Id].JobBaseType
			local playerJob = GameManager.Instance:GetPlayJobBaseID()
			local isMarchPlayerJob = false
			if JobType == 0 or JobType == playerJob then
				isMarchPlayerJob = true
			end

			if not (isMarchPlayerJob) then
				CantEquipLv = 2
			end
		end
	elseif cfg.ItemType == 5 then
		--装备等级
		local lvLimit = ConfigDataAccessor.AmuletEquip[cfg.Id].Level
		local playerLv = GameManager.Instance:GetPlayerLevel()
		if lvLimit > playerLv then
			CantEquipLv = 1
		end
	end

	return CantEquipLv
end

function BagWindow:GetLevel(BaseID, ItemType)
	if ItemType == 2 then
		--装备
		if ConfigDataAccessor.Equip[BaseID] ~= nil then
			return ConfigDataAccessor.Equip[BaseID].Level
		end
	elseif ItemType == 5 then
		--纹章
		if ConfigDataAccessor.HeraldryEquip[BaseID] ~= nil then
			return ConfigDataAccessor.HeraldryEquip[BaseID].Level
		end
	elseif ItemType == 6 then
		--星次石
		if ConfigDataAccessor.HeraldryEquip[BaseID] ~= nil then
			return ConfigDataAccessor.HeraldryEquip[BaseID].HeraldryLevel
		end
	end

	return 0
end

function BagWindow:GetDataBySpaceId3(spaceID1, spaceID2, spaceID3)
	for i = 0, self.allitems.Count - 1 do
		if self.allitems[i].SpaceId == spaceID1 or self.allitems[i].SpaceId == spaceID2 or self.allitems[i].SpaceId == spaceID3 then
			--local cfg=LocalDataManager.Instance:GetItemDataCell(self.allitems[i].BaseID);
			local cfg = ConfigDataAccessor.Item[self.allitems[i].BaseID]
			local it = {}

			it.BoxType = self:GetBoxType(cfg)

			it.EntityID = self.allitems[i].EntityID
			it.BaseID = self.allitems[i].BaseID
			it.Num = self.allitems[i].Num
			it.IsNew = self.allitems[i].IsNew
			it.SortProp = cfg.SortProp
			it.ItemType = cfg.ItemType
			it.SubType = cfg.SubType
			it.Quality = cfg.Quality
			it.CantEquipLv = self:GetCantEquipLv(it.EntityID, cfg)
			it.Level = self:GetLevel(it.BaseID, it.ItemType)
			table.insert(self.ItemDataList, it)
		end
	end
end

--判断是否出up和new
function BagWindow:CheckCanUp(data)
	local entityID = data.EntityID
	local item = ConfigDataAccessor.Item[data.BaseID]

	if item == nil then
		return false, false
	end

	local itemType = item.ItemType

	local canUp = false
	local canNew = false

	if itemType == 2 then
		--判断是否满足等级，职业
		local canEquip = BusinessManager.Instance:CheckIsEquipCanEquiped(data.BaseID)
		if not (canEquip) then
			return false, false
		end

		local subType = item.SubType
		local power = self.EquipPower[subType]

		local itemData = BusinessManager.Instance:GetItemByItemEntityId(entityID)
		local powerNow = FightPowerManager.Instance:GetShowEquipPower(itemData)

		canUp = powerNow > power

		if power == 0 then
			canNew = true
		end
	end

	return canUp, canNew
end

function BagWindow:OnBagItemRefesh(tr, idx)
	for i = 1, #self.BagItemList do
		if self.BagItemList[i].transform == tr then
			self.BagItemList[i]:SetItem(self.BagSlotList[idx + 1], true) --idx是从0开始，如果BagSlotList是lua的table需要idx+1
			local canUp, canNew = self:CheckCanUp(self.BagSlotList[idx + 1])
			self.BagItemList[i]:SetUp(canUp)
			--self.BagItemList[i]:SetNew(canNew);
			return
		end
	end

	--tr.gameObject:SetActive(true);
	local bi = ItemGrid:New(tr.gameObject, self.tr_ItemRoot)
	bi:InitItem(
		true,
		function(ritem)
			self:OnItemClick(ritem)
		end
	)
	bi.gameObject:SetActive(true)
	bi:SetItem(self.BagSlotList[idx + 1], true) --idx是从0开始，如果BagSlotList是lua的table需要idx+1
	local canUp, canNew = self:CheckCanUp(self.BagSlotList[idx + 1])
	bi:SetUp(canUp)
	--bi:SetNew(canNew);
	table.insert(self.BagItemList, bi)
end

--[[
function BagWindow:SortItem(tbl)

if #tbl > 1 then

table.sort(tbl, function(a, b)


if a.SortProp < b.SortProp then
-- 按SortProp升序
return true
elseif a.SortProp == b.SortProp then
-- 如果SortProp相等，按Type升序
if a.ItemType < b.ItemType then
return true
elseif a.ItemType == b.ItemType then
-- 如果Type相等，按Id升序
return a.BaseID < b.BaseID
end
end
return false


end)

end

end
]]
function BagWindow:SortItem(tbl)
	if #tbl > 1 then
		table.sort(
			tbl,
			function(a, b)
				--宝箱排序
				if a.BoxType ~= b.BoxType then
					return a.BoxType > b.BoxType
				end

				if a.Quality ~= b.Quality then
					return a.Quality > b.Quality
				end

				if a.ItemType ~= b.ItemType then
					return a.ItemType < b.ItemType
				end

				if a.CantEquipLv ~= b.CantEquipLv then
					return a.CantEquipLv < b.CantEquipLv
				end

				if a.SubType ~= b.SubType then
					return a.SubType < b.SubType
				end

				if a.Level ~= b.Level then
					return a.Level > b.Level
				end

				return false
			end
		)
	end
end

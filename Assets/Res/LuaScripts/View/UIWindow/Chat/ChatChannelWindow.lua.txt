function ChatChannelWindow:OnAttrMenuJBtnClick(channel)
    self.CurrentChannel = channel
    for i = 1, #self.AttrMenuJBtns do
        local child = self.AttrMenuJBtns[i]
        local child2 = self.ContentList[i]
        if child ~= nil and child2 ~= nil then
            local isShow = child2.Channel == self.CurrentChannel
            child:MakeSelected(isShow)
            child2:MakeSelected(isShow)
            if isShow then
                self.CurrentContent = child2
            end
        end
    end

    -- local isShowInput = self.CurrentChannel == CS.ProtoMsg.ChatChannel.ChannelSystem or self.CurrentChannel == CS.ProtoMsg.ChatChannel.ChannelCall or self.CurrentChannel == CS.ProtoMsg.ChatChannel.ChannelPrivate;
    local isShowInput = true
    if self.CurrentContent ~= nil then
        if self.CurrentContent.chatCfg ~= nil then
            isShowInput = self.CurrentContent.chatCfg.AllowInput
        end
    end

    self:CheckCurChannelIsCanSend()
    self.Bottom:SetActive(isShowInput)
end

-- 检查频道是否可以发送
function ChatChannelWindow:CheckCurChannelIsCanSend()
    local isCanSend = true
    if self.CurrentChannel == CS.ProtoMsg.ChatChannel.ChannelUnion then
        -- 如果是冒险团频道，检查是否有冒险团
        isCanSend = BusinessManager.Instance:GetGuildID() ~= 0
    elseif self.CurrentChannel == CS.ProtoMsg.ChatChannel.ChannelTeam then
        -- 如果是队伍频道，检查是否有队伍
        local module = ModuleManager.Instance:GetModule(ModuleDef.TeamModule)
        local luatable = module:GetLuaTable()
        local teamInfo = luatable:GetTeamInfo()
        local hasTeam = false -- 是否有队伍
        if teamInfo ~= nil and teamInfo.TeamID ~= 0 and teamInfo.UserInfos.Count > 0 then
            hasTeam = true
        end
        isCanSend = hasTeam
    end
    self.sendBtn.UnClick = (not isCanSend)
end

-- 判断是否被冒险团禁言
function ChatChannelWindow:CheckIsSilenceInGuild()
    if self.CurrentChannel == CS.ProtoMsg.ChatChannel.ChannelUnion then
        local serverTime = GameManager.Instance:GetServerTimeStamp() / 1000
        local GuildProhibEndTime = BusinessManager.Instance:GetGuildProhibEndTime()
        local leftTime = GuildProhibEndTime - serverTime
        if leftTime > 0 then
            local timeStr = SecToStringColon2(leftTime)
            -- Util.ShowMessage("您已经被禁言， " .. timeStr .. " 后才可在冒险团发言");
            Util.ShowMessage(StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_IsSilenceInGuild"), {timeStr}))
            return true
        end
    end
    return false
end

-- 发送消息
function ChatChannelWindow:SendCurChatChannelData()
    if self.inputField.text == "" then
        return
    end
    -- 判断冒险团能否发送消息
    if self:CheckIsSilenceInGuild() then
        return
    end
    -- 检查频道是否能发送消息
    local isSend = self:CheckCurChannelIsSend()

    if isSend then
        self.inputField:SendData(self.CurrentChannel)
    end
end

-- 发送【输入框中的】消息
function ChatChannelWindow:SendDataByChatInput(ip)
    if ip == nil then
        return
    end

    -- 判断冒险团能否发送消息
    if self:CheckIsSilenceInGuild() then
        return
    end
    -- 检查频道是否能发送消息
    local isSend = self:CheckCurChannelIsSend()

    if isSend then
        local pid = -1;
        if self.module ~= nil then
            pid = self.module.selectPlayerId;
        end
        ip:SendData(self.CurrentChannel, pid)
    --self.inputField:Clear();
    end
end

-- 检查频道是否能发送消息
function ChatChannelWindow:CheckCurChannelIsSend()
    local isSend = true
    if self.CurrentContent ~= nil then
        isSend = self.CurrentContent:GetIsSend()
    end
    return isSend
end

-- 清空当前输入框信息
function ChatChannelWindow:ClearInput(lastTime)
    self.inputField:Clear()
    if self.CurrentContent ~= nil then
        self.CurrentContent:UpdateEndTime(lastTime)
    end
end

function ChatChannelWindow:SetShow(isShow, isPlayAnim)
    self.IsShow = isShow

    local alpha = 0
    if self.IsShow == true then
        alpha = 1
    end

    if isPlayAnim then
        if isShow then
            self:CheckCurChannelIsCanSend()
            self.PlayAnimation(
                "Open",
                function()
                    self.PanelCanvasGroup.alpha = alpha
                end
            )
        else
            self.PlayAnimation(
                "Out",
                function()
                    self.PanelCanvasGroup.alpha = alpha
                end
            )
        end
    else
        self.PanelCanvasGroup.alpha = alpha
    end

    self.PanelCanvasGroup.blocksRaycasts = isShow

    self.transform:SetAsLastSibling()

    UIManager.Instance:SetShowChatWindow(self.IsShow)
end

function ChatChannelWindow:AddMsgItem(chatMsgNotice)
    if ExtraData.IsMsgType(chatMsgNotice, CS.ProtoMsg.ChatExtraInfo.Types.EMsg.TeamGather) then
        return
    end

    for i = 1, #self.ContentList do
        self.ContentList[i]:AddItem(chatMsgNotice)
    end
end

function ChatChannelWindow:ClearAllChatItem()
    for i = 1, #self.ContentList do
        self.ContentList[i]:ClearAllItem()
    end
end

function ChatChannelWindow:SyncInput(ip)
    self.inputField:Sync(ip)
end

function ChatChannelWindow:SetBarrageChannelState(channel, isOpen)

        self.module:SetBarrageChannelState(channel, isOpen)
    end
end

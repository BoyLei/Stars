function ChatHudWidget:SetOpen(isOpen)
    self.IsOpen = isOpen;
    self.Transform.gameObject:SetActive(isOpen);
	print(string.format("HUD聊天打开:%s", isOpen))
end

function ChatHudWidget:SetShow(isShow)
    self.ScrollViewExtend:SetShow(isShow);
    self.IsShow = isShow;
	print(string.format("HUD聊天显隐:%s", isShow))
end

function ChatHudWidget:AddMsgs(CMNoticesList)
    self:SetShow(true);

    for i = 0, CMNoticesList.Count - 1 do
        local data = CMNoticesList[i];
        self:AddMsgItem(data);
    end
end

function ChatHudWidget:AddMsgItem(data)
    -- 过滤组队
    if not ExtraData.IsMsgType(data, CS.ProtoMsg.ChatExtraInfo.Types.EMsg.TeamGather) then
        -- 过滤好友
        local Channel = data.Channel;
        if Channel ~= CS.ProtoMsg.ChatChannel.ChannelPrivate then
            self:RemoveItemList();
            local item = self:GetItem();
            item:SetItem(data);
        end
    end
end

function ChatHudWidget:GetItem()
    local gob = newObject(self.HudChat);
    gob.transform:SetParent(self.Content);
    gob.transform.localPosition = Vector3.zero;
    gob.transform.localScale = Vector3.one;
    local item = ItemHudChat:New(gob);
    table.insert(self.ItemList, item);
    return item;
end

function ChatHudWidget:RemoveItemList()
    local count = #self.ItemList;
    if count >= 100 then
        table.remove(self.ItemList, 1);
        GameObject.Destroy(self.ItemList[1].GameObject);
    end
end

function ChatHudWidget:ClearItemByType(channel)
    local count = #self.ItemList;
    for i = count, 1, -1 do
        local child = self.ItemList[i];
        if child ~= nil then
            if child.Channel ~= nil then
                if child.Channel == channel then
                    -- 删除对应的item
                    GameObject.Destroy(self.ItemList[i].GameObject);
                end
            end
        end
    end
end

function ChatHudWidget:ClearAllChatItem()
    for i = 1, #self.ItemList do
		GameObject.Destroy(self.ItemList[i].GameObject);
	end
	self.ItemList = {};
end

return ChatHudWidget;

---@class ItemChatTime

local ItemChatTime = {}
ItemChatTime.__index = ItemChatTime

function ItemChatTime:New(obj)
	local self = {};
	setmetatable(self, ItemChatTime);

	self:InitControls(obj)

	return self;
end

function ItemChatTime:InitControls(obj)
	self.GameObject = obj;
	self.Transform = obj.transform;

	--聊天内容
	self.TimeText = self.Transform:Find("TimeText"):GetComponent(typeof(Text));

end

function ItemChatTime:SetItem(time)
	
	self.TimeText.text = TimeUtils.FormatShowTimeHM(time,":");

end

return ItemChatTime

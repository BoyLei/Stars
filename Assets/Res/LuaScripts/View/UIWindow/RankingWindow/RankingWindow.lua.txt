function RankingWindow:GetCurJob(jobId)
	local jobCfg = ConfigDataAccessor.Job;
	for k,v in pairs(jobCfg) do
		if v.BaseJob == jobId and v.ID%10 == 0 then
			return v;
		end
	end
	return nil;
end

function RankingWindow:OnItemRefesh(tr, idx)
    local rankInfo = self.rankInfos[idx + 1];

    for i = 1, #self.rankItemList, 1 do
        if self.rankItemList[i].transform == tr then
            self.rankItemList[i]:Refresh(rankInfo, self.curSelIndex, self.curSelRankType);
            return;
        end
    end

    local rankingItem = RankingItem:New(tr.gameObject);
    rankingItem:SetClickCb(function(id)
        self:OnClickRankingItem(id);
    end);

    tr.gameObject:SetActive(true);
    rankingItem:Refresh(rankInfo, self.curSelIndex, self.curSelRankType);

    table.insert(self.rankItemList, rankingItem);
end

function RankingWindow:OnClickRankingItem(id)
	--刷新选中
	--刷新左边栏
	self.curSelIndex = id;
	for _,v in pairs(self.rankItemList) do
		v:SetSelect(v.curData.RankNum == self.curSelIndex);
	end

	self:SelectItem(self.rankInfos[id]);
end

function RankingWindow:InitData(data)
	if data == nil then
		return;
	end
	
	-- self.myNameText
	-- self.percentText
	-- self.nameText
	-- self.statusText
	--刷新所有排行列表
	
	self.curSelIndex = 1;

	self.rankInfos = {};
	local ranks = data.PlayerRanks;
	for i=0, ranks.Count-1 do
		--ranks[i].RankNum
		--ranks[i].RankScore
		--ranks[i].OfflineData

		self.rankInfos[i+1] = ranks[i];
	end
	
	self.RankScorllView:ContinueScroll(#self.rankInfos, false, 0);
	
	--self.powerRankObj:SetActive(self.curSelRankType ~= 6);
	--self.towerRankObj:SetActive(self.curSelRankType == 6);

	self.titlePowerObj:SetActive(self.curSelRankType ~= 6);
	self.titleTowerObj:SetActive(self.curSelRankType == 6);
	
	if self.curSelRankType == 6 or self.curSelRankType == 1 then
		self.jobTypesGo:SetActive(false);
		self.jobSelBtn.gameObject:SetActive(false);
	else
		self:SelectItem(self.rankInfos[1]);
		--self.jobTypesGo:SetActive(true);
		self.jobSelBtn.gameObject:SetActive(true);
	end
	self.loopIndex = 1;
	self:StartLoop(true);
	
	if #self.rankInfos > 0 then
		self.selNameText.text = self.rankInfos[self.curSelIndex].OfflineData.UserName;
	end

	local str = "--";
	if data.MyRankNum <= 100 and data.MyRankNum>0 then
		str = tostring(data.MyRankNum);
	elseif data.MyRankNum <= data.TotalNum and data.MyRankNum>0 then
		local per = math.floor(data.MyRankNum / data.TotalNum * 100);
		str = string.format("%d", per).."%";
	else
	end
	self.percentText.text = str;
	self.nameText.text = GameManager.Instance:GetPlayerName();
	
	self.rewardBtn.gameObject:SetActive(data.MyAwardStatus == 1);

	--local selfBattleScore=GameManager.Instance:GetAllPower(0, false);
	if data.MyRankScore == 0 then
		self.statusText.text = "-"
	else
		if self.curSelRankType == 6 then
			self.statusText.text = tostring(data.MyRankScore);
		else
			self.statusText.text = GetTotalPowerRankLevel(data.MyRankScore);
		end
	end
	local rcl =ConfigDataAccessor.RankCommonList;
	for k,v in pairs(rcl) do
		if k == self.curSelRankType then
			local strType = "";
			if v.RefreshType == 1 then
				strType = LanguageHelp:GetLanguageByKey("Local_Str_Time_second");
			elseif v.RefreshType == 2 then
				strType = LanguageHelp:GetLanguageByKey("Local_Str_Time_minute");
			elseif v.RefreshType == 3 then
				strType = LanguageHelp:GetLanguageByKey("Local_Str_Time_hour");
			elseif v.RefreshType == 4 then
				strType = LanguageHelp:GetLanguageByKey("Local_Str_Time_day");
			elseif v.RefreshType == 5 then
				strType = LanguageHelp:GetLanguageByKey("Local_Str_Time_week");
			end

			local timeStamp=GetTimeStamp(data.UpdateTime);
			self.descText.text = StringFormat(LanguageHelp:GetLanguageByKey("RankWindow_12"),{v.RefreshParam, strType,timeStamp}  );
			--self.descText.text = string.format("每%d%s刷新；数据截止至：%s", v.RefreshParam, strType, GetTimeStamp(data.UpdateTime));
			break;
		end
	end
end

function RankingWindow:SelectItem(data)
	--if data == nil then
		--return;
	--end
	
	if #self.rankInfos > 0 then
		self.selNameText.text = self.rankInfos[self.curSelIndex].OfflineData.UserName;
	else
		self.selNameText.text = "";
	end
	
	--local cfg = ConfigDataAccessor.Partner[curID];
	self.rawImg.texture = ResourceManager.Instance.RenderTexture;

	if self.jobModel ~= nil then
		BusinessManager.Instance:CloseModel(1, self.jobModel);
		self.jobModel = nil;
	end
	if self.partnerModel ~= nil then
		BusinessManager.Instance:CloseModel(2, self.partnerModel);
		self.partnerModel = nil;
	end
	
	if #self.rankInfos == 0 then
		return;
	end
	--显示职业模型
	self.jobModel = BusinessManager.Instance:ShowModel(1, ConfigDataAccessor.Job[data.OfflineData.JobID].AvatarID, Vector3(0.176,-0.425,0.01));
	--显示伙伴模型
	self.partnerModel = BusinessManager.Instance:ShowModel(2, 90101, Vector3(1.412,-0.641,0.714));

end

function RankingWindow:StartLoop(isBegin)
	self.rawImg.gameObject.transform.localScale = Vector3.one
	if self.tw_delay~= nil then
		self.tw_delay:Kill();
		self.tw_delay = nil;
	end

	local delay = ConfigDataAccessor.SystemList[236].Value/1000;
	if isBegin then
		delay = 0;
	end
	self.tw_delay = DOVirtual.DelayedCall(delay, function()
		self:OnClickRankingItem(self.loopIndex);
		self.loopIndex = self.loopIndex + 1;
		if self.loopIndex > 3 or self.loopIndex > #self.rankInfos then
			self.loopIndex = 1;
		end

		self:StartLoop(false);
	end);
end

function RankingWindow:StopLoop()
	if self.tw_delay~= nil then
		self.tw_delay:Kill();
		self.tw_delay = nil;
	end
end

function RankingWindow:OnOpen(arg)
	if not self.IsOpen then
		UIManager.Instance:SetRawCamrea(true);
	end
	self.IsOpen = true;
end

function RankingWindow:OnClose(arg)
	self:CloseWindow();
	UIManager.Instance:SetRawCamrea(false);
	self:StopLoop();
	self:CloseModel()

	self.module.rankWindowView = nil
end

function RankingWindow:CloseModel()
	self.rawImg.gameObject.transform.localScale = Vector3.zero
	if self.tw_delay~= nil then
		self.tw_delay:Kill();
		self.tw_delay = nil;
	end
	if self.jobModel ~= nil then
		BusinessManager.Instance:CloseModel(1, self.jobModel);
		self.jobModel = nil;
	end
	if self.partnerModel ~= nil then
		BusinessManager.Instance:CloseModel(2, self.partnerModel);
		self.partnerModel = nil;
	end
end
function RankingWindow:CloseWindow()
	self.IsOpen = false;
	self.module.rankWindowView = nil
end

return RankingWindow

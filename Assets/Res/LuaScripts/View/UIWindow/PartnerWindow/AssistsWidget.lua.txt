function AssistsWidget:CleanAllSelected()
	--无限列表
	for i=1,#self.PartnerItems do
		self.PartnerItems[i]:OnSelect(false);
	end
end

function AssistsWidget:OnSelect(selectIndex)
	self.SelectIndex=selectIndex;
	self:CleanAllSelected();
	self:SetPartnerInfo();
end

function AssistsWidget:OnItemRefesh(tr,idx)
	if self.PartnerItems==nil then
		return;
	end
	for i=1,#self.PartnerItems do
		if self.PartnerItems[i].transform == tr then
			self.PartnerItems[i]:UpdateContent(self.Partners[idx+1],self.SelectIndex);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
			return
		end
	end
	local bi = PartnerAssistItem.new(tr.gameObject,self.m_ItemRoot,self.AtlasPath,function(index)
			self:OnSelect(index)
		end,function (data)
			return  self:CompareAttr(data)
		end);
	bi.prefab:SetActive(true);
	bi:UpdateContent(self.Partners[idx+1],self.SelectIndex);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
	table.insert(self.PartnerItems, bi);
end

function AssistsWidget:Sort()
	-- 状态权重映射表
	local stateWeights = {
		[CS.ProtoMsg.PartnerState.InBattle] = 1,
		[CS.ProtoMsg.PartnerState.Concretization] = 1,
		[CS.ProtoMsg.PartnerState.Assist] = 2
	}

	-- 辅助函数，获取伙伴的质量，如果伙伴信息不存在则返回0
	local function getQuality(partnerIndex)
		local config = ConfigDataAccessor.Partner[partnerIndex]
		return config and config.Quality or 0
	end

	-- 排序函数
	table.sort(self.Partners, function(p1, p2)
			-- 获取状态权重，如果状态未知则默认为0
			local t1 = stateWeights[p1.data.State] or 0
			local t2 = stateWeights[p2.data.State] or 0

			-- 如果状态权重不同，根据权重排序
			if t1 > t2 then return true end
			if t1 < t2 then return false end

			-- 状态权重相同，比较等级
			local lv1 = p1.Level or 0
			local lv2 = p2.Level or 0

			if lv1 > lv2 then return true end
			if lv1 < lv2 then return false end

			-- 等级相同，比较质量
			local quality1 = getQuality(p1.data.Index)
			local quality2 = getQuality(p2.data.Index)

			return quality1 > quality2
		end)
end

---比较属性
function AssistsWidget:CompareAttr(data)
	local list={}
	if self.Slot_Qualifications~=nil and #self.Slot_Qualifications>0 then
		for i = 1, #self.Slot_Qualifications do
			if self.SlotPartner ==nil then
				local t= data.PartnerData.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue
				if t>0 then
					table.insert(list,{id=self.Slot_Qualifications[i],isup=1})
				else
					table.insert(list,{id=self.Slot_Qualifications[i],isup=0})
				end

			else
				if data.Index== self.SlotPartner.Index then
					table.insert(list,{id=self.Slot_Qualifications[i],isup=0})
				else
					local isup=	data.PartnerData.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue>self.SlotPartner.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue;
					if isup then
						table.insert(list,{id=self.Slot_Qualifications[i],isup=1})
					else
						if data.PartnerData.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue==self.SlotPartner.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue then
							table.insert(list,{id=self.Slot_Qualifications[i],isup=0})
						else
							table.insert(list,{id=self.Slot_Qualifications[i],isup=-1})
						end
					end
				end
			end
		end
	end
	return list;
end


function AssistsWidget:GetSelectPartner()
	if self.Partners==nil then
		return nil
	end

	for _k,_v in pairs(self.Partners) do
		if _v.data.Index== self.SelectIndex then
			return _v.data
		end
	end
end

function AssistsWidget:SetAtrribuite()
	local playerJob=GameManager.Instance:GetPlayJobBaseID();

	self.m_CurDesc.text=""
	self.m_NextDesc.text=""
	for i = 1, #self.CurrentAttrs do
		self.CurrentAttrs[i].gameObject:SetActive(false)
	end

	if self.SlotPartner==nil then
		for i = 1, #self.Slot_Qualifications do
			if i<3 then
				local objz=self.CurrentAttrs[i].transform:Find("Text1"):GetComponent(typeof(Text));
				local obja=self.CurrentAttrs[i].transform:Find("Text2"):GetComponent(typeof(Text));
				local AoiAttr= ConfigDataAccessor.ParConversion[self.Slot_Qualifications[i]]
				if AoiAttr~=nil then
					objz.text=AoiAttr.Name..":0"
				end

				local config1=  ConfigDataAccessor.GetAttrByID(self.Slot_Nature[i]);
				local attrName1 = "";
				if config1 ~= nil then
					attrName1 = config1.Name;
				end
				obja.text=attrName1..":0"


				self.CurrentAttrs[i].gameObject:SetActive(true)

			end
		end
	else
		local SlotPartnerAptitudes={}
		for i = 1, #self.Slot_Qualifications do
			table.insert(SlotPartnerAptitudes,self.SlotPartner.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue)
		end


		local info= ConfigDataAccessor.GetPartnerAssitConfig(playerJob,self.Index,SlotPartnerAptitudes);
		if info==nil then
			return
		end

		for i = 1, #self.Slot_Qualifications do
			if i<3 then
				local objz=self.CurrentAttrs[i].transform:Find("Text1"):GetComponent(typeof(Text));
				local obja=self.CurrentAttrs[i].transform:Find("Text2"):GetComponent(typeof(Text));

				local AoiAttr= ConfigDataAccessor.ParConversion[self.Slot_Qualifications[i]]
				if AoiAttr~=nil then
					objz.text=AoiAttr.Name..":"..self.SlotPartner.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue.."<color=#00ff00>".."(+"..(info.Ratio[i]/100).."%)".."</color>"
				end

				local config1=  ConfigDataAccessor.GetAttrByID(self.Slot_Nature[i]);
				local attrName1 = "";
				if config1 ~= nil then
					attrName1 = config1.Name;
				end
				local va=self.SlotPartner.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue*info.Ratio[i]*0.0001
				obja.text=attrName1..":"..math.floor(va).."<color=#00ff00>".. "(+"..(info.Ratio[i]/100).."%)".."</color>"



				self.CurrentAttrs[i].gameObject:SetActive(true)

			end
		end
	end

	---替换后
	for i = 1, #self.ReplacetAttrs do
		self.ReplacetAttrs[i].gameObject:SetActive(false)
	end
	local data= self:GetSelectPartner();
	if data==nil then
		return
	end
	local DataPartnerAptitudes={}
	for i = 1, #self.Slot_Qualifications do
		table.insert(DataPartnerAptitudes,data.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue)
	end

	local info= ConfigDataAccessor.GetPartnerAssitConfig(playerJob,self.Index,DataPartnerAptitudes);
	if info==nil then
		return
	end


	self.m_CurDesc.text=info.Desc

	for i = 1, #self.Slot_Qualifications do
		if i<3 then
			local objz=self.ReplacetAttrs[i].transform:Find("Text1"):GetComponent(typeof(Text));
			local obja=self.ReplacetAttrs[i].transform:Find("Text2"):GetComponent(typeof(Text));
			local AoiAttr= ConfigDataAccessor.ParConversion[self.Slot_Qualifications[i]]

			if AoiAttr~=nil then
				objz.text=AoiAttr.Name..":"..data.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue.."<color=#00ff00>".."(+"..(info.Ratio[i]/100).."%)".."</color>"
			end

			local config1=  ConfigDataAccessor.GetAttrByID(self.Slot_Nature[i]);
			local attrName1 = "";
			if config1 ~= nil then
				attrName1 = config1.Name;
			end
			local showAtr=data.AMD.Aptitudes[self.Slot_Qualifications[i]].BaseValue*info.Ratio[i]*0.0001;
			obja.text=attrName1..":"..math.floor(showAtr).."<color=#00ff00>".."(+"..(info.Ratio[i]/100).."%)".."</color>"


			self.ReplacetAttrs[i].gameObject:SetActive(true)
		end
	end
	local next= ConfigDataAccessor.GetPartnerAssitConfigByID(playerJob,self.Index,info.Id+1);
	if next~=nil then
		self.m_NextDesc.text=next.Desc
	end
end

function AssistsWidget:GetQualityName(quality)
	if quality== 1 then
		return "Partner_quality_n"
	elseif quality== 2 then
		return "Partner_quality_r"
	elseif quality== 3 then
		return "Partner_quality_sr"
	elseif quality== 4 then
		return "Partner_quality_ssr"
	end
end

function AssistsWidget:GetStateName(state,id)
	local result=""

	if self.SlotPartner==nil then
		if state== CS.ProtoMsg.PartnerState.Assist  then
			result= LanguageHelp:GetLanguageByKey("Local_Str_alreadyGoIntoBattle")
		else
			---点击空各自进入
			result= LanguageHelp:GetLanguageByKey("Local_Str_assist_in_fighting")
		end


	else
		--闲置 ，助战
		if state~= CS.ProtoMsg.PartnerState.Assist  then
			result= LanguageHelp:GetLanguageByKey("Local_Str_Replace")
		else
			result= LanguageHelp:GetLanguageByKey("Local_Str_alreadyGoIntoBattle")
			if self.SlotPartner.Index==id then
				result= LanguageHelp:GetLanguageByKey("Local_Str_Downarray")
			end

		end

	end
	return result;

end

function AssistsWidget:OnClickStatBtn()

	local data= self:GetSelectPartner();
	if data==nil then
		return
	end
	local Index=0;
	if data.State==CS.ProtoMsg.PartnerState.InBattle or data.State==CS.ProtoMsg.PartnerState.Concretization then
		Index=11
	end

	if Index==0 then
		self:DoAction(data);
	else
		UIAPI.ShowMsgBox(Index,function (EventName)
				if EventName=="SURE" then
					self:DoAction(data);
				end
			end)
	end

end

function AssistsWidget:DoAction(data)

	if self.SlotPartner==nil then
		---上阵
		LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule, "OnPartnerAssistReq", {data.Index,self.Index,data.CurSeat});

	else
		if self.SlotPartner.Index==data.Index then
			-- LanguageHelp:GetLanguageByKey("Local_Str_Downarray")
			local config=ConfigDataAccessor.Partner[data.Index];
			if config~=nil then
				--local text=	config.Name.."取消助战"
				local text=	StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_271"), {config.Name})

				Util.ShowMessage(text);

			end

			LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule, "PartnerFallReq", data.Index);
		else
			if data.state ~= CS.ProtoMsg.PartnerState.Assist  then
				LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule, "OnPartnerAssistReq", {data.Index,self.Index,data.CurSeat});
			end
		end
	end
end

function PartnerDetailWidget:OnOpenPartnerEquip(equipType, istakeoff)
	self.curEquipType = equipType;

	if self.partnerEquipTips == nil then
		self.partnerEquipTips = PartnerEquipTips:ctor(self.partnerEquipTipsGO);
	end
	if self.partnerEquipItems == nil then
		self.partnerEquipItems = PartnerEquipItems:ctor(self.equipItemsGO);
	end

	local data = self:GetSelectPartner();
	if data == nil then
		return
	end

	local equipedEquip = BusinessManager.Instance:GetPartnerEquipedEquip(data.ID, equipType);

	self.equipGO:SetActive(true);
	if equipedEquip == nil then
		self.partnerEquipTips:Hide();
	else
		self.partnerEquipTips:Show(equipedEquip.BaseID, equipedEquip.BaseID, equipedEquip.EntityID, data.ID);
	end
	self.partnerEquipItems:Show(equipType, self, equipedEquip, istakeoff);

	self.m_PartnerRawImage.transform.parent = self.equipGO.transform;
	self.equipTypesGO.transform.parent = self.equipGO.transform;
end

function PartnerDetailWidget:OnTakeOff()
	if self.partnerEquipTips ~= nil then
		self.partnerEquipTips:OnTakeOff();
		self:InitEquip();
	end
end

function PartnerDetailWidget:RefreshEquips(istakeoff)
	self:OnOpenPartnerEquip(self.curEquipType, istakeoff)
	self:InitEquip();
end

function PartnerDetailWidget:OnClosePartnerEquip()
	self.equipGO:SetActive(false);
	for j = 1, 3 do
		self.equipItems[j].selfGo:SetActive(false);
	end

	self.m_PartnerRawImage.transform.parent = self.m_leftTrans;
	self.equipTypesGO.transform.parent = self.m_leftTrans;
end

function PartnerDetailWidget:InitEquip()
	local data = self:GetSelectPartner();
	for i = 1, 3 do
		self.equipItems[i].icon.gameObject:SetActive(false);
		for j = 1, 3 do
			self.equipItems[i].stars[j]:SetActive(false);
		end
		self.equipItems[i].bgRare.gameObject:SetActive(false);
	end

	if data == nil then
		return
	end
	for i = 1, 3 do
		local equipedEquip = BusinessManager.Instance:GetPartnerEquipedEquip(data.ID, i);

		if equipedEquip ~= nil then
			self.equipItems[i].icon.gameObject:SetActive(true);

			local baseID = equipedEquip.BaseID;

			local cfg2 = ConfigDataAccessor.Item[baseID];
			self.equipItems[i].bgRare.gameObject:SetActive(true);
			ResourceHelper:LoadIconByAtlasAsync(self.equipItems[i].bgRare, ResourceHelper.AtlasPathCommonItem,
				"Bag_Grid_sq" .. cfg2.Quality, false)

			local itemData = ConfigDataAccessor.Item[equipedEquip.BaseID]
			ResourceHelper:LoadIconAsync(self.equipItems[i].icon, "", itemData.Icon, false)

			local partnerEquip = ConfigDataAccessor.PartnerEquip;
			if partnerEquip[baseID] ~= nil then
				for j = 1, 3 do
					self.equipItems[i].stars[j]:SetActive(j <= partnerEquip[baseID].AdvancedLevel);
				end
			end
		end
	end
end

function PartnerDetailWidget:OnItemRefesh(tr, idx)
	if self.PartnerItems == nil then
		return;
	end
	for i = 1, #self.PartnerItems do
		if self.PartnerItems[i].transform == tr then
			self.PartnerItems[i]:UpdateContent(self.Partners[idx + 1], self.SelectIndex); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
			return
		end
	end
	local bi = PartnerHeadItem.new(tr.gameObject, self.m_PartnerRoot, self.AtlasPath, function(index)
			self:OnSelect(index)
		end);
	bi.prefab:SetActive(true);
	bi:UpdateContent(self.Partners[idx + 1], self.SelectIndex, "PartnerDetail"); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
	table.insert(self.PartnerItems, bi);
end

function PartnerDetailWidget:CleanAllSelected()
	-- 无限列表
	for i = 1, #self.PartnerItems do
		self.PartnerItems[i]:OnSelect(false);
	end
end

function PartnerDetailWidget:OnSelect(selectIndex)
	self.SelectIndex = selectIndex;
	SetSelectPartner(self.SelectIndex)
	self:CleanAllSelected();
	self:SetPartnerInfo(false, 0);
end

function PartnerDetailWidget:GetQualityName(quality)
	if quality == 1 then
		return "Partner_quality_n"
	elseif quality == 2 then
		return "Partner_quality_r"
	elseif quality == 3 then
		return "Partner_quality_sr"
	elseif quality == 4 then
		return "Partner_quality_ssr"
	end
end

function PartnerDetailWidget:SetPartnerInfo(upgrade, lv)
	local data = self:GetSelectPartner();
	if data == nil then
		return
	end
	self.m_PartnerInfo:SetActive(self.m_Menus.CurrentIndex ~= 3)
	local config = ConfigDataAccessor.Partner[data.Index];
	for i = 1, #self.Stars do
		if i <= data.CurStar then
			self.Stars[i]:SetActive(true)
		else
			self.Stars[i]:SetActive(false)
		end
	end

	if config ~= nil then
		self.PartnerName.text = config.Name
		self:GetSprite(self:GetQualityName(config.Quality), function(sp)
				self.m_Quality.sprite = sp
			end)

		local avatarDataCell = ConfigDataAccessor.Avatar[config.AvatarID];
		if avatarDataCell ~= nil then
			self:SetModel(avatarDataCell, String2Vector3(config.ModelScale), String2Vector3(config.ModelRot),
				String2Vector3(config.ModelPos), config.ModelAnima);
		end

		self.CertificationText.text = tostring(data.AptitudeBreakStep);
		local PartnerNature = ConfigDataAccessor.PartnerNature[config.Nature][data.Level]
		if PartnerNature ~= nil then
			self.Power.text = tostring(PartnerNature.Power);
		end
	end

	self:InitEquip();

	self.CurrentPart:OnShow(data, lv);
end

function PartnerDetailWidget:SetModel(avatarDataCell, scale, localRotate, localPos, animName)
	if self.m_avatarDataCell ~= avatarDataCell then
		------- 设置模型
		self:PushDisplayGob();
		self.m_avatarDataCell = avatarDataCell;
		ResourceManager.Instance:PopGameObject("Roles/Template/Display_Model", function(go)
				if go == nil then
					return;
				end
				if self.m_avatarDataCell == nil then
					ResourceManager.Instance:PushGameObject("Roles/Template/Display_Model", go);
					return;
				end
				UIManager.Instance:SetModel2UI2(go, self.m_avatarDataCell.AvatarID, CS.StarProjectDef.ThreeDModelPos.Raw, scale,
					Vector3.zero, localRotate, localPos, animName);
				go:SetActive(true);
				self.m_ModelViewDisplay = go:GetComponent(typeof(RoleViewDisplay));
				-- self.m_DisplayRawing:SetModelView(self.m_ModelViewDisplay);

				if self.m_ModelViewDisplay ~= nil then
					local rotation = Quaternion.Euler(0, 0, 0);
					self.m_ModelViewDisplay:SetRotation(rotation);
					self.m_DisplayRawimg:SetState(true);
				end

				------- 设置模型
				self.m_PartnerRawImage.texture = ResourceManager.Instance.RenderTexture;
			end);

	end
	if self.m_ModelViewDisplay ~= nil then
		local rotation = Quaternion.Euler(0, 0, 0);
		self.m_ModelViewDisplay:SetRotation(rotation);
		self.m_DisplayRawimg:SetState(true);
	end
end

function PartnerDetailWidget:PushDisplayGob()
	if self.m_ModelViewDisplay ~= nil then
		ResourceManager.Instance:PushGameObject("Roles/Template/Display_Model", self.m_ModelViewDisplay.gameObject);
		-- GameObject.Destroy(self.m_ModelViewDisplay.gameObject);
		self.m_ModelViewDisplay = nil;
		self.m_DisplayRawimg:SetModelView(self.m_ModelViewDisplay);
	end
end

function PartnerDetailWidget:OnMenusItemClick(index)
	if index == 1 then
		-- 基础
		self.CurrentPart:OnHide();
		self.CurrentPart = self.PartnerBasePart;
		local data = self:GetSelectPartner();
		self.CurrentPart:OnShow(data, 0);
	elseif index == 2 then
		self.CurrentPart:OnHide();
		self.CurrentPart = self.CertificationPart;
		local data = self:GetSelectPartner();
		self.CurrentPart:OnShow(data, 0);

		-- 资质
	elseif index == 3 then
		-- 技能
		self.CurrentPart:OnHide();
		self.CurrentPart = self.PartnerSkillPart;
		local data = self:GetSelectPartner();
		self.CurrentPart:OnShow(data, 0);

	end
	self.SubTag = index;
	self.m_Menus:Check(index);
	self:SetPartnerInfo(false, 0);
end

function PartnerDetailWidget:Sort()
	-- 出战状态》品质》星级》战力
	table.sort(self.Partners, function(p1, p2)
			local t1 = 0;
			local t2 = 0;
			if p1.State == CS.ProtoMsg.PartnerState.InBattle or p1.State == CS.ProtoMsg.PartnerState.Concretization then
				t1 = 1;
			end

			if p2.State == CS.ProtoMsg.PartnerState.InBattle or p2.State == CS.ProtoMsg.PartnerState.Concretization then
				t2 = 1;
			end

			if t1 > t2 then
				return true
			elseif t1 == t2 then

				local config1 = ConfigDataAccessor.Partner[p1.Index];
				local config2 = ConfigDataAccessor.Partner[p2.Index];
				if config1.Quality > config2.Quality then
					return true
				elseif config1.Quality == config2.Quality then
					if p1.CurStar > p2.CurStar then
						return true
					elseif p1.CurStar == p2.CurStar then
						local PartnerNature1 = ConfigDataAccessor.PartnerNature[config1.Nature][p1.Level]
						local PartnerNature2 = ConfigDataAccessor.PartnerNature[config2.Nature][p2.Level]
						if PartnerNature1 == nil then
							return false
						end

						if PartnerNature2 == nil then
							return false
						end

						return PartnerNature1.Power > PartnerNature2.Power
					end
				end
			end
			return false;
		end)
end

function PartnerDetailWidget:OnOpen(arg)
	self.IsOpen = true;
	self.PartnerBasePart:OnHide();
	self.CertificationPart:OnHide();
	self.PartnerSkillPart:OnHide();
	self.m_Menus:Init(self.SubTag, function(index)
			self:OnMenusItemClick(index);

		end);
	self.Partners = arg.Partners
	self.Module = arg.Module;
	-- if self.Module:GetOldTabIndex()~=2 then
	local id=GetSelectPartner();
	if id==0 or id==nil then
		self.SelectIndex = arg.SelectIndex
		SetSelectPartner(self.SelectIndex)
	else
		self.SelectIndex=id;
	end
	-- end
	self:Sort()
	self.m_ScrollView:ContinueScroll(#self.Partners);

	if self.SubTag == 1 then
		self.CurrentPart = self.PartnerBasePart;
	elseif self.SubTag == 2 then
		self.CurrentPart = self.CertificationPart;
	elseif self.SubTag == 3 then
		self.CurrentPart = self.PartnerSkillPart;
	end

	self:SetPartnerInfo(false, 0);
	UIManager.Instance:SetRawCamrea(true);

	if arg.SubTag ~= nil and arg.SubTag > 0 then
		self:OnMenusItemClick(arg.SubTag);
	end
end

function PartnerDetailWidget:GetSelectPartner()
	if self.Partners == nil then
		return nil
	end

	for _k, _v in pairs(self.Partners) do
		if _v.Index == self.SelectIndex then
			return _v
		end
	end
end

function PartnerDetailWidget:OnPartnerUpLevel(index, lv)
	if self.SelectIndex == index and self.IsOpen then
		self.Partners = self.Module:GetAllPartners();
		self:Sort()
		self.m_ScrollView:ContinueScroll(#self.Partners);
		self:SetPartnerInfo(true, lv);
		-- local data= self:GetSelectPartner();
		-- self.CurrentPart:OnShow(data,self.m_PartnerRawImage.texture);
	end
end

function PartnerDetailWidget:OnPartnerUpgrade(index)

end

function PartnerStarRiseWidget:Sort()
    -- 出战状态》品质》星级》战力
    table.sort(self.Partners, function(p1, p2)
        local t1 = 0;
        local t2 = 0;
        if p1.State == CS.ProtoMsg.PartnerState.InBattle or p1.State == CS.ProtoMsg.PartnerState.Concretization then
            t1 = 1;
        end

        if p2.State == CS.ProtoMsg.PartnerState.InBattle or p2.State == CS.ProtoMsg.PartnerState.Concretization then
            t2 = 1;
        end

        if t1 > t2 then
            return true
        elseif t1 == t2 then
            local config1 = ConfigDataAccessor.Partner[p1.Index];
            local config2 = ConfigDataAccessor.Partner[p2.Index];
            if config1.Quality > config2.Quality then
                return true
            elseif config1.Quality == config2.Quality then
                if p1.CurStar > p2.CurStar then
                    return true
                elseif p1.CurStar == p2.CurStar then
                    local PartnerNature1 = ConfigDataAccessor.PartnerNature[config1.Nature][p1.Level]
                    local PartnerNature2 = ConfigDataAccessor.PartnerNature[config2.Nature][p2.Level]
                    if PartnerNature1 == nil then
                        return false
                    end

                    if PartnerNature2 == nil then
                        return false
                    end

                    return PartnerNature1.Power > PartnerNature2.Power
                end
            end
        end
        return false;
    end)
end

function PartnerStarRiseWidget:FreshPartners()
    if self.Partners ~= nil then
        Log("PartnerStarRiseWidget OnPartnerFresh");
        -- self:OnFresh();
        self:Sort()
        self.m_ScrollView:ContinueScroll(#self.Partners);
    end
end

function PartnerStarRiseWidget:OnOnPartnerUpStar(isSuccess)
    if isSuccess then
        if self.BtnRiseClickFlag == 1 then
            self.BtnRiseClickFlag = 2;
            self:OnFresh();
        end
    else
        self.BtnRiseClickFlag = 2;
    end
end

function PartnerStarRiseWidget:OnEnable()
    -- 脏数据更新
    CS.StarProject.GlobalEvent.OnPartnerFresh:AddListener(function(id)
        self:FreshPartners();
    end);

    CS.StarProject.GlobalEvent.OnPartnerUpStar:AddListener(function(isSuccess)
        self:OnOnPartnerUpStar(isSuccess);
    end);
end

function PartnerStarRiseWidget:OnDisable()
    LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule, "OnForceTab", 3);
    -- 脏数据更新
    CS.StarProject.GlobalEvent.OnPartnerFresh:RemoveListener(function(id)
        self:FreshPartners();
    end);

    CS.StarProject.GlobalEvent.OnPartnerUpStar:RemoveListener(function(isSuccess)
        self:OnOnPartnerUpStar(isSuccess);
    end);
end

function PartnerStarRiseWidget:OnDestroy()
end

function PartnerStarRiseWidget:OnOpen(arg)
    self.Partners = arg.Partners
	local id=GetSelectPartner();
	if id==0 or id==nil then
		self.SelectIndex = arg.SelectIndex
		SetSelectPartner(self.SelectIndex)
	else
		self.SelectIndex=id;
	end
	
    --self.SelectIndex = arg.SelectIndex
    self:Sort()
    self.m_ScrollView:ContinueScroll(#self.Partners);
    self:SetCurPartnerData();

function PartnerStarRiseWidget:SetCurPartnerData()
    if self.Partners == nil then
        Log("PartnerStarRiseWidget:SetCurPartnerData self.Partners=nil err");
        return;
    end

    self.CurSelectPartnerData = self:GetSelectPartner();
    if self.CurSelectPartnerData == nil then
        Log("PartnerStarRiseWidget:SetCurPartnerData self.CurSelectPartnerData=nil err");
        return;
    end

    self.CurPartnerCfg = ConfigDataAccessor.Partner[self.CurSelectPartnerData.Index];
    if self.CurPartnerCfg == nil then
        Log("PartnerStarRiseWidget:SetCurPartnerData CurPartnerCfg=nil err");
        return;
    end

    self.CurPartnerUpStarCfg = ConfigDataAccessor.Item[self.CurPartnerCfg.UpStarItem];
end

function PartnerStarRiseWidget:BaseDataIsSure()
    if self.Partners == nil then
        Log("PartnerStarRiseWidget:BaseDataIsSure self.Partners=nil err");
        return false;
    end

    if self.CurSelectPartnerData == nil then
        Log("PartnerStarRiseWidget:BaseDataIsSure self.CurSelectPartnerData=nil err");
        return false;
    end

    if self.CurPartnerCfg == nil then
        Log("PartnerStarRiseWidget:BaseDataIsSure CurPartnerCfg=nil err");
        return false;
    end
    return true;
end

        return;
    end

        self.Stars[i]:SetActive(i <= self.CurSelectPartnerData.CurStar);

-- 获取选中的伙伴信息
function PartnerStarRiseWidget:GetSelectPartner()
    if self.Partners == nil then
        return nil
    end

    for _k, _v in pairs(self.Partners) do
        if _v.Index == self.SelectIndex then
            return _v;
        end
    end
end

function PartnerStarRiseWidget:OnItemRefesh(tr, idx)
    if self.PartnerItems == nil then
        return;
    end
    for i = 1, #self.PartnerItems do
        if self.PartnerItems[i].transform == tr then
            self.PartnerItems[i]:UpdateContent(self.Partners[idx + 1], self.SelectIndex); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end
    local bi = PartnerHeadItem.new(tr.gameObject, self.m_PartnerRoot, self.AtlasPath, self.PartnerHeadItemClickFun);
    bi.prefab:SetActive(true);
    bi:UpdateContent(self.Partners[idx + 1], self.SelectIndex, "StarRise"); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.PartnerItems, bi);
end

function PartnerStarRiseWidget:CleanAllSelected()
    -- 无限列表
    for i = 1, #self.PartnerItems do
        self.PartnerItems[i]:OnSelect(false);
    end
end

function PartnerStarRiseWidget:OnSelect(selectIndex)
    self.SelectIndex = selectIndex;
	SetSelectPartner(self.SelectIndex)
    self:CleanAllSelected();
    self:SetCurPartnerData();
    self:SetPartnerInfo();
    self:SetRiseStarItem();
end

-- 获得品质-颜色图标
function PartnerStarRiseWidget:GetQualityName(quality)
    if quality == 1 then
        return "Partner_quality_n"
    elseif quality == 2 then
        return "Partner_quality_r"
    elseif quality == 3 then
        return "Partner_quality_sr"
    elseif quality == 4 then
        return "Partner_quality_ssr"
    end
end

-- 设置模型
function PartnerStarRiseWidget:SetModel(avatarDataCell, scale, localRotate, localPos, animName)
    if self.m_avatarDataCell ~= avatarDataCell then
        ------- 设置模型
        self:PushDisplayGob();
        self.m_avatarDataCell = avatarDataCell;
        ResourceManager.Instance:PopGameObject("Roles/Template/Display_Model", function(go)
            if go == nil then
                return;
            end
            if self.m_avatarDataCell == nil then

                return;
            end
            UIManager.Instance:SetModel2UI2(go, self.m_avatarDataCell.AvatarID, CS.StarProjectDef.ThreeDModelPos.Raw, scale,
                Vector3.zero, localRotate, localPos, animName);
            go:SetActive(true);
            self.m_ModelViewDisplay = go:GetComponent(typeof(RoleViewDisplay));
            -- self.m_DisplayRawing:SetModelView(self.m_ModelViewDisplay);

            if self.m_ModelViewDisplay ~= nil then
                local rotation = Quaternion.Euler(0, 0, 0);
                self.m_ModelViewDisplay:SetRotation(rotation);
                self.m_DisplayRawimg:SetState(true);
            end

            ------- 设置模型
            self.m_PartnerRawImage.texture = ResourceManager.Instance.RenderTexture;
        end);
    end
    if self.m_ModelViewDisplay ~= nil then
        local rotation = Quaternion.Euler(0, 0, 0);

        self.m_DisplayRawimg:SetState(true);

-- 升星后的脏数据更新
function PartnerStarRiseWidget:OnFresh()
    -- 脏数据会直接改变BattleTeamModule里面的引用数据
    -- 所以这里不需要查找更新了那个的差量信息
    self:SetCurPartnerData();
    self:SetPartnerInfo();
    self:SetRiseStarItem();

    local star = self.CurSelectPartnerData.CurStar;
    self.LastStarExpendCfg = self:GetStarExpendCfg(star);
    -- 播放觉醒成功的界面
    self:ShowStarRiseFinishPanel();

    self.BtnRiseClickFlag = 0;
end

function PartnerStarRiseWidget:ShowStarRiseFinishPanel()

            self.ClickStarRiseFinishPanelFun);
    else
        self.StarRiseFinishPanel:Init(self.CurSelectPartnerData, self.CurPartnerCfg, self.LastStarExpendCfg);
    end
end

-- 设置升星item
function PartnerStarRiseWidget:SetRiseStarItem()
    local isSure = self:BaseDataIsSure();
    if isSure == false then
        Log("PartnerStarRiseWidget:SetRiseStarItem isSure=false err");
        return;
    end

    -- 根据品质和星级获得升星的
    local quality = self.CurPartnerCfg.Quality;
    self:GetStarExpendCfgs(quality);

    local star = self.CurSelectPartnerData.CurStar;
    self.NextStarExpendCfg = self:GetStarExpendCfg(star + 1);

    -- 设置6个Item
    for i = 1, #self.m_RiseStarItems do
        local item = self.m_RiseStarItems[i];
        local data = self.CurQualityStarExpendCfg[i];
        if item ~= nil and data ~= nil then
            -- state: 【-1:未解锁（不能解锁）】【0:未解锁（可解锁）】【1:已解锁】
            local state = -1;
            if self.NextStarExpendCfg ~= nil and data.Id == self.NextStarExpendCfg.Id then
                state = 0;
            end
            if star >= data.Star then
                state = 1;
            end
            item:SetData(data, state);
        end
    end

    -- 设置消耗材料
    -- self.ExpendItemIcon.sprite = self:GetSprite();
    if self.CurPartnerUpStarCfg ~= nil then
        ResourceHelper:LoadIconAsync(self.ExpendItemIcon, "", self.CurPartnerUpStarCfg.Icon,
            false);
    end

    if self.NextStarExpendCfg ~= nil then
        --self.ExpendItemCount.text = "x" .. self.NextStarExpendCfg.Item_num .. " <size=34>觉醒</size>";
        self.ExpendItemCount.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_PartnerStarRiseWidget_NextStarExpendCfg"), {self.NextStarExpendCfg.Item_num});

    else
        Log("PartnerStarRiseWidget:SetRiseStarItem self.NextStarExpendCfg=false err");
        self.CurCoutText.text = "";
    end
    self.JBtnRise.gameObject:SetActive(self.NextStarExpendCfg ~= nil);
end

-- 获得当前品质的升星配置
function PartnerStarRiseWidget:GetStarExpendCfgs(quality)
    self.CurQualityStarExpendCfg = ConfigDataAccessor.StarExpend[quality];
end

-- 获得当前升星的配置
function PartnerStarRiseWidget:GetStarExpendCfg(star)
    for k, v in pairs(self.CurQualityStarExpendCfg) do
        if v.Star == star then
            return v;
        end
    end
    return nil;
end

-- 升星回调
function PartnerStarRiseWidget:OnJBtnRiseClick()
    if self.BtnRiseClickFlag == 1 then
        Log("PartnerStarRiseWidget:OnJBtnRiseClick self.BtnRiseClickFlag==1,, err");
        return;
    end

    local curTime = CS.SGF.Time.TimeUtils.ClientNowStampMilli;
    if self.BtnRiseClickLastTime ~= 0 then
        if (curTime - self.BtnRiseClickLastTime) < 1000 then
            Log("PartnerStarRiseWidget:OnJBtnRiseClick (curTime - self.BtnRiseClickLastTime) < 1000,, err");
            return;
        end
    end

    local isSure = self:BaseDataIsSure();
    if isSure == false then
        Log("PartnerStarRiseWidget:OnJBtnRiseClick isSure=false err");
        return;
    end

    if self.NextStarExpendCfg == nil then
        Log("PartnerStarRiseWidget:OnJBtnRiseClick self.NextStarExpendCfg=false err");
        return;
    end

    local itemId = self.CurPartnerCfg.UpStarItem;
    local itemNum = self.NextStarExpendCfg.Item_num;
    local itemCount = BusinessManager.Instance:GetCurrencyNum(itemId);
    if itemCount < itemNum then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_273"));
        return;
    end

    self.BtnRiseClickLastTime = curTime; -- 记录点击的时间
    self.BtnRiseClickFlag = 1;
    LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule, "OnPartnerUpStarReq", self.CurPartnerCfg.Id);
end

function PartnerEquipWidget:Awake()

    self.PartnerItems = {}

    self.m_ItemGrid = self:Find("Root/Left/ItemGrid").gameObject;
    self.m_PartnerRoot = self:Find("Root/Left/Scroll View/Mask/List");

    self.m_ScrollView = self:Find("Root/Left/Scroll View"):GetComponent(typeof(SuperScrollViewController));
    self.m_ScrollView:NewScroll(self.m_ItemGrid, 0, function(tr, idx)
        self:OnItemRefesh(tr, idx);
    end);

    -- 属性
    self.attrs = {}
    for i = 1, 9 do
        local eai = EquipUpgradeSlotAttr:New(self:Find("Root/Right/Info/Mask/List/Attr" .. i).gameObject);
        table.insert(self.attrs, eai);
    end

    -- 装备特技
    self.subattrs = {}
    for i = 1, 9 do
        local eai = EquipSubAttrItem:New(self:Find("Root/Right/Info/Mask/List/PartnerSubAttr" .. i).gameObject);
        table.insert(self.subattrs, eai);
    end

    -- 无特技
    self.go_nullSkill = self:Find("Root/Right/Info/Mask/List/NullSkill").gameObject

    self.go_info = self:Find("Root/Right/Info").gameObject;
    self.go_null = self:Find("Root/Right/Null").gameObject;

    self:Find("Root/PartnerEquip/click"):GetComponent(typeof(JButton)).OnClick = (function()
        -- 关闭itemtips
        LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnCloseTips", {});
    end);

    self.PartnerInfoPart = PartnerInfoPart:New(self:Find("Root/PartnerInfo").gameObject, self);
    self.PartnerEquipPart = PartnerEquipPart:New(self:Find("Root/PartnerEquip").gameObject, self);
end

function PartnerEquipWidget:Start()

    ---伙伴装备成功

    -- 监听伙伴穿装备
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.WearParEquipRetID, function(data)
        self:OnWearParEquipRet(data)
    end, nil);

    -- 监听伙伴脱装备
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.TakeOffParEquipRetID, function(data)
        self:OnTakeOffParEquipRet(data)
    end, nil);

end

end

end

end

-- 刷新单一装备属性
function PartnerEquipWidget:RefeshEquipAttr(baseID)

    if baseID == 0 then
        -- 没有装备
        self.go_info:SetActive(false);
        self.go_null:SetActive(true);
    else
        self.go_info:SetActive(true);
        self.go_null:SetActive(false);
        -- 显示装备信息

        local Nature = ConfigDataAccessor.PartnerEquip[baseID].Nature;
        local Value = ConfigDataAccessor.PartnerEquip[baseID].Value;

        local Effect = ConfigDataAccessor.PartnerEquip[baseID].Effect;

        -- 主属性
        for i = 1, #self.attrs do
            self.attrs[i].gameObject:SetActive(false);
        end

        for i = 1, #Nature do
            self.attrs[i].gameObject:SetActive(true);
            self.attrs[i]:RefeshAttrProb(Nature[i], Value[i], Value[i], true);
        end

        -- 副属性
        for i = 1, #self.subattrs do
            self.subattrs[i].gameObject:SetActive(false);
        end
        for i = 1, #Effect do
            if Effect[i] > 0 then
                self.subattrs[i].gameObject:SetActive(true);
                self.subattrs[i]:RefeshParEquipAttrProb(Effect[i]);
            end
        end

        -- 暂时先屏蔽
        -- self.go_nullSkill:SetActive(#Effect<=0);

    end

end

-- 刷新所有装备属性
function PartnerEquipWidget:RefeshAllEquipAttr()

    local NatureAll = {};
    local ValueAll = {};

    local EffectAll = {};

    -- 装备
    if self:GetSelectPartner() ~= nil then
        for i = 1, 3 do
            local data = BusinessManager.Instance:GetPartnerEquipedEquip(self:GetSelectPartner().ID, i);
            if data ~= nil then
                local Nature = ConfigDataAccessor.PartnerEquip[data.BaseID].Nature;
                local Value = ConfigDataAccessor.PartnerEquip[data.BaseID].Value;
                local Effect = ConfigDataAccessor.PartnerEquip[data.BaseID].Effect;

                for j = 1, #Nature do
                    local haveSame = false;
                    for p = 1, #NatureAll do
                        if Nature[j] == NatureAll[p] then
                            haveSame = true;
                            ValueAll[p] = ValueAll[p] + Value[j];
                            break
                        end
                    end
                    if not (haveSame) then
                        table.insert(NatureAll, Nature[j]);
                        table.insert(ValueAll, Value[j]);
                    end
                end

                for j = 1, #Effect do
                    local haveSame = false;
                    for p = 1, #EffectAll do
                        if EffectAll[p] == Effect[j] then
                            haveSame = true;
                        end
                    end
                    if not (haveSame) then
                        table.insert(EffectAll, Effect[j]);
                    end
                end

            end
        end
    end

    if #NatureAll == 0 then
        -- 没有装备
        self.go_info:SetActive(false);
        self.go_null:SetActive(true);
    else
        -- 显示装备信息
        self.go_info:SetActive(true);
        self.go_null:SetActive(false);

        -- 主属性
        for i = 1, #self.attrs do
            self.attrs[i].gameObject:SetActive(false);
        end

        for i = 1, #NatureAll do
            self.attrs[i].gameObject:SetActive(true);
            self.attrs[i]:RefeshAttrProb(NatureAll[i], ValueAll[i], ValueAll[i], true);
        end

        -- 副属性
        for i = 1, #self.subattrs do
            self.subattrs[i].gameObject:SetActive(false);
        end
        for i = 1, #EffectAll do
            if EffectAll[i] > 0 then
                self.subattrs[i].gameObject:SetActive(true);
                self.subattrs[i]:RefeshParEquipAttrProb(EffectAll[i]);
            end
        end

    end
end

function PartnerEquipWidget:OnItemRefesh(tr, idx)

    if self.PartnerItems == nil then
        return;
    end
    for i = 1, #self.PartnerItems do
        if self.PartnerItems[i].transform == tr then
            self.PartnerItems[i]:UpdateContent(self.Partners[idx + 1], self.SelectIndex); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end
    local bi = PartnerHeadItem.new(tr.gameObject, self.m_PartnerRoot, self.AtlasPath, function(index)
        self:OnSelect(index)
    end);
    bi.prefab:SetActive(true);
    bi:UpdateContent(self.Partners[idx + 1], self.SelectIndex, "PartnerEquip"); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.PartnerItems, bi);
end

function PartnerEquipWidget:OnSelect(selectIndex)
    -- 关闭itemtips
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnCloseTips", {});

    self.SelectIndex = selectIndex;
    self:CleanAllSelected();
    self:SetPartnerInfo();
end

function PartnerEquipWidget:SetPartnerInfo()
    local data = self:GetSelectPartner();
    if data == nil then
        return
    end

    self:RefeshAllEquipAttr();
    self.PartnerEquipPart:Show(false);

    self.PartnerInfoPart:SetPartnerInfo(data);
    self.PartnerInfoPart:Show(true);
end

function PartnerEquipWidget:SetPartnerEquip(idx)
    self.PartnerInfoPart:Show(false);

    self.PartnerEquipPart:SetPartnerEquip(idx);
    self.PartnerEquipPart:Show(true);
end

function PartnerEquipWidget:Sort()
    -- 出战状态》品质》星级》战力
    table.sort(self.Partners, function(p1, p2)
        local t1 = 0;
        local t2 = 0;
        if p1.State == CS.ProtoMsg.PartnerState.InBattle or p1.State == CS.ProtoMsg.PartnerState.Concretization then
            t1 = 1;
        end

        if p2.State == CS.ProtoMsg.PartnerState.InBattle or p2.State == CS.ProtoMsg.PartnerState.Concretization then
            t2 = 1;
        end

        if t1 > t2 then
            return true
        elseif t1 == t2 then

            local config1 = ConfigDataAccessor.Partner[p1.Index];
            local config2 = ConfigDataAccessor.Partner[p2.Index];
            if config1.Quality > config2.Quality then
                return true
            elseif config1.Quality == config2.Quality then
                if p1.CurStar > p2.CurStar then
                    return true
                elseif p1.CurStar == p2.CurStar then
                    local PartnerNature1 = ConfigDataAccessor.PartnerNature[config1.Nature][p1.Level]
                    local PartnerNature2 = ConfigDataAccessor.PartnerNature[config2.Nature][p2.Level]
                    if PartnerNature1 == nil then
                        return false
                    end

                    if PartnerNature2 == nil then
                        return false
                    end

                    return PartnerNature1.Power > PartnerNature2.Power
                end
            end
        end
        return false;
    end)
end

function PartnerEquipWidget:OnOpen(arg)
    self.Partners = arg.Partners
    self.SelectIndex = arg.SelectIndex
    self:Sort();
    self.m_ScrollView:ContinueScroll(#self.Partners);
    self.CurrentPart = self.PartnerBasePart;
    self:SetPartnerInfo();
    UIManager.Instance:SetRawCamrea(true);
end

function PartnerEquipWidget:GetSelectPartner()
    if self.Partners == nil then
        return nil
    end

    for _k, _v in pairs(self.Partners) do
        if _v.Index == self.SelectIndex then
            return _v
        end
    end
end

function PartnerEquipWidget:GetPartnerBaseIdByEntityID(entityID)

    if entityID == nil then
        return 0;
    end

    for _k, _v in pairs(self.Partners) do
        if _v.ID == entityID then
            return _v.Index
        end
    end

    return 0;
end

function PartnerEquipWidget:OnClose(arg)

    self.PartnerInfoPart:PushDisplayGob();
    UIManager.Instance:SetRawCamrea(false);
    self.Partners = nil;
    self.SelectIndex = -1;
    self.Index = nil; ---槽位索引
    self.PartnerInfoPart.m_ModelDataCell = nil;
    self.PartnerInfoPart.m_ModelViewDisplay = nil;
    self.PartnerInfoPart.m_avatarDataCell = nil;
end

-- 穿装备回调
function PartnerEquipWidget:OnWearParEquipRet(data)
    print("穿伙伴装备回调");

    local pb = data.data;
    local ret = pb.Ret; -- 1成功 14失败
    if ret == CS.ProtoMsg.WearParEquipRet.Types.Result.Ok then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_219"));
        self:RefeshAll();
    end

end

-- 脱装备回调
function PartnerEquipWidget:OnTakeOffParEquipRet(data)
    print("脱伙伴装备回调");

    local pb = data.data;
    local ret = pb.Ret; -- 1成功 14失败
    if ret == CS.ProtoMsg.TakeOffParEquipRet.Types.Result.Ok then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_220"));
        self:RefeshAll();
    end

end

function PartnerEquipWidget:RefeshAll()
    self:RefeshAllEquipAttr();
    self.PartnerInfoPart:SetPartnerInfo(self:GetSelectPartner());
    self.PartnerEquipPart:SetPartnerEquip();
end

return PartnerEquipWidget

---@class EquipSelItem
local EquipSelItem = {}
EquipSelItem.__index = EquipSelItem

function EquipSelItem:ctor(go, cb)
	local self = {};
	setmetatable(self, EquipSelItem);
	
	self.gameObject = go;
    self.transform = go.transform;

	self.selBtn = self.transform:GetComponent(typeof(Button));
	self.selBtn.onClick:AddListener(function() cb(); end);

    self.selGo = self.transform:Find("Sel").gameObject;
    self.nameText = self.transform:Find("Name"):GetComponent(typeof(Text));
    self.dealText = self.transform:Find("Deal/Text"):GetComponent(typeof(Text));

    self.stars = {};
    for i=1,3 do
        local star = self.transform:Find("ItemGrid/Stars/Star"..i).gameObject;
		self.stars[i] = star;
    end

	self.numText = self.transform:Find("ItemGrid/Num"):GetComponent(typeof(Text));
	self.icon = self.transform:Find("ItemGrid/Icon"):GetComponent(typeof(Image));
	self.bgRare = self.transform:Find("ItemGrid/BgRare"):GetComponent(typeof(Image));

	return self;
end

function EquipSelItem:SetEquipSel(sel)

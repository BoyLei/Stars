---@class RiseStarAttrItem

RiseStarAttrItem = class("RiseStarAttrItem",UIPanel)

---表现层
RiseStarAttrItem.AttrName = nil;
RiseStarAttrItem.AttrNum = nil;

function RiseStarAttrItem:ctor(go)
	
	self.prefab = go;
	self.AttrName=self:Find("AttrName"):GetComponent(typeof(Text));
	self.AttrNum=self:Find("AttrNum"):GetComponent(typeof(Text));
	
end

function RiseStarAttrItem:Init(AttrName,AttrNum)
	
	self.AttrName.text = AttrName;
	self.AttrNum.text = AttrNum;
	
	self.prefab.gameObject:SetActive(true);
	
end

return RiseStarAttrItem;

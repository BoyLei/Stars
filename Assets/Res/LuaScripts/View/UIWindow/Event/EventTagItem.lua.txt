local EventTagItem = {}

EventTagItem.__index = EventTagItem


function EventTagItem:New(_go)
    local self = {}
    setmetatable(self, EventTagItem)

    self.gameObject = _go
    self.transform = _go.transform

    self.Name = self.transform:Find("name"):GetComponent(typeof(Text))


    return self --返回自身
end

function EventTagItem:InitItem()
    self.IsUsed=false;
    self.gameObject:SetActive(false);
end


function EventTagItem:SetItem(name)
    self.Name.text=name;

    self.IsUsed=true;
    self.gameObject:SetActive(true);
end

function EventTagItem:SetUse(b)
    self.IsUsed=b;
    self.gameObject:SetActive(b);
end


return EventTagItem

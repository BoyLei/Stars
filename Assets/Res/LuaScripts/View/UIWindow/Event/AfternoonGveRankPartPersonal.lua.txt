local AfternoonGveRankPartPersonal = {}

AfternoonGveRankPartPersonal.__index = AfternoonGveRankPartPersonal

local AfternoonGveRankPartPersonalItem=require("View/Uiwindow/Event/AfternoonGveRankPartPersonalItem")

function AfternoonGveRankPartPersonal:New(_go) 
    local self = {};    
    setmetatable(self, AfternoonGveRankPartPersonal); 
    

    self.gameObject=_go;
    self.transform=_go.transform;


    --道具列表
    self.AfternoonGveRankPartPersonalItemList={};
	--无限列表
	self.go_AfternoonGveRankPartPersonalItem = self.transform:Find("GngPersonalRankItem").gameObject;

	self.tr_ItemRoot=self.transform:Find("Scroll View/Mask/List");
	self.go_AfternoonGveRankPartPersonalItem:SetActive(false);
	self.go_AfternoonGveRankPartPersonalItem.transform:SetParent(self.transform);
	self.AfternoonGveRankPartPersonalScorllView=self.transform:Find("Scroll View"):GetComponent(typeof(SuperScrollViewController));
	self.AfternoonGveRankPartPersonalScorllView:NewScroll(self.go_AfternoonGveRankPartPersonalItem, 0, function (tr,idx)
		self:OnAfternoonGveRankPartPersonalItemRefesh(tr,idx);
	end);


    self:Show(false);

    return self;    --返回自身
end



function AfternoonGveRankPartPersonal:OnAfternoonGveRankPartPersonalItemRefesh(tr,idx)

	for i=1,#self.AfternoonGveRankPartPersonalItemList do
		if self.AfternoonGveRankPartPersonalItemList[i].transform == tr then
            self.AfternoonGveRankPartPersonalItemList[i]:SetItem(self.ItemDataList[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
		end

	end

    local bi = AfternoonGveRankPartPersonalItem:New(tr.gameObject);

	bi.gameObject:SetActive(true);
	bi:SetItem(self.ItemDataList[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1

    table.insert(self.AfternoonGveRankPartPersonalItemList, bi);

end


function AfternoonGveRankPartPersonal:Show(b,data,needFitter)
    self.gameObject:SetActive(b);
    if b then

		self.NeedFitter=needFitter;
        self:Refesh(data);


    end
end



function AfternoonGveRankPartPersonal:Refesh(data)

    if self.ItemDataList==nil then
        self.ItemDataList ={}
    else
        for k,_ in pairs(self.ItemDataList) do self.ItemDataList[k] = nil end
    end


    for i=0,data.Count-1 do
		if self.NeedFitter and data[i].GuildName~= BusinessManager.Instance:GetGuildName() then

		else
        	table.insert( self.ItemDataList, data[i] );
		end
    end
	
	for i=data.Count,7 do
		local d={}
		d.Rank=-1;
		table.insert( self.ItemDataList, d);
	end

	--self:SortItem(self.ItemDataList);
	self.AfternoonGveRankPartPersonalScorllView:ContinueScroll(#self.ItemDataList);

end


function AfternoonGveRankPartPersonal:SortItem(tbl)

	if #tbl > 1 then

		table.sort(tbl, function(a, b)
			--升序
			if a.ScoOneP < b.ScoOneP then
				return true
			end
			return false
		end)

	end

end


return AfternoonGveRankPartPersonal;

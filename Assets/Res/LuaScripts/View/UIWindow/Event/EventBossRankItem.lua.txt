local EventBossRankItem = {}

EventBossRankItem.__index = EventBossRankItem
local ObjectPool= require("Common/Helper/ObjectPool")

function EventBossRankItem:New(_go) 
    local self = {};    
    setmetatable(self, EventBossRankItem); 
    

    self.gameObject=_go;
    self.transform=_go.transform;


    self.bg=self.transform:Find("bg"):GetComponent(typeof(Image));

    self.rank=self.transform:Find("Rank"):GetComponent(typeof(Text));

    self.name=self.transform:Find("name"):GetComponent(typeof(Text));
    self.perTex=self.transform:Find("per"):GetComponent(typeof(Text));


    return self;    --返回自身
end



function EventBossRankItem:InitItem()
    self.IsUsed=false;
    self.gameObject:SetActive(false);
end


function EventBossRankItem:SetUse(b)
    self.IsUsed=b;
    self.gameObject:SetActive(b);
end

function EventBossRankItem:RefeshItem(name,per,isHaveReward,rank)


    self:SetUse(true);

    self.rank.text="0"..rank;

    self.name.text=name;
    self.bg.fillAmount=per;
    self.perTex.text=""..(math.ceil(per*1000)/10).."%";

end




return EventBossRankItem;

local AfternoonGveRankItem = {}

AfternoonGveRankItem.__index = AfternoonGveRankItem

local ItemGrid=require("View/Item/ItemGrid");


function AfternoonGveRankItem:New(_go) 
    local self = {};    
    setmetatable(self, AfternoonGveRankItem); 
    

    self.gameObject=_go;
    self.transform=_go.transform;

    self.rankBg=self.transform:Find("RankBg"):GetComponent(typeof(Image));
    self.rank=self.transform:Find("rank"):GetComponent(typeof(Text));

    --道具
    self.items={}
    for i=1,6 do
        local it = ItemGrid:New(self.transform:Find("Items/ItemGrid"..i).gameObject);
        it:InitItem(false,function(ritem)  
            self:OnItemClick(ritem);
        end);
        table.insert( self.items,it);
    end


    return self;    --返回自身
end


function AfternoonGveRankItem:OnItemClick(ritem)
	LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips",{ ritem.BaseID,false ,0,ritem.transform,0});
end

function AfternoonGveRankItem:SetItem(data)

    self.rank.text=data.rankTex;

    if data.rank<4 then
        self.rankBg.gameObject:SetActive(true);
    else
        self.rankBg.gameObject:SetActive(false);
    end


    --消耗道具
	for i=1,#self.items do
        self.items[i].gameObject:SetActive(false);
	end

    
    --消耗道具
	for i=1,#data.reward do
        self.items[i].gameObject:SetActive(true);
        if data.num~= nil then
            self.items[i]:SetItemNum(data.reward[i],data.num[i]);
        else
            self.items[i]:SetItemNum(data.reward[i],0);
        end
	end

end


function AfternoonGveRankItem:SetNull()
    self.rankBg.gameObject:SetActive(false);
    self.rank.text=LanguageHelp:GetLanguageByKey("Local_Str_Unranked");
end




return AfternoonGveRankItem;

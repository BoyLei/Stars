local ItemAttrDeclare = {}

ItemAttrDeclare.__index = ItemAttrDeclare;


local ItemAttrDeclareSon = require("View/UIWindow/PlayerAttrWindow/ItemAttrDeclareSon");

function ItemAttrDeclare:New(_go)
	local self = {};
	setmetatable(self, ItemAttrDeclare);
	self.Transform = _go.transform;
	self.TypeText = self.Transform:Find("Top/Image/TextTitle"):GetComponent(typeof(Text));
	self.Layout = self.Transform:Find("DownLayout");

	self.ItemAttr = nil;

	self.BaseAttrList = {};
	self.BaseAttrItemList = {};

	self.ItemAttrDeclareSon = nil;
	self.Func = nil;

	self.ItemBaseAttrClickFun = function(data)
		self:OnItemBaseAttrClick(data);
	end

	return self;
end

function ItemAttrDeclare:SetItem(type, baseAttrList, item, mainPlayerLevel, func)
	local typeName = E_AttrOwnModuleType[type];
	self.TypeText.text = typeName;
	self.BaseAttrList = baseAttrList;
	self.ItemAttrDeclareSon = item;
	self.BaseAttrItemList = {};
	for k, v in pairs(self.BaseAttrList) do
		local gob = newObject(self.ItemAttrDeclareSon);
		self:CreateItemBaseAttr(gob, v, mainPlayerLevel);
		gob.transform:SetParent(self.Layout);
		gob.transform.localPosition = Vector3.zero;
		gob.transform.localScale = Vector3.one;
	end
	self.Func = func;
end

function ItemAttrDeclare:CreateItemBaseAttr(gameObject, data, mainPlayerLevel)
	local item = ItemAttrDeclareSon.new(gameObject);
	item:UpdateContent(data, self.ItemBaseAttrClickFun, mainPlayerLevel);
	item.Transform.gameObject:SetActive(true);
	table.insert(self.BaseAttrItemList, item);
end

function ItemAttrDeclare:OnItemBaseAttrClick(data)
	self:CleanBaseAttrAllSelected();
	self:CheckItemByAttrId(data.AttrId);
end

-- 清空所有基础属性的选中
function ItemAttrDeclare:CleanBaseAttrAllSelected()
	for i = 1, #self.BaseAttrItemList do
		self.BaseAttrItemList[i]:OnSelect(false);
	end
end

function ItemAttrDeclare:CheckItemByAttrId(id)
	for i = 1, #self.BaseAttrItemList do
		local item = self.BaseAttrItemList[i];
		if item.AttrData ~= nil and item.AttrData.AttrId == id then
			item:OnSelect(true);
			self.Func(item.Transform.gameObject);
			return;
		end
	end
end

return ItemAttrDeclare;

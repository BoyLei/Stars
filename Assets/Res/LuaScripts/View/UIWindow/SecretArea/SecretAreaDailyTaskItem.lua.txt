---@class SecretAreaDailyTaskItem
SecretAreaDailyTaskItem = class("SecretAreaDailyTaskItem", UIPanel)

SecretAreaDailyTaskItem.TaskID = -1;
SecretAreaDailyTaskItem.Cfg = nil;
SecretAreaDailyTaskItem.CurProgress = 0;
SecretAreaDailyTaskItem.IsGet = false;
SecretAreaDailyTaskItem.TargetNum = 1;

function SecretAreaDailyTaskItem:ctor(go, parent, atlas)
    self.prefab = go;
    self.AtlasPath = atlas;
    self.Name = self:Find("Tile"):GetComponent(typeof(Text));
    self.AwardBtn = self:Find("AwardBtn"):GetComponent(typeof(JButton));
    self.AwardBtnText = self:Find("AwardBtn/Text"):GetComponent(typeof(Text));
    self.m_ItemPreviewGo = self:Find("ItemPreview").gameObject;
    self.HasGet = self:Find("HasGet").gameObject
    self.m_ItemPreview = StructAccessor.ItemPreview.new(self.m_ItemPreviewGo);
    self.AwardBtn.OnClick = (function()
        self:GetAward();
    end);
end

function SecretAreaDailyTaskItem:GetAward()
    local isfull = self.CurProgress >= self.TargetNum;
    if not self.IsGet or isfull then
        LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "SendPSGetDailyRewardReq", self.TaskID);
    end
end

function SecretAreaDailyTaskItem:FreshItem()
    if self.Cfg == nil then
        return
    end

    local params = {};
    table.insert(params, self.CurProgress);
    for i, v in ipairs(self.Cfg.Paramas) do
        table.insert(params, v);
    end

    local matches = {}
    for match in string.gmatch(self.Cfg.TaskDes, "%{(.-)%}") do
        table.insert(matches, tonumber(match))
    end

    --for i, match in ipairs(matches) do
        --params[i] = self.Cfg.Paramas[match];
    --end

    self.Name.text = StringFormat(self.Cfg.TaskDes, params);

    if self.IsGet then
        self.HasGet:SetActive(true)
        self.AwardBtn.gameObject:SetActive(false)
    else
        self.HasGet:SetActive(false)
        self.AwardBtn.gameObject:SetActive(true)
        if self.CurProgress >= self.TargetNum then
            self.AwardBtnText.text = LanguageHelp:GetLanguageByKey("Local_Str_GetReward")
            self.AwardBtn.Image.color = Color(1, 1, 1, 1);
        else
            self.AwardBtnText.text = LanguageHelp:GetLanguageByKey("Local_Str_Unfinished")
            self.AwardBtn.Image.color = Color(1, 1, 1, 0);
        end
    end

    local items = self.Cfg.TaskReward;
    local nums = self.Cfg.TaskRewardNum;
    local CostItems = {}
    for i = 1, #items do
        table.insert(CostItems, {
            ItemID = items[i],
            ItemNum = nums[i]
        })
    end

    self.m_ItemPreview:SetItems(CostItems);
end

-- 数据更新
function SecretAreaDailyTaskItem:UpdateContent(item)
    if item ~= nil then
        self.TaskID = item.TaskID;
        self.Cfg = item.Cfg;
        self.CurProgress = item.CurProgress;
        self.TargetNum = item.TargetNum;
        self.IsGet = item.IsGet;
    end

    self:FreshItem();
end

return SecretAreaDailyTaskItem;

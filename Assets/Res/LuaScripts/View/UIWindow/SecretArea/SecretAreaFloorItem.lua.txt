---@class SecretAreaFloorItem
SecretAreaFloorItem = class("SecretAreaFloorItem", UIPanel)

SecretAreaFloorItem.m_IsLock = true;
SecretAreaFloorItem.m_Source = 0; -- 0 主界面 1 胜率界面

SecretAreaFloorItem.Btn = nil;
SecretAreaFloorItem.Scale = nil;
SecretAreaFloorItem.Selected = nil;
SecretAreaFloorItem.Star = nil;
SecretAreaFloorItem.Floor = nil;
SecretAreaFloorItem.Lock = nil;
SecretAreaFloorItem.Finish = nil;

function SecretAreaFloorItem:ctor(go, parent, atlas, callBack, source)
    self.prefab = go;
    self.AtlasPath = atlas;
    self.CallBack = callBack;
    self.m_Source = source;

    self.Btn = go:GetComponent(typeof(JButton));
    self.BtnFun = function()
        self:OnClickHandler();
    end
    self.Btn.OnClick = self.BtnFun;

    if self.m_Source == 1 then
        self.Scale = self:Find("Scale").gameObject;
        self.Selected = self:Find("Scale/Selected").gameObject;
        self.Star = self:Find("Scale/Star").gameObject;
        self.Floor = self:Find("Scale/Floor"):GetComponent(typeof(Text));
        self.Lock = self:Find("Scale/Lock").gameObject;
        self.Finish = self:Find("Scale/Finish").gameObject;
    else
        self.Selected = self:Find("Selected").gameObject;
        self.Star = self:Find("Star").gameObject;
        self.Floor = self:Find("Floor"):GetComponent(typeof(Text));
        self.Lock = self:Find("Lock").gameObject;
    end
end

-- 点击事件
function SecretAreaFloorItem:OnClickHandler()
    ----- 现在未解锁的也可以点击
    -- if self.m_IsLock then
    -- return;
    -- end
    self.CallBack(self.Index);
    -- self:OnSelect(true);
end

-- 设置选中状态
function SecretAreaFloorItem:OnSelect(select)
    if self.m_Source == 1 then
        -- self.Selected:SetActive(select);
        local scale = Vector3.one;
        if select then
            scale = scale * 1.1;
        end
        self.Selected:SetActive(select and (not self.m_IsLock)); -- 光圈是：选中状态&&已解锁状态
        self.Scale.transform.localScale = scale;
    else
        --self.Selected:SetActive(select and (not self.m_IsLock)); -- 光圈是：选中状态&&已解锁状态
        self.Selected:SetActive(select); -- 光圈是：选中状态&&已解锁状态

    end
end

-- 数据更新
function SecretAreaFloorItem:UpdateContent(data, selectIndex, maxFloor, isLock)
    self.Index = data;
    self.m_IsLock = isLock;
    if data == selectIndex then
        self:OnClickHandler();
    else
        self:OnSelect(false);
    end
    self.Lock:SetActive(self.m_IsLock);
    self.Star:SetActive((not self.m_IsLock));
    if self.m_Source == 1 then
        self.Finish:SetActive(maxFloor > self.Index);
    end
    -- self.Floor.text = "第" .. self.Index .. "层";
    self.Floor.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_limitFloor"), {self.Index});
end

return SecretAreaFloorItem;

]] local SecretAreaAchieveItem = require("View/UIWindow/SecretArea/SecretAreaAchieveItem")
SecretAreaAchieveWidget = class("SecretAreaAchieveWidget", UIPanel)
SecretAreaAchieveWidget.m_Close = nil;
SecretAreaAchieveWidget.m_ScrollView = nil;
SecretAreaAchieveWidget.m_AchieveItem = nil;
SecretAreaAchieveWidget.m_Menus = nil;

-- 数据
SecretAreaAchieveWidget.ItemList = {};
SecretAreaAchieveWidget.ItemDatas = {}
SecretAreaAchieveWidget.TabIndex = 1;
SecretAreaAchieveWidget.MissionDatas = nil;
local OnAchieveChangeFunc = nil;
function SecretAreaAchieveWidget:Awake()
    self.m_Close = self:Find("Center/TopCenter/Close"):GetComponent(typeof(JButton));
    self.m_ScrollView = self:Find("Center/ScrollView"):GetComponent(typeof(SuperScrollViewController));
    self.m_AchieveItem = self:Find("Center/AchieveItem").gameObject
    self.m_Menus = self:Find("Center/TopCenter/TopLeft/Menus"):GetComponent(typeof(StarUITabMenu));

    self.m_ScrollView:NewScroll(self.m_AchieveItem, 0, function(tr, idx)
        self:OnItemRefesh(tr, idx);
    end);

    self.m_Close.OnClick = (function()
        self.Close();
    end);
    OnAchieveChangeFunc = function(arg)
        self:OnAchieveChange(arg);
    end
end

function SecretAreaAchieveWidget:OnAchieveChange(arg)
    self:OnFresh(arg);
end

function SecretAreaAchieveWidget:OnItemRefesh(tr, idx)
    if self.ItemList == nil then
        return;
    end
    for i = 1, #self.ItemList do
        if self.ItemList[i].transform == tr then
            self.ItemList[i]:UpdateContent(self.ItemDatas[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end

    local bi = SecretAreaAchieveItem.new(tr.gameObject, self.m_ItemRoot, self.AtlasPath);
    bi.prefab:SetActive(true);
    bi:UpdateContent(self.ItemDatas[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.ItemList, bi);
end

function SecretAreaAchieveWidget:Start()

end

function SecretAreaAchieveWidget:OnEnable()
    CS.StarProject.GlobalEvent.OnPersonSecretAchieveChange:AddListener(OnAchieveChangeFunc);
end

function SecretAreaAchieveWidget:OnDisable()
    CS.StarProject.GlobalEvent.OnPersonSecretAchieveChange:RemoveListener(OnAchieveChangeFunc);
end

function SecretAreaAchieveWidget:OnDestroy()

end

function SecretAreaAchieveWidget:OnMenusItemClick(index)
    self.TabIndex = index;
    self.m_Menus:Check(index);

    self:SetInfo();
end

function SecretAreaAchieveWidget:SetInfo()
    self.ItemDatas = {}
    local list = ConfigDataAccessor.SeasonAchieve
    if list ~= nil then
        for k, v in pairs(list) do
            if v ~= nil then
                if v.AchieveType == self.TabIndex then
                    table.insert(self.ItemDatas, {
                        config = v,
                        data = self.MissionDatas[v.AchieveID]
                    })
                end
            end
        end
    end

    -- table.sort(self.ItemDatas, function(p1, p2)
    --     return p1.config.AchieveID < p2.config.AchieveID;
    -- end);
    -- 可领取-未完成-已完成
    table.sort(self.ItemDatas, function(p1, p2)
        local state1 = p1.data.State
        local state2 = p2.data.State

        local achieveID1 = p1.config.AchieveID
        local achieveID2 = p2.config.AchieveID

        if state1 and state2 then
            -- 降序
            return achieveID1 < achieveID2;
        else
            if state1 and state2 == false then
                -- 降序
                return false;
            elseif state1 == false and state2 then
                return true;
            else
                local canGetP1 = p1.data.Value >= p1.config.TargetNum
                local canGetP2 = p2.data.Value >= p2.config.TargetNum
                if canGetP1 and canGetP2 then
                    return achieveID1 < achieveID2;
                elseif canGetP1 and canGetP2 == false then
                    return true;
                elseif canGetP1 == false and canGetP2 then
                    return false;
                else
                    return achieveID1 < achieveID2;
                end
            end
        end
    end);

    self.m_ScrollView:ContinueScroll(#self.ItemDatas);
end

function SecretAreaAchieveWidget:OnFresh(arg)
    self.MissionDatas = {}
    local list = ConfigDataAccessor.SeasonAchieve
    if list ~= nil then
        for k, v in pairs(list) do
            if v ~= nil then
                local value = 0;
                if arg ~= nil and arg.AchieveList ~= nil then
                    if arg.AchieveList[v.SubType] ~= nil then
                        value = arg.AchieveList[v.SubType]
                    end
                end
                local state = false;
                if arg ~= nil and arg.GotAchieveList ~= nil then
                    if arg.GotAchieveList[k] ~= nil then
                        state = arg.GotAchieveList[k]
                    end
                end
                self.MissionDatas[k] = {
                    config = v,
                    State = state,
                    Value = value
                }
            end
        end
    end
    self.m_Menus:Init(self.TabIndex, function(index)
        self:OnMenusItemClick(index);
    end);
    self:SetInfo();
end

function SecretAreaAchieveWidget:OnOpen(arg)
    self.TabIndex = 1;
    self:OnFresh(arg);
end

function SecretAreaAchieveWidget:OnClose(arg)
    LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "RefreshTickets")
end

return SecretAreaAchieveWidget

SecretAreaSuccessTipsWidget=class("SecretAreaSuccessTipsWidget", UIPanel)

SecretAreaSuccessTipsWidget.IsOpen = false;

function SecretAreaSuccessTipsWidget:Awake()
	self.transform = self.prefab.transform;
	self.gameObject = self.transform.gameObject;
	self.onCloseDestroy(true)

	self.Time = self.transform:Find("bg/Time"):GetComponent(typeof(Text));
	
	-- update事件
	self.UpdateFun = function ()
		self:UpdateTime();
	end
end

function SecretAreaSuccessTipsWidget:Start()

end

function SecretAreaSuccessTipsWidget:OnEnable()

end

function SecretAreaSuccessTipsWidget:OnDisable()

end

function SecretAreaSuccessTipsWidget:OnDestroy()
	
end

function SecretAreaSuccessTipsWidget:OnOpen(arg)
	MonoHelper.RemoveUpdateListener(self.UpdateFun);
	self.IsOpen = true;
	self.m_Countdown = 3;
	--self.Time.text = "用时" .. SecToStringColon2(arg.ResData.UseTime / 1000); -- 通关时间
	self.Time.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaSuccessTipsWidget_Time"), {SecToStringColon2(arg.ResData.UseTime / 1000)});

	MonoHelper.AddUpdateListener(self.UpdateFun);
end

function SecretAreaSuccessTipsWidget:OnClose(arg)
	self.IsOpen = false;
	MonoHelper.RemoveUpdateListener(self.UpdateFun);
end

-- 更新倒计时
function SecretAreaSuccessTipsWidget:UpdateTime()
	if self.m_Countdown > 0 and self.IsOpen then
		local fixedDeltaTime = Time.deltaTime;
		self.m_Countdown = self.m_Countdown - fixedDeltaTime;
		if self.m_Countdown <= 0 then
			self.Close();
		end
	end
end


return  SecretAreaSuccessTipsWidget;

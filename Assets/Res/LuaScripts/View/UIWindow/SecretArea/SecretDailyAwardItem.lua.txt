---@class SecretDailyAwardItem
SecretDailyAwardItem = class("SecretDailyAwardItem", UIPanel)
SecretDailyAwardItem.Name = nil;
SecretDailyAwardItem.m_ItemPreviewGo = nil;
SecretDailyAwardItem.m_ItemPreview = nil;
SecretDailyAwardItem.Image = nil;
function SecretDailyAwardItem:ctor(go, parent, atlas)
    self.prefab = go;
    self.AtlasPath = atlas;
    self.Name = self:Find("Tile"):GetComponent(typeof(Text));
    self.Image = self:Find("Image"):GetComponent(typeof(Image));
    self.m_ItemPreviewGo = self:Find("ItemPreview").gameObject;
    self.m_ItemPreview = StructAccessor.ItemPreview.new(self.m_ItemPreviewGo);
end

function SecretDailyAwardItem:GetColor(rank)
    if rank == 1 then
        return "FF000E"
    elseif rank == 2 then
        return "00A0FF"
    elseif rank == 3 then
        return "00D450"
    else
        return "ffffeb"
    end
end
-- 数据更新
function SecretDailyAwardItem:UpdateContent(data)
    self.Name.text = data.Tile
    local colorname = self:GetColor(data.RankID);
    local color = ColorDefine.Instance:GetColor(colorname);
    self.Image.color = color

    local items = data.RewardID;
    local nums = data.RewardNum;

    local CostItems = {}
    for i = 1, #items do
        table.insert(CostItems, {
            ItemID = items[i],
            ItemNum = nums[i]
        })
    end
    self.m_ItemPreview:SetItems(CostItems);
end

return SecretDailyAwardItem;

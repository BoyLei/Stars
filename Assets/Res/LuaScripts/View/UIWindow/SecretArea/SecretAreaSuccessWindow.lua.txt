function SecretAreaSuccessWindow:OnOpen(arg)
	self.IsOpen = true;
	self:InitData(arg);
	-- 倒计时
	MonoHelper.RemoveUpdateListener(self.UpdateFun);
	MonoHelper.AddUpdateListener(self.UpdateFun);

function SecretAreaSuccessWindow:OnClose(arg)

	MonoHelper.RemoveUpdateListener(self.UpdateFun);
	self:CloseWindow();
end

-- 更新倒计时
function SecretAreaSuccessWindow:UpdateTime()
	if self.m_Countdown > 0 and self.IsOpen then
		local fixedDeltaTime = Time.deltaTime;
		self.m_Countdown = self.m_Countdown - fixedDeltaTime;
		if self.m_Countdown <= 0 then
			-- 倒计时结束触发自动退出副本
			self:OnClickExitJBtn();
		end
	end
end

-- 设置层数
function SecretAreaSuccessWindow:SetFloors(UnlockMaxFloor)
	local startFloor = UnlockMaxFloor;
	if startFloor >= self.m_MaxFloors then
		startFloor = startFloor - 2; -- 如果是最后一层就【-2】
	elseif startFloor > 1 then
		startFloor = startFloor - 2; -- 如果大于1就【-1】[-1之前是把当前选中放在了最左边]【-2是放中间了】
	else
		-- 否则就不变
	end
	self.SuperScrollViewController:ContinueScroll(#self.m_Floors, false, startFloor);
end

-- 设置奖励
function SecretAreaSuccessWindow:SetDropRewardData(CommonDrop)
	-- 奖励显示
	for i = 0, CommonDrop.Count - 1 do
		local infos = CommonDrop[i].Infos;
		for j = 0, infos.Count - 1 do
			local data = infos[j];
			local itemCfg = ConfigDataAccessor.Item[data.Itemid];
			local quality = 1;
			local itemType = 1;
			if itemCfg ~= nil then
				quality = itemCfg.Quality;
				itemType = itemCfg.ItemType;
			end
			--Log("11111111111道具id=" .. data.Itemid .. ",道具品质=" .. quality)

			if self.m_AllArward[data.Itemid] == nil then
				local item = {
					ItemID = data.Itemid,
					Itemid = data.Itemid,
					ItemNum = data.ItemNum,
					Quality = quality,
					EntityID = data.EntityID,
					IsFirstPassReward = false
				};
				self.m_AllArward[data.Itemid] = item;
			else
				if itemType == 2 then
					-- 如果是角色装备，不叠加
					local item = {
						ItemID = data.Itemid,
						Itemid = data.Itemid,
						ItemNum = data.ItemNum,
						Quality = quality,
						EntityID = data.EntityID,
						IsFirstPassReward = false
					};
					local index = data.Itemid .. i .. j;
					self.m_AllArward[index] = item;
				else
					self.m_AllArward[data.Itemid].ItemNum = self.m_AllArward[data.Itemid].ItemNum + data.ItemNum;
				end
			end
		end
	end
end

function SecretAreaSuccessWindow:SetFirstRewardData(FirstReward)
	if FirstReward ~= nil and FirstReward.Data ~= nil then
		local index = 1;
		for k, v in pairs(FirstReward.Data) do
			local itemCfg = ConfigDataAccessor.Item[k];
			local quality = 1;
			local itemType = 1;
			if itemCfg ~= nil then
				quality = itemCfg.Quality;
				itemType = itemCfg.ItemType;
			end
			--Log("22222222222道具id=" .. k .. ",道具品质=" .. quality)

			if self.m_AllArward[k] == nil then
				local item = {
					ItemID = k,
					Itemid = k,
					ItemNum = v,
					Quality = quality,
					EntityID = 0,
					IsFirstPassReward = true
				};
				self.m_AllArward[k] = item;
			else
				if itemType == 2 then
					-- 如果是角色装备，不叠加
					local item = {
						ItemID = k,
						Itemid = k,
						ItemNum = v,
						Quality = quality,
						EntityID = 0,
						IsFirstPassReward = true
					};
					local index = k .. v .. index;
					self.m_AllArward[index] = item;
				else
					self.m_AllArward[k].ItemNum = self.m_AllArward[k].ItemNum + v;
				end
				index = index + 1;
			end
		end
	end

end

-- 奖励排序，高-》低
function SecretAreaSuccessWindow:SortReward(rewardList)
	-- 生成一个用于排序的临时数组
	local tempArray = {}
	for key, value in pairs(rewardList) do
		table.insert(tempArray, value)
	end
	-- 定义比较函数，按照Quality从大到小排序
	local function compare(a, b)
		if a.IsFirstPassReward and b.IsFirstPassReward then
			return a.Quality > b.Quality
		else
			if a.IsFirstPassReward and b.IsFirstPassReward == false then
				return true;
			elseif a.IsFirstPassReward == false and b.IsFirstPassReward then
				return false;
			else
				return a.Quality > b.Quality
			end
		end
	end
	-- 对临时数组进行排序
	table.sort(tempArray, compare)
	return tempArray
end

function SecretAreaSuccessWindow:OnItemRefesh(tr, idx)
	if self.SuperScrollViewControllerItems == nil then
		return;
	end
	local floor = self.m_Floors[idx + 1]; -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
	local maxFloor = 1;
	if self.m_CueUnLockMaxFloor > 0 then
		maxFloor = self.m_CueUnLockMaxFloor;
	end
	local isLock = floor > maxFloor;
	for i = 1, #self.SuperScrollViewControllerItems do
		if self.SuperScrollViewControllerItems[i].prefab.transform == tr then
			self.SuperScrollViewControllerItems[i]:UpdateContent(floor, self.SelectIndex, self.m_CueUnLockMaxFloor,
				isLock); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
			return;
		end
	end

	local sp = SecretAreaFloorItem.new(tr.gameObject, self.SuperScrollViewControllerRoot, self.AtlasPath,
		self.FloorClickFun, 1);
	sp.prefab:SetActive(true);
	sp:UpdateContent(floor, self.SelectIndex, self.m_CueUnLockMaxFloor, isLock);
	table.insert(self.SuperScrollViewControllerItems, sp);
end

function SecretAreaSuccessWindow:SetAllItemSelectState()
	for i = 1, #self.SuperScrollViewControllerItems do
		local isSelect = false;
		if self.SuperScrollViewControllerItems[i].Index == self.SelectIndex then
			isSelect = true;
		end
		self.SuperScrollViewControllerItems[i]:OnSelect(isSelect);
	end
end

function SecretAreaSuccessWindow:OnSelect(selectIndex)
	self.SelectIndex = selectIndex;
	self:SetAllItemSelectState();
	-- 修改当前层数信息
	self:SetCurrentFloorData();
end

-- 设置当前层数信息
function SecretAreaSuccessWindow:SetCurrentFloorData()
	-- 当前层数
	local floor = self.SelectIndex;
	-- 当前层数配置
	self.m_CurrentFloorCfg = ConfigDataAccessor.SecretCopy[floor];
	-- 设置进入按钮的状态
	self:SetEnterBtnState(floor);
	-- 设置门票数量
	self:SetTicketsNum();
end

-- 设置进入按钮的状态
function SecretAreaSuccessWindow:SetEnterBtnState(floor)
	-- 按钮设置
	local alpha = 0.3;
	if self.m_CueUnLockMaxFloor == 0 and floor == 1 then
		alpha = 1;
	elseif self.m_CueUnLockMaxFloor > 0 and self.m_CueUnLockMaxFloor >= floor then
		alpha = 1;
	end

	if self.m_CurrentFloorCfg.Grade > self.m_CurPlayerLevel then
		alpha = 0.3;
	end

	-- 需求：【客户端】首次通关拟态梦境后，隐藏界面的下一层按钮，玩家单击退出后直接回到主hud场景（特殊处理）
	-- 当前是第一层&&历史最高也是第一层
	if self.m_ChallengeFloor == 1 and self.m_LastUnlockFloor == 2 then
		alpha = 0;
	end

	self.JBtnSureCanvasGroup.alpha = alpha;
	self.JBtnSureCanvasGroup.blocksRaycasts = alpha == 1;

end

-- 设置门票数量
function SecretAreaSuccessWindow:SetTicketsNum(floor)
	if self.m_CurrentFloorCfg == nil then
		return;
	end
	self.m_TicketsCount = BusinessManager.Instance:GetCurrencyNum(self.m_CurrentFloorCfg.TicketID);
end

-- 进入提示回调
function SecretAreaSuccessWindow:EnterTipsCallBack()
	if self.m_TicketsCount >= self.m_NeedTicketCount then
		LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "SendEnterPersonSercetReq", self.SelectIndex);
	else
		local cfg = ConfigDataAccessor.Item[self.m_CurrentFloorCfg.TicketID];
		local name = self.m_CurrentFloorCfg.TicketID;
		if cfg ~= nil then
			name = cfg.Name;
		end
		local isFirst = false;
		for i, v in pairs(ConfigDataAccessor.TicketSynthesize) do
			if v.ID == self.m_CurrentFloorCfg.TicketID then
				isFirst = true;
			end
			break
		end
		if isFirst then
			CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(StringFormat(
					LanguageHelp:GetLanguageByKey("Local_Str_ShopItem_buyErrorMsg"), {name}));
			return;
		end
		-- 打开合成提示
		UIAPI.ShowMsgBox(33, function(EventName)
				if EventName == "SURE" then
					self:TicketTipsCallBack();
				end
			end, name);
	end
end

-- 点击翻页按钮
function SecretAreaSuccessWindow:OnClickPagingJBtn(num)
	-- 计算偏移量
	local offsetNum = self.m_VisibleMiddleFloor - 2;
	if num < 0 then
		offsetNum = -(self.m_VisibleFloorLength + self.m_VisibleMiddleFloor - 1);
	end
	-- 判断是否会过边界
	local middleFloor = self:GetMiddleFloorIndex();
	local startFloor = middleFloor + offsetNum;
	if startFloor < 0 then
		startFloor = 0;
	end
	if (startFloor + self.m_VisibleFloorLength) > self.m_MaxFloors then
		startFloor = self.m_MaxFloors - self.m_VisibleFloorLength;
	end
	self.SuperScrollViewController:NevigateByIdx(startFloor);
end

-- 获取当前中间的层数下标
function SecretAreaSuccessWindow:GetMiddleFloorIndex()
	if self.SuperScrollViewControllerItems == nil or self.SuperScrollViewControllerItems == {} then
		return 0;
	end
	local minX = 99999;
	local middleIndex = 0;
	local floorMiddlePointX = math.abs(self.MiddleGod.transform.position.x); -- 层数item中心点X坐标
	for i = 1, #self.SuperScrollViewControllerItems do
		local child = self.SuperScrollViewControllerItems[i];
		if child ~= nil then
			local itemWorldPosX = child.Scale.transform.position.x;
			local x = math.abs(math.abs(itemWorldPosX) - floorMiddlePointX);
			if x < minX then
				middleIndex = child.Index;
				minX = x;
			end
		end
	end
	return middleIndex;
end

-- 初始化宝箱百分比
function SecretAreaTaskWidget:InitBoxsTimePercent()
    self.m_BoxsTimePercent = {};
    for k, v in pairs(ConfigDataAccessor.SecretSettle) do
        local percent = v.RemainTime * 1000 / self.m_MaxTime;
        table.insert(self.m_BoxsTimePercent, percent);
    end
end

function SecretAreaTaskWidget:SetBoxsPos()
    local v2 = Vector2.zero;
    for i = 1, #self.PointOns do
        local item = self.PointOns[i];
        local data = self.m_BoxsTimePercent[i];
        if item ~= nil and data ~= nil then
            v2.x = (data * (self.m_MaxTime / 1000)) - 6;
            local rect = item.transform:GetComponent(typeof(RectTransform));
            rect.anchoredPosition = v2;
            item.enabled = true;
        end
    end
end

-- 计算宝箱点的透明度
function SecretAreaTaskWidget:CalculateBoxPointAlpha()
    for i = 1, #self.PointOns do
        local item = self.PointOns[i];
        if item ~= nil then
            local index = #self.PointOns + 1 - i;
            item.enabled = self.m_BoxCount >= index;
        end
    end
    -- self.BoxAnim:SetActive(false);
    -- self.BoxAnim:SetActive(true);
    self.BoxAnim:Play("ImgBox_Anim");
    DOVirtual.DelayedCall(0.35, function()
        self.BoxCount.text = "x" .. self.m_BoxCount;
    end);
    if self.m_BoxCount > 0 then
        local tipsStr = LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaTaskWidget_BoxTimeTips2");
        CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(tipsStr);
    end
end

-- 计算宝箱数量
function SecretAreaTaskWidget:CalculateBoxCount()
    for k, v in pairs(ConfigDataAccessor.SecretSettle) do
        local child = v;
        if self.m_Countdown >= child.RemainTime * 1000 then
            if child.BoxNum ~= self.m_BoxCount then
                self.m_BoxCount = child.BoxNum;
                self.m_BoxCountTips = child.BoxNum;
            end
            return;
        end
    end
    self.m_BoxCount = 0;
end

function SecretAreaTaskWidget:SetBoxTimeTipsCache(boxNum, time)
    local key = boxNum .. "_" .. time;
    local isContains = table.ContainsValue(self.BoxTimeTipsCache, key);
    if isContains == false then
        table.insert(self.BoxTimeTipsCache, key)
    end
end

function SecretAreaTaskWidget:GetBoxTimeTipsCache(boxNum, time)
    local key = boxNum .. "_" .. time;
    return table.ContainsValue(self.BoxTimeTipsCache, key);
end

-- 宝箱减少提示
function SecretAreaTaskWidget:BoxTimeTips()
    if self.m_BoxCountTips == 1 then
        return;
    end
    for k, v in pairs(ConfigDataAccessor.SecretSettle) do
        local child = v;
        if child.BoxNum == self.m_BoxCountTips then
            local tipsTime5 = (self.m_Countdown - 5 * 1000) <= (child.RemainTime * 1000);
            local tipsTime10 = (self.m_Countdown - 10 * 1000) <= (child.RemainTime * 1000);
            local tipsStr = "";
            if tipsTime5 and self:GetBoxTimeTipsCache(child.BoxNum, 5) == false then
                tipsStr =
                    StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaTaskWidget_BoxTimeTips1"), {5});
                self:SetBoxTimeTipsCache(child.BoxNum, 5);
            elseif tipsTime10 and self:GetBoxTimeTipsCache(child.BoxNum, 10) == false then
                tipsStr = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaTaskWidget_BoxTimeTips1"),
                    {10});
                self:SetBoxTimeTipsCache(child.BoxNum, 10);
            end
            if tipsStr ~= "" then
                CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(tipsStr);
            end
        end
    end
end

-- 更新任务阶段
function SecretAreaTaskWidget:UpdateTaskStage(Stage)
    if self.m_TaskStageServer == Stage then
        return;
    end

    self.m_TaskStageServer = Stage;
    if self.m_TaskStageServer == 1 then
        self:SetTaskState(E_SecretTaskStageType.Task2);
    elseif self.m_TaskStageServer == 2 then
        self:SetTaskState(E_SecretTaskStageType.Task3);
    end

    if self.m_TaskStageServer == 2 then
        LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "OpenEctypeWarningWidget",
            {2, LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaTaskWidget_BOSSTips")});
        -- CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaTaskWidget_BOSSTips"));
        -- 通知小地图显示boss位置信息
        CS.StarProject.GlobalEvent.OnPersonSecretBossAppearRet:Invoke(true);
    end

    -- self:SetTask1State();	-- 阶段是1时，服务器标识【击杀任务】进行中
    -- self:SetTask2State(false);	-- 阶段是2时，服务器标识BOSS刚开始
end

-- 更新怪物进度
function SecretAreaTaskWidget:UpdateMonsterProgress(progress)
    -- 这边过滤是因为，如果已经超过100%进度了，服务器还会同步100%+的数
    -- 那我们这边就过滤，，不管了
    -- -- if self.m_Task1Stage == 0 then
    -- -- 	self:SetMainTaskShow(true)
    -- -- 	self:SetOtherTaskShow(false, 1)
    -- -- 	self.Countdown.gameObject:SetActive(true);
    -- -- end
    -- if self.m_Task1Stage == E_SecretTaskStageType.Task0 or self.m_Task1Stage == E_SecretTaskStageType.Task1 then
    -- 	self.Countdown.gameObject:SetActive(true);
    -- end
    if self.m_Task1Progress >= self.m_MaxProgress then
        return;
    end
    local pro = progress;
    if progress > self.m_MaxProgress then
        pro = self.m_MaxProgress;
    end
    self.m_Task1Progress = pro;
    local showProgress = self.m_Task1Progress / self.m_MaxProgress;
    if showProgress > 1 then
        showProgress = 1;
    end
    -- self.MonsterProgress.fillAmount = showProgress;
    -- self.MonsterProgress:DOFillAmount(showProgress, 2);
    if self.SliderTween ~= nil then
        self.SliderTween:Kill();
    end
    if self.SlideAreaTween ~= nil then
        self.SlideAreaTween:Kill();
    end
    if self.m_Task1Progress == 0 then
        self.MonsterProgressSlider.value = 0;
        self.SlideArea:SetActive(false);
    else
        self.SliderTween = UIDOTweenExt.SliderDOValue(self.MonsterProgressSlider, showProgress, 2, false);
        self.SlideArea:SetActive(true);
        self.SlideAreaTween = DOVirtual.DelayedCall(2, function()
            self.SlideArea:SetActive(false);
        end);
    end

    showProgress = showProgress * 100;
    showProgress = math.floor(showProgress);
    self.ProgressNum.text = showProgress .. "%";
    if self.m_Task1Progress >= self.m_MaxProgress then
        -- 秘境任务【进度】完成
        -- self:SetTask1State(true);
        self.Task1Anim:Play("SecretAreaTaskWidget_Task1_Anim_100%");
        DOVirtual.DelayedCall(1, function()
            self:SetTaskState(E_SecretTaskStageType.Task3);
        end);
    end
end

-- 更新倒计时
function SecretAreaTaskWidget:UpdateCountdown(endTime)
    -- 当前时间-结束时间
    self.m_Countdown = endTime - GameManager.Instance:GetServerTimeStamp();
    if self.m_Countdown > 0 then
        self.Countdown.text = SecToStringColon2(self.m_Countdown / 1000);
    else
        -- self.Countdown.text = "0分0秒";
        self.Countdown.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Time1"), {0, 0});
    end
    local ratio = self.m_Countdown / self.m_MaxTime;
    if ratio > 1 then
        ratio = 1;
    end
    self.TimeProgress:SetWidth(self.m_ProgressMaxWidth * ratio);

    if self.m_Task1Stage == E_SecretTaskStageType.Task0 or self.m_Task1Stage == E_SecretTaskStageType.Task1 then
        self.Countdown.text = LanguageHelp:GetLanguageByKey("Local_Str_no_start")
    end
end

function SecretAreaTaskWidget:InitCountdown()
     -- 当前时间-结束时间
     self.m_Countdown = self.m_MaxTime;
     if self.m_Countdown > 0 then
         self.Countdown.text = SecToStringColon2(self.m_Countdown / 1000);
     else
         -- self.Countdown.text = "0分0秒";
         self.Countdown.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Time1"), {0, 0});
     end
     local ratio = self.m_Countdown / self.m_MaxTime;
     if ratio > 1 then
         ratio = 1;
     end
     self.TimeProgress:SetWidth(self.m_ProgressMaxWidth * ratio);
 
     if self.m_Task1Stage == E_SecretTaskStageType.Task0 or self.m_Task1Stage == E_SecretTaskStageType.Task1 then
         self.Countdown.text = LanguageHelp:GetLanguageByKey("Local_Str_no_start")
     end
end

-- 更新倒计时
function SecretAreaTaskWidget:UpdateTime()
    if self.m_Countdown > 0 and self.IsOpen and self.m_AllTaskFinish == false and self.m_StartCountdown then
        local fixedDeltaTime = Time.deltaTime * 1000;
        self.m_Countdown = self.m_Countdown - fixedDeltaTime;
        if self.m_Countdown <= 0 then
            self.Countdown.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Time1"), {0, 0});
            self.TimeProgress:SetWidth(0);
        else
            local tipsTime5 = (self.m_Countdown - 5 * 1000) <= 0;
            local tipsTime10 = (self.m_Countdown - 10 * 1000) <= 0;
            local tipsStr = "";
            if tipsTime5 and self:GetBoxTimeTipsCache(99, 5) == false then
                tipsStr =
                    StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaTaskWidget_EndTimeTips1"), {5});
                self:SetBoxTimeTipsCache(99, 5);
            elseif tipsTime10 and self:GetBoxTimeTipsCache(99, 10) == false then
                tipsStr = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaTaskWidget_EndTimeTips1"),
                    {10});
                self:SetBoxTimeTipsCache(99, 10);
            end
            if tipsStr ~= "" then
                CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(tipsStr);
            end

            self.Countdown.text = SecToStringColon2(self.m_Countdown / 1000);
            local ratio = self.m_Countdown / self.m_MaxTime;
            if ratio > 1 then
                ratio = 1;
            end
            self.TimeProgress:SetWidth(self.m_ProgressMaxWidth * ratio);
        end
        local oldBoxCount = self.m_BoxCount;
        self:CalculateBoxCount();
        self:BoxTimeTips();
        if oldBoxCount ~= self.m_BoxCount then
            -- 宝箱数量有变化
            self:CalculateBoxPointAlpha();
        end
    end
end

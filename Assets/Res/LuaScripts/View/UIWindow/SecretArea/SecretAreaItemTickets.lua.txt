---@class SecretAreaItemTickets

local SecretAreaItemTickets = {}

SecretAreaItemTickets.__index = SecretAreaItemTickets

function SecretAreaItemTickets:New(_go)
    local self = {}
    setmetatable(self, SecretAreaItemTickets)

    self.gameObject = _go
    self.transform = _go.transform

    self.Icon = self.transform:Find("icon"):GetComponent(typeof(Image))
    self.Num = self.transform:Find("num"):GetComponent(typeof(Text))
    self.Floor = self.transform:Find("floor"):GetComponent(typeof(Text))

    self.ItemID = 0

    return self
end

function SecretAreaItemTickets:SetItem(itemId)
    self.ItemID = itemId
    if self.ItemID == 0 then
        return
    end

    local cfg = ConfigDataAccessor.Item[self.ItemID]
    if cfg ~= nil then
        -- 加载图片
        ResourceHelper:LoadIconAsync(self.Icon, "", cfg.Icon, false)
    end

    local startFloor = 0;
    local endFloor = 0;
    for i = 1, #ConfigDataAccessor.SecretCopy do
        local child = ConfigDataAccessor.SecretCopy[i];
        if child ~= nil then
            local TicketID = child.TicketID;
            if TicketID == self.ItemID then
                if startFloor == 0 then
                    startFloor = child.Floor;
                end
                endFloor = child.Floor;
            end
        end
    end
    --self.Floor.text = "第" .. startFloor .. "层~第" .. endFloor .. "层";
    self.Floor.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_limitFloor"), {startFloor}) .. "~" .. StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_limitFloor"), {endFloor});

    if startFloor == 0 and startFloor == endFloor then
        self.gameObject:SetActive(false);
    end
end

function SecretAreaItemTickets:UpdateNum()
    if self.ItemID == 0 then
        return
    end
    -- 获取背包里面的数量
    self.Num.text = BusinessManager.Instance:GetCurrencyNum(self.ItemID)
end

return SecretAreaItemTickets

SecretAreaFailWindow = class("SecretAreaFailWindow", UIPanel)


SecretAreaFailWindow.IsOpen = false;

SecretAreaFailWindow.m_IsRevive = false;	-- 是否复活
SecretAreaFailWindow.m_Countdown = 20;

function SecretAreaFailWindow:Awake()
	self.transform = self.prefab.transform;
	self.gameObject = self.transform.gameObject;
	self.onCloseDestroy(false)

	local FrameTransForm = self.transform:Find("Cut4Cam90/AssemblyRatio2").transform;
	self.Bg = FrameTransForm:Find("Center/bg").gameObject;
	self.Title = FrameTransForm:Find("Center/bg/Title"):GetComponent(typeof(Text));
	self.Title2 = FrameTransForm:Find("Center/bg/Title2").gameObject;
	self.MapName = FrameTransForm:Find("Center/bg/MapName"):GetComponent(typeof(Text));
	
	-- 失败【结算】
	self.FailPanel = FrameTransForm:Find("Center/FailPanel").gameObject;
	self.FailTipsCountdown = FrameTransForm:Find("Center/FailPanel/Countdown"):GetComponent(typeof(Text));
	-- 【确定】【J按钮】
	self.JBtnSure = FrameTransForm:Find("Center/FailPanel/JBtnSure"):GetComponent(typeof(JButton));
	self.JBtnSureFun = function()
		self:OnClickSureJBtn();
	end
	self.JBtnSure.OnClick = self.JBtnSureFun;
	-- 失败【复活】
	self.RevivePanel = FrameTransForm:Find("Center/RevivePanel").gameObject;
	self.ReviveTipsCountdown = FrameTransForm:Find("Center/RevivePanel/Countdown"):GetComponent(typeof(Text));
	-- 【复活】【J按钮】
	self.JBtnRevive = FrameTransForm:Find("Center/RevivePanel/JBtnRevive"):GetComponent(typeof(JButton));
	self.JBtnReviveFun = function()
		self:OnClickReviveJBtn();
	end
	self.JBtnRevive.OnClick = self.JBtnReviveFun;
	-- 倒计时
	
	-- update事件
	self.UpdateFun = function ()
		self:UpdateTime();
	end
	MonoHelper.AddUpdateListener(self.UpdateFun);
end

function SecretAreaFailWindow:BindModule(module)
	self.module = module;
end

function SecretAreaFailWindow:Start()

end

function SecretAreaFailWindow:OnEnable()
	CS.StarProject.GlobalEvent.OnPersonSecretRevive:AddListener(function (isFinish)
			self:HandleOnPersonSecretRevive(isFinish);
		end);
end

function SecretAreaFailWindow:OnDisable()
	CS.StarProject.GlobalEvent.OnPersonSecretRevive:RemoveListener(function (isFinish)
			self:HandleOnPersonSecretRevive(isFinish);
		end);
end

function SecretAreaFailWindow:OnDestroy()
	MonoHelper.RemoveUpdateListener(self.UpdateFun);
end

function SecretAreaFailWindow:OnOpen(arg)
	self.IsOpen = true;
	self:InitData(arg);
end

function SecretAreaFailWindow:OnClose(arg)
	self.IsOpen = false;
	self:CloseWindow();
end

function SecretAreaFailWindow:InitData(arg)
	self.m_IsRevive = arg.IsRevive;
	
	self.FailPanel:SetActive((not self.m_IsRevive));
	self.RevivePanel:SetActive(self.m_IsRevive);
	self.Title2:SetActive(self.m_IsRevive);
	self.MapName.text = "";
	self.m_Countdown = SystemConstConfigs.SecretCopy_ExitTime;
	local v3 = Vector3.zero;
	if self.m_IsRevive then
		-- 复活界面
		local reviveCount = arg.ReviveCount + 1;
		local reviveCfg = ConfigDataAccessor.SecretRevive[reviveCount];
		if reviveCfg ~= nil then
			self.m_Countdown = reviveCfg.DecSecond;
		end
		--self.ReviveTipsCountdown.text = "时间进度 +" .. self.m_Countdown .. "s";
		self.ReviveTipsCountdown.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaFailWindow_ReviveTipsCountdown"), {self.m_Countdown});

		v3.y = 110;
		self.Bg.transform.localPosition = v3;
		self.Title.text =LanguageHelp:GetLanguageByKey("Local_Str_Waiting_for_resurrection")
	else
		-- 失败界面
		local floor = arg.Floor;
		--self.MapName.text = "拟态梦境：第" .. floor .. "层";
		self.MapName.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaFailWindow_MapName"), {floor});

		--self.FailTipsCountdown.text = "将在 " .. self.m_Countdown .. "s 后离开副本";
		self.FailTipsCountdown.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaFailWindow_FailTipsCountdown"), {self.m_Countdown});

		v3.y = 285;
		self.Bg.transform.localPosition = v3;
		self.Title.text =LanguageHelp:GetLanguageByKey("Local_Str_Challenge_failure")
	end
	
	self.PlayAnimation("SecretAreaFailWindow_Open", nil);
end

-- 更新倒计时
function SecretAreaFailWindow:UpdateTime()
	if self.m_Countdown > 0 and self.IsOpen then
		local fixedDeltaTime = Time.deltaTime;
		self.m_Countdown = self.m_Countdown - fixedDeltaTime;
		local times = math.floor(self.m_Countdown + 0.5);
		if self.m_IsRevive then
			-- 复活界面
			--self.ReviveTipsCountdown.text = "时间进度 +" .. times .. "s";
			self.ReviveTipsCountdown.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaFailWindow_ReviveTipsCountdown"), {times});
		else
			---- 失败界面
			--self.FailTipsCountdown.text = "将在 " .. times .. "s 后离开副本";
			self.FailTipsCountdown.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaFailWindow_FailTipsCountdown"), {times});
		end
		
		if self.m_Countdown <= 0 then
			-- 【复活】倒计时结束 服务器会自动复活
			-- 【失败】服务器会主动切场景
			-- 倒计时没有了就自动关闭界面
			if self.m_IsRevive then
				self:OnClickReviveJBtn();
			else
				self:OnClickSureJBtn();
			end
		end
	end
end

function SecretAreaFailWindow:HandleOnPersonSecretRevive(isFinish)
	-- TODO: 失败了，怎么处理？
	self.Close();
end

----------------------------------------------------- 按钮点击事件 【STAR】 ------------------------------------------

-- 点击【复活】按钮
function SecretAreaFailWindow:OnClickReviveJBtn()
	-- 发送复活请求
	-- 这里要等服务器的回复
	--self.Close();
	LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "SendPersonSecretReviveReq", nil);
end

-- 点击【确定】按钮
function SecretAreaFailWindow:OnClickSureJBtn()
	-- 发送退出副本请求
	-- 这里要等服务器的回复
	--self.Close();
	LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "SendLeaveInstanceReq", nil);
end

----------------------------------------------------- 按钮点击事件 【END】 ------------------------------------------

function SecretAreaFailWindow:CloseDestroyWindow()
	self.onCloseDestroy(true)
	self.Close();
	self.module.FailWindow = nil;
end

-- 关闭界面
function SecretAreaFailWindow:CloseWindow()

end


return SecretAreaFailWindow;

local SecretDailyAwardItem = require("View/UIWindow/SecretArea/SecretDailyAwardItem")

SecretAreaRankWidget = class("SecretAreaRankWidget", UIPanel)
SecretAreaRankWidget.IsOpen = false
SecretAreaRankWidget.m_Close = nil;
SecretAreaRankWidget.m_Tips = nil;
SecretAreaRankWidget.m_ScrollView = nil;
SecretAreaRankWidget.m_AwardItem = nil;
SecretAreaRankWidget.m_Cur_Rank = nil;  -- 当前赛季排名
SecretAreaRankWidget.m_Next_Rank = nil; -- 下段位排名
SecretAreaRankWidget.m_Cur_ItemPreview = nil;
SecretAreaRankWidget.m_Next_ItemPreview = nil;
SecretAreaRankWidget.m_Cur_ItemPreviewGo = nil;
SecretAreaRankWidget.m_Next_ItemPreviewGo = nil;

SecretAreaRankWidget.m_Rank = nil;
SecretAreaRankWidget.m_MaxRank = nil;
--数据
SecretAreaRankWidget.ItemList = {};
SecretAreaRankWidget.ItemDatas = {}


function SecretAreaRankWidget:Awake()
	self.m_Close = self:Find("Center/TopCenter/Close"):GetComponent(typeof(JButton));
	self.m_Tips = self:Find("Center/TopCenter/tips"):GetComponent(typeof(Text));
	self.m_ScrollView = self:Find("Center/ScrollView"):GetComponent(typeof(SuperScrollViewController));
	self.m_AwardItem = self:Find("Center/AwardItem").gameObject
	self.m_Cur_Rank = self:Find("Center/CurrentAward/Info"):GetComponent(typeof(Text));
	self.m_Next_Rank = self:Find("Center/NextAward/Rank/Info"):GetComponent(typeof(Text));
	self.m_Cur_ItemPreviewGo = self:Find("Center/CurrentAward/ItemPreview"):GetComponent(typeof(RectTransform));
	self.m_Next_ItemPreviewGo = self:Find("Center/NextAward/Rank/ItemPreview"):GetComponent(typeof(RectTransform));
	self.m_Rank = self:Find("Center/NextAward/Rank").gameObject
	self.m_MaxRank = self:Find("Center/NextAward/MaxRank").gameObject
	self.m_ScrollView:NewScroll(self.m_AwardItem, 0, function(tr, idx)
		self:OnItemRefesh(tr, idx);
	end);

	self.m_Cur_ItemPreview = StructAccessor.ItemPreview.new(self.m_Cur_ItemPreviewGo);
	self.m_Cur_ItemPreview:SetSize(406, 94, 10)

	self.m_Next_ItemPreview = StructAccessor.ItemPreview.new(self.m_Next_ItemPreviewGo);
	self.m_Next_ItemPreview:SetSize(406, 94, 10)

	self.m_Close.OnClick = (function()
		self.Close();
	end);
end

function SecretAreaRankWidget:OnItemRefesh(tr, idx)
	if self.ItemList == nil then
		return;
	end
	for i = 1, #self.ItemList do
		if self.ItemList[i].transform == tr then
			self.ItemList[i]:UpdateContent(self.ItemDatas[idx + 1]); --idx是从0开始，如果ItemDataList是lua的table需要idx+1
			return
		end
	end

	local bi = SecretDailyAwardItem.new(tr.gameObject, self.m_ItemRoot, self.AtlasPath);
	bi.prefab:SetActive(true);
	bi:UpdateContent(self.ItemDatas[idx + 1]); --idx是从0开始，如果ItemDataList是lua的table需要idx+1
	table.insert(self.ItemList, bi);
end

function SecretAreaRankWidget:Start()
end

function SecretAreaRankWidget:OnEnable()
end

function SecretAreaRankWidget:OnDisable()
end

function SecretAreaRankWidget:OnDestroy()
end

function SecretAreaRankWidget:OnOpen(arg)
	self.IsOpen = true
	self:InitData(arg);
end

function SecretAreaRankWidget:InitData(arg)
	self.ItemDatas = {}
	local CurSeasonID = arg.CurSeasonID;
	if CurSeasonID == 0 or CurSeasonID == nil then
		CurSeasonID = 1;
	end
	local RankAwardID = 1001;
	local SeasonSecretCfg = ConfigDataAccessor.SeasonSecret[CurSeasonID];
	if SeasonSecretCfg ~= nil then
		RankAwardID = SeasonSecretCfg.RankAward;
	end

	local rank = arg.Rank;
	if rank == 0 or rank == nil then
		rank = 99999;
	end

	local curCfg = nil;
	local nextCfg = nil;

	local list = ConfigDataAccessor.SecretRankClient[RankAwardID];
	if list ~= nil then
		local count = #list;
		for i = 1, #list do
			local dt = list[i]
			if dt ~= nil then
				if rank <= dt.Rank and curCfg == nil then
					curCfg = dt;
				end
				if curCfg == nil and i == count then
					-- 被迫选中最后一个
					nextCfg = dt;
				end
				table.insert(self.ItemDatas,
					{
						Tile = dt.RankTitle,
						RankID = dt.Rank,
						RewardID = dt.RewardID,
						RewardNum = dt.RewardNum
					})
			end
		end
	end

	self.m_ScrollView:ContinueScroll(#self.ItemDatas);

	self:ShowCurrentRank(curCfg, rank)
	if curCfg ~= nil and curCfg.Rank == 1 then
		self.m_Rank:SetActive(false)
		self.m_MaxRank:SetActive(true)
		--self:ShowNextRank(list[curid]);
	else
		self.m_Rank:SetActive(true)
		self.m_MaxRank:SetActive(false)
		if nextCfg == nil and curCfg ~= nil then
			nextCfg = list[curCfg.RankID - 1];
		end
		self:ShowNextRank(nextCfg);
	end

	self:SetRankTime();
end

function SecretAreaRankWidget:ShowCurrentRank(data, rank)
	self.m_Cur_Rank.text = rank
	local CostItems = {}
	if data ~= nil then
		local items = data.RewardID;
		local nums = data.RewardNum;
		for i = 1, #items do
			table.insert(CostItems, { ItemID = items[i], ItemNum = nums[i] })
		end
	end

	self.m_Cur_ItemPreview:SetItems(CostItems);
end

function SecretAreaRankWidget:ShowNextRank(data)
	if data == nil then
		return
	end
	self.m_Next_Rank.text = data.Rank
	local items = data.RewardID;
	local nums = data.RewardNum;
	local CostItems = {}
	for i = 1, #items do
		table.insert(CostItems, { ItemID = items[i], ItemNum = nums[i] })
	end

	self.m_Next_ItemPreview:SetItems(CostItems);
end

function SecretAreaRankWidget:SetRankTime()
	local now = math.floor(TimeUtils.GetServerTimeNow() / 1000);
	local tiemTable = FormatUnixTime2Date(now);
	if tiemTable ~= nil then
		local month = tiemTable.month;
		local day = tiemTable.day;
		--local text = "赛季时间：" .. month .. ".1~" .. month .. ".15（赛季结束后可领取奖励）";
		local text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaRankWidget_Tips1"), {month, month});

		if day > 15 then
			local lastDay = GetTotalDaysByYearAndMonth(tiemTable.year, month);
			--text = "赛季时间：" .. month .. ".16~" .. month .. "." .. lastDay .. "（赛季结束后可领取奖励）";
		 	text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaRankWidget_Tips2"), {month, month, lastDay});
		end
		self.m_Tips.text = text;
	else
		self.m_Tips.text = "";
	end
end

function SecretAreaRankWidget:OnClose(arg)
	self.IsOpen = false
	LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "RefreshTickets")
end

return SecretAreaRankWidget

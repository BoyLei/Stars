---@class SecretAreaAchieveItem
SecretAreaAchieveItem = class("SecretAreaAchieveItem", UIPanel)
SecretAreaAchieveItem.Name = nil;
SecretAreaAchieveItem.Desc = nil;
SecretAreaAchieveItem.Process = nil;
SecretAreaAchieveItem.AwardBtn = nil;
SecretAreaAchieveItem.AwardBtnText = nil;
SecretAreaAchieveItem.m_ItemPreviewGo = nil;
SecretAreaAchieveItem.m_ItemPreview = nil;
SecretAreaAchieveItem.ID = nil;
SecretAreaAchieveItem.State = 0;
SecretAreaAchieveItem.config = nil;
SecretAreaAchieveItem.ProcessNum = 0;
SecretAreaAchieveItem.HasGet = nil;

function SecretAreaAchieveItem:ctor(go, parent, atlas)
    self.prefab = go;
    self.AtlasPath = atlas;
    self.Name = self:Find("Tile"):GetComponent(typeof(Text));
    self.Desc = self:Find("Desc"):GetComponent(typeof(Text));
    self.Process = self:Find("Process"):GetComponent(typeof(Text));
    self.AwardBtn = self:Find("AwardBtn"):GetComponent(typeof(JButton));
    self.AwardBtnText = self:Find("AwardBtn/Text"):GetComponent(typeof(Text));
    self.m_ItemPreviewGo = self:Find("ItemPreview").gameObject;
    self.HasGet = self:Find("HasGet").gameObject
    self.m_ItemPreview = StructAccessor.ItemPreview.new(self.m_ItemPreviewGo);
    self.AwardBtn.OnClick = (function()
        self:GetAward();
    end);
end

function SecretAreaAchieveItem:GetAward()
    local isfull = self.ProcessNum >= tonumber(self.config.TargetNum);
    if not self.State or isfull then
        LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "SendPersonSecretAchieveAwardReq", self.ID);
    end
end

function SecretAreaAchieveItem:FreshItem(state, value)
    if self.config == nil then
        return
    end

    self.State = state
    self.ProcessNum = value;

	-- 探索任务特殊只显示（0/1）或（1/1）
	if self.config.AchieveType == 1 then
		local curNum = 0;
		if self.ProcessNum >= tonumber(self.config.TargetNum) then
			curNum = 1;
		end
		self.Process.text = "(" .. curNum .. "/1)";
	else
		self.Process.text = "(" .. self.ProcessNum .. "/" .. self.config.TargetNum .. ")";
	end
    self.ID = self.config.AchieveID
    self.Name.text = self.config.AchieveTitle
    self.Desc.text = self.config.AchieveDec

    -- if self.State==0 then
    -- if self.ProcessNum>=tonumber(self.config.TargetNum) then
    -- self.AwardBtnText.text="领取"
    -- else
    -- self.AwardBtnText.text="未完成"
    -- end
    -- elseif self.State==1 then
    -- self.AwardBtnText.text="领取"
    -- else
    -- self.AwardBtnText.text="已领取"
    -- end

    if self.State then
        self.HasGet:SetActive(true)
        self.AwardBtn.gameObject:SetActive(false)

        -- self.AwardBtnText.text="已领取"
        -- self.AwardBtn.Image.color.a=0;
    else
        self.HasGet:SetActive(false)
        self.AwardBtn.gameObject:SetActive(true)
        if self.ProcessNum >= tonumber(self.config.TargetNum) then
            self.AwardBtnText.text = LanguageHelp:GetLanguageByKey("Local_Str_GetReward")
            self.AwardBtn.Image.color = Color(1, 1, 1, 1);
        else
            self.AwardBtnText.text = LanguageHelp:GetLanguageByKey("Local_Str_Unfinished")
            self.AwardBtn.Image.color = Color(1, 1, 1, 0);
        end
    end
    -- self.AwardBtn.Image:SetGray(not self.State==1)
    local items = self.config.RewardID;
    local nums = self.config.RewardNum;

    local CostItems = {}
    for i = 1, #items do
        table.insert(CostItems, {
            ItemID = items[i],
            ItemNum = nums[i]
        })
    end

    self.m_ItemPreview:SetItems(CostItems);
end

-- 数据更新
function SecretAreaAchieveItem:UpdateContent(item)
    self.config = item.config -- 配置表数据
    local state = 0;
    local value = 0;
    if item.data ~= nil then
        state = item.data.State
        value = item.data.Value
    end

    self:FreshItem(state, value);
end

return SecretAreaAchieveItem;

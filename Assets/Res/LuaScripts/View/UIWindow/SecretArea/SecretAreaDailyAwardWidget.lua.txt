local SecretDailyTaskItem = require("View/UIWindow/SecretArea/SecretAreaDailyTaskItem")
SecretAreaDailyAwardWidget = class("SecretAreaDailyAwardWidget", UIPanel)
SecretAreaDailyAwardWidget.m_Close = nil;
SecretAreaDailyAwardWidget.m_AwardScrollView = nil;
SecretAreaDailyAwardWidget.m_AwardItem = nil;
SecretAreaDailyAwardWidget.m_Name = nil;
SecretAreaDailyAwardWidget.m_Info = nil;
SecretAreaDailyAwardWidget.m_GetBtn = nil;
SecretAreaDailyAwardWidget.m_ItemPreviewGo = nil;
SecretAreaDailyAwardWidget.m_ItemPreview = nil;
SecretAreaDailyAwardWidget.AwardGo = nil;
SecretAreaDailyAwardWidget.NextGet = nil;
-- 数据
SecretAreaDailyAwardWidget.RewardItemList = {};
SecretAreaDailyAwardWidget.RewardItemDatas = {};

SecretAreaDailyAwardWidget.TaskItemList = {};
SecretAreaDailyAwardWidget.TaskItemDatas = {};

SecretAreaDailyAwardWidget.PassFloor = 0;
SecretAreaDailyAwardWidget.LastPassFloor = 0;

SecretAreaDailyAwardWidget.TabIndex = 1;

local OnDailyAwardFunc = nil;
local OnDailyTaskFunc = nil;


function SecretAreaDailyAwardWidget:Awake()
    self.m_Close = self:Find("Center/TopCenter/Close"):GetComponent(typeof(JButton));
    self.m_Menus = self:Find("Center/TopCenter/Menus"):GetComponent(typeof(StarUITabMenu));

    self.m_TaskScrollView = self:Find("Center/TaskScrollView"):GetComponent(typeof(SuperScrollViewController));
    self.m_TaskItem = self:Find("Center/DailyTaskItem").gameObject

    self.m_AwardScrollView = self:Find("Center/AwardScrollView"):GetComponent(typeof(SuperScrollViewController));
    self.m_AwardItem = self:Find("Center/DailyAwardItem").gameObject
    self.AwardGo = self:Find("Center/CurrentAward").gameObject
    self.NextGet = self:Find("Center/CurrentAward/NextGetBtn"):GetComponent(typeof(JButton));
    self.m_Name = self:Find("Center/CurrentAward/InfoText"):GetComponent(typeof(Text));
    self.m_Info = self:Find("Center/CurrentAward/Info"):GetComponent(typeof(Text));
    self.m_ItemPreviewGo = self:Find("Center/CurrentAward/ItemPreview").gameObject;
    self.m_GetBtn = self:Find("Center/CurrentAward/GetBtn"):GetComponent(typeof(JButton));
    self.m_ItemPreview = StructAccessor.ItemPreview.new(self.m_ItemPreviewGo);
    self.m_ItemPreview:SetSize(396, 94, 10)

    self.m_TaskScrollView:NewScroll(self.m_TaskItem, 0, function(tr, idx)
        self:OnTaskItemRefesh(tr, idx);
    end);

    self.m_AwardScrollView:NewScroll(self.m_AwardItem, 0, function(tr, idx)
        self:OnAwardItemRefesh(tr, idx);
    end);

    self.m_Close.OnClick = (function()
        self.Close();
    end);

    self.NextGet.OnClick = (function()
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_222"))
    end);

    self.m_GetBtn.OnClick = (function()
        LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "OnSendPersonSecretDayAwardReq", {});
    end);

    OnDailyAwardFunc = function(floor, lastfloor)
        self:OnDailyAward(floor, lastfloor)
    end

    OnDailyTaskFunc = function(DailyChallangeCfg)
        self:OnDailyTask(DailyChallangeCfg)
    end

end

-- 刷新每日奖励信息
function SecretAreaDailyAwardWidget:OnDailyAward(floor, lastfloor)
    self.PassFloor = floor;
    self.LastPassFloor = lastfloor;
    self:FreshDailyAward();
end

-- 刷新每日奖励信息
function SecretAreaDailyAwardWidget:OnDailyTask(DailyChallangeCfg)
    self.DailyChallangeCfg = DailyChallangeCfg
    self:FreshDailyTask()
end

function SecretAreaDailyAwardWidget:Start()
end

function SecretAreaDailyAwardWidget:OnEnable()
    CS.StarProject.GlobalEvent.OnPersonSecretAwardRet:AddListener(OnDailyAwardFunc);
    CS.StarProject.GlobalEvent.OnPersonSecretDailyTaskRet:AddListener(OnDailyTaskFunc);
end

function SecretAreaDailyAwardWidget:OnDisable()
    CS.StarProject.GlobalEvent.OnPersonSecretAwardRet:RemoveListener(OnDailyAwardFunc);
    CS.StarProject.GlobalEvent.OnPersonSecretDailyTaskRet:RemoveListener(OnDailyTaskFunc);
end

function SecretAreaDailyAwardWidget:OnDestroy()
end

function SecretAreaDailyAwardWidget:OnOpen(arg)
    self.PassFloor = arg.PassFloor ---今日最高层数
    self.LastPassFloor = arg.LastPassFloor ---昨日最高层数
    self:FreshDailyAward();

    self.DailyChallangeCfg = arg.DailyChallangeCfg
    self:FreshDailyTask()

    self:OnInitMenu();
end

function SecretAreaDailyAwardWidget:OnInitMenu()
    self.m_Menus:Init(self.TabIndex, function(index)
        self:OnMenusItemClick(index);
    end);
end

function SecretAreaDailyAwardWidget:OnMenusItemClick(index)
    self.TabIndex = index;
    self.m_Menus:Check(index);

    self.m_TaskScrollView.gameObject:SetActive(self.TabIndex == 1);
    self.m_AwardScrollView.gameObject:SetActive(self.TabIndex == 2);
    self.AwardGo:SetActive(self.TabIndex == 2);
end

---------------------------------------------------- 每日任务

function SecretAreaDailyAwardWidget:FreshDailyTask()
    self.TaskItemDatas = {}
	if self.DailyChallangeCfg ~= nil then
		for k, v in pairs(self.DailyChallangeCfg) do
            table.insert(self.TaskItemDatas, v);
        end
	end

    table.sort(self.TaskItemDatas, function(p1, p2)
        local state1 = p1.IsGet
        local state2 = p2.IsGet

        local achieveID1 = p1.TaskID
        local achieveID2 = p2.TaskID

        if state1 and state2 then
            -- 降序
            return achieveID1 < achieveID2;
        else
            if state1 and state2 == false then
                -- 降序
                return false;
            elseif state1 == false and state2 then
                return true;
            else
                local canGetP1 = p1.CurProgress >= p1.TargetNum
                local canGetP2 = p2.CurProgress >= p2.TargetNum
                if canGetP1 and canGetP2 then
                    return achieveID1 < achieveID2;
                elseif canGetP1 and canGetP2 == false then
                    return true;
                elseif canGetP1 == false and canGetP2 then
                    return false;
                else
                    return achieveID1 < achieveID2;
                end
            end
        end
    end);

    self.m_TaskScrollView:ContinueScroll(#self.TaskItemDatas);
end

function SecretAreaDailyAwardWidget:OnTaskItemRefesh(tr, idx)
    if self.TaskItemList == nil then
        return;
    end
    for i = 1, #self.TaskItemList do
        if self.TaskItemList[i].transform == tr then
            self.TaskItemList[i]:UpdateContent(self.TaskItemDatas[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end

    local bi = SecretDailyTaskItem.new(tr.gameObject, nil, self.AtlasPath);
    bi.prefab:SetActive(true);
    bi:UpdateContent(self.TaskItemDatas[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.TaskItemList, bi);
end

---------------------------------------------------- 每日奖励

function SecretAreaDailyAwardWidget:FreshDailyAward()
    self.RewardItemDatas = {}
    local start = 1
    local list = ConfigDataAccessor.SecretDailyAward

    local lstIndex = 0;
    local cIndex = 0;

    local ls = {}
    for k, v in pairs(list) do
        table.insert(ls, v.Floor)
        if self.LastPassFloor <= v.Floor then
            if v.Floor < lstIndex or lstIndex == 0 then
                lstIndex = v.Floor
            end
        end
        if self.PassFloor <= v.Floor then
            if v.Floor < cIndex or cIndex == 0 then
                cIndex = v.Floor
            end
        end
    end
    table.sort(ls, function(f1, f2)
        return f1 < f2
    end);
    if list ~= nil then
        for i = 1, #ls do
            local v = list[ls[i]]
            if v ~= nil then
                local info = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_limitFloor"),
                    {start .. "~" .. v.Floor});
                table.insert(self.RewardItemDatas, {
                    Floor = v.Floor,
                    AwardID = v.Award,
                    Tile = info,
                    RewardID = v.RewardID,
                    RewardNum = v.RewardNum
                })
                start = v.Floor + 1
            end
        end
    end

    table.sort(self.RewardItemDatas, function(p1, p2)
        return p1.Floor < p2.Floor;
    end);

    self.m_AwardScrollView:ContinueScroll(#self.RewardItemDatas);

    local id = 0;
    if self.LastPassFloor > 0 then
        -- 今日未领奖励
        self.m_Name.text = LanguageHelp:GetLanguageByKey("Local_Str_Tips43")
        self.m_Info.text = self.LastPassFloor
        self.m_GetBtn.gameObject:SetActive(true)
        id = lstIndex
        self.m_ItemPreviewGo:SetActive(true)
        self.NextGet.gameObject:SetActive(false);
    else
        -- 今日已领奖励
        self.m_GetBtn.gameObject:SetActive(false)
        self.m_Name.text = LanguageHelp:GetLanguageByKey("Local_Str_Tips44")
        if self.PassFloor > 0 then
            self.m_Info.text = self.PassFloor
            self.m_ItemPreviewGo:SetActive(true)
            self.NextGet.gameObject:SetActive(true);
        else
            self.m_Info.text = "0"
            self.m_ItemPreviewGo:SetActive(false);
            self.NextGet.gameObject:SetActive(false);

        end
        id = cIndex
    end
    self:ShowAward(id)
end

function SecretAreaDailyAwardWidget:OnAwardItemRefesh(tr, idx)
    if self.RewardItemList == nil then
        return;
    end
    for i = 1, #self.RewardItemList do
        if self.RewardItemList[i].transform == tr then
            self.RewardItemList[i]:UpdateContent(self.RewardItemDatas[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end

    local bi = SecretDailyAwardItem.new(tr.gameObject, self.m_ItemRoot, self.AtlasPath);
    bi.prefab:SetActive(true);
    bi:UpdateContent(self.RewardItemDatas[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.RewardItemList, bi);
end

function SecretAreaDailyAwardWidget:ShowAward(id)
    if id <= 0 then
        return
    end
    local config = ConfigDataAccessor.SecretDailyAward[id]
    if config ~= nil then
        local items = config.RewardID;
        local nums = config.RewardNum;

        local CostItems = {}
        for i = 1, #items do
            table.insert(CostItems, {
                ItemID = items[i],
                ItemNum = nums[i]
            })
        end
        self.m_ItemPreview:SetItems(CostItems);
    end
end

---------------------------------------------------- 每日奖励

function SecretAreaDailyAwardWidget:OnClose(arg)
    LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "RefreshTickets")
end

return SecretAreaDailyAwardWidget

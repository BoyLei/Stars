-- 获取奖励信息
function SecretAreaWindow:GetRewardList(rewards, rewardNums, isFirstGet)
    local tempList = {}
    for i = 1, #rewards do
        local itemId = rewards[i]
        local itemNum = 1
        if i <= #rewardNums then
            itemNum = rewardNums[i]
        end
        local itemCfg = ConfigDataAccessor.Item[itemId]
        if tempList[itemId] == nil then
            local quality = 1
            if itemCfg ~= nil then
                quality = itemCfg.Quality
            end
            local item = {
                ItemID = itemId,
                ItemNum = itemNum,
                Quality = quality,
                IsFirstPassReward = isFirstGet
            }
            tempList[itemId] = item
        else
            tempList[itemId] = tempList[itemId].ItemNum + itemNum
        end
    end
    -- 奖励排序，高-》低
    local sortTempList = self:SortReward(tempList)
    return sortTempList
end

function SecretAreaWindow:SetRewardItemPreview(ItemPreview, CostItems)
    ItemPreview:SetItems(CostItems)
end

-- 奖励排序，高-》低
function SecretAreaWindow:SortReward(rewardList)
    -- 生成一个用于排序的临时数组
    local tempArray = {}
    for key, value in pairs(rewardList) do
        table.insert(tempArray, value)
    end
    -- 定义比较函数，按照Quality从大到小排序
    local function compare(a, b)
        if a.Quality ~= b.Quality then
            return a.Quality > b.Quality
        end
        return false
    end
    -- 对临时数组进行排序
    table.sort(tempArray, compare)
    return tempArray
end

    self.CurTicketsNum.text = self.m_TicketsCount
end

-- 设置中间的层数
function SecretAreaWindow:SetMiddleFloorPos(middleFloor)
    local startFloor = 0 -- 这个要计算一下开始层数的偏移量
    if middleFloor > self.m_VisibleMiddleFloor then
        startFloor = middleFloor - self.m_VisibleMiddleFloor
    end
    self.SuperScrollViewController:ContinueScroll(#self.m_Floors, false, startFloor)
end

function SecretAreaWindow:OnItemRefesh(tr, idx)
    if self.SuperScrollViewControllerItems == nil then
        return
    end
    local floor = self.m_Floors[idx + 1] -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    local maxFloor = 1
    if self.UnlockFloor > 0 then
        maxFloor = self.UnlockFloor
    end
    local isLock = floor > maxFloor
    for i = 1, #self.SuperScrollViewControllerItems do
        if self.SuperScrollViewControllerItems[i].prefab.transform == tr then
            self.SuperScrollViewControllerItems[i]:UpdateContent(floor, self.SelectIndex, self.UnlockFloor, isLock) -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end
    local sp =
        SecretAreaFloorItem.new(
        tr.gameObject,
        self.SuperScrollViewControllerRoot,
        self.AtlasPath,
        function(index)
            self:OnSelect(index)
        end,
        0
    )
    sp.prefab:SetActive(true)
    sp:UpdateContent(floor, self.SelectIndex, self.UnlockFloor, isLock)
    table.insert(self.SuperScrollViewControllerItems, sp)
end

function SecretAreaWindow:SetAllItemSelectState()
    for i = 1, #self.SuperScrollViewControllerItems do
        local isSelect = false
        if self.SuperScrollViewControllerItems[i].Index == self.SelectIndex then
            isSelect = true
        end
        self.SuperScrollViewControllerItems[i]:OnSelect(isSelect)
    end
end

function SecretAreaWindow:OnSelect(selectIndex)
    self.SelectIndex = selectIndex
    self:SetAllItemSelectState()
    -- 修改当前层数信息
    self:SetCurrentFloorData()

    self.PlayAnimation("SecretAreaWindow_Anim_QieHuan", nil)
    self.m_bgAnimation:Play("SecretAreaWindow_Bg_Anim_QieHuan")
end

-- 更新层数
function SecretAreaWindow:InitFloors()
    self.m_Floors = {}
    self.m_MaxFloors = 1
    local length = #ConfigDataAccessor.SecretCopy
    for i = 1, length do
        self.m_Floors[i] = i
        -- 获取最大的层数
        if i >= length then
            self.m_MaxFloors = ConfigDataAccessor.SecretCopy[i].Floor
        end
    end
end

-- 初始化门票列表
function SecretAreaWindow:InitTicketsList()
    for i, v in pairs(ConfigDataAccessor.TicketSynthesize) do
        if v.MaterialID ~= 0 then
            local gob = newObject(self.ItemTickets)
            gob.transform:SetParent(self.TicketsListPanel)
            gob.transform.localPosition = Vector3.zero
            gob.transform.localScale = Vector3.one
            local item = SecretAreaItemTickets:New(gob)
            table.insert(self.m_TicketsItemList, item)
            item:SetItem(v.MaterialID)
        end
    end
end

-- 更新门票信息
function SecretAreaWindow:UpdateTicket()
    for i = 1, #self.m_TicketsItemList do
        local child = self.m_TicketsItemList[i]
        child:UpdateNum()
    end
end

-- 进入提示回调
function SecretAreaWindow:EnterTipsCallBack()
    if self.m_TicketsCount >= self.m_NeedTicketCount then
        self:EnterFloorCloseWindow()
    else
        local cfg = ConfigDataAccessor.Item[self.m_CurrentFloorCfg.TicketID]
        local name = self.m_CurrentFloorCfg.TicketID
        if cfg ~= nil then
            name = cfg.Name
        end
        local isFirst = false
        for i, v in pairs(ConfigDataAccessor.TicketSynthesize) do
            if v.ID == self.m_CurrentFloorCfg.TicketID then
                isFirst = true
            end
            break
        end
        if isFirst then
            CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ShopItem_buyErrorMsg"), {name}))
            return
        end
        -- 打开合成提示
        UIAPI.ShowMsgBox(
            33,
            function(EventName)
                if EventName == "SURE" then
                    self:TicketTipsCallBack()
                end
            end,
            name
        )
    end
end

-- 点击翻页按钮
function SecretAreaWindow:OnClickPagingJBtn(num)
    self:OnClickTicketsArrowJBtn(false)
    -- 计算偏移量
    local offsetNum = self.m_VisibleMiddleFloor - 2
    if num < 0 then
        offsetNum = -(self.m_VisibleFloorLength + self.m_VisibleMiddleFloor - 1)
    end
    -- 判断是否会过边界
    local middleFloor = self:GetMiddleFloorIndex()
    local startFloor = middleFloor + offsetNum
    if startFloor < 0 then
        startFloor = 0
    end
    if (startFloor + self.m_VisibleFloorLength) > self.m_MaxFloors then
        startFloor = self.m_MaxFloors - self.m_VisibleFloorLength
    end
    self.SuperScrollViewController:NevigateByIdx(startFloor)
end

-- 获取当前中间的层数下标
function SecretAreaWindow:GetMiddleFloorIndex()
    if self.SuperScrollViewControllerItems == nil or self.SuperScrollViewControllerItems == {} then
        return 0
    end
    local minY = 99999
    local middleIndexY = 0
    local floorMiddlePointY = math.abs(self.MiddleGod.transform.position.y) -- 层数item中心点X坐标
    for i = 1, #self.SuperScrollViewControllerItems do
        local child = self.SuperScrollViewControllerItems[i]
        if child ~= nil then
            local itemWorldPosY = child.prefab.transform.position.y
            local y = math.abs(math.abs(itemWorldPosY) - floorMiddlePointY)
            if y < minY then
                middleIndexY = child.Index
                minY = y
            end
        end
    end
    return middleIndexY
end

local WorldLineDot = {}

WorldLineDot.__index = WorldLineDot


function WorldLineDot:New(_go) 
    local self = {};    
    setmetatable(self, WorldLineDot); 
    
    self.gameObject=_go;
    self.transform=_go.transform;

	self.Rect=self.transform:GetComponent(typeof(RectTransform));
	

    return self;    --返回自身
end


function WorldLineDot:InitItem()
    self.IsUsed=false;
    self.gameObject:SetActive(false);
end


function WorldLineDot:SetItem(pos)

	self.Rect.anchoredPosition=pos;
	self.IsUsed=true;
    self.gameObject:SetActive(true);
end

function WorldLineDot:SetUse(b)
    self.IsUsed=b;
    self.gameObject:SetActive(b);
end

return WorldLineDot;

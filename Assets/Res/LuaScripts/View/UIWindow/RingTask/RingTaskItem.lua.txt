local RingTaskItem = {}

RingTaskItem.__index = RingTaskItem

local ItemGrid = require("View/Item/ItemGrid")

function RingTaskItem:New(_go)
    local self = {}
    setmetatable(self, RingTaskItem)

    self.gameObject = _go
    self.transform = _go.transform

    self.TaskName = self.transform:Find("taskName"):GetComponent(typeof(Text))
    self.Name = self.transform:Find("name"):GetComponent(typeof(Text))
    self.Desc = self.transform:Find("Scroll View/Viewport/Content/desc"):GetComponent(typeof(Text))

    self.Head = self.transform:Find("Head"):GetComponent(typeof(Image))

    self.anim = self.transform:GetComponent(typeof(Animator))

    --道具
    self.items = {}
    for i = 1, 4 do
        local it = ItemGrid:New(self.transform:Find("Items/ItemGrid" .. i).gameObject)
        it:InitItem(
            false,
            function(ritem)
                self:OnItemClick(ritem)
            end
        )
        table.insert(self.items, it)
    end

    self.go_item = self.transform:Find("Items").gameObject
    self.go_item2 = self.transform:Find("Items2").gameObject

    --道具
    self.items2 = {}
    for i = 1, 5 do
        local it = ItemGrid:New(self.transform:Find("Items2/Viewport/Content/ItemGrid" .. i).gameObject)
        it:InitItem(
            false,
            function(ritem)
                self:OnItemClick(ritem)
            end
        )
        table.insert(self.items2, it)
    end

    self.go_BtnJump = self.transform:Find("BtnJump").gameObject
    self.go_Finish = self.transform:Find("Finish").gameObject
    self.go_NotBegin = self.transform:Find("NotBegin").gameObject

    --点击寻路
    self.go_BtnJump:GetComponent(typeof(JButton)).OnClick = (function()
        --寻路去目标点

        local config = TaskHelper.GetTaskConfig(self.TaskId)
        if config ~= nil then
            local TaskInfo = TaskHelper.GetRunningTask(self.TaskId)

            --任务进行中
            if TaskInfo.TaskState == CS.ProtoMsg.TaskStateEnum.TaskStageRunning then
                local taskTarget = TaskHelper.GetTaskTarget(self.TaskId)
                if taskTarget ~= nil then
                    CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(self.TaskId)

                    UIManager.Instance:CloseWindow("Ectype/Prefab/GamePlayChoiceWindow")
                    UIManager.Instance:CloseWindow("RingTask/RingTaskWindow")
                    TaskHelper.FindPathByTask(
                        taskTarget.MapID,
                        taskTarget.FindPath,
                        taskTarget.TargetID,
                        function(result)
                            CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(0)
                            if result and (taskTarget.FindPath.FindPathType == CS.Task.E_FindPath.NPC or taskTarget.FindPath.FindPathType == CS.Task.E_FindPath.InterAction) then
                                local EventID = TaskHelper.GetTaskRunningEventID(self.TaskId)
                                self:OnFinishPathFinding(taskTarget.TargetID, self.TaskId, EventID, taskTarget.taskType)
                            end
                        end
                    )
                end
            elseif TaskInfo.TaskState == CS.ProtoMsg.TaskStateEnum.TaskStageFinished then
                CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(self.TaskId)
                UIManager.Instance:CloseWindow("Ectype/Prefab/GamePlayChoiceWindow")
                UIManager.Instance:CloseWindow("RingTask/RingTaskWindow")
                --任务已完成未提交
                TaskHelper.FindPathByNpc(
                    config.TaskFinishInfo.FinishTaskMapID,
                    config.TaskFinishInfo.FinishTaskNPC,
                    1,
                    function(result)
                        CS.StarProject.GlobalEvent.SelectTaskItem:Invoke(0)

                        if result then
                            self:OnFinishPathFinding(config.TaskFinishInfo.FinishTaskNPC, self.TaskId, 0, CS.Task.TaskType.Dialogue)
                        end
                    end
                )
            end
        end
    end)

    return self --返回自身
end

function RingTaskItem:OnFinishPathFinding(targetid, SourceID, SourceSubID, taskType)
    if taskType == CS.Task.TaskType.Dialogue then
        self:OnOpenDialogue(targetid, SourceID, SourceSubID)
    elseif taskType == CS.Task.TaskType.InterAction then
        self:OnInterAction(targetid, SourceID, SourceSubID)
    end
end

function RingTaskItem:OnInterAction(objectID, SourceID, SourceSubID)
    local uid = GameManager.Instance:GetObjectEntityIDByConfig(objectID)
    BusinessManager.Instance:TurnToEntity(uid)
    LuaModuleDispatcher:SendMessage(ModuleDef.InterActionModule, "QueryInter", uid, true, SourceID, SourceSubID)
end

function RingTaskItem:OnOpenDialogue(npcID, SourceID, SourceSubID)
    local lobbyService = CS.ProtoMsg.LobbyServiceReq()
    lobbyService.MapID = GameManager.Instance:GetCurMapId()
    lobbyService.UID = GameManager.Instance:GetNPCEntityIDByConfig(npcID)
    lobbyService.ObjID = npcID
    lobbyService.SourceType = CS.ProtoMsg.MapServiceSourceTypeEnum.Task
    lobbyService.SourceID = SourceID
    lobbyService.SourceSubID = SourceSubID
    lobbyService.ObjType = CS.ProtoMsg.ServiceObjEnum.ObjNpc
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, lobbyService, false)
end

function RingTaskItem:OnItemClick(ritem)
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {ritem.BaseID, false, 0, ritem.transform, 0})
end

function RingTaskItem:InitItem(onItemClick)
    self.OnItemClick = onItemClick
end

function RingTaskItem:SetNpcInfo(npcID)
    local npcConfig = ConfigDataAccessor.Npc[npcID]

    if npcConfig ~= nil then
        self.Name.text = npcConfig.Name

        local avataID = npcConfig.AvatarID
        -- 头像
        local avatarDataCell = ConfigDataAccessor.Avatar[avataID]
        if avatarDataCell ~= nil then
            local headCfg = ConfigDataAccessor.ModelHead[avatarDataCell.HeadID]
            if headCfg ~= nil then
                --self.Head.sprite = self:GetSpriteByAtlas(headCfg.HeadAtlasName, headCfg.MinHead)

                --self.Head.sprite = AtlasManager.Instance:GetSprite("ui/icons/atlas/iconrole", headCfg.MinHead)
                ResourceHelper:LoadIconByAtlasAsync(self.Head, ResourceHelper.AtlasPathRole, headCfg.MinHead, false)
            --self.HeadIcon:SetNativeSize()
            end
        end
    end
end

function RingTaskItem:SetItem(id)
    local isFinish = false

    if id ~= nil and id > 0 then
        local idNow = id
        local isLast = false
        local configNow = TaskHelper.GetTaskConfig(idNow)
        local baseNow = nil
        if configNow ~= nil then
            --[[
                        if configNow.TaskFinishInfo.Targets.Count > 0 then
                self:SetNpcInfo(configNow.TaskFinishInfo.Targets[0].Target.TargetID)
            end
            ]]
            baseNow = configNow.Base
        else
            isLast = true
        end

        --读取配置表的环任务
        local ringTaskCfg = ConfigDataAccessor.GetRingTaskConfigById(id)
        self:SetNpcInfo(ringTaskCfg.Npc)

        while not (isLast) do
            --任务状态
            if TaskHelper.ContainTask(idNow) then
                --正在进行
                self.go_BtnJump:SetActive(true)
                self.go_Finish:SetActive(false)
                self.go_NotBegin:SetActive(false)
                self.TaskId = idNow
                isLast = true
            elseif TaskHelper.IsTaskFinsh(idNow) then
                if baseNow.NextTaskID == 0 then
                    --已经完成
                    self.go_BtnJump:SetActive(false)
                    self.go_Finish:SetActive(true)
                    self.go_NotBegin:SetActive(false)
                    self.TaskId = idNow
                    isLast = true
                    isFinish = true
                else
                    --继续往下查
                    configNow = TaskHelper.GetTaskConfig(baseNow.NextTaskID)
                    if configNow ~= nil then
                        idNow = baseNow.NextTaskID
                        baseNow = configNow.Base
                    else
                        --结束
                        baseNow = nil
                        isLast = true
                        print("任务ID：" .. baseNow.NextTaskID .. "不存在")
                    end
                end
            else
                --还没开始
                self.go_BtnJump:SetActive(false)
                self.go_Finish:SetActive(false)
                self.go_NotBegin:SetActive(true)
                self.TaskId = idNow
                isLast = true
            end
        end

        if baseNow ~= nil then
            if self.go_NotBegin.activeSelf then
                self.Desc.text = LanguageHelp:GetLanguageByKey("Local_Str_Tips42")
                self.TaskName.text = configNow.Base.TaskName
            else
                self.TaskName.text = configNow.Base.TaskName
                --[[
                self.Desc.text = "·" .. baseNow.Desc

                local taskTarget = TaskHelper.GetTaskTarget(self.TaskId)
                if taskTarget ~= nil then
                    self.Desc.text = self.Desc.text .. "(" .. taskTarget.CurProgress .. "/" .. taskTarget.MaxCount .. ")"
                end
                ]]
                --根据配置表显示
                if isFinish then
                    --已完成
                    self.Desc.text = ringTaskCfg.RingTask_Peroration
                else
                    --进行时
                    self.Desc.text = ringTaskCfg.RingTask_Preamble
                end
            end
        end

        local aNum = #ringTaskCfg.AwardsDisplay

        if aNum > 3 then
            self.go_item:SetActive(false)
            self.go_item2:SetActive(true)

            for i = 1, #self.items2 do
                self.items2[i].gameObject:SetActive(false)
            end
            local idx = 0
            for i = 1, #ringTaskCfg.AwardsDisplay do
                idx = idx + 1
                if idx <= #self.items2 then
                    self.items2[idx].gameObject:SetActive(true)
                    self.items2[idx]:SetItemNum(ringTaskCfg.AwardsDisplay[i], ringTaskCfg.AwardsDisplayNum[i])

                    self.items2[idx].go_Equip:SetActive(self.go_Finish.activeSelf)
                end
            end
        else
            self.go_item:SetActive(true)
            self.go_item2:SetActive(false)

            for i = 1, #self.items do
                self.items[i].gameObject:SetActive(false)
            end
            local idx = 0
            for i = 1, #ringTaskCfg.AwardsDisplay do
                idx = idx + 1
                if idx <= #self.items then
                    self.items[idx].gameObject:SetActive(true)
                    self.items[idx]:SetItemNum(ringTaskCfg.AwardsDisplay[i], ringTaskCfg.AwardsDisplayNum[i])

                    self.items[idx].go_Equip:SetActive(self.go_Finish.activeSelf)
                end
            end
        end

    --self.anim:Play("Lock")
    --self.anim:Play("Complete")
    end

    return isFinish
end

function RingTaskItem:PlayUnLockAnim()
    self.anim:Play("Lock")
end

function RingTaskItem:PlayCompleteAnim()
    self.anim:Play("Complete")
end

return RingTaskItem

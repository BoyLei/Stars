function SkillSwitchItemNew:Refresh(skillId, open, talentId, lv, curSel)
    self.skillId = skillId
    self.talentID = talentId
    self.skillLV = lv
    self:Show(true)

    -- open表示 是否已经装在在 技能槽中
    self.usedGo:SetActive(open);

    -- self.dressed:SetActive(open)

    self:SetSel(false)
    self:MarkHighLight(false)

    local skillName = nil
    -- TODO: DL
    -- 擦, 发现这块 原代码逻辑有问题, 一个天赋 可能有多个 技能, 所以 此处默认采用第一个 可能有问题
    -- 应该采用 能够显示的 那个技能.
    local id = ConfigDataAccessor.TalentList[talentId].BattleActive[1]
    for _, v in pairs(ConfigDataAccessor.SkillDescList) do
        if v.SkillID == id and v.Level == lv then
            ResourceHelper:LoadIconByAtlasAsync(self.skillIcon, ResourceHelper.AtlasPathSkill, v.IconPath, false)
            skillName = v.SkillName
            break
        end
    end

    -- 设置天赋 (目前装载界面不显示)

    for _, v in pairs(ConfigDataAccessor.TalentList) do
        if talentId == v.ID then
            self:RefreshTalent(v.TalentType)

            local playerLevel = GameManager.Instance:GetPlayerLevel();
            if playerLevel >= v.LimitLv then
                self:Show(true);
            else
                self:Show(false);
            end

        end
    end

    local maxLevel = 0
    for k, v in pairs(ConfigDataAccessor.SkillLevelList) do
        if v.SkillID == skillId then
            if v.Skill_Level > maxLevel then
                maxLevel = v.Skill_Level
            end
        end
    end
    self.percentText.text = lv --[[..'/'..maxLevel]] -- ;
    self.nameText.text = skillName

    -- self.selGo:SetActive(skillId == curSel);
end

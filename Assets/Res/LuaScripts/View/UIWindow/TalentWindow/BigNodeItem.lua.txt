function BigNodeItem:InitData(id, lv, preUnlock, usePointEnough)
	self.nodeId = id;
    local attriTTNode = ConfigDataAccessor.AttriTTNode[id];

    self.talentmsgcb =  function(data)
        if data.data.NodeID == id then 
			LuaModuleDispatcher:SendMessage(ModuleDef.TalentModule, "OnOpenNodeInfo", {self.nodeId, self.transform, 0});
        end
    end
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.TalentNodeActiveRetID,self.talentmsgcb,nil)
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.TalentNodeActiveRetID, self.talentmsgcb,nil);

    self.status1:SetActive(false);
    self.status2:SetActive(false);
    self.status3:SetActive(false);
    self.status4:SetActive(false);
	
	-- for _,v in pairs(self.allEff) do
	-- 	self:RemoveBindItem(v);
	-- end
	
	local upgrade = false;
	if self.curLv == nil then
		self.curLv = lv;
	elseif self.curLv ~= lv then
		upgrade = true;
		self.curLv = lv;
	end
	
    self.nodeIcon.color = Color(1, 1, 1);
    if not preUnlock or not usePointEnough then
        self.status1:SetActive(true);
        self.nodeIcon.color = Color(0.33, 0.33, 0.33);
        self:RemoveAllbind()
    elseif lv ==0 then
        local itemsEnough = true;
        local items, nums = self:GetItems(attriTTNode, 1);

        local talentModule = ModuleManager.Instance:GetModule(ModuleDef.TalentModule);
        local leavePoint = talentModule:GetLuaTable().talentPoint;

        for k, v in pairs(items) do
            if v == 113 then
                if leavePoint < nums[k] then
                    itemsEnough = false;
                end
            else
                local itemNum = BusinessManager.Instance:GetCurrencyNum(v);
                if itemNum < nums[k] then
                    itemsEnough = false;
                end
            end
        end

        if itemsEnough then
            self.nodeIcon.color = Color(1, 1, 1);
            
           -- self.allEff[1]:SetActive(true);
            self:RemoveBindByType(2,3)
            self:ShowEffect(self.allEff[1])
			self.status3:SetActive(true);
        else
            self:RemoveAllbind()
            self.nodeIcon.color = Color(0.56, 0.56, 0.56);
			self.status1:SetActive(true);
        end
    else
        self.status4:SetActive(true);
		
		if upgrade then
            self:RemoveBindByType(1,3)
            self:ShowEffect(self.allEff[2])
			--self.allEff[2]:SetActive(true);
        else
            self:RemoveAllbind()
        end
    end

    if lv ~= nil and lv >= attriTTNode.MaxLevel then
        self.levelText.gameObject:SetActive(false);
    else
        self.levelText.gameObject:SetActive(true);
        self.levelText.text = lv..'/'..attriTTNode.MaxLevel;
    end

    self.levelText.gameObject:SetActive(false);
	
	--加载属性icon
	local attributes = ConfigDataAccessor.Attribute;
	for k, v in pairs(attributes) do
		if v.AttrId == attriTTNode.AttrID then
			if v.Icon ~= nil then
				ResourceHelper:LoadIconByAtlasAsync(self.nodeIcon, ResourceHelper.AtlasPathIconAttribute, v.Icon, false)
			end
			break
		end
	end
end

function BigNodeItem:GetItems(attriTTNode, lv)
    if lv == 1 then
        return attriTTNode.FirLvUseItem, attriTTNode.FirLvNum
    elseif lv == 2 then
        return attriTTNode.SecUseItem, attriTTNode.SecLvNum
    elseif lv == 3 then
        return attriTTNode.ThrLvUseItem, attriTTNode.ThrLvNum
    end
    return {}, {};
end

function BigNodeItem:Clear()
    self.super:Clear()
    NetworkManager.Instance:OffMessageEnum(
        CS.MsgIDEnum.TalentNodeActiveRetID,
        self.talentmsgcb,
        nil
    )
end

return BigNodeItem

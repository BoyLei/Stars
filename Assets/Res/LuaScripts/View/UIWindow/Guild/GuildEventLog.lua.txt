local GuildEventLog = {}

GuildEventLog.__index = GuildEventLog

local GuildEventLogItem = require("View/Uiwindow/Guild/GuildEventLogItem")

function GuildEventLog:New(_go)
    local self = {}
    setmetatable(self, GuildEventLog)

    self.gameObject = _go
    self.transform = _go.transform

    --道具列表
    self.EventLogItemList = {}
    --无限列表
    self.go_EventLogItem = self.transform:Find("Cut4Cam90/Scroll View/Mask/List/GuildEventLogItem").gameObject

    self.tr_ItemRoot = self.transform:Find("Cut4Cam90/Scroll View/Mask/List")
    self.go_EventLogItem:SetActive(false)
    self.go_EventLogItem.transform:SetParent(self.transform)
    self.EventLogScorllView = self.transform:Find("Cut4Cam90/Scroll View"):GetComponent(typeof(SuperScrollViewController))
    self.EventLogScorllView:NewScroll(
        self.go_EventLogItem,
        0,
        function(tr, idx)
            self:OnEventLogItemRefesh(tr, idx)
        end
    )

    --关闭按钮
    self.transform:Find("Cut4Cam90/BtnClose"):GetComponent(typeof(JButton)).OnClick = (function()
        self:Show(false)
    end)

    self:Show(false)

    return self --返回自身
end

function GuildEventLog:OnEventLogItemRefesh(tr, idx)
    for i = 1, #self.EventLogItemList do
        if self.EventLogItemList[i].transform == tr then
            self.EventLogItemList[i]:SetItem(self.ItemDataList[idx + 1]) --idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end

    local bi = GuildEventLogItem:New(tr.gameObject)

    bi.gameObject:SetActive(true)
    bi:SetItem(self.ItemDataList[idx + 1]) --idx是从0开始，如果ItemDataList是lua的table需要idx+1

    table.insert(self.EventLogItemList, bi)
end

function GuildEventLog:Show(b, gEvent)
    self.gameObject:SetActive(b)
    if b then
        self.GEvent = gEvent
        self:Refesh()
    end
end

function GuildEventLog:Refesh()
    if self.ItemDataList == nil then
        self.ItemDataList = {}
    else
        for k, _ in pairs(self.ItemDataList) do
            self.ItemDataList[k] = nil
        end
    end

    --刷新数据 EventInfo
    for i = 0, self.GEvent.Count - 1 do
        local data = {}

        data.EventStr = self.GEvent[i].EventStr --List
        data.EventTime = self.GEvent[i].EventTime --ulong
        data.EventID = self.GEvent[i].EventID --int

        table.insert(self.ItemDataList, data)
    end

    self:SortItem(self.ItemDataList)
    self.EventLogScorllView:ContinueScroll(#self.ItemDataList)
end

function GuildEventLog:SortItem(tbl)
    if #tbl > 1 then
        table.sort(
            tbl,
            function(a, b)
                --时间降序
                if a.EventTime > b.EventTime then
                    return true
                end
                return false
            end
        )
    end
end

return GuildEventLog

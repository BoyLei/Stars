local GuildMemberItem = {}

GuildMemberItem.__index = GuildMemberItem

function GuildMemberItem:New(_go,onClick) 
    local self = {};    
    setmetatable(self, GuildMemberItem); 
    

    self.gameObject=_go;
    self.transform=_go.transform;

    self.Name=self.transform:Find("TitleBg/name"):GetComponent(typeof(Text));
    self.Lv=self.transform:Find("TitleBg/lv"):GetComponent(typeof(Text));
    self.Duty=self.transform:Find("TitleBg/duty"):GetComponent(typeof(Text));
    self.Job=self.transform:Find("TitleBg/job"):GetComponent(typeof(Text));
    self.BattlePoint=self.transform:Find("TitleBg/battlePoint/battlePoint"):GetComponent(typeof(Text));
    self.ActivePoint=self.transform:Find("TitleBg/activePoint"):GetComponent(typeof(Text));
    self.State=self.transform:Find("TitleBg/state"):GetComponent(typeof(Text));

    self.aLv=self.transform:Find("TitleBg/aLv"):GetComponent(typeof(Text));
    self.transform:GetComponent(typeof(JButton)).OnClick=(function()
        if onClick~=nil then
            onClick(self);
        end

	end);

    return self;    --返回自身
end



function GuildMemberItem:SetItem(info)


    self.Info=info;


    self.PID=info.PID;


    --self.Lv.text=info.Level .. "级";
    self.Lv.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LvLimit3"), {info.Level});

    
    local cfg=ConfigDataAccessor.AdvGradeExp[info.RiskLevel];
    if cfg~=nil then
	    self.aLv.text=cfg.AdvClass..GetRomaNum(cfg.AdvOrder);
    else
        self.aLv.text="";
    end

    self.PosLevelID=info.PosLevelID;
    self.Duty.text=ConfigDataAccessor.GuildTitle[self.PosLevelID].Comments;

    --等CD
    local serverTime=GameManager.Instance:GetServerTimeStamp()/1000;
    local ProhibEndTime=info.BanChatTime;--禁言结束时间
    local leftTime=ProhibEndTime-serverTime;

    if leftTime>0 then
        --local timeStr=SecToStringColon2(leftTime);
        --Util.ShowMessage("您之前修改过冒险团名， "..timeStr.." 内无法修改过冒险团名");
        --do return; end

        --self.Name.text=info.UserName.."(禁言中)";
        self.Name.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Tips37"), {info.UserName});


    else
        self.Name.text=info.UserName;
    end


    if ConfigDataAccessor.Job[info.ModelID]~=nil then
        self.Job.text=ConfigDataAccessor.Job[info.ModelID].Name ;
    else
        self.Job.text=""..info.ModelID ;
    end

    self.BattlePoint.text=info.Power;
    self.ActivePoint.text=info.Activation;

    if info.IsOnline then
        self.State.text="<color=#8CE071>".. LanguageHelp:GetLanguageByKey("Local_Str_GuildMemberItem_State") .."</color>";


        --超过2分钟就是离线
        if info.LogOutTime>0 then
            --等CD
            local serverTime=GameManager.Instance:GetServerTimeStamp()/1000;
            local LogOutTime=info.LogOutTime;--上次修改时间
            local leftTime=serverTime-LogOutTime;

            if leftTime>120 then
                local timeStr=SecToStringColon4(leftTime);
                --self.State.text= timeStr.."前";
                self.State.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_GuildMemberItem_State2"), {timeStr});

            end
        end


    else
        if info.LogOutTime>0 then


            --等CD
            local serverTime=GameManager.Instance:GetServerTimeStamp()/1000;
            local LogOutTime=info.LogOutTime;--上次修改时间
            local leftTime=serverTime-LogOutTime;

            if leftTime>0 then
                local timeStr=SecToStringColon4(leftTime);
                --self.State.text= timeStr.."前";
                self.State.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_GuildMemberItem_State2"), {timeStr});

            else
                self.State.text=LanguageHelp:GetLanguageByKey("Local_Str_GuildMemberItem_State1");
            end


        else
            self.State.text=LanguageHelp:GetLanguageByKey("Local_Str_GuildMemberItem_State1");
        end
    end


end


return GuildMemberItem;

    self.module = m
end

function DonateWindow:InitItems()
    self.transform = self.prefab.transform
    self.gameObject = self.transform.gameObject

    self.m_CloseBtn = self.transform:Find("Cut4Cam90/AssemblyRatio2/L/CloseBtn/CloseBtn"):GetComponent(typeof(Button))
    self.m_CloseBtn.onClick:AddListener(
        function()
            self:OnClickCloseBtn()
        end
    )

    self.go_BtnFinish = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/BtnFinish").gameObject
    self.go_BtnNotFinish = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/BtnNotFinish").gameObject
    self.go_Finish = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/GoFinish").gameObject

    -- 点击完成委托
    self.go_BtnFinish:GetComponent(typeof(JButton)).OnClick = function()
        if self.ItemSet > 0 then
            if self.Enough then
                -- 打开选择页面
                local data = {}
                data.DonateOrderListId = self.ConfigIndex
                data.OrderIndex = self.OrderIndex

                UIManager.Instance:OpenWidget("Donate/DonateSumbitWidget", false, data, nil, CS.StarProjectDef.MainPageCommond.HideNone, false, false)
            end
        else
            if self.Enough then
                -- 直接提交

                local guildDonateSubmitReq = CS.ProtoMsg.GuildDonateSubmitReq()
                guildDonateSubmitReq.OrderIndex = self.OrderIndex
                CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, guildDonateSubmitReq, false)
            else
                Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_No_Item"))
            end
        end
    end

    -- 点击刷新委托
    self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/BtnRefesh"):GetComponent(typeof(JButton)).OnClick = function()
        if self.IsDailyMax then
            local cfg = ConfigDataAccessor.Item[self.RefeshPriceId]
            local itemName = cfg.Name
            local itemNum = self.RefeshPriceNum

            local haveNum = BusinessManager.Instance:GetCurrencyNum(self.RefeshPriceId)
            local isItemEnough = (haveNum >= itemNum)

            CS.StarProjectDef.UIAPI.ShowMsgBox(
                52,
                function(str)
                    if str == "SURE" then
                        if not (isItemEnough) then
                            -- Util.ShowMessage(itemName.."不足");
                            Util.ShowMessage(StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ShopItem_buyErrorMsg"), {itemName}))
                        else
                            local refreshGuildDonateOrderReq = CS.ProtoMsg.RefreshGuildDonateOrderReq()
                            refreshGuildDonateOrderReq.IsMoney = self.IsDailyMax
                            CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, refreshGuildDonateOrderReq, false)
                        end
                    else
                    end
                end,
                itemNum,
                itemName
            )
        else
            CS.StarProjectDef.UIAPI.ShowMsgBox(
                51,
                function(str)
                    if str == "SURE" then
                        local refreshGuildDonateOrderReq = CS.ProtoMsg.RefreshGuildDonateOrderReq()
                        refreshGuildDonateOrderReq.IsMoney = self.IsDailyMax
                        CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, refreshGuildDonateOrderReq, false)
                    else
                    end
                end
            )
        end
    end

    self.DonateTaskItems = {}
    for i = 1, 9 do
        local dti =
            DonateTaskItem:New(
            self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/DonateTaskItem" .. i).gameObject,
            function(item)
                self:OnTaskItemClick(item)
            end,
            i - 1
        )
        table.insert(self.DonateTaskItems, dti)
    end

    self.Rewards = {}
    for i = 1, 3 do
        local rd = {}
        for j = 1, 2 do
            local dti = ItemGrid:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/g" .. i .. "/Reward" .. j).gameObject)
            dti:InitItem(
                false,
                function(ritem)
                    self:OnProgressItemClick(ritem, i)
                end
            )
            table.insert(rd, dti)
        end
        table.insert(self.Rewards, rd)
    end

    self.NotEnoughTex = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Left"):GetComponent(typeof(Text))

    self.Lv = self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/Lv"):GetComponent(typeof(Text))

    self.refeshTime = self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/refeshTime"):GetComponent(typeof(Text))
    self.refeshPrice = self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/refeshIcon/refeshPrice"):GetComponent(typeof(Text))
    self.refeshIcon = self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/refeshIcon"):GetComponent(typeof(Image))

    self.Name = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Name"):GetComponent(typeof(Text))

    self.Title = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Title"):GetComponent(typeof(Text))
    self.Desc = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Desc"):GetComponent(typeof(Text))
    self.DescName = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/DescName"):GetComponent(typeof(Text))

    self.Head = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Hero/Bg/head"):GetComponent(typeof(Image))
    self.HeadBg = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Hero/Bg"):GetComponent(typeof(Image))
    self.GoHead = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Hero").gameObject

    self.Icon = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Icon"):GetComponent(typeof(Image))

    -- 点击信息
    self.go_BtnInfo = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/BtnInfo").gameObject
    self.go_BtnInfo:GetComponent(typeof(JButton)).OnClick = function()
        LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {self.BaseID, false, 0, self.go_BtnInfo.transform, 0})
    end

    -- 点击信息
    self.go_BtnTips = self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/BtnTips").gameObject
    self.go_BtnTips:GetComponent(typeof(JButton)).OnClick = function()
        CS.StarProjectDef.UIAPI.ShowGameplayInfo(115)
    end

    self.Left = self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Left"):GetComponent(typeof(Text))

    self.Slider = self.transform:Find("Cut4Cam90/AssemblyRatio2/Midst/slider"):GetComponent(typeof(RectTransform))
    self.SliderW = self.Slider.sizeDelta.x
    self.SliderH = self.Slider.sizeDelta.y

    -- 道具
    self.items = {}
    for i = 1, 5 do
        local it = ItemGrid:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/R/Items/ItemGrid" .. i).gameObject)
        it:InitItem(
            false,
            function(ritem)
                self:OnItemClick(ritem)
            end
        )
        table.insert(self.items, it)
    end
end

function DonateWindow:OnItemClick(ritem)
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {ritem.BaseID, false, 0, ritem.transform, 0})
end

function DonateWindow:OnProgressItemClick(ritem, idx)
    local aimIdx = 4

    if idx == 1 then
        aimIdx = 4
    elseif idx == 2 then
        aimIdx = 6
    elseif idx == 3 then
        aimIdx = 9
    end

    if aimIdx > self.GetNumNow then
        LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {ritem.BaseID, false, 0, ritem.transform, 0})
    else
        local getGuildDonateCountRewardReq = CS.ProtoMsg.GetGuildDonateCountRewardReq()
        getGuildDonateCountRewardReq.Count = aimIdx
        CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, getGuildDonateCountRewardReq, false)
    end
end

function DonateWindow:OnTaskItemClick(item)
    for i = 1, #self.DonateTaskItems do
        self.DonateTaskItems[i]:MakeSelected(false)
    end

    item:MakeSelected(true)

    self.OrderIndex = item.OrderIndex

    self.HaveNum = item.HaveNum
    self.NeedNum = item.NeedNum
    self.Enough = item.IsEnough
    self:RefeshRight(item.ConfigIndex, item.IsSubmited)
end

function DonateWindow:Open()
    local guildDonateMD = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.GuildDonate):GetGuildDonateMD()

    if guildDonateMD ~= nil then
        local curGetList = guildDonateMD.CurModelList.CurGetList -- 当前已领取奖励数据
        local orderList = guildDonateMD.OrderList.OrderList -- 订单列表
        local todayCount = guildDonateMD.TodayCount -- 今日已刷新次数

        local maxCount = SystemConstConfigs.Donate_FreeRefreshCount

        self.RefeshPriceId = 4
        self.RefeshPriceNum = SystemConstConfigs.Donate_RefreshCost

        if todayCount >= maxCount then
            self.refeshTime.gameObject:SetActive(false)
            self.refeshIcon.gameObject:SetActive(true)

            -- ICON
            local cfg = ConfigDataAccessor.Item[self.RefeshPriceId]
            ResourceHelper:LoadIconAsync(self.refeshIcon, "", cfg.Icon, false)

            self.refeshPrice.text = SystemConstConfigs.Donate_RefreshCost

            self.IsDailyMax = true
        else
            self.IsDailyMax = false

            self.refeshIcon.gameObject:SetActive(false)
            self.refeshTime.gameObject:SetActive(true)
            self.refeshTime.text = "(" .. (maxCount - todayCount) .. "/" .. maxCount .. ")"
        end

        local GetNum = 0
        for i = 0, orderList.Count - 1 do
            if i + 1 <= #self.DonateTaskItems then
                self.DonateTaskItems[i + 1]:SetItem(orderList[i])
                if orderList[i].IsSubmited then
                    GetNum = GetNum + 1
                end
            end
        end

        self.GetNumNow = GetNum
        self.Lv.text = GetNum

        local per = GetNum / 9
        --[[
		if GetNum>=9 then
			self.Slider.sizeDelta=Vector2(self.SliderW,self.SliderH);
		else
			self.Slider.sizeDelta=Vector2(110*GetNum,self.SliderH);
		end
		]]
        self.Slider.sizeDelta = Vector2(110 * GetNum, self.SliderH)

        -- 初始化

        -- 冒险等级
        local RiskInfo = GameManager.Instance.M_MainPlayerCtrlBase.Data.Attrs:GetProtoValue(CS.StarProjectDef.AOIAttrDefine.RiskInfo)
        if RiskInfo ~= nil then
            local curRiskLevel = RiskInfo.CurRiskLevel
            local rews = nil
            if IsOpenAdvLv() then
                rews = ConfigDataAccessor.GetDonateOrderRewardAdvLv(curRiskLevel)
            else
                local playerLv = GameManager.Instance:GetPlayerLevel()
                rews = ConfigDataAccessor.GetDonateOrderReward(playerLv)
            end

            for i = 1, #self.Rewards do
                for j = 1, #self.Rewards[i] do
                    self.Rewards[i][j].gameObject:SetActive(false)
                end
            end

            for i = 1, #rews do
                if i <= #self.Rewards then
                    local dCfg = rews[i]
                    for j = 1, #dCfg.Reward do
                        if j <= #self.Rewards[i] then
                            self.Rewards[i][j].gameObject:SetActive(true)
                            self.Rewards[i][j]:SetItemNum(dCfg.Reward[j], dCfg.RewardNum[j])
                            self.Rewards[i][j]:MakeSelected(false)

                            local aimIdx = 4
                            if i == 1 then
                                aimIdx = 4
                            elseif i == 2 then
                                aimIdx = 6
                            elseif i == 3 then
                                aimIdx = 9
                            end
                            self.Rewards[i][j].go_Equip:SetActive(aimIdx <= self.GetNumNow)
                        end
                    end
                end
            end

            -- 奖励道具[全部是固定的]
            for i = 1, #self.items do
                self.items[i].gameObject:SetActive(false)
            end
            local dCfg1 = nil
            if IsOpenAdvLv() then
                dCfg1 = ConfigDataAccessor.GetDonateOrderRewardEachAdvLv(curRiskLevel)
            else
                local playerLv = GameManager.Instance:GetPlayerLevel()
                dCfg1 = ConfigDataAccessor.GetDonateOrderRewardEach(playerLv)
            end

            if dCfg1 ~= nil then
                for i = 1, #dCfg1.Reward do
                    self.items[i].gameObject:SetActive(true)
                    self.items[i]:SetItemNum(dCfg1.Reward[i], dCfg1.RewardNum[i])
                end
            end
        end

        --[[
		for i=2,4 do
			local dCfg=ConfigDataAccessor.DonateOrderReward[i];
			self.Rewards[i-1]:SetItemNum(dCfg.Reward[1],dCfg.RewardNum[1])
			self.Rewards[i-1]:MakeSelected(false);
		end
		]]
        -- 打勾
        for i = 0, curGetList.Count - 1 do
            local idx = curGetList[i]

            local aimIdx = 1
            if idx == 4 then
                aimIdx = 1
            elseif idx == 6 then
                aimIdx = 2
            elseif idx == 9 then
                aimIdx = 3
            end

            for j = 1, #self.Rewards[aimIdx] do
                self.Rewards[aimIdx][j].go_Equip:SetActive(false)
                self.Rewards[aimIdx][j]:MakeSelected(true)
            end
            -- self.Rewards[aimIdx]:MakeSelected(true);
        end

        -- self:OnTaskItemClick(self.DonateTaskItems[self.OrderIndex+1]);

        local isFindNotFinish = false

        for i = 1, #self.DonateTaskItems do
            if not (self.DonateTaskItems[i].IsSubmited) then
                isFindNotFinish = true
                self:OnTaskItemClick(self.DonateTaskItems[i])
                break
            end
        end

        if not (isFindNotFinish) then
            self:OnTaskItemClick(self.DonateTaskItems[1])
        end
    end
end

function DonateWindow:RefeshRight(configIndex, isSubmited)
    self.ConfigIndex = configIndex

    local dCfg = ConfigDataAccessor.DonateOrderList[configIndex]

    self.ItemSet = dCfg.ItemSet

    if dCfg.ItemSet > 0 then
        -- 道具集
        local sCfg = ConfigDataAccessor.DonateItemSet[dCfg.ItemSet]

        ResourceHelper:LoadIconAsync(self.Icon, "", sCfg.Icon, false)

        self.Name.text = sCfg.ShowName

        self.go_BtnInfo:SetActive(false)
    else
        -- 单个道具
        local cfg = ConfigDataAccessor.Item[dCfg.Item]
        ResourceHelper:LoadIconAsync(self.Icon, "", cfg.Icon, false)

        self.Name.text = cfg.Name

        self.BaseID = dCfg.Item
        self.go_BtnInfo:SetActive(true)
    end

    -- self.go_NotEnough:SetActive(not(self.Enough));

    if isSubmited then
        self.go_BtnFinish:SetActive(false)
        self.go_BtnNotFinish:SetActive(false)
        self.go_Finish:SetActive(true)
        self.NotEnoughTex.text = ""
    else
        if self.Enough then
            self.NotEnoughTex.text = "<color=#25903e>" .. self.HaveNum .. "</color>/" .. self.NeedNum
        else
            self.NotEnoughTex.text = "<color=#d04545>" .. self.HaveNum .. "</color>/" .. self.NeedNum
        end

        self.go_BtnFinish:SetActive(self.Enough)
        self.go_BtnNotFinish:SetActive(not (self.Enough))
        self.go_Finish:SetActive(false)
    end

    for i = 1, #self.items do
        self.items[i]:MakeSelected(isSubmited)
    end

    if dCfg.Spine_id ~= "" then
        self.GoHead:SetActive(true)

        local avatarDataCell = ConfigDataAccessor.Avatar[tonumber(dCfg.Spine_id)]
        if avatarDataCell ~= nil then
            local headCfg = ConfigDataAccessor.ModelHead[avatarDataCell.HeadID]
            if headCfg ~= nil then
                ResourceHelper:LoadIconByAtlasAsync(self.Head, headCfg.HeadAtlasName, headCfg.MinHead, false)
            end
        end
    else
        self.GoHead:SetActive(false)
    end

    self.Title.text = dCfg.ForceName
    self.Desc.text = dCfg.Text

    if dCfg.NPCname ~= "" then
        self.DescName.text = "— " .. dCfg.NPCname
    else
        self.DescName.text = ""
    end
end

function DonateWindow:Refesh()
    self:Open()
end

function DonateWindow:OnGuildDonateSubmitRet(data)
    if data == nil then
        return
    end

    Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_108"))
    self:Refesh()
end

function DonateWindow:OnRefreshGuildDonateOrderRet(data)
    if data == nil then
        return
    end

    Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_109"))
    self:Refesh()
end

function DonateWindow:OnGetGuildDonateCountRewardRet(data)
    if data == nil then
        return
    end

    Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_110"))
    self:Refesh()
end

    print("DonateWindow:Start")
end

function DonateWindow:OnEnable()
end

function DonateWindow:OnDisable()
end

function DonateWindow:OnDestroy()
end

function DonateWindow:OnOpen(arg)
    self.IsOpen = true

    self.OrderIndex = 0

    self:Refesh()

    self.message1 = function(data)
        self:OnGuildDonateSubmitRet(data)
    end
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.GuildDonateSubmitRetID, self.message1, nil)

    self.message2 = function(data)
        self:OnRefreshGuildDonateOrderRet(data)
    end
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.RefreshGuildDonateOrderRetID, self.message2, nil)

    self.message3 = function(data)
        self:OnGetGuildDonateCountRewardRet(data)
    end
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.GetGuildDonateCountRewardRetID, self.message3, nil)
end

function DonateWindow:OnClose(arg)
    -- self:CloseWindow();
end

function DonateWindow:CleanAllSelected()
end

function DonateWindow:OnClickCloseBtn()
    --[[
	if self.module.IsOpenTips then
		self.module.IsOpenTips=false;
		self.module:CloseItemTips();

		self:CleanAllSelected();
	end
	]]
    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.GuildDonateSubmitRetID, self.message1, nil)

    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.RefreshGuildDonateOrderRetID, self.message2, nil)

    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.GetGuildDonateCountRewardRetID, self.message3, nil)

    self.Close()

    --[[
	self.PlayAnimation("DonateWindowAnim_Close", function ()
		self.idxNow=-1;
		self.IsOpen=false;
		self.module:OnCloseAdventureLevel();

		self.Close();
	end)
	]]
end

return DonateWindow

function ArenaSettleWindow:SetCurUIHandler()
	CloseBtnFunc = function() self:OnClickCloseBtn(); end;
	ContinueChallengeBtnFunc = function() self:OnClickContinueChallengeBtn(); end;

	self.m_CloseBtn.OnClick = CloseBtnFunc;
	self.m_ContinueChallengeBtn.OnClick = ContinueChallengeBtnFunc;
end

function ArenaSettleWindow:BindModule(m)
	self.Module = m
end

function ArenaSettleWindow:Awake()
	self:InitAllComponent();
	self:SetCurUIHandler();
end

function ArenaSettleWindow:RenderAllView(isInit)
	local data = self.Args
	self.m_Success:SetActive(data.IsWin)
	self.m_Fail:SetActive(not data.IsWin)

	-- 先展示排名变化再展示奖励列表
	self:RenderRankContent(data)
	self.m_RankContentRoot:SetActive(true)
	self.m_RewardsRoot:SetActive(false)
	self.m_UpRankRoot:SetActive(data.IsRankChanged)
	self.m_NoChangeRankRoot:SetActive(not data.IsRankChanged)
	self.m_UpRankText.text = tostring(data.TargetRank - data.MyRank)

	self.m_NoReward:SetActive(not data.HasReward)
	self:RenderRewardPart(data)
	--self.m_RemainCountText.text = string.format(LanguageHelp:GetLanguageByKey("Local_Str_strRemain"), data.RemainCount) 
	self.m_RemainCountText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_strRemain"), {data.RemainCount});

	self.timer = DOVirtual.DelayedCall(4, function ()
		self.m_RankContentRoot:SetActive(false)
		self.m_RewardsRoot:SetActive(true)
	end)
end

-- 职业图标&立绘显示
function ArenaSettleWindow:RenderJobImage(jobId, imageBg, imgIcon, imageHead)
	-- 职业类型icon设置
	local jobshi = math.floor(jobId / 10)
	if imageBg ~= nil then
		local jobTypeBgName = "Icon_ArmBg" .. jobshi
		ResourceHelper:LoadIconByAtlasAsync(imageBg, ResourceHelper.AtlasPathRole, jobTypeBgName, false)
	end

	if imgIcon ~= nil then
		local jobTypeName = "Icon_Arm" .. jobshi
		ResourceHelper:LoadIconByAtlasAsync(imgIcon, ResourceHelper.AtlasPathRole, jobTypeName, false)
	end

	-- 头像
	if imageHead ~= nil then
		local jobCfg = ConfigDataAccessor.Job[jobId]
    	if jobCfg ~= nil then
	        local avatarDataCell = ConfigDataAccessor.Avatar[jobCfg.AvatarID]
	        if avatarDataCell ~= nil then
	            local headCfg = ConfigDataAccessor.ModelHead[avatarDataCell.HeadID]
	            if headCfg ~= nil then
	            	ResourceHelper:LoadIconByAtlasAsync(imageHead, headCfg.HeadAtlasName, headCfg.MinHead, false)
	            end
	        end
        end
    end
end

function ArenaSettleWindow:RenderRankContent(data)
	self.m_MyNickNameText.text = data.MyName
	--self.m_MyLevelText.text = string.format(strLevel, data.MyLevel)
	self.m_MyLevelText.text=LanguageHelp:GetLanguageByKey("Local_Str_Level")..": "..data.MyLevel;

	--self.m_MyRankText.text = string.format(strRank, data.MyRank)
	self.m_MyRankText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ArenaSettleWindow_strRank"), {data.MyRank});

	self:RenderJobImage(data.MyJobId, nil, self.m_MyJobTypeIconImage, self.m_MyHeadIconImage)

	self.m_TargetNickNameText.text = data.TargetName
	--self.m_TargetLevelText.text = string.format(strLevel, data.TargetLevel)
	self.m_TargetLevelText.text=LanguageHelp:GetLanguageByKey("Local_Str_Level")..": "..data.TargetLevel;

	--self.m_TargetRankText.text = string.format(strRank, data.TargetRank)
	self.m_TargetRankText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ArenaSettleWindow_strRank"), {data.TargetRank});

	self:RenderJobImage(data.TargetJobId, nil, self.m_TargetJobTypeIconImage, self.m_TargetHeadIconImage)
end

function ArenaSettleWindow:RenderRewardPart(data)
	local hasRewardData = data ~= nil and data.FinalDrop ~= nil and data.FinalDrop.Count > 0
	self.m_RewardItemPreviewGo:SetActive(hasRewardData)
	if not hasRewardData then return end

	self.CostItems={}
	for i = 0, data.FinalDrop.Count-1 do
		local v = data.FinalDrop[i]
		table.insert(self.CostItems,{ItemID=v.Index,ItemNum=v.Count})
	end

	self.m_RewardItemPreview:SetItems(self.CostItems);
end

function ArenaSettleWindow:OnDestroy()

function ArenaSettleWindow:OnOpen(arg)
	self.Args = arg
	self:RenderAllView()
end

function ArenaSettleWindow:OnClose(arg)
	if self.timer ~= nil then
		self.timer:Kill()
	end
end

function ArenaSettleWindow:OnClickCloseBtn()
	LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "SendLeaveInstanceReq", nil);
	self:Close()
end

function ArenaSettleWindow:OnClickContinueChallengeBtn()
	local message = CS.ProtoMsg.ArenaGetMyBaseInfoReq()
    NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, message, false)
end

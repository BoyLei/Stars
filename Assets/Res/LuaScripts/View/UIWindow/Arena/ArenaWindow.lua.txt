function ArenaWindow:OnFreshPlayerList(data)
	-- self.ItemDatas=data
	self.ItemDatas=self.module.ChallengePlayerDataList
	-- table.print(self.ItemDatas)
	-- 默认显示3个比自己高的1个比自己低的
	if #self.ItemDatas == 16 then
		self.m_ScrollView:ContinueScroll(#self.ItemDatas,false, 11);
	-- 新号没有比自己低的玩家，数据为14条
	elseif #self.ItemDatas == 14 then
		self.m_ScrollView:ContinueScroll(#self.ItemDatas,false, 10);
	else
		self.m_ScrollView:ContinueScroll(#self.ItemDatas,true);
	end
end

-- 职业图标&立绘显示
function ArenaWindow:RenderJobImage(jobId, imageBg, imgIcon, imageHead)
	-- 职业类型icon设置
	local jobshi = math.floor(jobId / 10)
	if imageBg ~= nil then
		local jobTypeBgName = "Icon_ArmBg" .. jobshi
		ResourceHelper:LoadIconByAtlasAsync(imageBg, ResourceHelper.AtlasPathRole, jobTypeBgName, false)
	end

	if imgIcon ~= nil then
		local jobTypeName = "Icon_Arm" .. jobshi
		ResourceHelper:LoadIconByAtlasAsync(imgIcon, ResourceHelper.AtlasPathRole, jobTypeName, false)
	end

	-- 职业立绘
	if imageHead ~= nil then
		local jobCfg = ConfigDataAccessor.Job[jobId]
    	if jobCfg ~= nil then
	        local avatarDataCell = ConfigDataAccessor.Avatar[jobCfg.AvatarID]
	        if avatarDataCell ~= nil then
	            local headCfg = ConfigDataAccessor.ModelHead[avatarDataCell.HeadID]
	            if headCfg ~= nil then
	            	ResourceHelper:LoadIconByAtlasAsync(imageHead, headCfg.HeadAtlasName, headCfg.MaxHead, false)
	            end
	        end
        end
    end
end

function ArenaWindow:RenderMyRankInfo()
	local data = self.module.MyRankInfoData
	if data == nil then return end
	
	self.m_MineNameText.text = data.PlayerName
	self.m_MineLevelContentText.text = "Lv."..tostring(data.Level)

	self.m_MineFightContentText.text = GetPlayerTotalPowerRankLevel(false) 

	self.m_MineRankContentText.text = tostring(data.Rank)
	self:RenderJobImage(data.JobId, nil, self.m_MineOccupationImage)
end

function ArenaWindow:RenderChallengeReward()
	local config= ConfigDataAccessor.PlayOpenConditions[4]
	if config==nil then return end

	--奖励
	local Items=ConfigDataAccessor.DropShow[config.AwardID];
	self.CostItems={}
	if Items~=nil then
		for i, v in ipairs(Items) do
			table.insert(self.CostItems,{ItemID=v.ItemID,ItemNum=v.Num})
		end
	end
	self.m_ItemPreview:SetItems(self.CostItems);

	--self.m_ChallengeRewardTipText.text = string.format(LanguageHelp:GetLanguageByKey("Local_Str_strChallengeRewardTip"),self.module.FreeChallengeCount)
	self.m_ChallengeRewardTipText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_strChallengeRewardTip"), {self.module.FreeChallengeCount});

end

function ArenaWindow:OnItemRefesh(tr,idx)
	if self.ItemList ==nil then return end
	for i=1,#self.ItemList do
		if self.ItemList[i].transform == tr then
            self.ItemList[i]:SetItem(self.ItemDatas[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
		end
	end

    local bi = ArenaPlayerItem:New(tr.gameObject);
	bi.gameObject:SetActive(true);
	bi:SetItem(self.ItemDatas[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1

    table.insert(self.ItemList, bi);

end

function ArenaWindow:RenderChallengeAndRewardCount()
	local info = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.ComCount):GetData(17)
	local challengedCount = info.AddCount - info.RemainCount

	--self.m_RemainContentText.text = string.format(LanguageHelp:GetLanguageByKey("Local_Str_strRemainContent"), info.RemainCount, info.AddCount)
	self.m_RemainContentText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_strRemainContent"), {info.RemainCount, info.AddCount});

	local canRewardCount = challengedCount >= self.module.FreeChallengeCount and 0 or self.module.FreeChallengeCount - challengedCount
	self.m_TreasureCountText.text = tostring(canRewardCount)
	self.m_TreasureBoxReceivedAll:SetActive(canRewardCount <= 0)
end

function ArenaWindow:RenderRankRewardPart(data)
	local hasRewardData = data ~= nil and data.FinalDrop ~= nil and data.FinalDrop.Count > 0
	self.m_RankRewardPartRoot:SetActive(hasRewardData)
	if not hasRewardData then return end

	--self.m_RankRewardTimeText.text = string.format(strRankRewardTime, TimeToFullString2(TimeUtils.Stamp2DataTime(data.MyLastAwardTime)))
	self.m_RankRewardTimeText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_strRankRewardTime"), {TimeToFullString2(TimeUtils.Stamp2DataTime(data.MyLastAwardTime))});

	--self.m_RankRewardText.text = string.format(strRankReward, data.MyRankNum) 
	self.m_RankRewardText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_strRankReward"), {data.MyRankNum});

	self.CostItems={}
	for i = 0, data.FinalDrop.Count-1 do
		local v = data.FinalDrop[i]
		table.insert(self.CostItems,{ItemID=v.Index,ItemNum=v.Count})
	end
	
	self.m_RewardItemPreview:SetItems(self.CostItems);
end

function ArenaWindow:RenderBuyPartRoot(isShow)
	self.m_BuyCountPartRoot:SetActive(isShow)
	if not isShow then return end

	--self.m_BuyTipText.text = string.format(LanguageHelp:GetLanguageByKey("Local_Str_strBuyTipText"),self.module:GetCanBuyChallengeCount())
	self.m_BuyTipText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_strBuyTipText"), {self.module:GetCanBuyChallengeCount()});

	local cfg = ConfigDataAccessor.Item[self.CostCurrencyId];
	ResourceHelper:LoadIconAsync(self.m_CostcurrencyImage, "", cfg.Icon, false);
	self:RenderBuyCount()
end

function ArenaWindow:RenderBuyCount()
	self.m_BuyCountText.text = tostring(self.BuyCount)
	self.m_CostCurrencyText.text = tostring(self.module:GetCostCurrencyNum(self.BuyCount))
end

function ArenaWindow:OnClickAddRemainCountBtn()
	if self.module:GetCanBuyChallengeCount() <= 0 then
		Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_149"))
		return
	end
	self.BuyCount = 1
	self:RenderBuyPartRoot(true)
end

function ArenaWindow:OnClickBuyConfirmBtn()
	if self.BuyCount <= 0 then return end
	
	local costNum = self.module:GetCostCurrencyNum(self.BuyCount)
	local ownNum = BusinessManager.Instance:GetCurrencyNum(self.CostCurrencyId)

	if ownNum < costNum then
		Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_81"))
		return
	end

	self.module:ArenaBuyChallengeTimesReq(self.BuyCount)
	self.BuyCount = 0

	self.m_BuyCountPartRoot:SetActive(false)
end

function ArenaWindow:OnClickAddBtn()
	if self.BuyCount >= self.module:GetCanBuyChallengeCount() then return end;
	self.BuyCount = self.BuyCount + 1
	self:RenderBuyCount()
end

function ArenaWindow:OnClickSubBtn()
	if self.BuyCount <= 1 then return end;
	self.BuyCount = self.BuyCount - 1
	self:RenderBuyCount()
end

function ArenaWindow:OnClickBattleRecordBtn()
	self.module:ArenaGetMyFightLogReq()
end

function ArenaWindow:OnClickChangeChallengeBtn()
	if self.module:GetCurrentChallengeCount() > 0 then
		self.module:ArenaGetFightListReq()
	else
		if self.module:GetCanBuyChallengeCount() > 0 then
			self.OnClickAddRemainCountBtn()
		else
			Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_261"))
		end
	end
end

function PartnerEquipComposeWindow:OnClickAddBtn()
	if self.Full then
		print("点击  11")
		self:CleanItem();
	else
		print("点击  22")
		self.SlotItemEntitys[self.CurrentItemID]={}
		--可交易次数
		--可交易次数更高
		--可交易次数更低
		--同类型优先选择未装备的魂石
		--self.CurrentItemEntityID

		local tempTradedCount=0;
		local item=BusinessManager.Instance:GetItemByItemEntityId(self.CurrentItemEntityID)
		if item~=nil then
			tempTradedCount=item.TradedCount
		end

		local list={}
		local items=BusinessManager.Instance:GetAllItemsByItemBaseId(self.CurrentItemID)

		if items.Count>0 then
			for k, v in pairs(items) do

				local data={}
				data.data=v;

				if v.HeroID==0 then

					data.Equip=1
				else
					data.Equip=0
				end
				data.TradedCount=v.TradedCount
				if v.TradedCount==tempTradedCount then
					data.Same=1;
				else
					data.Same=0;
				end

				table.insert(list,data)
			end

			table.sort(list,function (item1,item2)
					if item1.Same==item2.Same then
						if item1.Equip==item2.Equip then
							return item1.TradedCount<item2.TradedCount
						else
							return item1.Equip<item2.Equip
						end
					else
						return item1.Same<item2.Same
					end
				end)

			print(#list)
			print(items.Count)
			self:CleanItem();

			local cnt=0;
			for k, v in pairs(list) do
				if cnt<3 then
					self:SetComposeItem(k,v.data.BaseID,v.data.ID)
				end
				cnt=cnt+1
			end
		end
	end
end

function PartnerEquipComposeWindow:IsSelect(itemID,entityID)
	local list=self.SlotItemEntitys[itemID]
	if list~=nil then
		for k, v in pairs(list) do
			if tostring(v.value)==tostring(entityID) then
				return  true
			end
		end
	end
	return  false
end

function PartnerEquipComposeWindow:OnClickEmpty(index)
	local SelectIndex=0;
	local list={}
	local items=BusinessManager.Instance:GetAllItemsByItemBaseId(self.CurrentItemID)
	for k, v in pairs(items) do
		
		if v.HeroID==0 then

			if not self:IsSelect(v.BaseID,v.ID) then
				if SelectIndex==0 then
					SelectIndex=tostring(v.ID)
				end
			end
			table.insert(list,v)
		end
	end
	
	self.PartnerEquipSelect:OnOpen({SlotIndex=index,ItemID=self.CurrentItemID},
		list,
		SelectIndex,
		function ()
			self:OnEquipSelectClose()
		end)
end

function PartnerEquipComposeWindow:SetSelectID(ID,entityID)
	self.CurrentItemID=ID
	self.CurrentItemEntityID=entityID
	local itemid=self:GetNextID(ID)
	if itemid==0 then
		print("找不到合成道具"..ID )
		return 
	end
	local item=ConfigDataAccessor.Item[itemid]
	if item~=nil then
		self.m_name.text=item.Name;
		ResourceHelper:LoadIconAsync(self.m_icon, "", item.Icon, false)
	end

	local equipCfg=ConfigDataAccessor.PartnerEquip[itemid]
	if equipCfg~=nil then
		for k, v in pairs(self.Stars) do
			v:SetActive(k<=equipCfg.AdvancedLevel)
		end
	end


	local item_Cfg=ConfigDataAccessor.Item[ID]
	local itemEqip_Cfg=ConfigDataAccessor.PartnerEquip[ID]
	local itemEqip_Next_Cfg=ConfigDataAccessor.PartnerEquip[itemid]


	self.m_Cur_Power.text=itemEqip_Cfg.Power
	self.m_Next_Power.text=itemEqip_Next_Cfg.Power

	local datas={}
	for i = 1, #itemEqip_Cfg.QualiFications do
		table.insert(datas,{ID=itemEqip_Cfg.QualiFications[i],Cur=itemEqip_Cfg.QualiFicationsValue[i],Next=itemEqip_Next_Cfg.QualiFicationsValue[i],IsPlayer=false})
	end

	for i = 1, #itemEqip_Cfg.Nature do
		table.insert(datas,{ID=itemEqip_Cfg.Nature[i],Cur=itemEqip_Cfg.Value[i],Next=itemEqip_Next_Cfg.Value[i],IsPlayer=true})
	end

	for k, v in pairs(datas) do
		self:SetAtrribuite(k,v)
	end

	self.m_infomation:SetActive(false)
	local equipCfg=ConfigDataAccessor.PartnerEquip[ID]
	if equipCfg~=nil then
		
	if #equipCfg.Effect>0 then
		self.m_infomation:SetActive(true)
		local effectcfg=ConfigDataAccessor.PartnerEquipEffect[equipCfg.Effect[1]]
		if effectcfg~=nil then


			local PartnerType="";

			local PartnerTypeCfg=ConfigDataAccessor.PartnerType[effectcfg.PartnerType]
			if PartnerTypeCfg~=nil then
				PartnerType=PartnerTypeCfg.Name
			end

			if PartnerType== nil or PartnerType=="" or PartnerType=='' then
				self.m_SkillName.text=effectcfg.Name
			else
				self.m_SkillName.text=effectcfg.Name.."<"..PartnerType..">"
			end

			self.m_SkillLevel.text=StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LvLimit3"),{equipCfg.EquipLevel})
			self.m_SkillDesc.text=effectcfg.Desc
			--self.m_Skillicon=self:Find("Content/infomation/Skillicon"):GetComponent(typeof(Image));;

		end
	end
	end
	self:CleanItem();
	self:SetComposeItem(1,ID,entityID)

end

function PartnerEquipComposeWindow:OnSlotItemChange(itemID)
	local count=0;
	if self.SlotItemEntitys[itemID]~=nil then
		for k, v in pairs(self.SlotItemEntitys[itemID]) do
			if v~=nil then
				count=count+1
			end
		end
	end
	return count==3;
end

function PartnerEquipComposeWindow:SetComposeItem(index,itemID,entityID)

	if self.SlotItemEntitys[itemID] ==nil then
		self.SlotItemEntitys[itemID]={}
	end
	--table.insert(self.SlotItemEntitys[itemID],index,uint64.new(entityID))
	self.SlotItemEntitys[itemID][index]=uint64.new(entityID)
	local isFull=self:OnSlotItemChange(itemID);
	self.Full=isFull
	self.m_UpBtn.gameObject:SetActive(isFull)
	if isFull then
		self.m_addText.text=LanguageHelp:GetLanguageByKey("partner_sys_17")
	else
		self.m_addText.text=LanguageHelp:GetLanguageByKey("partner_sys_16")
	end

	local itemCfg=ConfigDataAccessor.Item[itemID]
	local equipCfg=ConfigDataAccessor.PartnerEquip[itemID]
	if itemCfg~=nil and equipCfg~=nil then
		local centerItem=self.Items[index]
		local rightItem=self.Right_Items[index]

		if centerItem~=nil then
			centerItem.Has:SetActive(true)
			local quality = itemCfg.Quality;
			if quality==99 then
				quality=1;
			end
			ResourceHelper:LoadIconByAtlasAsync(
				centerItem.di,
				"UI/Partner/Atlas/Partner1",
				"Partner_btn_" .. quality,
				false,
				function (s)
					centerItem.di.gameObject:SetActive(true)
				end
			)

			ResourceHelper:LoadIconAsync(centerItem.icon, "", itemCfg.Icon, false)
			if equipCfg.AdvancedLevel==1 then
				centerItem.Star1:SetActive(true)
				centerItem.Star2:SetActive(false)
				centerItem.Star3:SetActive(false)
			elseif  equipCfg.AdvancedLevel==2 then
				centerItem.Star1:SetActive(false)
				centerItem.Star2:SetActive(true)
				centerItem.Star3:SetActive(false)
			elseif  equipCfg.AdvancedLevel==3 then
				centerItem.Star1:SetActive(false)
				centerItem.Star2:SetActive(false)
				centerItem.Star3:SetActive(true)
			end
		end

		if rightItem~=nil then

			rightItem.Has:SetActive(true)
			local quality = itemCfg.Quality;
			if quality==99 then
				quality=1;
			end
			ResourceHelper:LoadIconByAtlasAsync(
				rightItem.di,
				ResourceHelper.AtlasPathCommonItem,
				"Bag_Grid_sq" .. quality,
				false,
				function (s)
					rightItem.di.gameObject:SetActive(true)
				end
			)

			ResourceHelper:LoadIconAsync(rightItem.icon, "", itemCfg.Icon, false)
		end

	end

end

function PartnerEquipComposeWindow:CleanItem()
	for k, v in pairs(self.Items) do
		v.Clean(v.empty,v.Has);
	end

	for k, v in pairs(self.Right_Items) do
		v.Clean(v.empty,v.Has);
	end
	
	--local isFull=self:OnSlotItemChange(itemID);
	self.Full=false
	self.m_UpBtn.gameObject:SetActive(self.Full)
	if self.Full then
		self.m_addText.text=LanguageHelp:GetLanguageByKey("partner_sys_17")
	else
		self.m_addText.text=LanguageHelp:GetLanguageByKey("partner_sys_16")
	end
end

function PartnerEquipComposeWindow:SetAtrribuite(index,data)
	local item=self.ItemAtrrs[index]
	if item~=nil and data~=nil then

		local Name=""
		if data.IsPlayer then

			local config1=  ConfigDataAccessor.GetAttrByID(data.ID);
			local attrName1 = "";
			if config1 ~= nil then
				attrName1 = config1.Name;
			end

			Name=attrName1..LanguageHelp:GetLanguageByKey("partner_sys_12")

		else
			local config=ConfigDataAccessor.GetAttrByID(data.ID)
			if config~=nil then
				Name=config.Name;
			end
		end
		item.name.text=Name;

		item.Cur.text=data.Cur
		item.Next.text=data.Next
		self:GetSpriteByAtlas("UI/Icons/Atlas/IconAttribute","Attribute_"..data.ID,function (sp)
				item.icon.sprite=sp
			end)
	end

end

function  PartnerEquipComposeWindow:SetItemEmpty(item,isEmpty)
	item.empty:SetActive(isEmpty)
	item.Sel:SetActive(not isEmpty)
	item.icon.gameObject:SetActive(not isEmpty)
	item.Star1:SetActive(not isEmpty)
	item.Star2:SetActive(not isEmpty)
	item.Star3:SetActive(not isEmpty)
end


function PartnerEquipComposeWindow:GetNextID(id)
	for k, v in pairs(ConfigDataAccessor.Compose) do
		if v.Id==id then
			return v.TargeID
		end
	end
	return 0
end

function Construct:DoChristen()
	local hasHeight=false;
	local config=ConfigDataAccessor.Partner[self.PartnerData.Index];
	if config==nil then
		return
	end
	--消耗组ID
	local qualificate=ConfigDataAccessor.QualificationsExpend[config.QualificationsExpendID][self.PartnerData.AptitudeBreakStep]
	if qualificate~=nil then

		local totalUnSave=0;
		local totalBase=0;
		for i = 1, #qualificate.QualiFications do

			local t=self.PartnerData.AMD.Aptitudes[qualificate.QualiFications[i]]
			totalUnSave=totalUnSave+t.UnSaveValue;
			totalBase=totalBase+t.BaseValue;
		end
		hasHeight=totalUnSave>totalBase
	end

	local eCost=ConfigDataAccessor.QualificationsExpend[config.QualificationsExpendID][self.PartnerData.AptitudeBreakStep]


	if hasHeight then
		UIAPI.ShowMsgBox(3,function (EventName)
				if EventName=="SURE" then
					if self:SuitCost(eCost) then
						LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule, "OnPartnerChristenReq",self.PartnerData.Index);
					end
				end
			end)
	else
		if self:SuitCost(eCost) then
			LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule, "OnPartnerChristenReq",self.PartnerData.Index);
		end
	end
end

function Construct:SuitCost(qualificate)
	local money=	BusinessManager.Instance:GetCurrencyNum(qualificate.Money[1])
	if money< qualificate.Money[2] then
		--金币不足
		local itemName=ConfigDataAccessor.Item[qualificate.Money[1]]
		if itemName~=nil then
			--Util.ShowMessage(itemName.Name.."不足");
			Util.ShowMessage(StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ShopItem_buyErrorMsg"), {itemName.Name}))
		end
		print("金币不足")
		return false;
	end

	local itemSuit=true;
	for i = 1, #qualificate.Item_id do
		local temp=BusinessManager.Instance:GetCurrencyNum(qualificate.Item_id[i])
		if temp< qualificate.Item_num[i] then
			itemSuit=false;
			break;
		end
	end

	if not itemSuit then
		Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_No_Item"));
		print(LanguageHelp:GetLanguageByKey("Local_Str_No_Item"))
		return false;
	end

	return true;
end

function Construct:SetRedPoint(data)
	if self.m_RedPoint~=nil then
		local partnerMgr = BusinessManager.Instance:GetPartnerMDMgr();
		if partnerMgr~=nil then
			local red=	partnerMgr:UpdatePartnerConstructRed(data)
			self.m_RedPoint:SetActive(red)
			self.RedPoint:SetActive(red)
		end
	end
end


function Construct:OnOpen(data)

	self.PartnerData=data;
	self:SetRedPoint(data);
	self.m_Tipszizhi:SetActive(false)
	self:SetFoldText(LanguageHelp:GetLanguageByKey("partner_sys_24"))
	self.IsLower=false;
	self.IsFull=false;
	--头顶信息
	self.PartnerInfo:OnOpen(data,true,true)
	self.m_zizhi:SetActive(not isEmpty)
	self.m_Stage.text=StringFormat(LanguageHelp:GetLanguageByKey("partner_sys_1"),{data.AptitudeBreakStep})
	local config=ConfigDataAccessor.Partner[data.Index];
	if config==nil then
		return
	end

	local qualificate=ConfigDataAccessor.Qualifications[config.QualificationID][data.AptitudeBreakStep]
	if qualificate==nil then
		return
	end
	self.Props={}
	local isEmpty=true;

	local totalold=0;
	local totalunsave=0;
	local max=0;
	for i = 1, #qualificate.QualiFications do
		max=max+qualificate.Value_max[i]
		local AoiAttr= ConfigDataAccessor.ParConversion[qualificate.QualiFications[i]]
		if AoiAttr~=nil then
			local t=data.AMD.Aptitudes[qualificate.QualiFications[i]]
			if t.UnSaveValue >0 then
				isEmpty=false;
			end
			totalold=totalold+t.BaseValue
			totalunsave=totalunsave+t.UnSaveValue
		end
	end


	for i = 1, #qualificate.QualiFications do
		local AoiAttr= ConfigDataAccessor.ParConversion[qualificate.QualiFications[i]]
		if AoiAttr~=nil then
			local t=data.AMD.Aptitudes[qualificate.QualiFications[i]]

			if self.AttrList[qualificate.QualiFications[i]]~=nil then
				self.AttrList[qualificate.QualiFications[i]]:OnOpen({PropValue=data.AMD.Aptitudes[qualificate.QualiFications[i]].UnSaveValue,
						OldValue=data.AMD.Aptitudes[qualificate.QualiFications[i]].BaseValue,
						PropMax=qualificate.Value_max[i]},isEmpty);
			end
		end
	end


	local isFull=max==totalold;
	self.IsFull=isFull;
	self.m_UpBtn.gameObject:SetActive(isEmpty)
	self.m_goBtn.gameObject:SetActive(not isEmpty)
	self.m_UpBtn1.gameObject:SetActive(not isEmpty)
	self.m_Back.gameObject:SetActive(not isEmpty)
	self.m_Tips.gameObject:SetActive(not isEmpty)
	self.IsLower=totalunsave<=totalold;
	self.m_ItemObj:SetActive(not isFull)
	self.m_MoneyObj:SetActive(not isFull)
	if isFull then
		self.m_zizhiText.text=LanguageHelp:GetLanguageByKey("partner_sys_8")
		self.m_UpBtnText.text=LanguageHelp:GetLanguageByKey("partner_sys_10")
	else
		self.m_UpBtnText.text=LanguageHelp:GetLanguageByKey("partner_sys_9")
		if isEmpty then
			self.m_zizhiText.text=""
		else
			if totalold==totalunsave then
				self.m_zizhiText.text=LanguageHelp:GetLanguageByKey("partner_sys_3")
			else
				if totalunsave>totalold then
					self.m_zizhiText.text=LanguageHelp:GetLanguageByKey("partner_sys_4")
				else
					self.m_zizhiText.text=LanguageHelp:GetLanguageByKey("partner_sys_5")
				end
			end
		end
	end
	local itemless=false
	local moneyless= false
	local lessStar=false
	if isFull then
		local qualificate=ConfigDataAccessor.Qualifications[config.QualificationID][data.AptitudeBreakStep]
		if qualificate~=nil then
			if qualificate.Item_id~=nil and #qualificate.Item_id>0 and qualificate.Item_num~=nil and #qualificate.Item_num>0 then
				itemless=self:SetCostItem(qualificate.Item_id[1],qualificate.Item_num[1])
			end
			
			if data.CurStar<qualificate.Star then
				lessStar=true
			end
			moneyless= self:SetCostMoney(qualificate.Money)
		end
	else
		
		
		local eCost=ConfigDataAccessor.QualificationsExpend[config.QualificationsExpendID][data.AptitudeBreakStep]
		itemless=self:SetCostItem(eCost.Item_id[1],eCost.Item_num[1])
		moneyless= self:SetCostMoney(eCost.Money)
	end
	
	
	
	local less=itemless or  moneyless
	
	self.m_UpBtn.UnClick=less
	self.m_UpBtn.ForbidClick=less
	
	--self.m_goBtn.UnClick=less
	--self.m_goBtn.ForbidClick=less
	
	self:UpdateState(qualificate.LevelGrade)
end

function Construct:UpdateState(needLevel)
	--玩家冒险等级
	local curRiskLevel =self.PartnerData.Level;

	if self.IsFull  and (curRiskLevel< needLevel) then
		self.m_btns.gameObject:SetActive(false)
		self.m_UnUpgrade:SetActive(true)
		self.m_UnUpgradeUpText.text=StringFormat( LanguageHelp:GetLanguageByKey("partner_sys_45"),{needLevel})
	else
		self.m_btns.gameObject:SetActive(true)
		self.m_UnUpgrade:SetActive(false)
	end
end

function Construct:SetCostItem(id,number)

	local ownCnt=BusinessManager.Instance:GetCurrencyNum(id)
	local less=ownCnt<number;

	local itemcfg= ConfigDataAccessor.Item[id];
	if itemcfg~=nil then
		for k, v in pairs(self.CostItemImage) do
			ResourceHelper:LoadIconAsync(v, "", itemcfg.Icon, false)
		end
	end

	for k, v in pairs(self.CostItemText) do

		local text=ownCnt.."/"..number
		if less then
			text= "<color=#FF0000>".. ownCnt.."/"..number.."</color>"
		end
		v.text=text;
	end
	return less

end

function Construct:SetCostMoney(money)

	local id=money[1]
	local number=money[2]

	local ownCnt=BusinessManager.Instance:GetCurrencyNum(id)
	local less=ownCnt<number;

	local itemcfg= ConfigDataAccessor.Item[id];
	if itemcfg~=nil then
		for k, v in pairs(self.CostMoneyImage) do
			ResourceHelper:LoadIconAsync(v, "", itemcfg.Icon, false)
		end
	end

	for k, v in pairs(self.CostMoneyText) do

		local text=number
		if less then
			text= "<color=#FF0000>"..number.."</color>"
		end

		v.text=text;
	end
	return less
end
function Construct:DoTopped()

	local config=ConfigDataAccessor.Partner[self.PartnerData.Index];
	if config~=nil then
		local qualificate=ConfigDataAccessor.Qualifications[config.QualificationID][self.PartnerData.AptitudeBreakStep]
		if qualificate~=nil then
			if self.PartnerData.CurStar<qualificate.Star then
				print(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_213"))
				Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_213"));
				return
			end
			LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule, "OnPartnerLimitBreakReq",self.PartnerData.Index);
		end
	end

end

function Tipszizhi:OnCloseTips()
	self.prefab:SetActive(false)
	if self.onCloseCb~=nil then
		self.onCloseCb()
	end
end

function Tipszizhi:OnOpen(data)
	self.prefab:SetActive(true)
	local config=ConfigDataAccessor.Partner[data.Index];
	if config==nil then
		return
	end
	local maxBreak=0;
--[data.AptitudeBreakStep]
	local qualificate=ConfigDataAccessor.Qualifications[config.QualificationID]
	if qualificate==nil then
		return
	end
	
	for k, v in pairs(qualificate) do
		if v.BreakStep> maxBreak then
			maxBreak=v.BreakStep
		end
	end
	
	---已经满了
	local isMax=data.AptitudeBreakStep>=maxBreak;
	
	self.m_Next:SetActive(not isMax)
	self.m_Max:SetActive(isMax)
	
	local curqualificate=qualificate[data.AptitudeBreakStep]
	local curMax=0;
	for k, v in pairs(curqualificate.Value_max) do
		curMax=curMax+v;
	end	
	
	self.m_CurStage.text=tostring(data.AptitudeBreakStep)
	self.m_CurMax.text=tostring(curMax)	
	
	local isFull=true;
	
	for i = 1, #curqualificate.QualiFications do
		local AoiAttr= ConfigDataAccessor.ParConversion[curqualificate.QualiFications[i]]
		if AoiAttr~=nil then
			local t=data.AMD.Aptitudes[curqualificate.QualiFications[i]]
			if t.BaseValue < curqualificate.Value_max[i] then
				isFull=false;
				break;
			end
		end
	end
	
	
	if not isMax then
		local nextqualificate=qualificate[data.AptitudeBreakStep+1]
		local nextMax=0;
		for k, v in pairs(nextqualificate.Value_max) do
			nextMax=nextMax+v;
		end
		self.m_NextStage.text=tostring(data.AptitudeBreakStep+1)
		self.m_NextMax.text=tostring(nextMax)
		
		self.Conditions[1].Content.text=StringFormat(LanguageHelp:GetLanguageByKey("partner_sys_6"),{data.AptitudeBreakStep})
		self.Conditions[1].State:SetActive(isFull)
		
		self.Conditions[2].Content.text=StringFormat(LanguageHelp:GetLanguageByKey("partner_sys_7"),{nextqualificate.Star})
		self.Conditions[2].State:SetActive(data.CurStar>=nextqualificate.Star)
	end
end

function SoulStone:ctor(go,red)

	self.m_None=self:Find("Main/None").gameObject;

	self.m_PartnerInfo=self:Find("Main/Top/PartnerInfo").gameObject;
	self.m_hecheng=self:Find("hecheng").gameObject;

		table.insert(self.m_List,{

				Text=self:Find("Main/Has/Content/List/Item"..i.."/Text"):GetComponent(typeof(Text)),

function SoulStone:OnPartnerEquipUPRetEvent(data)
	self:FreshSolts()
end

function  SoulStone:FreshSolts()
	local equip1= BusinessManager.Instance:GetPartnerEquipedEquip(self.PartnerData.ID,1);
	local equip2= BusinessManager.Instance:GetPartnerEquipedEquip(self.PartnerData.ID,2);
	local equip3= BusinessManager.Instance:GetPartnerEquipedEquip(self.PartnerData.ID,3);
	local isEmpty=(equip1==nil) and(equip2==nil) and(equip3==nil)
	self.m_None:SetActive(isEmpty)
	self.m_Has:SetActive(not isEmpty)
	self.m_Slots[1]:OnOpen(equip1)
	self.m_Slots[2]:OnOpen(equip2)
	self.m_Slots[3]:OnOpen(equip3)
	self:SetRedPoint(self.PartnerData)
	
end

function SoulStone:SetRedPoint(data)
	local equip1= BusinessManager.Instance:GetPartnerEquipedEquip(data.ID,1);
	local equip2= BusinessManager.Instance:GetPartnerEquipedEquip(data.ID,2);
	local equip3= BusinessManager.Instance:GetPartnerEquipedEquip(data.ID,3);
	
	local type1Cnt=0;
	local type2Cnt=0;
	local type3Cnt=0;
	--拿到所有的伙伴装备
	local items = BusinessManager.Instance:GetItemsByItemType(4, 0)
	for k, v in pairs(items) do
		local cfg = ConfigDataAccessor.Item[v.BaseID]
		if cfg.SubType == 1 then
			type1Cnt=type1Cnt+1;
		elseif cfg.SubType == 2 then
			type2Cnt=type2Cnt+1;
		elseif cfg.SubType == 3 then
			type3Cnt=type3Cnt+1;	
		end
	end

	local red=(equip1==nil and type1Cnt>0) or (equip2==nil and type2Cnt>0) or (equip3==nil and type3Cnt>0)
	self.RedPoint:SetActive(red)
end


function SoulStone:OnWearParEquipRet(data)
	self:FreshSolts()
end

function SoulStone:OnTakeOffParEquipRet(data)
	self:FreshSolts()
end


function SoulStone:ClickEmptySlot(id)

	for k, v in pairs(self.m_Slots) do
		v:SetSelect(k==id)
	end
	print("ClickEmptySlot"..id)
	LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule,"OnPartnerEquipWindow",{
			self.PartnerData.ID,
			self.PartnerData.Index,
			id,
			0,
			0,
			function ()
				self:OnCallBackForOtherWindow()
			end
			})
end

function SoulStone:SelectSlot(id,item)
	print("SelectSlot"..id)
	for k, v in pairs(self.m_Slots) do
		v:SetSelect(k==id)
	end
	self.SelectItem=item
	self:SetItem(item.BaseID,true)

	self.m_CurSelectID=id;
	self.m_CurSelectItemID=item.BaseID
	self.m_CurSelectItemEntityID=item.ID
	--LuaModuleDispatcher:SendMessage(ModuleDef.BattleTeamModule,"OnPartnerEquipWindow",{self.PartnerData.ID,self.PartnerData.Index,id,itemID})

end

	local TakeOffParEquipRetEvent=LuaModuleDispatcher:Event(ModuleDef.BattleTeamModule,"TakeOffParEquipRetEvent");

function SoulStone:OnOpen(arg)
	self.PartnerData=arg;
	self:BackMain();
	self.PartnerInfo:OnOpen(arg,true,false)

	local equip1= BusinessManager.Instance:GetPartnerEquipedEquip(self.PartnerData.ID,1);
	local equip2= BusinessManager.Instance:GetPartnerEquipedEquip(self.PartnerData.ID,2);
	local equip3= BusinessManager.Instance:GetPartnerEquipedEquip(self.PartnerData.ID,3);
	local isEmpty=(equip1==nil) and(equip2==nil) and(equip3==nil)
	self.m_None:SetActive(isEmpty)
	self.m_Has:SetActive(not isEmpty)
	self.m_Slots[1]:OnOpen(equip1)
	self.m_Slots[2]:OnOpen(equip2)
	self.m_Slots[3]:OnOpen(equip3)

	local item=nil;
	local idx=0;
	if not isEmpty then

		if equip1~=nil then
			self.m_Slots[1]:SetSelect(true)
			item=equip1;
			idx=1
		else
			if equip2~=nil then
				self.m_Slots[2]:SetSelect(true)
				item=equip2;
				idx=2
			else
				if equip3~=nil then
					self.m_Slots[3]:SetSelect(true)
					item=equip3;
					idx=3
				end
			end
		end
	end

	if item~=nil then
		self.SelectItem=item
		self.m_CurSelectID=idx
		self.m_CurSelectItemID=item.BaseID
		self:SetItem(item.BaseID,true)

	end
end

function SoulStone:SetItemCompose(index,Id,Current)
	local item=self.m_ItemGroups[index]
	if item~=nil then

		local config=ConfigDataAccessor.Item[Id]
		if config~=nil then
			ResourceHelper:LoadIconAsync(item.icon, "", config.Icon, false)

			self:GetSpriteByAtlas("UI/Partner/Atlas/Partner1","Partner_frameF_"..config.Quality,function (sp)
					item.Normal.sprite=sp
					item.Now.sprite=sp
				end)
		end
		item.ItemID=Id
		item.Sel:SetActive(index==Current)
		item.Now.gameObject:SetActive(index==Current)
	end
end

function SoulStone:GetNextID(id)
	for k, v in pairs(ConfigDataAccessor.Compose) do
		if v.Id==id then
			return v.TargeID
		end
	end
	return 0
end

function SoulStone:GetPreviousID(id)
	for k, v in pairs(ConfigDataAccessor.Compose) do
		if v.TargeID==id then
			return v.Id
		end
	end
	return 0
end

function SoulStone:SetItem(id,needSetItem)

	local equipCfg=ConfigDataAccessor.PartnerEquip[id]
	if equipCfg~=nil then

		local itemCfg=ConfigDataAccessor.Item[id]
		if itemCfg~=nil then
			self.m_ItemName.text=itemCfg.Name
		end
		if needSetItem then

			local itemList={}
			local LastitemID=id
			local LastIndex=equipCfg.AdvancedLevel

			if LastIndex==1 then
				itemList[equipCfg.AdvancedLevel]=id
				itemList[equipCfg.AdvancedLevel+1]=self:GetNextID(itemList[equipCfg.AdvancedLevel])
				itemList[equipCfg.AdvancedLevel+2]=self:GetNextID(itemList[equipCfg.AdvancedLevel+1])
			elseif LastIndex==2 then
				itemList[equipCfg.AdvancedLevel-1]=self:GetPreviousID(id)
				itemList[equipCfg.AdvancedLevel]=id
				itemList[equipCfg.AdvancedLevel+1]=self:GetNextID(id)
			elseif LastIndex==3 then
				itemList[equipCfg.AdvancedLevel]=id
				itemList[equipCfg.AdvancedLevel-1]=self:GetPreviousID(id)
				itemList[equipCfg.AdvancedLevel-2]=self:GetPreviousID(itemList[equipCfg.AdvancedLevel-1])
			end

			for k, v in pairs(itemList) do
				self:SetItemCompose(k,v,equipCfg.AdvancedLevel)
			end


			self.m_goBtn.gameObject:SetActive(equipCfg.AdvancedLevel<3)
		end
		--self.m_LevelLimit_tt.text=StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LvLimit"),{equipCfg.Level})
		--self:SetStar(equipCfg.AdvancedLevel)
		self.m_Battle.text=tostring(equipCfg.Power)
		local datas={}
		if #equipCfg.QualiFications>0 and #equipCfg.QualiFications==#equipCfg.QualiFicationsValue then
			for i = 1, #equipCfg.QualiFications do
				table.insert(datas,{IsPlayer=false,ID=equipCfg.QualiFications[i],Value=equipCfg.QualiFicationsValue[i]})
			end
		end

		if #equipCfg.Nature>0 and #equipCfg.Nature==#equipCfg.Value then
			for i = 1, #equipCfg.Nature do
				table.insert(datas,{IsPlayer=true,ID=equipCfg.Nature[i],Value=equipCfg.Value[i]})
			end
		end

		for k, v in pairs(datas) do
			self:SetPropItem(k,v)
		end
		self.m_infomation:SetActive(false)
		if #equipCfg.Effect>0 then
			self.m_infomation:SetActive(true)
			local effectcfg=ConfigDataAccessor.PartnerEquipEffect[equipCfg.Effect[1]]
			if effectcfg~=nil then

				local PartnerType="";

				local PartnerTypeCfg=ConfigDataAccessor.PartnerType[effectcfg.PartnerType]
				if PartnerTypeCfg~=nil then
					PartnerType=PartnerTypeCfg.Name
				end

				if PartnerType== nil or PartnerType=="" or PartnerType=='' then
					self.m_SkillName.text=effectcfg.Name
				else
					self.m_SkillName.text=effectcfg.Name.."<"..PartnerType..">"
				end

				--self.m_SkillName.text=effectcfg.Name
				self.m_SkillLevel.text=StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LvLimit3"),{equipCfg.EquipLevel})
				self.m_SkillDesc.text=effectcfg.Desc
				--self.m_Skillicon=self:Find("Content/infomation/Skillicon"):GetComponent(typeof(Image));;

			end
		end

	end

end

function SoulStone:SetPropItem(index,data)
	local item=self.m_List[index]
	if item~=nil and data~=nil then

		local Name=""
		if data.IsPlayer then
			local config1=  ConfigDataAccessor.GetAttrByID(data.ID);
			local attrName1 = "";
			if config1 ~= nil then
				attrName1 = config1.Name;
			end
			Name=attrName1..LanguageHelp:GetLanguageByKey("partner_sys_12")

		else
			local config=ConfigDataAccessor.GetAttrByID(data.ID)
			if config~=nil then
				Name=config.Name;
			end

			--读取伙伴表
			Name=ConfigDataAccessor.ParConversion[data.ID].Name
		end
		item.Text.text=Name;

		item.Value.text=data.Value

		self:GetSpriteByAtlas("UI/Icons/Atlas/IconAttribute","Attribute_"..data.ID,function (sp)
				item.Icon.sprite=sp
			end)

	end
end

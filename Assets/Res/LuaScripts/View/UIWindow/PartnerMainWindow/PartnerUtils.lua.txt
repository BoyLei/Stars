local PartnerUtils = {}


--(升级属性+资质*转化率(分段计算))*(1+星级加成)+装备属性

function CalculatePartnerProp(data)
	local config=ConfigDataAccessor.Partner[data.Index];

	local Props={}
	local temp={}
	local  PartnerNature=ConfigDataAccessor.PartnerNature[config.Nature][data.Level]
	if PartnerNature~=nil then
		for i = 1, #PartnerNature.AttrID do
			local AoiAttr= ConfigDataAccessor.GetAttrByID(PartnerNature.AttrID[i])
			if AoiAttr~=nil then
				temp[PartnerNature.AttrID[i]]={Equip=0,id=PartnerNature.AttrID[i],PropName=AoiAttr.Name,Base=PartnerNature.AttrValue[i],NewCer=0,Cer=0,CerRatio=0,Star=0}
			end
		end

		for k, v in pairs(data.AMD.Aptitudes) do
			local convert= ConfigDataAccessor.ParConversion[k];
			if convert~=nil then
				for k1, v1 in pairs(convert.Natures) do


					local tbs=v1
					table.sort(tbs,function(t1,t2)
							return t1.Range<t2.Range
						end)
					local index=1;
					local cnt=#tbs;
					local left=v.BaseValue;
					local value=0;
					while index<=cnt do
						local nextIndex=index+1;
						if nextIndex>cnt then
							nextIndex=index
						end
						local ratio=tbs[index].Ratio*0.0001
						local range=tbs[nextIndex].Range

						if left<range then
							value=value+left*ratio
							break;
						else
							value=value+range*ratio
							index=index+1;
							left=left-range;
						end
					end

					if temp[k1]~=nil then
						temp[k1].NewCer=temp[k1].NewCer+math.floor(value)
					end
				end
			end
		end

		---星级
		local starExpend= ConfigDataAccessor.StarExpend[config.Quality][data.CurStar]
		if starExpend~=nil then
			for k = 1, GetTableLen(starExpend.Nature) do
				if temp[starExpend.Nature[k]]~=nil then
					temp[starExpend.Nature[k]].Star=starExpend.Value[k];
				end
			end
		end

		--装备相关
		CalculateToAddEquipValue(data,temp)
	end


	for k, v in pairs(temp) do
		table.insert(Props,v)
	end
	table.sort(Props,function (p1,p2)
			return p1.id<p2.id;
		end);

	return Props;

end

function CalculateToAddEquipValue(data,temp)
	--装备
	if data~= nil then
		for i=1,3 do
			local data=BusinessManager.Instance:GetPartnerEquipedEquip(data.ID,i);
			if data~= nil then
				local Nature=ConfigDataAccessor.PartnerEquip[data.BaseID].Nature;
				local Value=ConfigDataAccessor.PartnerEquip[data.BaseID].Value;
				for j=1,#Nature do
					local t=temp[Nature[j]];
					if t~=nil then
						t.Equip=t.Equip+Value[j];
						temp[Nature[j]]=t;
					end
				end
			end
		end
	end
end


return  PartnerUtils;

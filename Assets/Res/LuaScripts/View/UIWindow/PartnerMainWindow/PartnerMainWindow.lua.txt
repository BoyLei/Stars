function PartnerMainWindow:OnItemRefesh(tr, idx)
    if self.PartnerItems == nil then
        return
    end
    for i = 1, #self.PartnerItems do
        if self.PartnerItems[i].transform == tr then
            self.PartnerItems[i]:UpdateContent(self.Partners[idx + 1], self.SelectIndex) -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end
    local bi =
        PartnerHead.new(
        tr.gameObject,
        self.m_PartnerRoot,
        self.AtlasPath,
        function(index)
            self:OnSelect(index)
        end
    )
    bi.prefab:SetActive(true)
    bi:UpdateContent(self.Partners[idx + 1], self.SelectIndex) -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.PartnerItems, bi)
end

function PartnerMainWindow:Sort()
	-- 状态权重映射表
	local stateWeights = {
		[CS.ProtoMsg.PartnerState.InBattle] = 2,
		[CS.ProtoMsg.PartnerState.Concretization] = 2,
		[CS.ProtoMsg.PartnerState.Assist] = 1
	}

	-- 辅助函数，获取伙伴的质量，如果伙伴信息不存在则返回0
	local function getQuality(partnerIndex)
		local config = ConfigDataAccessor.Partner[partnerIndex]
		return config and config.Quality or 0
	end

	-- 排序函数
	table.sort(self.Partners, function(p1, p2)
			-- 获取状态权重，如果状态未知则默认为0
			local t1 = stateWeights[p1.State] or 0
			local t2 = stateWeights[p2.State] or 0

			-- 如果状态权重不同，根据权重排序
			if t1 > t2 then return true end
			if t1 < t2 then return false end

			-- 状态权重相同，比较等级
			local lv1 = p1.Level or 0
			local lv2 = p2.Level or 0

			if lv1 > lv2 then return true end
			if lv1 < lv2 then return false end

			-- 等级相同，比较质量
			local quality1 = getQuality(p1.Index)
			local quality2 = getQuality(p2.Index)

			return quality1 >quality2
		end)
end
function PartnerMainWindow:CleanAllSelected()
    -- 无限列表
    for i = 1, #self.PartnerItems do
        self.PartnerItems[i]:OnSelect(false)
    end
end

function PartnerMainWindow:OnSelect(selectIndex)
    self.SelectIndex = selectIndex
    self:CleanAllSelected()
    self:SetPartnerInfo()
end

function PartnerMainWindow:ItemUseJump(jumpType, selectIdx)
    if jumpType == 1 then
        --打开伙伴升级（详情-基础）
        self:OpenDetail(true)
    elseif jumpType == 2 then
        --打开伙伴洗礼(详情-资质)
        self:OnClickConstruct()
    elseif jumpType == 3 then
        --打开伙伴觉醒（进阶）
        if selectIdx > 0 then
            self.SelectIndex = selectIdx
            self:OnSelect(self.SelectIndex)
        end
        self:OnClickAdvanced()
    elseif jumpType == 4 then
        --打开魂石伙伴装备
        self:OnClickSoulStone()
    end
end
function PartnerMainWindow:GetSelectPartner()
    if self.Partners == nil then
        return nil
    end

    for _k, _v in pairs(self.Partners) do
        if _v.Index == self.SelectIndex then
            return _v
        end
    end
end

function PartnerMainWindow:SetPartnerInfo()
    local data = self:GetSelectPartner()
    if data == nil then
        return
    end

    local config = ConfigDataAccessor.Partner[data.Index]

    if config ~= nil then
        local avatarDataCell = ConfigDataAccessor.Avatar[config.AvatarID]
        if avatarDataCell ~= nil then
            self:SetModel(avatarDataCell, String2Vector3(config.ModelScale), String2Vector3(config.ModelRot), String2Vector3(config.ModelPos), config.ModelAnima)
        end
    end

    if self.TabCls[self.TabIndex] ~= nil then
        self.TabCls[self.TabIndex]:OnEnable()
        self.TabCls[self.TabIndex]:OnOpen(data)
    end
	for k, v in pairs(self.TabCls) do
		v:SetRedPoint(data)
	end
end

function PartnerMainWindow:PushDisplayGob()
    if self.m_ModelViewDisplay ~= nil then
        ResourceManager.Instance:PushGameObject("Roles/Template/Display_Model", self.m_ModelViewDisplay.gameObject)
        -- GameObject.Destroy(self.m_ModelViewDisplay.gameObject);
        self.m_ModelViewDisplay = nil
        self.m_DisplayRawimg:SetModelView(self.m_ModelViewDisplay)
    end
end

function PartnerMainWindow:SetModel(avatarDataCell, scale, localRotate, localPos, animName)
    if self.m_avatarDataCell ~= avatarDataCell then
        ------- 设置模型
        self:PushDisplayGob()
        self.m_avatarDataCell = avatarDataCell
        ResourceManager.Instance:PopGameObject(
            "Roles/Template/Display_Model",
            function(go)
                if go == nil then
                    return
                end
                if self.m_avatarDataCell == nil then
                    ResourceManager.Instance:PushGameObject("Roles/Template/Display_Model", go)
                    return
                end
                UIManager.Instance:SetModel2UI2(go, self.m_avatarDataCell.AvatarID, CS.StarProjectDef.ThreeDModelPos.Raw, scale, Vector3.zero, localRotate, localPos, animName)
                go:SetActive(true)
                self.m_ModelViewDisplay = go:GetComponent(typeof(RoleViewDisplay))
                self.m_DisplayRawimg:SetModelView(self.m_ModelViewDisplay)

                if self.m_ModelViewDisplay ~= nil then
                    local rotation = Quaternion.Euler(0, 0, 0)
                    self.m_ModelViewDisplay:SetRotation(rotation)
                    self.m_DisplayRawimg:SetState(true)
                end

                ------- 设置模型
                self.m_RoleImage.texture = ResourceManager.Instance.RenderTexture
            end
        )
    end
    if self.m_ModelViewDisplay ~= nil then
        local rotation = Quaternion.Euler(0, 0, 0)
        self.m_ModelViewDisplay:SetRotation(rotation)
        self.m_DisplayRawimg:SetState(true)
    end
end

function PartnerMainWindow:OnDisable()

        for k, v in pairs(self.TabCls) do

                v:OnDisable()
            end

end

function PartnerMainWindow:OnOpen(arg)
	local BankItemIDListRetEvent=LuaModuleDispatcher:Event(ModuleDef.BattleTeamModule,"PartnerUpgradeNewRetEvent");
	BankItemIDListRetEvent:AddListener(PartnerUpgradeNewRetEventFunc)

    local partnerMgr = BusinessManager.Instance:GetPartnerMDMgr()
    local partners = partnerMgr:GetPartners()
    self.Partners = {}
    for k, v in pairs(partners) do
        table.insert(self.Partners, v)
    end

    self:Sort()
    self.SelectIndex = self.Partners[1].Index
    self.m_ScrollView:ContinueScroll(#self.Partners)
    self:OnTabIndexChange(1)
    self:SetPartnerInfo()
end

function PartnerMainWindow:OnClose(arg)
    UIManager.Instance:SetRawCamrea(false)
	local BankItemIDListRetEvent=LuaModuleDispatcher:Event(ModuleDef.BattleTeamModule,"PartnerUpgradeNewRetEvent");
	BankItemIDListRetEvent:RemoveListener(PartnerUpgradeNewRetEventFunc)
    self:PushDisplayGob()
end

function PartnerMainWindow:OnClickCloseBtn()
    self.Close()
end

function PartnerMainWindow:OnTabIndexChange(index)
    if self.TabIndex ~= nil and self.TabIndex > 0 and self.TabIndex < 5 then
        self.UnSelects[self.TabIndex]:SetActive(true)
        self.Selects[self.TabIndex]:SetActive(false)
    end

    self.TabCls[self.TabIndex]:OnDisable()

    if index > 0 and index < 5 then
        self.UnSelects[index]:SetActive(false)
        self.Selects[index]:SetActive(true)

        self.TabIndex = index
    end

    for k, v in pairs(self.TabsObjs) do
        v:SetActive(k == index)
    end

    self.m_BackText.text = self.MenuNames[index]
end

function PartnerMainWindow:OnClickDetail()

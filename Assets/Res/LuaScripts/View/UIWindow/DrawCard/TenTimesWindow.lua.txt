function TenTimesWindow:Init(data)
    if data == nil then
        return;
    end

    self.cardPoolID = data.PoolID;

    -- local items = data.GachaResult;
    -- for i = 1, 10 do
    --     local finalDrops = items[i - 1].FinalDrop;
    --     local finalDrop = finalDrops[0];
    --     local originDrop = items[i - 1].OriginDrop;
    --     self.allItems[i]:Init(originDrop, finalDrop);
    -- end

    local cardPool = ConfigDataAccessor.CardPool;
    self.costNum = 0;
    --65.6
    local languagetype = LanguageManager.Instance:GetLanguageType()
    local rect = self.discountGo.transform:GetComponent(typeof(RectTransform))
    if languagetype == 1 then 
        rect.anchoredPosition = Vector2(89,0)
    elseif languagetype == 2 then 
        rect.anchoredPosition = Vector2(124.3,-4.1)
    end

    if cardPool[self.cardPoolID].SaleRaito > 0 then
        --self.costText.text = "抽卡消耗：10		<color=#64EBA4>x" .. (math.floor(cardPool[self.cardPoolID].SaleRaito / 1000)) .. "</color>";
        self.costText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_TenTimesWindow_costText"), {(math.floor(cardPool[self.cardPoolID].SaleRaito / 1000)),10});

        self.discountGo:SetActive(true);

        self.costNum = math.floor(cardPool[self.cardPoolID].SaleRaito / 1000);
    else
        --self.costText.text = "抽卡消耗：10";
        self.costText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_TenTimesWindow_costText2"), {10});

        self.discountGo:SetActive(false);

        self.costNum = 10;
    end

    self:SetTenTimesBtn();

    self.curCardItems = {};
    local items = data.GachaResult;
    for i = 1, 10 do
		local finalDrops = 0;
		if items[i - 1].FinalDrop.Count > 0 then
			finalDrops = items[i - 1].FinalDrop[0].Index;
		end
        local originDrop = items[i - 1].OriginDrop.Index;
        if finalDrops ~= originDrop then
            self.curCardItems[i] = finalDrops;
        else
            self.curCardItems[i] = 0;
        end
    end
end

local DrawCardDetail = {}

function DrawCardDetail:New(obj)

	self.skipBtn.onClick:AddListener(function()
			LuaModuleDispatcher:SendMessage(ModuleDef.DrawCardModule, "OnOpenTenTimesWindow", {self.allData});

	self.fx1 = self.transform:Find("Cut4Cam90/AssemblyRatio2/Root/zhuanhuan/01/Fx");
	self.fx2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/Root/zhuanhuan/01/Fx01");

	self.zhDi2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/Root/zhuanhuan/01/Icon/di"):GetComponent(typeof(Image));
	self.zhIcon2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/Root/zhuanhuan/01/Icon/icon"):GetComponent(typeof(Image));

function DrawCardDetail:Init(data, index)
	-- self.ItemData=BusinessManager.Instance:GetItemByItemEntityId(self.EntityID);

	self.allData = data;
	self.curIndex = index;

	local gachas = data.GachaResult;
	self.skipBtn.gameObject:SetActive(gachas.Count > 1);

	local curData = gachas[self.curIndex];

	local originDrop = curData.OriginDrop;
	local finalDrop = curData.FinalDrop[0];

	local itemData = ConfigDataAccessor.Item[originDrop.Index];
	local str = itemData.Name;
	local nameStr =
	"<size=116>" .. str:sub(1, 3) .. "</size><size=50> </size><size=84> " .. str:sub(4, string.len(str)) ..
	"</size>";
	self.name.text = nameStr;
	local name1 = self.name.transform:Find("Name1"):GetComponent(typeof(Text));
	local name2 = self.name.transform:Find("Name2"):GetComponent(typeof(Text));
	name1.text = nameStr;
	name2.text = nameStr;

	local partnerData = ConfigDataAccessor.Partner[originDrop.Index];
	local lv = partnerData.Quality;
	self:ShowQuality(lv);
	self.desc.text = partnerData.GachaSlogen;
	
	if lv == 4 then
		self.drawAnimator:Play("SSR", 0, 0);
	elseif lv == 3 then
		self.drawAnimator:Play("SR", 0, 0);
	elseif lv == 2 then
		self.drawAnimator:Play("R", 0, 0);
	else
		self.drawAnimator:Play("N", 0, 0);
	end
	
	self.ssr.gameObject:SetActive(lv == 4);
	self.sr.gameObject:SetActive(lv == 3);
	self.r.gameObject:SetActive(lv == 2);
	self.n.gameObject:SetActive(lv == 1);

	self.new.gameObject:SetActive(originDrop.Index == finalDrop.Index);
	--if self.curModel ~= nil then
		--BusinessManager.Instance:CloseModel(0, self.curModel);
		--self.curModel = nil;
	--end
	self.curModel = self:ShowModel(partnerData.AvatarID, partnerData.AppearAnima);
	--UIManager.Instance:SetRawCamrea(true);

	local showFirst = false;
	local curRewards = {};
	if originDrop.Index ~= finalDrop.Index then
		curRewards[finalDrop.Index] = finalDrop.Count;
		showFirst = true;
	end
	local cardPool = ConfigDataAccessor.CardPool[data.PoolID];
	curRewards[cardPool.AssociateToken] = cardPool.AssociateTokenNum;

	self:ShowExtraRewards(curRewards, showFirst);
end

function DrawCardDetail:ShowQuality(lv)
	self.ssr.gameObject:SetActive(lv == 4);
	self.sr.gameObject:SetActive(lv == 3);
	self.r.gameObject:SetActive(lv == 2);
	self.n.gameObject:SetActive(lv == 1);

	self.bgSSR.gameObject:SetActive(lv == 4);
	self.bgSR.gameObject:SetActive(lv == 3);
	self.bgR.gameObject:SetActive(lv == 2);
	self.bgN.gameObject:SetActive(lv == 1);
end

function DrawCardDetail:CloseWindow()
	self.IsOpen = false;
	if self.curModel ~= nil then
		BusinessManager.Instance:CloseModel(0, self.curModel);
		self.curModel = nil;
	end
	--UIManager.Instance:SetRawCamrea(false);
end

function DrawCardDetail:PushDisplayGob()
	if self.m_ModelViewDisplay ~= nil then
		ResourceManager.Instance:PushGameObject("Roles/Template/Display_Model", self.m_ModelViewDisplay.gameObject);
		--GameObject.Destroy(self.m_ModelViewDisplay.gameObject);
		self.m_ModelViewDisplay = nil;
		--self.m_DisplayRawing:SetModelView(self.m_ModelViewDisplay);
	end
end

function DrawCardDetail:GetEffectName(lv)
	if lv == 4 then
		return "FX_UI_SSR_zhuanhuan01";
	elseif lv == 3 then
		return "FX_UI_SR_zhuanhuan01";
	elseif lv == 2 then
		return "FX_UI_R_zhuanhuan01";
	else
		return "FX_UI_N_zhuanhuan01";
	end
end

function DrawCardDetail:ShowExtraRewards(rewards, showFirst)
	self.rewardGo1:SetActive(showFirst);
	if showFirst then
		local curItem = nil;
		for k,v in pairs(rewards) do
			curItem = k;
			break;
		end
		local partnerData = ConfigDataAccessor.Partner[curItem];
		lv = partnerData.Quality;
		local strName = self:GetEffectName(lv);

		self.go_perfab1=ResourceManager.Instance:LoadGameObject("effects/ui/drawcard/"..strName);
		self.go_perfab1.transform:SetParent(self.fx1);
		self.go_perfab1.transform.localScale=Vector3(1,1,1);
		self.go_perfab1.transform.localPosition=Vector3(0,0,0);
		self.go_perfab1:SetActive(true);
		
		local strName = self:GetEffectName(lv);
		self.go_perfab2=ResourceManager.Instance:LoadGameObject("effects/ui/drawcard/"..strName);
		self.go_perfab2.transform:SetParent(self.fx2);
		self.go_perfab2.transform.localScale=Vector3(1,1,1);
		self.go_perfab2.transform.localPosition=Vector3(0,0,0);
		self.go_perfab2:SetActive(true);
	end


	
	local index = 1;
	for k,v in pairs(rewards) do
		local itcfg =ConfigDataAccessor.Item[k];
		if index == 1 and showFirst then
			local quality=itcfg.Quality; 
			ResourceHelper:LoadIconByAtlasAsync(
				self.zhDi1,
				ResourceHelper.AtlasPathCommonItem,
				"Bag_Grid_sq" .. quality,
				false
			)
			ResourceHelper:LoadIconAsync(self.zhIcon1, "", itcfg.Icon, false);
			self.nameTxt1.text = itcfg.Name;
			self.numTxt1 = "x"..v;
		else
			local quality=itcfg.Quality; 
			ResourceHelper:LoadIconByAtlasAsync(
				self.zhDi2,
				ResourceHelper.AtlasPathCommonItem,
				"Bag_Grid_sq" .. quality,
				false
			)
			ResourceHelper:LoadIconAsync(self.zhIcon2, "", itcfg.Icon, false);
			self.nameTxt2.text = itcfg.Name;
			self.numTxt2 = "x"..v;
		end
		index = index + 1;
	end
end

return DrawCardDetail

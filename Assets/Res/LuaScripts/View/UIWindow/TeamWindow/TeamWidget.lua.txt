function TeamWidget:ChangeCaptain()
    if self.tinfo == nil then
        return;
    end
    local changeTeamCaptainReq = CS.ProtoMsg.ChangeTeamCaptainReq();
    changeTeamCaptainReq.TeamID = self.tinfo.TeamID;
    changeTeamCaptainReq.OldCapEID = GameManager.Instance.mainPlayerId;
    changeTeamCaptainReq.NewCapEID = self.curSelMem.Eid;
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeSpace,
        changeTeamCaptainReq, false);
end

function TeamWidget:Gather()
    Util.ShowMessage(LanguageHelp:GetLanguageByKey("Team_GatherTips"));
    local extrainfos = CS.ProtoMsg.ChatExtraInfos();
    local chatExtraInfo = CS.ProtoMsg.ChatExtraInfo();
    chatExtraInfo.EType = CS.ProtoMsg.ChatExtraInfo.Types.EMsg.TeamGather;
    extrainfos.ChatExtraInfo:Add(chatExtraInfo);

    local sendChatMsgReq = CS.ProtoMsg.SendChatMsgReq();
    local targetID = GameManager.Instance.mainPlayerId;
    local playerName = GameManager.Instance:GetEntityNameById(targetID);
    sendChatMsgReq.ChatMsg = playerName;
    sendChatMsgReq.Channel = CS.ProtoMsg.ChatChannel.ChannelTeam;
    sendChatMsgReq.ChatTime = TimeUtils.ServerNowStampMilli;
    sendChatMsgReq.SendData:Add(ExtraData.ToByteString(extrainfos));
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby,
        sendChatMsgReq, false);
end

-- 判断是是否拥有某项权力
function TeamWidget:IsHaveGuildPower(powerID)

    --[[
	1=审批入会
	2=踢人
	3=禁言
	4=任职
	5=冒险团改名、修改公告、修改宣言、修改标签
	8=修改招人等级
	10=邀请入会
	]]

    -- 没有冒险团
    if BusinessManager.Instance:GetGuildID() == 0 then
        return false;
    end

    local posLevelID = BusinessManager.Instance:GetGuildPosLevel();

    local powerList = ConfigDataAccessor.GuildTitle[posLevelID].Power;

    for k, v in pairs(powerList) do
        if v == powerID then
            return true;
        end
    end

    return false;

end

function TeamWidget:CheckCanGuildInvite()
    local canInvite = self:IsHaveGuildPower(10);

    self.normalNormalConferenceBtn.gameObject:SetActive(canInvite);
    self.normalCaptainConferenceBtn.gameObject:SetActive(canInvite);
    self.captainNormalConferenceBtn.gameObject:SetActive(canInvite);

end

function TeamWidget:ReqCaptain()
    if self.tinfo == nil then
        return;
    end
    local requestCaptainReq = CS.ProtoMsg.RequestCaptainReq();
    requestCaptainReq.RequesterID = GameManager.Instance.mainPlayerId;
    local playerName = GameManager.Instance:GetEntityNameById(GameManager.Instance.mainPlayerId);
    requestCaptainReq.RequesterName = playerName;
    requestCaptainReq.TeamID = self.tinfo.TeamID;
    for i = 0, self.tinfo.UserInfos.Count - 1 do
        if self.tinfo.UserInfos[i].IsCaptain then
            requestCaptainReq.BeRequester = self.tinfo.UserInfos[i].Eid;
            break
        end
    end
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeSpace,
        requestCaptainReq, false);
    self.captainGo:SetActive(false);
end

function TeamWidget:PathFind()
    local selfdata = self:GetSelfData();
    if selfdata == nil then
        return;
    end
	--公会领地处理
	if self.curSelMem.MapID == 100 then 
		Util.ShowMessage(LanguageHelp:GetLanguageByKey("RetMsg_RoomSpaceClientStr2107"));
		return
	end
	
    if self.curSelMem.SpaceID == selfdata.SpaceID and self.curSelMem.SrvID == selfdata.SrvID  then
        TaskHelper.FindToDynamicTarget("TeamModule", self.curSelMem.Eid);
    else
        local mapChangeReq = CS.ProtoMsg.MapChangeReq();
        mapChangeReq.MapID = self.curSelMem.MapID;
        mapChangeReq.ServerID = self.curSelMem.SrvID;
        mapChangeReq.Reason = CS.ProtoMsg.ChangeReason.ChangeLine;
        mapChangeReq.SpaceID = self.curSelMem.SpaceID;
        mapChangeReq.SpType = CS.ProtoMsg.SpaceType.SpaceScene;
        CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby,
            mapChangeReq, false);

        LuaModuleDispatcher:SendMessage(ModuleDef.TeamModule, "StartTeamFollowChangeScene", self.curSelMem.Eid);
    end
end

function TeamWidget:GetSelfData()
    if self.tinfo == nil then
        return nil;
    end
    local members = self.tinfo.UserInfos;
    for i = 0, members.Count - 1 do
        local mem = members[i];
        if mem.Eid == GameManager.Instance.mainPlayerId then
            return mem;
        end
    end
    return nil;
end

function TeamWidget:ShowIsCaptain(isCaptain)
    self.captainCaptainGo:SetActive(false);
    self.captainNormalGo:SetActive(false);
    self.normalCaptainGo:SetActive(false);
    self.normalNormalGo:SetActive(false);
    self.normalSelfGo:SetActive(false);

    self.captainGo:SetActive(isCaptain);
    self.normalGo:SetActive(not isCaptain);
    self:ActiveTeamBg(true);

    self.closeWidgetBtn.gameObject:SetActive(true);

end

function TeamWidget:SetTeamInfo(data)
    self.tinfo = data;

    for _, v in pairs(self.teamMembers) do
        v:UnBind();
        GameObject.Destroy(v.gameObject);
    end
    self.teamMembers = {}

    if data == nil or data.UserInfos == nil then
        return;
    end

    local members = data.UserInfos;

    local mmLists = {};

    for i = 0, members.Count - 1 do
        local mem = members[i];
        if mem.IsCaptain then
            table.insert(mmLists, 1, mem);
            print("[Team] i: " .. i .. " 是组长, 插入 1号位");
        else
            table.insert(mmLists, mem);
            print("[Team] i: " .. i .. " 组员, 插入 " .. #mmLists);

        end

    end

    for i = 1, #mmLists do
        local mem = mmLists[i];

        local gob = newObject(self.teamItem);
        gob:SetActive(true);
        gob.transform:SetParent(self.teamItem.transform.parent);
        gob.transform.localPosition = Vector3.zero;
        gob.transform.localScale = Vector3.one;

        local tItem = TeamItem:New(gob, mem, function(cap)
            self:OnClickMem(cap);
        end);
        table.insert(self.teamMembers, tItem);
    end
end

function TeamWidget:AddFriend(PID)
    if PID == nil then
        return;
    end
    local FriendModule = ModuleManager.Instance:GetModule(ModuleDef.FriendModule);
    if FriendModule then
        local luatable = FriendModule:GetLuaTable()
        local friendRelation = luatable:GetFriendRelation(PID);
        if friendRelation == 0 then
            local pids = {PID};
            local key = "ItemRecommendFriend_" .. PID;
            LuaModuleDispatcher:SendMessage(ModuleDef.FriendModule, "OnSendApplyToAddFriendReq",
                {pids, key, self.OnApplyToAddFriendRespFun});
            elseif friendRelation == 1 then
                Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_PlayerIsFriend"));
            elseif friendRelation == 2 then
                Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_friend_request_fail_5"));
            end
    end
end

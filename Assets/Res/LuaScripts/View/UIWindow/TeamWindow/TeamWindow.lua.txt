function TeamWindow:Show(data)
    self.tinfo = data;
    self.pid = BusinessManager.Instance:GetUserPID();

    if data == nil then
        return;
    end

    local isCaptain = self.module:EntityIsCaptain(GameManager.Instance.mainPlayerId);
    self.AtuoAIToggle.gameObject:SetActive(isCaptain);
    if isCaptain then
        self.AtuoAIToggle.isOn = data.NeedRobot;
    end

    self:ShowMemberModels(self.tinfo.UserInfos);
    self.teamNameInput.text = self.tinfo.Desc;
    self:SetBaseUIData();
    -- UIManager.Instance:SetRawCamrea(true);

    -- self.limitText.text = self:GetAdvStr(self.tinfo.FightLimit) .. "  <color=#FCFCE6>等级：</color><color=#FFF6B8>" .. self.tinfo.LvLimit .. "级</color>";
    self.limitText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_TeamWindow_limitText"),
        {self:GetAdvStr(self.tinfo.FightLimit), self.tinfo.LvLimit});

    local teamPlayList = ConfigDataAccessor.TeamPlayList;
    self.targetText.text = LanguageHelp:GetLanguageByKey("Local_Str_Tips9");
    for k, v in pairs(teamPlayList) do
        if k == self.tinfo.PlayID then
            -- self.targetText.text = "目标:" .. v.PlayName;
            self.targetText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_TeamLobbyItem_targetText"),
                {v.PlayName});
            break
        end
    end

end

function TeamWindow:SetBaseUIData()
    local isCaptain = self.module:EntityIsCaptain(GameManager.Instance.mainPlayerId);

    self.applyListBtn.gameObject:SetActive(isCaptain);

    self.teamNameInput.interactable = isCaptain;

function TeamWindow:RefreshApplyNum()
    local applyCount = #self.module.allJoinTeamReqs;

    local tb = {};
    applyCount = 0;
    for _, v in pairs(self.module.allJoinTeamReqs) do
        if tb[v.UserInfo.Eid] == nil then
            tb[v.UserInfo.Eid] = 1;
            applyCount = applyCount + 1;
        end
    end

    self.applyNum.text = tostring(applyCount);
    self.applyNumGo:SetActive(applyCount > 0);
    self.applyNumGo:SetActive(#self.module.allJoinTeamReqs > 0);

    RedPointManager.Instance:RefreshRedPointCount(RedPointType.TeamGroup, applyCount);
end

function TeamWindow:ReqCaptain()
    if self.tinfo == nil then
        return;
    end
    local requestCaptainReq = CS.ProtoMsg.RequestCaptainReq();
    requestCaptainReq.RequesterID = GameManager.Instance.mainPlayerId;
    local playerName = GameManager.Instance:GetEntityNameById(GameManager.Instance.mainPlayerId);
    requestCaptainReq.RequesterName = playerName;
    requestCaptainReq.TeamID = self.tinfo.TeamID;
    for i = 0, self.tinfo.UserInfos.Count - 1 do
        if self.tinfo.UserInfos[i].IsCaptain then
            requestCaptainReq.BeRequester = self.tinfo.UserInfos[i].Eid;
            break
        end
    end
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeSpace,
        requestCaptainReq, false);
 --   self.captainGo:SetActive(false);
end

    local selfdata
    if self.tinfo.UserInfos then
        for i = 0, self.tinfo.UserInfos.Count - 1 do
            local userInfo = self.tinfo.UserInfos[i];
            if userInfo then
                if userInfo.PID == BusinessManager.Instance:GetUserPID() then
                    selfdata = userInfo
                    break
                end
            end

        end
    end
    if selfdata == nil or self.curSelMem == nil then
        return;
    end
    if self.curSelMem.SpaceID == selfdata.SpaceID and self.curSelMem.SrvID == selfdata.SrvID then
        TaskHelper.FindToDynamicTarget("TeamModule", self.curSelMem.Eid);
    else
        local mapChangeReq = CS.ProtoMsg.MapChangeReq();
        mapChangeReq.MapID = self.curSelMem.MapID;
        mapChangeReq.ServerID = self.curSelMem.SrvID;
        mapChangeReq.Reason = CS.ProtoMsg.ChangeReason.ChangeLine;
        mapChangeReq.SpaceID = self.curSelMem.SpaceID;
        mapChangeReq.SpType = CS.ProtoMsg.SpaceType.SpaceScene;
        CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby,
            mapChangeReq, false);

        LuaModuleDispatcher:SendMessage(ModuleDef.TeamModule, "StartTeamFollowChangeScene", self.curSelMem.Eid);
    end
end
function TeamWindow:Gather()
    Util.ShowMessage(LanguageHelp:GetLanguageByKey("Team_GatherTips"));
    local extrainfos = CS.ProtoMsg.ChatExtraInfos();
    local chatExtraInfo = CS.ProtoMsg.ChatExtraInfo();
    chatExtraInfo.EType = CS.ProtoMsg.ChatExtraInfo.Types.EMsg.TeamGather;
    extrainfos.ChatExtraInfo:Add(chatExtraInfo);

    local sendChatMsgReq = CS.ProtoMsg.SendChatMsgReq();
    local targetID = GameManager.Instance.mainPlayerId;
    local playerName = GameManager.Instance:GetEntityNameById(targetID);
    sendChatMsgReq.ChatMsg = playerName;
    sendChatMsgReq.Channel = CS.ProtoMsg.ChatChannel.ChannelTeam;
    sendChatMsgReq.ChatTime = TimeUtils.ServerNowStampMilli;
    sendChatMsgReq.SendData:Add(ExtraData.ToByteString(extrainfos));
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby,
        sendChatMsgReq, false);
end

function TeamWindow:AddFriend(PID)
    if PID == nil then
        return;
    end
    local FriendModule = ModuleManager.Instance:GetModule(ModuleDef.FriendModule);
    if FriendModule then
        local luatable = FriendModule:GetLuaTable()
        local friendRelation = luatable:GetFriendRelation(PID);
        if friendRelation == 0 then
            local pids = {PID};
            local key = "ItemRecommendFriend_" .. PID;
            LuaModuleDispatcher:SendMessage(ModuleDef.FriendModule, "OnSendApplyToAddFriendReq",
                {pids, key, self.OnApplyToAddFriendRespFun});
        elseif friendRelation == 1 then
            Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_PlayerIsFriend"));
        elseif friendRelation == 2 then
            Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_friend_request_fail_5"));
        end
    end
end

function MedicineItem:New(obj)

    setmetatable(self, MedicineItem)

    self:InitControls(obj)

    --end

function MedicineItem:Refresh(data, t, cb, rem, drugs)
    self.cb = cb
    self.data = data
    self.medType = t

    self.drugs = drugs
    --if t == 3 then
    --self.drugs = {};
    --local ids = drugs.Eitemid;
    --for i = 0, ids.Count - 1 do
    --table.insert(self.drugs, ids[i]);
    --end
    --end

    local itemData = ConfigDataAccessor.Item[data.BaseID]

    --self:LoadIcon(self.medicineIcon, itemData.Icon);
    ResourceHelper:LoadIconAsync(self.medicineIcon, "", itemData.Icon, false)

    self.numText.text = data.Num

    local cfg=ConfigDataAccessor.Item[data.BaseID];
    self.nameTxt.text = GetQualityColor(cfg.Quality)..itemData.Name.."</color>";

    self.descTxt.text = itemData.Desc
    --self.rareImg.sprite= AtlasManager.Instance:GetSprite("ui/common/atlas/commonitem","Bag_Grid_sq"..itemData.Quality);
    ResourceHelper:LoadIconByAtlasAsync(
        self.rareImg,
        ResourceHelper.AtlasPathCommonItem,
        "Bag_Grid_sq" .. itemData.Quality,
        false
    )
    self.addedObj.transform.parent.gameObject:SetActive(t == 3)
    self.addedObj:SetActive(t == 3 and rem)
    --self.removeBtn.gameObject:SetActive(t == 3 and rem);
end

function MedicineItem:Show(b)
    self.gameObject:SetActive(b)
end

return MedicineItem

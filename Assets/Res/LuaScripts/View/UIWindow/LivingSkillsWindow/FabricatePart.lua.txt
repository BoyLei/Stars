function FabricatePart:UnLockBtn()
	if self.SelectInfo ~= nil and self.SelectInfo.Config~=nil and self.SelectInfo.Config.UnlockItem>0 then
		LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips",
			{self.SelectInfo.Config.UnlockItem, false, nil, self.m_CostBgRare.transform, 0});
	end
end

function FabricatePart:OnTaskGoto()
	if self.BreakTaskID > 0 then
		local taskinfo = TaskHelper.GetTaskConfig(self.BreakTaskID)
		if taskinfo ~= nil then
			if taskinfo.TaskFinishInfo.Targets.Count > 0 then
				local target = taskinfo.TaskFinishInfo.Targets[0].Target;
				if target ~= nil and target.FindPath ~= nil then
					TaskHelper.FindPathByTask(target.MapID, target.FindPath,target.TargetID)
					LuaModuleDispatcher:SendMessage(ModuleDef.LivingSkillsModule, "CloseSkillsWindow", {});
				end
			end
		end
	end
end

function FabricatePart:OnLifeSkillUnlockFormulationRet(args)
	local RetMsgID = args.MsgRet.RetMsgID; -- 对应的消息号(这个可以判断是哪个协议)
	local RetCode = args.MsgRet.RetCode;
	self.WaitingServerRet = false;
	if RetCode == 0 then
		self:InitItemList();
		self:SetNeedCount();
	end
end

function FabricatePart:OnGridItemRefesh(tr, idx)
	if self.GridItemList == nil then
		return;
	end

	for i = 1, #self.GridItemList do
		if self.GridItemList[i].transform == tr then
			self.GridItemList[i]:UpdateContent(self.DataLists[idx + 1], self.CurrntSelect, function(Index)
					self:OnSelectCallBack(Index)
				end); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
			return
		end
	end

	local bi = GatherPartItem.new(tr.gameObject, self.AtlasPath);
	bi.prefab:SetActive(true);
	bi:UpdateContent(self.DataLists[idx + 1], self.CurrntSelect, function(Index)
			self:OnSelectCallBack(Index)
		end); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
	table.insert(self.GridItemList, bi);
end

function FabricatePart:OnOutPutGridItemRefesh(tr, idx)
	if self.GridItemOutPutList == nil then
		return;
	end
	for i = 1, #self.GridItemOutPutList do
		if self.GridItemOutPutList[i].transform == tr then
			self.GridItemOutPutList[i]:UpdateContent(self.OutputDataLists[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1

	bi:UpdateContent(self.OutputDataLists[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
	table.insert(self.GridItemOutPutList, bi);
end

function FabricatePart:OnSelectCallBack(Index)
	for i = 1, #self.GridItemList do
		if self.GridItemList[i] ~= nil then
			if self.GridItemList[i].ID == Index.Config.ID then
				self.GridItemList[i]:Select(true)
			else
				self.GridItemList[i]:Select(false)
			end
		end
	end
	self.CurrntSelect = Index;
	self.SelectInfo = Index
	self.CreatID = Index.Config.ID;
	local less = false;
	local cfgs = Index.Config;
	if cfgs ~= nil then
		self.CollectSpeed = 100;
		-- 生效等级
		if self.AdditionActiveLV > 0 and cfgs.Level <= self.AdditionActiveLV then
			self.CollectSpeed = cfgs.CollectSpeed;
		end
		self.OutputDataLists = {}
		local config = ConfigDataAccessor.Item[cfgs.BasicOutput]
		if config ~= nil then
			self.m_Name.text = GetQualityColor(config.Quality) .. config.Name .. "</color>"
			local quality = config.Quality;
			ResourceHelper:LoadIconByAtlasAsync(self.m_BgRare, ResourceHelper.AtlasPathCommonItem,
				"Bag_Grid_sq" .. quality, false)
			ResourceHelper:LoadIconAsync(self.m_Icon, "", config.Icon, false)
			self.m_Desc.text = config.Desc
		else
			LogError("找不到配置" .. cfgs.BasicOutput)
		end

		if cfgs.UnlockItem > 0 then
			local costCfg = ConfigDataAccessor.Item[cfgs.UnlockItem]
			if costCfg ~= nil then

				local quality = costCfg.Quality;
				ResourceHelper:LoadIconByAtlasAsync(self.m_CostBgRare, ResourceHelper.AtlasPathCommonItem,
					"Bag_Grid_sq" .. quality, false)
				ResourceHelper:LoadIconAsync(self.m_CostIcon, "", costCfg.Icon, false)

				local ownCnt = BusinessManager.Instance:GetCurrencyNum(cfgs.UnlockItem)
				if ownCnt < 1 then
					self.m_CostNum.text = "<color=#ff0000>" .. ownCnt .. "</color>/1"

				else
					self.m_CostNum.text = ownCnt .. "/1";
				end
			else
				LogError("找不到配置" .. cfgs.UnlockItem)
			end
		end
		
		self.m_Info.text=cfgs.CreatSource
		
		-- self.m_Level.text=cfgs.Level.."级"
		self.m_Level.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LvLimit3"), {cfgs.Level});

		-- self.m_ChanChuText.text="单次产出:"..cfgs.BasicOutputNum
		self.m_ChanChuText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_FabricatePart_ChanChuText"),
			{cfgs.BasicOutputNum});

		local ownCnt = BusinessManager.Instance:GetCurrencyNum(cfgs.BasicOutput)
		if ownCnt > 999 then
			-- self.m_OwnText.text="拥有:999+"
			self.m_OwnText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SellCount_OwnCount"), {":999+"});

		else
			-- self.m_OwnText.text="拥有:"..ownCnt
			self.m_OwnText.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SellCount_OwnCount"),
				{":" .. ownCnt});
		end

		---产出
		table.insert(self.OutputDataLists, {
				ItemID = cfgs.RareOutput,
				ItemNum = cfgs.RareOutputNum
			})

		self.m_OutputScrollView:ContinueScroll(#self.OutputDataLists);

		-- 输入
		self.CreateCnt = 0;
		self.MaxCnt = cfgs.OnceMaxCreateNum;
		self.InputDataLists = {}

		for i = 1, #cfgs.CostItem do
			local ownCnt = BusinessManager.Instance:GetCurrencyNum(cfgs.CostItem[i])
			local t = ownCnt / cfgs.CostItemNum[i]
			if t < self.MaxCnt then
				self.MaxCnt = math.floor(t);
			end
			if ownCnt < cfgs.CostItemNum[i] then
				less = true;
			end
			table.insert(self.InputDataLists, {
					ItemID = cfgs.CostItem[i],
					ItemNum = cfgs.CostItemNum[i]
				})
		end

		-- 采集活力值
		local energy = GameManager.Instance:GetPlayerCollectEnergy();
		local t = energy / cfgs.EnergyCost
		if t < self.MaxCnt then
			self.MaxCnt = math.floor(t);
		end

		if self.MaxCnt < 1 then
			self.MinCnt = 0;
			self.CreateCnt = 0;
			self.m_right.UnClick = true;
			self.m_jian.UnClick = true;
			self.m_left.UnClick = true;
			self.m_jia.UnClick = true;
			self.m_Num.text = ""
		else
			self.MinCnt = 1;
			self.m_right.UnClick = false;
			self.m_jian.UnClick = false;
			self.m_left.UnClick = false;
			self.m_jia.UnClick = false;
			self.m_Num.text = "1"
			self.CreateCnt = 1;
		end
		self.m_InputScrollView:ContinueScroll(#self.InputDataLists);
	end

	local click = false;
	if Index.UnLock == 1 then
		if cfgs ~= nil and cfgs.UnlockItem>0 then
			local ownCnt = BusinessManager.Instance:GetCurrencyNum(cfgs.UnlockItem)
			if ownCnt < 1 then
				click = true;
			end
		end

		self.m_FabricateText.text = LanguageHelp:GetLanguageByKey("Local_Str_Unlock")
		self.m_CostItem:SetActive(true)
		self.m_Count:SetActive(false)
		self.m_CostEngery:SetActive(false)
		self.m_FabricateBtn.UnClick = click;
		self.m_FabricateBtn.Image.raycastTarget = not click;
	elseif Index.UnLock == 2 then
		self.m_FabricateText.text = LanguageHelp:GetLanguageByKey("Local_Str_Make")
		self.m_CostItem:SetActive(false)
		self.m_Count:SetActive(true)
		self.m_CostEngery:SetActive(true)
		self:SetConditionText(cfgs.EnergyCost, cfgs.Level, less)
	end

end

function FabricatePart:SetNeedCount()
	if self.CurrntSelect ~= nil then
		local cfgs = self.CurrntSelect.Config;

		self.InputDataLists = {}

		for i = 1, #cfgs.CostItem do
			local ownCnt = BusinessManager.Instance:GetCurrencyNum(cfgs.CostItem[i])
			local t = ownCnt / cfgs.CostItemNum[i]
			table.insert(self.InputDataLists, {
					ItemID = cfgs.CostItem[i],
					ItemNum = math.floor(tonumber(cfgs.CostItemNum[i]) * self.CreateCnt)
				})
		end
		self.m_InputScrollView:ContinueScroll(#self.InputDataLists);
	end
end

function FabricatePart:SetConditionText(EnergyCost, Level, less)
	-- local text=EnergyCost.."/次"
	local text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LimitTime"), {EnergyCost});

	local bset = false;
	local sceneType = GameManager.Instance:GetCurMapType()
	if sceneType == CS.ProtoMsg.SpaceType.SpaceScene then
		if less then
			text = LanguageHelp:GetLanguageByKey("Local_Str_item_insufficient_1");
			bset = true
		else
			local job = BaseJob + self.CurrentJob;
			local lifeSkillMgr = BusinessManager.Instance.M_LifeSkillUserDataMDMgr
			if lifeSkillMgr ~= nil then
				local jobInfo = lifeSkillMgr:GetData(job)
				if jobInfo ~= nil then
					if Level <= jobInfo.SkillLevel then
						local energy = GameManager.Instance:GetPlayerCollectEnergy();
						if energy < EnergyCost then
							text = LanguageHelp:GetLanguageByKey("Local_Str_item_insufficient_2");
							bset = true
						end
					else
						text = LanguageHelp:GetLanguageByKey("Local_Str_item_insufficient_3");
						bset = true
					end
				end
			end
		end
	else
		text = LanguageHelp:GetLanguageByKey("Local_Str_Tips38")
		bset = true
	end

	self.m_CostEngeryIcon:SetActive(not bset)
	self.m_FabricateBtn.UnClick = bset
	self.m_FabricateBtn.Image.raycastTarget = not bset;
	self.m_Cost.text = text
end

function FabricatePart:OnTabMenuItemClick(index)
	if self.CurrentJob == index then
		return
	end
	self.m_Tabs:Check(index);
	self.CurrentJob = index;
	self:InitItemList();
end

function FabricatePart:OnOpen(arg)
	self.CurrentJob = 1;
	local selectID = 0;
	if arg ~= nil then
		self.CurrentJob = arg.JobID % 200
		selectID = arg.MineID
	end

	self.m_Tabs:Init(self.CurrentJob, function(index)
			self:OnTabMenuItemClick(index);
		end)
	self:InitItemList(selectID);
end

function FabricatePart:GetConfig(job, level)
	local configs = ConfigDataAccessor.LifeSkillJobList;
	for k, v in pairs(configs) do
		if v.SkillID == job and v.Level == level then
			return v
		end
	end
	return nil
end

function FabricatePart:IsUnLock(jobInfo, id)
	if jobInfo ~= nil then
		if jobInfo.Formulation ~= nil and jobInfo.Formulation.CreateIdMap ~= nil then
			if jobInfo.Formulation.CreateIdMap:ContainsKey(id) then
				return jobInfo.Formulation.CreateIdMap[id];
			end
		end
	end
	return nil;
end

function FabricatePart:InitItemList(arg)
	local job = BaseJob + self.CurrentJob;
	self.DataLists = {}
	self.CurrntSelect = nil;
	local jobInfo = BusinessManager.Instance:GetLifeJobSkill(job)
	local configs = ConfigDataAccessor.LifeSkillJobList;

	local MaxLevel = 0;
	self.BreakTaskID = 0
	local CurrentCfg = nil;
	for k, v in pairs(configs) do
		if v.SkillID == job and v.Level == jobInfo.SkillLevel then
			CurrentCfg = v;
		end
		if v.SkillID == job and v.Level >= jobInfo.SkillLevel then

			if v.BreakQuest ~= 0 then
				self.BreakTaskID = v.BreakQuest
				MaxLevel = v.Level + 1;
				break
			end
		end
	end

	local cfgs = ConfigDataAccessor.LifeSkillCreateList;
	for k, v in pairs(cfgs) do

		if v.SkillID == job and v.Level < MaxLevel then

			local ulock = self:IsUnLock(jobInfo, v.ID);
			local display = 1;
			local cf = self:GetConfig(job, v.Level);
			if cf ~= nil then
				display = cf.DisplayGroup;
			end
			local LockType = 1;
			if ulock then
				LockType = 2;
			end
			table.insert(self.DataLists, {
					Config = v,
					UnLock = LockType,
					DisplayGroup = display,
					Level = jobInfo.SkillLevel
				})
		end
	end

	-- 生效等级
	self.AdditionActiveLV = 0;
	if CurrentCfg ~= nil then
		self.AdditionActiveLV = CurrentCfg.AdditionActiveLV
		if jobInfo.SkillExp < CurrentCfg.NeedEX then
			self.m_ExpObj:SetActive(true)
			self.m_Break:SetActive(false);
			self.m_Exp.value = jobInfo.SkillExp / CurrentCfg.NeedEX
			self.m_ExpValue.text = jobInfo.SkillExp .. "/" .. CurrentCfg.NeedEX
			-- self.m_JobLv.text=jobInfo.SkillLevel.."级"
			self.m_JobLv.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LvLimit3"), {jobInfo.SkillLevel});

		else
			if self.BreakTaskID > 0 then
				local taskinfo = TaskHelper.GetTaskConfig(self.BreakTaskID)
				if taskinfo ~= nil then
					self.m_TaskName.text = "[" .. taskinfo.Base.TaskName .. "]"
				end
			end
			self.m_Break:SetActive(true);
			self.m_ExpObj:SetActive(false)
			-- self.m_JobLv.text=jobInfo.SkillLevel.."级".."(经验已满)"
			self.m_JobLv.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_FabricatePart_JobLv"),
				{jobInfo.SkillLevel});

		end
	end

	self:SortItems();
	if arg == 0 or arg == nil then
		self.CurrntSelect = self.DataLists[1];
	else
		for i = 1, #self.DataLists do
			if self.DataLists[i].Config.ID == arg then
				self.CurrntSelect = self.DataLists[i];
			end
		end
	end

	for k, v in pairs(self.GridItemList) do
		v:ClearData();
	end
	-- GridItemList
	self.m_ScrollView:ContinueScroll(#self.DataLists);
	self:OnSelectCallBack(self.CurrntSelect)
	self.m_JobName.text = JobConfig[job].JobName
	self:GetSprite(JobConfig[job].SpriteName, function(sp)
			self.m_JobIcon.sprite = sp
		end)

end

function FabricatePart:SortItems()
	if self.DataLists == nil or #self.DataLists == 0 then
		return
	end
	-- 解锁>>区间>> 道具品质>> 等级
	table.sort(self.DataLists, function(a, b)

			if a.UnLock > b.UnLock then
				return true
			elseif a.UnLock == b.UnLock then
				if a.DisplayGroup > b.DisplayGroup then
					return true
				elseif a.DisplayGroup == b.DisplayGroup then

					if a.Level > b.Level then
						return true
					elseif a.Level == b.Level then
						local a_Quality = 0;
						local b_Quality = 0;
						local aCfg = ConfigDataAccessor.Item[a.Config.BasicOutput];
						if aCfg ~= nil then
							a_Quality = aCfg.Quality;
						else
							LogError("读取配置失败 " .. a.Config.BasicOutput)
						end

						local bCfg = ConfigDataAccessor.Item[b.Config.BasicOutput];
						if bCfg ~= nil then
							b_Quality = bCfg.Quality;
						else
							LogError("读取配置失败 " .. b.Config.BasicOutput)
						end

						return a_Quality > b_Quality
					end

				end
			end
		end);

end

				Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Tips39"))
			end

			Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Tips40"))
		end
	end

	-- OpenLifeSkillCreatWidget
end

function FabricatePart:OnClickleft()
	if self.CreateCnt ~= self.MinCnt then
		self.CreateCnt = self.MinCnt
		self:OnCounttChange();
	end
end

function FabricatePart:OnClickright()
	if self.CreateCnt ~= self.MaxCnt then
		self.CreateCnt = self.MaxCnt
		self:OnCounttChange();
	end
end

function FabricatePart:OnClickjian()
	if self.CreateCnt <= self.MinCnt then
		return
	end
	self.CreateCnt = self.CreateCnt - 1;
	self:OnCounttChange();
end

function FabricatePart:OnClickjia()
	if self.CreateCnt >= self.MaxCnt then
		return
	end
	self.CreateCnt = self.CreateCnt + 1;
	self:OnCounttChange();
end

function FabricatePart:OnCounttChange()
	self.m_Num.text = tostring(self.CreateCnt)
	self:SetNeedCount();
end

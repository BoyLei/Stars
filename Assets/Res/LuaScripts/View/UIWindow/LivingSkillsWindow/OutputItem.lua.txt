OutputItem=class("OutputItem",UIPanel);
OutputItem.m_Selected=nil;
OutputItem.m_BgRare=nil;
OutputItem.m_Icon=nil;
OutputItem.m_Name=nil;
OutputItem.m_Desc=nil;
OutputItem.ItemID=0;

function OutputItem:ctor(go, atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_BgRare=self:Find("BgRare"):GetComponent(typeof(Image));
	self.m_Icon=self:Find("Icon"):GetComponent(typeof(Image));
	self.m_Name=self:Find("Name"):GetComponent(typeof(Text));
	self.m_Desc=self:Find("Desc"):GetComponent(typeof(Text));
	self.Btn = self.prefab:GetComponent(typeof(JButton));
	self.Btn.OnClick = (function() self:OnClickHandler() end);
end

--点击事件
function OutputItem:OnClickHandler()
	LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips",
		{ self.ItemID, false, nil, self.m_Icon.transform, 0 });

end

function OutputItem:UpdateContent(arg,Desc)
	self.ItemID=arg.ItemID;
	local config=ConfigDataAccessor.Item[self.ItemID]
	if config~=nil then
		self.m_Name.text=GetQualityColor(config.Quality).. config.Name.."</color>"

		local quality=config.Quality;
		ResourceHelper:LoadIconByAtlasAsync(
			self.m_BgRare,
			ResourceHelper.AtlasPathCommonItem,
			"Bag_Grid_sq"..quality,
			false
		)
		ResourceHelper:LoadIconAsync(self.m_Icon, "", config.Icon, false)
	end
	self.m_Desc.text=Desc;

end

return  OutputItem

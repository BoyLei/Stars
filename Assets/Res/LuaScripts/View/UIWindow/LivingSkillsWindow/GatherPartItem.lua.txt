GatherPartItem=class("GatherPartItem",UIPanel);
GatherPartItem.m_Selected=nil;
GatherPartItem.m_BgRare=nil;
GatherPartItem.m_Icon=nil;
GatherPartItem.m_Name=nil;
GatherPartItem.m_Level=nil;
GatherPartItem.m_Btn=nil;
GatherPartItem.OnSelectCallBack=nil;
GatherPartItem.ItemID=0;
GatherPartItem.CurrentLevel=0;
GatherPartItem.ID=0;
local SelectionFunc=nil;

GatherPartItem.Data=nil;

function GatherPartItem:ctor(go, atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_Selected=self:Find("Selected").gameObject;
	self.m_BgRare=self:Find("BgRare"):GetComponent(typeof(Image));
	self.m_Icon=self:Find("Icon"):GetComponent(typeof(Image));
	self.m_Name=self:Find("Name"):GetComponent(typeof(Text));
	self.m_Level=self:Find("Level"):GetComponent(typeof(Text));	
	self.m_Btn=self.prefab:GetComponent(typeof(JButton));
	SelectionFunc=function() self:OnClickSelectBtn(); end;
	self.m_Btn.OnClick= SelectionFunc;
end

function GatherPartItem:OnClickSelectBtn()
	if self.m_Selected.activeInHierarchy then
		return 
	end
	if self.OnSelectCallBack~=nil then
		self.OnSelectCallBack(self.Data);
		self:Select(true)
	end
	
end

function GatherPartItem:Select(select)
	self.m_Selected:SetActive(select)
end

function GatherPartItem:ClearData()
	self.OnSelectCallBack=nil;
	self.ItemID=0;
	self.CurrentLevel=0;
	self.ID=0;
	self.Data=nil;
	self:Select(false)
end

function GatherPartItem:UpdateContent(arg,select,cb)
	self.Data=arg;
	self.ID=self.Data.Config.ID
	self.OnSelectCallBack=cb;
	self.ItemID=self.Data.Config.BasicOutput;
	self.CurrentLevel=arg.Level;
	self.m_Level.text=StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_LvLimit3"),{self.Data.Config.Level})
	self:Select(self.Data.Config.ID==select);
	local config=ConfigDataAccessor.Item[self.ItemID]
	if config~=nil then
		self.m_Name.text=GetQualityColor(config.Quality).. config.Name.."</color>"

		local quality=config.Quality;
		ResourceHelper:LoadIconByAtlasAsync(
			self.m_BgRare,
			ResourceHelper.AtlasPathCommonItem,
			"Bag_Grid_sq"..quality,
			false
		)
		ResourceHelper:LoadIconAsync(self.m_Icon, "", config.Icon, false)
	end

end

return  GatherPartItem

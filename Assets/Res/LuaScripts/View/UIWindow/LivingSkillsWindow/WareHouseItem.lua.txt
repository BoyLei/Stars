WareHouseItem=class("WareHouseItem",UIPanel)
WareHouseItem.m_Has=nil;
WareHouseItem.m_Lock=nil;
WareHouseItem.m_Empty=nil;
WareHouseItem.m_Selected=nil;
WareHouseItem.m_BgRare=nil;
WareHouseItem.m_Icon=nil;
WareHouseItem.m_Num=nil;
WareHouseItem.m_Btn=nil;
WareHouseItem.ItemID=0;

local SelectionFunc=nil;
WareHouseItem.Data=nil;

function WareHouseItem:ctor(go, atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_Has=self:Find("Have").gameObject;
	self.m_Empty=self:Find("Null").gameObject;
	self.m_Selected=self:Find("Have/Selected").gameObject;
	self.m_BgRare=self:Find("Have/BgRare"):GetComponent(typeof(Image));
	self.m_Icon=self:Find("Have/Icon"):GetComponent(typeof(Image));
	self.m_Num=self:Find("Have/Num"):GetComponent(typeof(Text));
	self.m_Btn=self.prefab:GetComponent(typeof(JButton));
	SelectionFunc=function() self:OnClickSelectBtn(); end;
	self.m_Btn.OnClick= SelectionFunc;
end

function WareHouseItem:OnClickSelectBtn()
	if self.Data==nil or self.ItemID==nil then
		return 
	end
	LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips",
		{ self.ItemID, false, nil, self.m_Icon.transform, 0 });
end


function WareHouseItem:UpdateContent(arg)
	self.Data=arg.Data;

	--是否锁定
	local IsLock=self.Data==nil;

	if IsLock then
		self.m_Empty:SetActive(true);
		self.m_Has:SetActive(false);
		return
	else
		self.m_Empty:SetActive(false);
		self.m_Has:SetActive(true);
	end

	self.ItemID=self.Data.BaseID;
	local config=ConfigDataAccessor.Item[self.ItemID]
	if config~=nil then
		--self.m_Name.text=config.Name
		local quality=config.Quality;
		ResourceHelper:LoadIconByAtlasAsync(
			self.m_BgRare,
			ResourceHelper.AtlasPathCommonItem,
			"Bag_Grid_sq"..quality,
			false
		)
		ResourceHelper:LoadIconAsync(self.m_Icon, "", config.Icon, true)
	end

	self.m_Num.text=self.Data.Num	
end

return  WareHouseItem

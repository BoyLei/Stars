---@class DailyTeamBossItem
DailyTeamBossItem = class("DailyTeamBossItem",UIPanel)
DailyTeamBossItem.m_Name=nil;
DailyTeamBossItem.m_Level=nil;
DailyTeamBossItem.m_Head=nil;

function DailyTeamBossItem:ctor(go,atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_Name=self:Find("Name"):GetComponent(typeof(Text))
	self.m_Head=self:Find("Head"):GetComponent(typeof(Image))
	self.m_Level=self:Find("Level"):GetComponent(typeof(Text))
end


--数据更新
function DailyTeamBossItem:UpdateContent(data)
	self.m_Name.text=data.Name;
	self.m_Level.text="Lv"..data.Level;
	--self:GetSprite(data.Head,function (sp)
			--self.m_Head=sp
	--end)
end

return  DailyTeamBossItem;

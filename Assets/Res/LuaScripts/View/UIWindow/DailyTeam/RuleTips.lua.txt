---@class RuleTips
RuleTips = class("RuleTips",UIPanel)

RuleTips.m_Icon=nil;
RuleTips.CloseBtn=nil;
RuleTips.m_Name=nil;
RuleTips.m_Desc=nil;

RuleTips.ID=0;
RuleTips.VerticalLayoutGroup=nil;
RuleTips.ContentContentSizeFitter=nil;
RuleTips.RectTrans=nil;
local OnClickFunc=nil

function RuleTips:ctor(go,atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_Icon=self:Find("Info/Mask/List/Info/Icon"):GetComponent(typeof(Image))
	self.m_Name=self:Find("Info/Mask/List/Info/Name"):GetComponent(typeof(Text))
	self.m_Desc=self:Find("Info/Mask/List/Desc"):GetComponent(typeof(Text))
	self.VerticalLayoutGroup=self:Find("Info/Mask/List"):GetComponent(typeof(VerticalLayoutGroup))
	self.ContentContentSizeFitter=self:Find("Info/Mask/List"):GetComponent(typeof(ContentSizeFitter))
	self.RectTrans=self:Find("Info/Mask/List"):GetComponent(typeof(RectTransform))
	self.CloseBtn=self:Find("Close"):GetComponent(typeof(JButton))
	OnClickFunc=function() self:OnClickRule(); end
	self.CloseBtn.OnClick=OnClickFunc;

end

function RuleTips:OnClickRule()
	self:OnHide()
end

--数据更新
function RuleTips:OnShow(id,pos)
	self.ID=id
	
	local config=ConfigDataAccessor.LevelAffix[id]
	if config~=nil then
		self.m_Name.text=config.Name
		self.m_Desc.text=config.Desc
		self:GetSprite(config.LittleIcon,function (sp)
				self.m_Icon.sprite=sp
		end)

		self.ContentContentSizeFitter:SetLayoutVertical();
		self.VerticalLayoutGroup:CalculateLayoutInputVertical();
		CS.UnityEngine.UI.LayoutRebuilder.ForceRebuildLayoutImmediate(self.RectTrans);
	end
	
	self.prefab.transform.position=pos
	self.prefab:SetActive(true)

end


function RuleTips:OnHide()
	self.prefab:SetActive(false)
end
return  RuleTips;

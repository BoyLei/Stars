---@class DailyTeamHeadItem
DailyTeamHeadItem = class("DailyTeamHeadItem",UIPanel)

DailyTeamHeadItem.m_Name=nil;
DailyTeamHeadItem.m_Head=nil;
DailyTeamHeadItem.m_Flag=nil;
DailyTeamHeadItem.m_Confirm=nil;
DailyTeamHeadItem.m_Cap=nil;
function DailyTeamHeadItem:ctor(go,atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_Name=self:Find("Name"):GetComponent(typeof(Text))
	self.m_Head=self:Find("Head"):GetComponent(typeof(Image))
	self.m_Flag=self:Find("Flag").gameObject
	self.m_Confirm=self:Find("Confirm").gameObject
	self.m_Cap=self:Find("Cap").gameObject
end


--数据更新
function DailyTeamHeadItem:OnShow(data,isShow)
	self.prefab:SetActive(true)
	self.m_Flag:SetActive(isShow)
	self.m_Confirm:SetActive(not isShow)
	self.m_Name.text=data.UserName


	if data.IsCaptain then
		self.m_Cap:SetActive(true)
	else
		self.m_Cap:SetActive(false)
	end
	
	local jobDataCell = ConfigDataAccessor.Job[data.JobId];
	if jobDataCell ~= nil then
		local avatarDataCell= ConfigDataAccessor.Avatar[jobDataCell.AvatarID];
		if avatarDataCell ~= nil then
			local headCfg = ConfigDataAccessor.ModelHead[avatarDataCell.HeadID];
			if headCfg ~= nil then
				--local sp=AtlasManager.Instance:GetSprite(headCfg.HeadAtlasName,headCfg.MinHead);
				-- if sp~=nil and self.m_Head~=nil then
				-- 	self.m_Head.sprite=sp
				-- end
				ResourceHelper:LoadIconByAtlasAsync(
					self.m_Head,
					headCfg.HeadAtlasName,
					headCfg.MinHead,
					false
				)
			end
		end
	end

end

function DailyTeamHeadItem:OnHide()
	self.prefab:SetActive(false)
end

return  DailyTeamHeadItem;

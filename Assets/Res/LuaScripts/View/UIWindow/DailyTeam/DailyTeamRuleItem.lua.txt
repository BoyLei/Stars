---@class DailyTeamRuleItem
DailyTeamRuleItem = class("DailyTeamRuleItem",UIPanel)
DailyTeamRuleItem.m_Head=nil;
DailyTeamRuleItem.ID=0;
DailyTeamRuleItem.Btn=nil;
DailyTeamRuleItem.CallBack=nil;
local OnClickRuleFunc=nil
function DailyTeamRuleItem:ctor(go,atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.m_Head=self:Find("BgRare"):GetComponent(typeof(Image))
	self.Btn=self.prefab.transform:GetComponent(typeof(JButton))
	OnClickRuleFunc=function() self:OnClickRule(); end
	self.Btn.OnClick=OnClickRuleFunc;

end

function DailyTeamRuleItem:OnClickRule()
	if self.CallBack~=nil then
		self.CallBack(self.ID,self.prefab.transform.position)
	end
	--LuaModuleDispatcher:SendMessage(ModuleDef.DailyTeamModule,"OnShowDailyTeamRuleWindow",self.ID)
end

--数据更新
function DailyTeamRuleItem:UpdateContent(data,cb)
	self.ID=data.ID
	self:GetSprite(data.Icon,function (sp)
			self.m_Head.sprite=sp		
	end)
	self.CallBack=cb
end

return  DailyTeamRuleItem;

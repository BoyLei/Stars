---@class DailyTeamItem
DailyTeamItem = class("DailyTeamItem",UIPanel)

DailyTeamItem.m_TaskName=nil;
DailyTeamItem.m_Content=nil;
DailyTeamItem.m_BossIcon=nil;
DailyTeamItem.m_select=nil;
DailyTeamItem.Btn=nil;
DailyTeamItem.Data=nil;
DailyTeamItem.IsFind=false;
DailyTeamItem.IsShow=false;
local SceneMapLoadComplete=nil;
local BtnFun=nil;
function DailyTeamItem:ctor(go,atlas)
	self.prefab = go;
	self.AtlasPath = atlas;

	self.Btn = go.transform:GetComponent(typeof(JButton));
	BtnFun = function()
		self:OnClickHandler();
	end

	self.Btn.OnClick = BtnFun;
	self.m_select = self:Find("select").gameObject;
	self.m_BossIcon = self:Find("BossIcon"):GetComponent(typeof(Image));
	self.m_TaskName = self:Find("TaskName"):GetComponent(typeof(Text));
	self.m_Content = self:Find("Content"):GetComponent(typeof(Text));
	SceneMapLoadComplete=function(mapid)
		self:OnSceneMapLoadComplete(mapid);
	end

	self.IsFind=false;
	self.IsShow=false;
	CS.StarProject.GlobalEvent.OnSceneMapConfigLoad:AddListener(SceneMapLoadComplete);

end

function DailyTeamItem:OnSceneMapLoadComplete(mapid)
	if not self.IsShow then
		return
	end
	if self.Data~=nil and self.Data.MapID==mapid then
		if not self.IsFind then
			TaskHelper.FindPathByArea(self.Data.MapID,self.Data.AreaID,1,nil);
		end
	end
end

-- 设置选中状态
function DailyTeamItem:OnSelect(select)
	self.m_select:SetActive(select);
end

--数据更新
function DailyTeamItem:OnShow(data)

	self.Data=data;
	local bossName=""
	local config= ConfigDataAccessor.Monster[data.BossID]
	if config~=nil then
		bossName=config.Name
		local avatar=ConfigDataAccessor.Avatar[config.AvatarID]
		if avatar~=nil then
			local headcfg=ConfigDataAccessor.ModelHead[avatar.HeadID]
			if headcfg~=nil then
				self:GetSpriteByAtlas(headcfg.HeadAtlasName,headcfg.MinHead,function (sp)
						self.m_BossIcon.sprite=sp
						self.m_BossIcon:SetNativeSize();
				end);
				
			end
		end
	end
	--self.m_TaskName.text="<color=#00c6ff>[集结]</color>".."击败"..bossName
	self.m_TaskName.text = LanguageHelp:GetLanguageByKey("Local_Str_DailyTeamItem_TaskName");

	--self.m_Content.text="寻找"..bossName.."(0/1)"
	self.m_Content.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_DailyTeamItem_Content"), {bossName});

	self.prefab:SetActive(true)
	self.IsShow=true;

end

function DailyTeamItem:OnHide()
	self.prefab:SetActive(false)
	self.Data=nil;
	self.IsShow=false;
	self.IsFind=false;
end

return  DailyTeamItem;

DailyTeamRuleWindow=class("DailyTeamRuleWindow",UIPanel)
DailyTeamRuleWindow.m_Desc1=nil;
DailyTeamRuleWindow.m_Desc2=nil;
DailyTeamRuleWindow.m_Root=nil;
DailyTeamRuleWindow.m_BuffIcon=nil;
DailyTeamRuleWindow.m_Name=nil;
DailyTeamRuleWindow.m_SureBtn=nil;
local SureBtnFunc=nil;
local AutoClose=5;


function DailyTeamRuleWindow:Awake()
	self.m_Desc1=self:Find("Cut4Cam90/AssemblyRatio2/Center/Desc1"):GetComponent(typeof(Text));
	self.m_Desc2=self:Find("Cut4Cam90/AssemblyRatio2/Center/Desc2"):GetComponent(typeof(Text));
	self.m_Root=self:Find("Cut4Cam90/AssemblyRatio2/Center/Root").gameObject;
	self.m_BuffIcon=self:Find("Cut4Cam90/AssemblyRatio2/Center/Root/Item/Card"):GetComponent(typeof(Image));
	self.m_Name=self:Find("Cut4Cam90/AssemblyRatio2/Center/Root/Item/Name"):GetComponent(typeof(Text));
	self.m_SureBtn=self:Find("BackGround"):GetComponent(typeof(JButton));
	SureBtnFunc=function() self:OnClickSureBtn(); end;
end

function DailyTeamRuleWindow:Start()
	self.m_SureBtn.OnClick= SureBtnFunc;
end

function DailyTeamRuleWindow:OnEnable()

end

function DailyTeamRuleWindow:OnDisable()

end

function DailyTeamRuleWindow:OnDestroy()

end

function DailyTeamRuleWindow:OnOpen(arg)
	local id=arg;
	local config=ConfigDataAccessor.LevelAffix[id];
	if config~=nil then
		self.m_Desc1.text=config.Desc
		self.m_Name.text=config.Name
		ResourceHelper:LoadIconAsync(self.m_BuffIcon, "", config.Icon, false)

		-- ResourceManager.Instance:LoadSpriteAsync(config.Icon, function (sp)
		-- 		if self.m_BuffIcon ~= nil and sp ~= nil then
		-- 			self.m_BuffIcon.sprite = sp;
		-- 		end
		-- 	end)
	end
	
	DelayInvoker.DelayInvoke(self,AutoClose,function ()
			self.Close();	
	 end);
end

function DailyTeamRuleWindow:OnClose(arg)

end

function DailyTeamRuleWindow:OnClickSureBtn()
	self.Close();
end



return  DailyTeamRuleWindow

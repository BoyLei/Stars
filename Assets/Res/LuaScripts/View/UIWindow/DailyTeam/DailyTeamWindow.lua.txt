function  DailyTeamWindow:OnBossItemRefesh(tr,idx)
	if self.BossItemList==nil then
		return;
	end
	for i=1,#self.BossItemList do
		if self.BossItemList[i].transform == tr then
			self.BossItemList[i]:UpdateContent(self.BossItemDatas[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
			return
		end
	end

	local bi = DailyTeamBossItem.new(tr.gameObject,self.AtlasPath);
	bi.prefab:SetActive(true);
	bi:UpdateContent(self.BossItemDatas[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
	table.insert(self.BossItemList, bi);
end


function DailyTeamWindow:OnBuffItemRefesh(tr,idx)
	if self.BuffItemList==nil then
		return;
	end
	for i=1,#self.BuffItemList do
		if self.BuffItemList[i].transform == tr then
			self.BuffItemList[i]:UpdateContent(self.BuffItemDatas[idx+1],function (id,pos)
					self:OnClickRuleItem(id,pos)
				end);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
			return
		end
	end

	local bi = DailyTeamRuleItem.new(tr.gameObject,self.AtlasPath);
	bi.prefab:SetActive(true);
	bi:UpdateContent(self.BuffItemDatas[idx+1],function (id,pos)
			self:OnClickRuleItem(id,pos)
		end);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
	table.insert(self.BuffItemList, bi);
end

function DailyTeamWindow:InitCount(config)

	if config~=nil then

		-- 冒险等级
		local curRiskLevel = GameManager.Instance.M_MainPlayerCtrlBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,CS.StarProjectDef.AOIAttrDefine.PlayerLevel)

		if config.UnlockLevel> curRiskLevel then


			--local msg="冒险等级达到 ";
			local msg = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaWindow_NotEnterTips2"), {config.UnlockLevel});

			self.m_Tickets1.text=msg

			self.m_JBtnTickets:SetActive(true)
			self.m_AwardNum.gameObject:SetActive(false)
			self.m_FightBtn.gameObject:SetActive(false)
		else
			local count=0;
			local module= ModuleManager.Instance:GetModule(ModuleDef.GamePlayChoiceModule)
			if module~=nil then
				local luamodule=module:GetLuaTable()
				if luamodule~=nil then
					local countinfo= luamodule:GetComCountInfo(CountConfigID);
					if countinfo~=nil then
						count=countinfo.RemainCount
					end
				end
			end
			--self.m_AwardNum.text="今日剩余次数:"..count
			self.m_AwardNum.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_DailyTeamWindow_AwardNum"), {count});

			self.m_JBtnTickets:SetActive(false)
			self.m_AwardNum.gameObject:SetActive(true)
			self.m_FightBtn.gameObject:SetActive(true)
		end


	end

end

function DailyTeamWindow:InitDropdown()
	self.m_Dropdown:ClearOptions();
	self.DropMap={}
	local curRiskLevel = GameManager.Instance.M_MainPlayerCtrlBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,CS.StarProjectDef.AOIAttrDefine.PlayerLevel)
	self.CurrentDropIndex=0;
	local level=0;
	local list= List_String()
	local config= ConfigDataAccessor.TeamDailyCopy
	for i = 0, #config do
		local area=StringFormat(LanguageHelp:GetLanguageByKey("DailyTeamWindow_SYS_1"), {config[i].UnlockLevel,config[i].AreaName});
		list:Add(area)
		self.DropMap[i]=	config[i].ID
		if curRiskLevel >=config[i].UnlockLevel then
			if config[i].UnlockLevel>level then
				level=config[i].UnlockLevel
				self.CurrentDropIndex=list.Count-1
			end
		end
	end
	self.m_Dropdown:AddOptions(list);
	self.m_Dropdown.value=self.CurrentDropIndex
	self:OnFresh();
end


function DailyTeamWindow:InitBoss(BossGroupID,level)
	self.BossItemDatas={}
	local ms=ConfigDataAccessor.CopyBoss[BossGroupID]
	if ms~=nil then
		for k, v in pairs(ms) do
			if v~=nil then
				local monster=	ConfigDataAccessor.Monster[v.ID]
				if monster~=nil then
					table.insert(self.BossItemDatas,{Name=monster.Name,Level=level,Head=monster.HeadIcon})
				end
			end
		end
	end
	self.m_BossScrollView:ContinueScroll(#self.BossItemDatas);
end

function DailyTeamWindow:InitRule(RuleGroupID)
	self.BuffItemDatas={}
	local ms=ConfigDataAccessor.CopyRule[RuleGroupID]
	if ms~=nil then
		for k, v in pairs(ms) do
			if v~=nil then
				local lf=	ConfigDataAccessor.LevelAffix[v.RuleID]
				if lf~=nil then
					table.insert(self.BuffItemDatas,{ID=v.RuleID,Icon=lf.LittleIcon})
				end
			end
		end
	end
	self.m_BuffScrollView:ContinueScroll(#self.BuffItemDatas);
end

function DailyTeamWindow:InitAward(awards)

	local CostItems={}
	for k, v in pairs(awards) do
		table.insert(CostItems,{ItemID=v,ItemNum=1})
	end
	self.m_ItemPreview:SetItems(CostItems);
end

function DailyTeamWindow:OnClickFightBtn()
	local id=self.DropMap[self.CurrentDropIndex]
	local config= ConfigDataAccessor.TeamDailyCopy[id]
	if config~=nil then
		local curRiskLevel = GameManager.Instance.M_MainPlayerCtrlBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,CS.StarProjectDef.AOIAttrDefine.PlayerLevel)

			if config.UnlockLevel> curRiskLevel then
				--local msg="冒险等级"..config.AdvGrade.."可解锁"
				local msg=StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaWindow_NotEnterTips2"), {config.UnlockLevel})

				Util.ShowMessage(msg);
				return
			end
	

		--1 没有队伍 创建队伍
		--2、有队伍 且是队长  则开始挑战
		--3、有队伍 不是队长  飘字提示 仅有队长才能发起挑战
		local module=ModuleManager.Instance:GetModule(ModuleDef.TeamModule);
		local luatable=module:GetLuaTable();
		local teamInfo=luatable:GetTeamInfo();
		local hasTeam=false;
		local isCaptain=false;
		if teamInfo~=nil and teamInfo.TeamID~=0 and teamInfo.UserInfos.Count>0 then
			hasTeam=true;
			for i = 0,teamInfo.UserInfos.Count -1 do
				local info=teamInfo.UserInfos[i];
				if info~=nil then
					if info.IsCaptain then
						if info.Eid==GameManager.Instance.mainPlayerId then
							isCaptain=true;
						end
					end
				end
			end
		end
		if hasTeam then
			if isCaptain then

				LuaModuleDispatcher:SendMessage(ModuleDef.DailyTeamModule, "OnTeamDailyStartChallengeReq",id);
			else
				Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_152"));
			end
		else
			-- 队长集合
			UIAPI.ShowMsgBox(16, function(EventName)
					if EventName == "SURE" then
						self:QueryTeamInfo(config.PlayID);

					end
				end)
		end
	end
end

function DailyTeamWindow:QueryTeamInfo(playID)
	--TeamLobbyWindow
	local getAllTeamInfoReq = CS.ProtoMsg.GetAllTeamInfoReq();
	getAllTeamInfoReq.Select = CS.ProtoMsg.SelectTeamType.Null;
	getAllTeamInfoReq.PlayID=playID
	CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeSpace,
		getAllTeamInfoReq, false);
end

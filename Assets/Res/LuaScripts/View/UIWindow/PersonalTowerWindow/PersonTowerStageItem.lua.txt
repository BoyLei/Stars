---@class PersonTowerStageItem

PersonTowerStageItem = class("PersonTowerStageItem", UIPanel)

PersonTowerStageItem.m_IsLock = true;

function PersonTowerStageItem:ctor(go, parent, atlas)
	self.prefab = go;
	self.AtlasPath = atlas;
	self.Btn = self:Find("Btn"):GetComponent(typeof(JButton));
	self.ImgNotArrive = self:Find("ImgNotArrive").gameObject;
	self.ObjClaimed = self:Find("Claimed").gameObject;
	self.BtnFun = function()
		self:OnClickHandler();
	end
	self.Btn.OnClick = self.BtnFun;
	self.TxtTitle = self:Find("Text"):GetComponent(typeof(Text));
	self.OnPersonTowerReward = function(data)
		self:OnPersonTowerRewardFunc(data);
	end
	local PersonTowerTargetChangedEvent=LuaModuleDispatcher:Event(ModuleDef.PersonalTowerModule,"PersonTowerReward");
	PersonTowerTargetChangedEvent:AddListener(self.OnPersonTowerReward)
end


--点击事件
function PersonTowerStageItem:OnClickHandler()
	LuaModuleDispatcher:SendMessage(ModuleDef.PersonalTowerModule, "SendPersonTowerAwardReq", {Floor = self.Index });
end

-- 设置选中状态
function PersonTowerStageItem:OnSelect(select)
	local scale = Vector3.one;
	if select then
		scale = scale * 1.1;
	end
	self.Selected:SetActive(select and (not self.m_IsLock)); -- 光圈是：选中状态&&已解锁状态
end
function PersonTowerStageItem:OnPersonTowerRewardFunc(floor)
	if floor == self.Index then
		self.Btn.gameObject:SetActive(false)
	end
end
--数据更新
function PersonTowerStageItem:UpdateContent(floor, isReceived,curFloor)
	self.Index = floor;
	self.Btn.gameObject:SetActive(not isReceived and curFloor>self.Index)
	self.ImgNotArrive.gameObject:SetActive(not isReceived and curFloor<= self.Index)
	self.ObjClaimed.gameObject:SetActive(isReceived)
	local stageItem =  ConfigDataAccessor.TowerStage[self.Index]
	if stageItem then
		for i = 1, 3 do
			local config = ConfigDataAccessor.Item[stageItem.StageAward[i]]
			local btnRewardItem = self:Find("Item"..i):GetComponent(typeof(JButton))
			if config then
				local imgQuality = btnRewardItem.gameObject.transform:Find("BgRare"):GetComponent(typeof(Image))
				local imgIcon = btnRewardItem.gameObject.transform:Find("Icon"):GetComponent(typeof(Image))
				local txtNum = btnRewardItem.gameObject.transform:Find("Text"):GetComponent(typeof(Text))
				local quality = config.Quality;
				ResourceHelper:LoadIconByAtlasAsync(
						imgQuality,
						ResourceHelper.AtlasPathCommonItem,
						"Bag_Grid_sq" .. quality,
						false
				)

				ResourceHelper:LoadIconAsync(imgIcon, "", config.Icon, false)
				txtNum.text = stageItem.StageAwardNum[i]
				btnRewardItem.gameObject:SetActive(true)
				btnRewardItem.OnClick=(function()
					LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips",
							{ stageItem.StageAward[i], false, nil, imgIcon.transform, 0 });
				end);
			else
				btnRewardItem.gameObject:SetActive(false)
			end
		end
	end
	--self.TxtTitle.text =  self.Index.." 层通关奖励"
	self.TxtTitle.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_PersonTowerStageItem_TxtTitle"), {self.Index});

end

return PersonTowerStageItem;

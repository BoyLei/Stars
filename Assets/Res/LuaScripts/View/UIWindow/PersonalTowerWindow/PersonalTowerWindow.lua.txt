function PersonalTowerWindow:InitStageAwardList(stageAwardList)
    local startFloor = 0 -- 跳转的index
    self.StageAwardList = {}
    self.NextStageFloor = math.ceil(self.UnlockFloor / 5) * 5
    local  stageConfig = ConfigDataAccessor.TowerStage
    if stageConfig then
        for m, n in pairs(stageConfig) do
            local find = false
            local stageItem = {}
            if stageAwardList then
                for i, v in ipairs(stageAwardList) do
                    if v.stageId == n.ID then
                        stageItem.Floor = v.stageId
                        stageItem.IsReceived = v.isReceived
                        table.insert(self.StageAwardList,stageItem)
                        find = true
                        break
                    end
                end
            end
            if not find then
                stageItem.Floor = n.ID
                stageItem.IsReceived = false
                table.insert(self.StageAwardList,stageItem)
            end
        end
    end
    table.sort(self.StageAwardList,function (a,b)
        return a.Floor<b.Floor
    end)
    local haveReward = false
    for i, v in ipairs(self.StageAwardList) do
        if v.IsReceived == false and self.UnlockFloor >= v.Floor then
            startFloor = i
            haveReward = true
            break
        end
    end
    if not haveReward then
        for i, v in ipairs(self.StageAwardList) do
            if v.IsReceived == false then
                startFloor = i
                break
            end
        end
    end
    local nextStageData  = ConfigDataAccessor.TowerStage[self.NextStageFloor]
    if nextStageData then
        for i = 1, 3 do
            local config = ConfigDataAccessor.Item[nextStageData.StageAward[i]]
            local btnRewardItem = self.ObjNextAwards:Find("Item"..i):GetComponent(typeof(JButton))
            if config then
                local imgQuality = btnRewardItem.gameObject.transform:Find("BgRare"):GetComponent(typeof(Image))
                local imgIcon = btnRewardItem.gameObject.transform:Find("Icon"):GetComponent(typeof(Image))
                local txtNum = btnRewardItem.gameObject.transform:Find("Text"):GetComponent(typeof(Text))
                local quality = config.Quality;
                ResourceHelper:LoadIconByAtlasAsync(
                        imgQuality,
                        ResourceHelper.AtlasPathCommonItem,
                        "Bag_Grid_sq" .. quality,
                        false
                )

                ResourceHelper:LoadIconAsync(imgIcon, "", config.Icon, false)
                txtNum.text = nextStageData.StageAwardNum[i]
                btnRewardItem.gameObject:SetActive(true)
                btnRewardItem.OnClick=(function()
                    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips",
                            { nextStageData.StageAward[i], false, nil, imgIcon.transform, 0 });
                end);

            else
                btnRewardItem.gameObject:SetActive(false)
            end
        end
        local offsetFloor = 5 - (self.UnlockFloor-1)%5
        --self.TxtNextStage.text =  string.format("再挑战 <color=#FFF191>%d</color> 层可领取",offsetFloor)
        self.TxtNextStage.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_PersonalTowerWindow_TxtNextStage"), {offsetFloor});

        self.ObjNextAwards.gameObject:SetActive(true)
    else
        self.ObjNextAwards.gameObject:SetActive(false)
    end

    if startFloor <= self.m_VisibleMiddleFloor then
        startFloor = 1
    end
    self.SuperScrollViewControllerStage:ContinueScroll(#self.StageAwardList, false, startFloor - 1)
end

-- 初始化界面信息
function PersonalTowerWindow:InitData(arg)
    self:UpdataAllData(arg.PassFloor)
    -- 当前层数（首次打开是已解锁最高层数）
    for i, v in ipairs(self.m_Floors) do
        if self.UnlockFloor == v then
            self.SelectIndex = i
            break
        end
    end
    if self.SelectIndex <= 0 then
        self.SelectIndex = 1
    end
    self:SetMiddleFloorPos(self.SelectIndex)
    self:InitStageAwardList(arg.StageAwardList)
end

-- 更新秘境全量数据
function PersonalTowerWindow:UpdataAllData(passFloor)
    local minFloor,maxFloor
    local length = #ConfigDataAccessor.TowerCopy
    if passFloor + 10 >= length then
        maxFloor = length
    else
        maxFloor = passFloor + 11
    end
    if passFloor - 2 <= 0 then
        minFloor = 1
    else
        minFloor = passFloor - 1
    end
    self.m_Floors = {}
    self.StageAwardList = {}


    for i = minFloor, maxFloor do
        table.insert( self.m_Floors,i)
    end

    -- 设置已解锁的最高层数
    self.UnlockFloor = passFloor + 1
    if self.UnlockFloor > maxFloor then
        self.UnlockFloor = maxFloor
    end
end
function PersonalTowerWindow:OnItemRefesh(tr, idx)
    if self.SuperScrollViewControllerItems == nil then
        return
    end
    local floor = self.m_Floors[idx + 1] --idx是从0开始，如果ItemDataList是lua的table需要idx+1
    local maxFloor = 1
    if self.UnlockFloor > 0 then
        maxFloor = self.UnlockFloor
    end
    local isLock = floor > maxFloor
    for i = 1, #self.SuperScrollViewControllerItems do
        if self.SuperScrollViewControllerItems[i].prefab.transform == tr then
            self.SuperScrollViewControllerItems[i]:UpdateContent(floor, self.SelectIndex, self.UnlockFloor, isLock) --idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end
    local sp =
    PersonTowerFloorItem.new(
            tr.gameObject,
            self.SuperScrollViewControllerRoot,
            self.AtlasPath,
            function(index)
                self:OnSelect(index)
            end
    )
    sp.prefab:SetActive(true)
    sp:UpdateContent(floor, self.m_Floors[self.SelectIndex], self.UnlockFloor, isLock)
    table.insert(self.SuperScrollViewControllerItems, sp)
end
function PersonalTowerWindow:OnStageItemRefesh(tr, idx)
    if self.SuperScrollViewControllerItemsStage == nil then
        return
    end
    local stageItem = self.StageAwardList[idx+1] --idx是从0开始，如果ItemDataList是lua的table需要idx+1
    for i = 1, #self.SuperScrollViewControllerItemsStage do
        if self.SuperScrollViewControllerItemsStage[i].prefab.transform == tr then
            self.SuperScrollViewControllerItemsStage[i]:UpdateContent(stageItem.Floor, stageItem.IsReceived,self.UnlockFloor) --idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end
    local sp =
    PersonTowerStageItem.new(
            tr.gameObject,
            self.SuperScrollViewControllerRootStage,
            self.AtlasPath
    )
    sp.prefab:SetActive(true)
    sp:UpdateContent(stageItem.Floor, stageItem.IsReceived,self.UnlockFloor)
    table.insert(self.SuperScrollViewControllerItemsStage, sp)
end
-- 设置中间的层数
function PersonalTowerWindow:SetMiddleFloorPos(middleFloor)
    local startFloor = 0 -- 这个要计算一下开始层数的偏移量
    if middleFloor > self.m_VisibleMiddleFloor then
        startFloor = middleFloor - self.m_VisibleMiddleFloor
    end
    self.SuperScrollViewController:ContinueScroll(#self.m_Floors, false, startFloor)
end
function PersonalTowerWindow:OnSelect(selectIndex)
    for i, v in ipairs(self.m_Floors) do
        if selectIndex == v then
            self.SelectIndex = i
            if self.AnimBg then
                self.AnimBg:CrossFade("Open", 0, 0, 0)
                self.AnimBg:SetTrigger("Select")
            end
            break
        end
    end
    self:SetAllItemSelectState()
    -- 修改当前层数信息
    self:SetCurrentFloorData()
end

-- 设置当前层数信息
function PersonalTowerWindow:SetCurrentFloorData()
    -- 当前层数
    local floor = self.m_Floors[self.SelectIndex]
    -- 当前层数配置
    self.m_CurrentFloorCfg = ConfigDataAccessor.TowerCopy[floor]
    self.CurrentFloor.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_limitFloor"), {floor});
    local playerLevel = GameManager.Instance:GetPlayerLevel();

    if self.m_CurrentFloorCfg.ConditionID == 0 then
        self.CurLevel.text = playerLevel;
        self.EnterLevel.text = "--"
    else
        local conditionCfg
        ---- 推荐战力
        for i, v in pairs(ConfigDataAccessor.CMcondition) do
            if v.GroupID == self.m_CurrentFloorCfg.ConditionID then
                conditionCfg = v
            end
        end

        if conditionCfg then
            if tonumber(conditionCfg.Args1) > playerLevel then
                self.CurLevel.text = string.format("<color=#c06666>%d</color>" ,playerLevel)
            else
                self.CurLevel.text = playerLevel
            end

            self.EnterLevel.text = conditionCfg.Args1
        end
    end


    ---- 常规奖励
    self:SetConventionReward()
    -- 按钮设置
    local show = false
    local isMete=GameManager.Instance:IsConditionMete(self.m_CurrentFloorCfg.ConditionID);--判断是否满足
    if isMete then
        show = true
    end
    self.TextEnterTips.gameObject:SetActive(not show)
    self.JBtnEnter.gameObject:SetActive(self.UnlockFloor == floor)
    self.JBtnEnterFinish.gameObject:SetActive(self.UnlockFloor> floor)
    if floor ~=  self.UnlockFloor then
        local notHave = not LocalCacheManager.Instance:Has("PersonTowerLevel_"..self.UnlockFloor);
        if notHave then
            LocalCacheManager.Instance:SetValueType("PersonTowerLevel_"..self.UnlockFloor, true, CacheType.Normal);
            RedPointManager.Instance:TriggerConditionTypeBoolValue(RedPointConditionType.PersonTowerLevel, false);
        end
    end
end
function PersonalTowerWindow:SetAllItemSelectState()
    for i = 1, #self.SuperScrollViewControllerItems do
        local isSelect = false
        if self.SuperScrollViewControllerItems[i].Index == self.SelectIndex then
            isSelect = true
        end
        self.SuperScrollViewControllerItems[i]:OnSelect(isSelect)
    end
end
-- 设置常规奖励显示
function PersonalTowerWindow:SetConventionReward()
    if self.m_CurrentFloorCfg == nil then
        return
    end
    -- 获取道具信息
    --local sortTempList = self:GetRewardList(self.m_CurrentFloorCfg.RewardID, self.m_CurrentFloorCfg.RewardNum)
    local sortTempList = self:GetRewardList(self.m_CurrentFloorCfg.CommonAward, self.m_CurrentFloorCfg.CommonAwardNum)
    -- 设置奖励信息
    self:SetRewardItemPreview(self.ConventionRewardItemPreview, sortTempList);
end
function PersonalTowerWindow:SetRewardItemPreview(ItemPreview, CostItems)
    ItemPreview:SetItems(CostItems)
end
-- 获取奖励信息
function PersonalTowerWindow:GetRewardList(rewards, rewardNums)
    local tempList = {}
    for i = 1, #rewards do
        local itemId = rewards[i]
        local itemNum = 1
        if i <= #rewardNums then
            itemNum = rewardNums[i]
        end
        local itemCfg = ConfigDataAccessor.Item[itemId]
        if itemCfg then
            if tempList[itemId] == nil then
                local quality = 1
                if itemCfg ~= nil then
                    quality = itemCfg.Quality
                end
                local item = { ItemID = itemId, ItemNum = itemNum, Quality = quality }
                tempList[itemId] = item
            else
                tempList[itemId] = tempList[itemId].ItemNum + itemNum
            end
        end
    end
    -- 奖励排序，高-》低
    local sortTempList = self:SortReward(tempList)
    return sortTempList
end
-- 奖励排序，高-》低
function PersonalTowerWindow:SortReward(rewardList)
    -- 生成一个用于排序的临时数组
    local tempArray = {}
    for key, value in pairs(rewardList) do
        table.insert(tempArray, value)
    end
    -- 定义比较函数，按照Quality从大到小排序
    local function compare(a, b)
        if a.Quality ~= b.Quality then
            return a.Quality > b.Quality
        end
        return false
    end
    -- 对临时数组进行排序
    table.sort(tempArray, compare)
    return tempArray
end

-- 点击进入按钮
function PersonalTowerWindow:OnClickEnterJBtn()
    if GameManager.Instance:IsConditionMete(self.m_CurrentFloorCfg.ConditionID) then
        local notHave = not LocalCacheManager.Instance:Has("PersonTowerLevel_"..self.UnlockFloor);
        if notHave then
            LocalCacheManager.Instance:SetValueType("PersonTowerLevel_"..self.UnlockFloor, true, CacheType.Normal);
            RedPointManager.Instance:TriggerConditionTypeBoolValue(RedPointConditionType.PersonTowerLevel, false);
        end
        LuaModuleDispatcher:SendMessage(ModuleDef.PersonalTowerModule, "SendEnterPersonTowerLevelReq", self.UnlockFloor)
    else
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_221"));
    end
end

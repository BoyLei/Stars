AvgBookBox = class("AvgBookBox", UIPanel)
AvgBookBox.m_TalkBoxContent = nil
AvgBookBox.m_TalkBoxName = nil

function AvgBookBox:Awake()
    self.m_TalkBoxContent = self:Find("Anim/Scroll View/Viewport/Content/TalkBoxContent"):GetComponent(typeof(Text))
    self.m_TalkBoxName = self:Find("Anim/TalkBoxName"):GetComponent(typeof(Text))
    self.m_TalkBoxName2 = self:Find("Anim/TalkBoxName2"):GetComponent(typeof(Text))
    self.rect_grid = self:Find("Anim/Scroll View/Viewport/Content"):GetComponent(typeof(RectTransform))
    self:Find("Anim/BtnClose"):GetComponent(typeof(JButton)).OnClick = (function()
        self:FinishTalk()
    end)

    self.module = nil
end

function AvgBookBox:BindModule(m)
    print(".............")
    self.module = m
end

function AvgBookBox:Start()
end

function AvgBookBox:OnEnable()
end

function AvgBookBox:OnDisable()
end

function AvgBookBox:OnDestroy()
end

function AvgBookBox:OnOpen(arg)
end

function AvgBookBox:OnClose(arg)
end

function AvgBookBox:GetTalkContentById(idx)
    --self.talkConetntNow = LocalDataManager.Instance:GetCommonDialogDataCell(idx);
    self.talkConetntNow = self.module:CommonDialogDataCellById(idx)

    self:RefeshTalkInfo()
end

function AvgBookBox:RefeshTalkInfo()
    --显示名字 用#分隔
    local arr = string.split(self.talkConetntNow.Name2, "#")
    self.m_TalkBoxName.text = arr[1]
    if #arr > 1 then
        self.m_TalkBoxName2.text = arr[2]
    else
        self.m_TalkBoxName2.text = ""
    end
    self.rect_grid.anchoredPosition = Vector2(self.rect_grid.anchoredPosition.x, 0)
    --显示内容

    self.m_TalkBoxContent.text = self.talkConetntNow.Dialog_desc2

    --播放语音
    if self.talkConetntNow.Voice_id ~= 0 then
    --Util.PlayAvgVoice("Weapon_sword__pickup");
    end
end

function AvgBookBox:FinishTalk()
    UIManager.Instance:CloseWidget("Avg/AvgBookBox")

    self.module:OnFinishAvgBookBox()
end

return AvgBookBox

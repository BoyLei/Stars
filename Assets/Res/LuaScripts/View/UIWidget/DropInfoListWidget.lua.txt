DropInfoListWidget = class("DropInfoListWidget", UIPanel)
DropInfoListWidget.Items = {}

local DropInfoListItem = require("View/Item/DropInfoListItem")

function DropInfoListWidget:Awake()
    self.transform = self.prefab.transform
    self.gameObject = self.prefab

    self.usedCount = 0

    for i = 1, 5 do
        local _go = self.transform:Find("Grid/DropInfoListItem" .. i).gameObject
        local it =
            DropInfoListItem:New(
            _go,
            function()
                self.usedCount = self.usedCount - 1
                if self.usedCount < 0 then
                    self.usedCount = 0
                end
            end
        )
        table.insert(self.Items, it)
    end

    self.go_TaskReward = self.transform:Find("TaskReward").gameObject
    self.go_TaskReward:SetActive(false)
    self.rect_TaskReward = self.go_TaskReward:GetComponent(typeof(RectTransform))
end

function DropInfoListWidget:BindModule(m)
    print("DropInfoListWidget:BindModule")
    self.module = m

    MonoHelper.AddUpdateListener(
        function()
            self:PopCheck()
        end
    )

    --self:PopCheck();
end

function DropInfoListWidget:PopCheck()
    self:Pop()
end

function DropInfoListWidget:Show(b)
    if b then
        --coroutine.resume(self.co);
    else
        --coroutine.wait(self.co);
    end
end

function DropInfoListWidget:Pop()
    local it = self:GetUnUsed()

    --找到池子里没有使用的
    if it ~= nil then
        local data = self.module:PopFront()
        if data ~= nil then
            self.transform:SetAsFirstSibling()

            local itemId = data.Itemid
            local num = data.ItemNum

            if data.AwardType == 1 then
                --1秒关闭
                if self.tw ~= nil then
                    self.tw:Kill()
                    self.tw = nil
                end
                self.tw =
                    DOVirtual.DelayedCall(
                    1,
                    function()
                        self.go_TaskReward:SetActive(false)
                    end
                )
                self.go_TaskReward:SetActive(true)
                self.rect_TaskReward.anchoredPosition = Vector2(self.rect_TaskReward.anchoredPosition.x, -183 + 45 * (self.usedCount + 1))
            else
                if self.tw ~= nil then
                    self.tw:Kill()
                    self.tw = nil
                end
                self.go_TaskReward:SetActive(false)
            end

            --local cfg=LocalDataManager.Instance:GetItemDataCell(itemId);
            local cfg = ConfigDataAccessor.Item[itemId]
            local popInfo = {}
            popInfo.Desc = cfg.Name .. " x" .. num
            popInfo.ItemType = cfg.ItemType
            popInfo.Quality = cfg.Quality
            popInfo.Icon = cfg.Icon

            self:MoveUsedUp()
            it:SetItem(popInfo)
            self.usedCount = self.usedCount + 1

            SoundManager.Instance:PlayWwiseAudio("UI_Drop_Bag_Sound", false, CS.StarProjectDef.E_SoundNTFtype.MyListener_SystemSound, nil)
        end
    end
end

function DropInfoListWidget:MoveUsedUp()
    for i = 1, #self.Items do
        local it = self.Items[i]
        if it.IsUsed then
            it:MoveUp()
        end
    end
end

function DropInfoListWidget:CleanAll()
    for i = 1, #self.Items do
        local it = self.Items[i]
        it:UnUse()
    end
end

function DropInfoListWidget:GetUnUsed()
    for i = 1, #self.Items do
        local it = self.Items[i]
        if not (it.IsUsed) then
            return it
        end
    end

    return nil
end

function DropInfoListWidget:Start()
end

function DropInfoListWidget:OnDisable()
end

function DropInfoListWidget:OnDestroy()

function DropInfoListWidget:OnOpen(arg)

function DropInfoListWidget:OnClose(arg)
    self.module.dropInfoListWidgetView = nil
end

return DropInfoListWidget

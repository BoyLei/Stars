ObjectInteractiveWidget = class("ObjectInteractiveWidget", UIPanel)
ObjectInteractiveWidget.Items = {};

local ObjectPool = require("Common/Helper/ObjectPool")

function ObjectInteractiveWidget:Awake()
	self.transform = self.prefab.transform;
	self.gameObject = self.prefab;

	local _go = self.transform:Find("Grid/ObjectInteractiveItem").gameObject;
	self.objectInteractivePool = ObjectPool:New(3, "View/Item/ObjectInteractiveItem", _go, _go.transform.parent,
		--生成回调
		function(ritem)
			ritem:InitItem(
			);
		end,
		--回收回调
		function(ritem)
			ritem:SetUse(false);
		end
	)
end

--清空全部
function ObjectInteractiveWidget:ClearAllItem()
	self.objectInteractivePool:RecycleAll();
end

--刷新客户端的本地服务
-- Type     -- 本地实体类型
-- ID       -- 交互物件ID
-- TriggerKey  -- 交互回调key
-- EntityKey  -- 特殊的实体key
function ObjectInteractiveWidget:RefeshLocalInteractive(Type, ID, TriggerKey, EntityKey)
	if ID == 0 then
		return;
	end
	local item = self.objectInteractivePool:Spawn();
	local cfg = ConfigDataAccessor.Interact[ID]; --配置
	if cfg ~= nil then
		local name = cfg.ModeltalkeName;
		local hasService = false;
		--选项图标（待完成）
		local icon = "Icon_Submit_01";

		if Type== 1 then
			--// 宝箱
			icon = "Icon_Interactive";

		elseif Type== 2 then
			--// 传送门

		elseif Type== 3 then
			--// 通缉实体
			icon = "Icon_Talk_01";
		elseif Type== 3 then
			--// 召唤物
			icon = "Icon_Talk_01";
		end
		
		local light=false;
		if cfg.IsAutoInter ==2 then
			light=true
		end
		
		item:SetItem(name, icon, ID, function(ritem)
				CS.StarProject.GlobalEvent.onClickLocalEntity:Invoke(Type, ID, TriggerKey, EntityKey);
			end, hasService,0,light);
	else
		print("找不到本地的交互物件配置", ID);
	end
end

--刷新交互列表
function ObjectInteractiveWidget:RefeshInteractive(list)
	self.objectInteractivePool:RecycleAll();
	for i = 0, list.Count - 1 do
		local item = self.objectInteractivePool:Spawn();

		local entityID = list[i];

		local BaseID = GameManager.Instance:GetEntityCfgID(entityID);

		local EntityType = GameManager.Instance:GetEntityType(entityID);

		if EntityType == CS.StarProjectDef.E_EntityType.Npc then
			local cfg = ConfigDataAccessor.Npc[BaseID]; --配置
			if cfg ~= nil then
				local name = cfg.Name;
				--选项图标（待完成）
				local icon = "Icon_Talk_01";
				if BusinessManager.Instance:IsNpcCanService(BaseID) then
					icon = "Icon_Task_01";
				end

				item:SetItem(name, icon, list[i], function(ritem)
						self:OnInteractiveToNpcClick(ritem)
					end, false)
			else
				print("找不到怪物配置", BaseID)
			end
		else
			local cfg = ConfigDataAccessor.Interact[BaseID]; --配置
			if cfg ~= nil then
				local name = cfg.ModeltalkeName;
				local hasService = false;
				local light=false
				if cfg.IsAutoInter==2 then
					light=true
				end
				--选项图标（待完成）
				local icon = "Icon_Submit_01";
				if BusinessManager.Instance:IsObjectCanService(BaseID) then
					icon = "Icon_Task_01";
					hasService = true;
				end

				item:SetItem(name, icon, list[i], function(ritem)
						self:OnInteractiveToInteractClick(ritem)
					end, hasService,0,light)
			else
				print("找不到交互物件配置", BaseID)
			end
		end







		--item:SetItemBak(list[i]);
	end
end

--点击交互按钮
function ObjectInteractiveWidget:OnInteractiveToInteractClick(ritem)
	if ritem ~= nil then
		--旋转面向交互物

		BusinessManager.Instance:TurnToEntity(ritem.EntityID);
		LuaModuleDispatcher:SendMessage(ModuleDef.InterActionModule, "QueryInter", ritem.EntityID);

		--[[

		BusinessManager.Instance:MoveToEntity(ritem.EntityID,function (isFindPath)
		if isFindPath then
		LuaModuleDispatcher:SendMessage(ModuleDef.InterActionModule, "QueryInter",ritem.EntityID);
		end
		end);

		]]
	end
end

--选择NPC的服务
function ObjectInteractiveWidget:CheckNpcInteractiveEffect(ri)
	--基础服务
	if ri.IsBaseService then
		local lobbyServiceReq = CS.ProtoMsg.LobbyServiceReq();
		lobbyServiceReq.MapID = GameManager.Instance:GetCurMapId();
		lobbyServiceReq.UID = ri.NPCEnityID;
		lobbyServiceReq.ObjID = ri.BaseID;
		lobbyServiceReq.Index = ri.Index;
		lobbyServiceReq.ObjType=CS.ProtoMsg.ServiceObjEnum.ObjNpc
		CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby,
			lobbyServiceReq, false);
	else
		--注册的服务

		if ri.Service ~= nil then
			local lobbyService = CS.ProtoMsg.LobbyServiceReq();
			lobbyService.MapID = GameManager.Instance:GetCurMapId();
			lobbyService.UID = ri.NPCEnityID;
			lobbyService.ObjID = ri.BaseID;
			lobbyService.SourceType = CS.ProtoMsg.MapServiceSourceTypeEnum.Task;
			lobbyService.SourceID = ri.Service.ServiceSourceID;
			lobbyService.SourceSubID = ri.Service.ServiceSourceSubID;
			lobbyService.ObjType=CS.ProtoMsg.ServiceObjEnum.ObjNpc
			CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby,
				lobbyService, false);
		end
	end
end

		BusinessManager.Instance:MoveToEntity(ritem.EntityID, function()
				local BaseID = GameManager.Instance:GetEntityCfgID(ritem.EntityID);
				local cfg = ConfigDataAccessor.Npc[BaseID]; --配置
				if cfg ~= nil then
					if ritem.IsJustTalk then
						if self:CheckIsHaveOnlyOneCommonService(BaseID) and ((#cfg.Dialog == 1 and cfg.Dialog[1] == 0) or #cfg.Dialog == 0) then
							--直接发请求（只有一个基础服务,并且对话为空的时候）
							local lobbyServiceReq = CS.ProtoMsg.LobbyServiceReq();
							lobbyServiceReq.MapID = GameManager.Instance:GetCurMapId();
							lobbyServiceReq.UID = ritem.EntityID;
							lobbyServiceReq.ObjID = BaseID;
							lobbyServiceReq.Index = 1;
							lobbyServiceReq.ObjType=CS.ProtoMsg.ServiceObjEnum.ObjNpc
							CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(
								CS.SGF.Network.ServerType.ServerTypeLobby, lobbyServiceReq, false);
						else
							--显示开场白
							--随机对话聊天


							if cfg.DialogGroup ~= {} and #cfg.DialogGroup>0 then
								local talkId=ConfigDataAccessor.GetNpcDialogIdByGroupID(cfg.DialogGroup[1]);
								if talkId>0 then
									LuaModuleDispatcher:SendMessage(ModuleDef.AvgLuaModule, "OnOpenAvgTalkBox",{talkId,nil,ritem.EntityID,true,ritem.Index});
								end
							else
								if cfg.Dialog ~= {} and #cfg.Dialog>0 then
									local talkIdx=math.random(1,#cfg.Dialog);
									LuaModuleDispatcher:SendMessage(ModuleDef.AvgLuaModule, "OnOpenAvgTalkBox",{cfg.Dialog[talkIdx],nil,ritem.EntityID,true,ritem.Index});
								else
									print("ObjectInteractiveWidget OnInteractiveToNpcClick cfg.Dialog ={} 没有对话服务配置 配置没找到");
								end
							end

						end
					else
						self:CheckNpcInteractiveEffect(ritem);
					end
				else
					print("ObjectInteractiveWidget OnInteractiveToNpcClick ritem" .. ritem.EntityID .. ",,NPC 配置没找到");
				end
			end);
	end
end




--判断是否只有唯一服务
function ObjectInteractiveWidget:CheckIsHaveOnlyOneCommonService(BaseID)
	local isHave = false;


	--local cfg=LocalDataManager.Instance:GetNPCDataCell(self.module.BaseID);--配置
	local cfg = ConfigDataAccessor.Npc[BaseID];

	local commonEffect = cfg.CommonEffect;

	local commonCondition = cfg.CommonCondition;


	if #commonEffect > 1 or #commonEffect == 0 then
		return isHave;
	end

	--基础服务
	for i = 1, #commonEffect do
		local canShow = false;
		--没有条件
		if commonCondition[i] == 0 then
			canShow = true;
		else
			--local cc=LocalDataManager.Instance:GetCMconditionDataCell(commonCondition[i]);
			--local cc = ConfigDataAccessor.CMcondition[commonCondition[i]];
			local isMete = GameManager.Instance:IsConditionMete(commonCondition[i]); --判断是否满足
			--local isMete = GameManager.Instance:IsConditionMete(cc.GroupID); --判断是否满足
			if isMete then
				canShow = true;
			else
				--不满足时显示
				--if cc.IsShow then
				--	canShow = true;
				--end
			end
		end

		if canShow then
			if commonEffect[i] ~= 0 then
				isHave = true;
			end
		end
	end

	--[[
	--任务服务
	local services=BusinessManager.Instance:GetNpcOtherService(self.module.BaseID);

	for i=0,services.Count-1 do
	local service=services[i];

	--local ce=LocalDataManager.Instance:GetCMeffectDataCell(service.EffectID);
	local ce=ConfigDataAccessor.CMeffect[service.EffectID];
	local selectItem = self.effectSelectPool:Spawn();
	local tex="";
	local serviceType=4;
	if ce~=nil then
	tex=ce.Name
	serviceType =ce.Effecttype;
	end
	if service.ServiceName~=nil and service.ServiceName~="" then
	tex=service.ServiceName;
	end
	selectItem:SetItem(tex,0,false,service,serviceType);
	end
	]]


	return isHave;
end

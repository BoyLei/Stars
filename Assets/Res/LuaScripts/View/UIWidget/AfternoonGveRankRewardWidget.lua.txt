AfternoonGveRankRewardWidget=class("AfternoonGveRankRewardWidget",UIPanel)

local AfternoonGveRankItem=require("View/Uiwindow/Event/AfternoonGveRankItem")
local SelectBtn= require("View/Item/SelectBtn")

function AfternoonGveRankRewardWidget:Awake()
	self.transform=self.prefab.transform;
	self.gameObject=self.transform.gameObject;

	self.transform:Find("Cut4Cam90/AssemblyRatio2/Info/BtnClose"):GetComponent(typeof(JButton)).OnClick=(function()
		self:OnPlayBack();
	end);

	self.MyRankItem=AfternoonGveRankItem:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/Info/Rank/MyRankItem").gameObject);

	--菜单按钮
	self.MenuBtns={};

	local btn1=SelectBtn:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/Info/tag1").gameObject);
	btn1:InitItem(1,function (idx)
		self:OnMenuBtnClick(idx);
	end);
	table.insert(self.MenuBtns,btn1);

	local btn2=SelectBtn:New(self.transform:Find("Cut4Cam90/AssemblyRatio2/Info/tag2").gameObject);
	btn2:InitItem(2,function (idx)
		self:OnMenuBtnClick(idx);
	end);
	table.insert( self.MenuBtns,btn2);


    --道具列表
    self.AfternoonGveRankItemList={};
	--无限列表
	self.go_AfternoonGveRankItem = self.transform:Find("Cut4Cam90/AssemblyRatio2/Info/Rank/Scroll View/Mask/List/Item").gameObject;

	self.tr_ItemRoot=self.transform:Find("Cut4Cam90/AssemblyRatio2/Info/Rank/Scroll View/Mask/List");
	self.go_AfternoonGveRankItem:SetActive(false);
	self.go_AfternoonGveRankItem.transform:SetParent(self.transform);
	self.EventRewardPersonalScorllView=self.transform:Find("Cut4Cam90/AssemblyRatio2/Info/Rank/Scroll View"):GetComponent(typeof(SuperScrollViewController));
	self.EventRewardPersonalScorllView:NewScroll(self.go_AfternoonGveRankItem, 0, function (tr,idx)
		self:OnAfternoonGveRankItemRefesh(tr,idx);
	end);



end




function AfternoonGveRankRewardWidget:OnMenuBtnClick(idx)

	if self.idxNow==idx then
		return;
	end

	
	self.idxNow=idx;

	for i=1,#self.MenuBtns do
		self.MenuBtns[i]:MakeSelected(false);
	end
	self.MenuBtns[idx]:MakeSelected(true);

	self:Refesh();

end


function AfternoonGveRankRewardWidget:OnAfternoonGveRankItemRefesh(tr,idx)

	for i=1,#self.AfternoonGveRankItemList do
		if self.AfternoonGveRankItemList[i].transform == tr then
            self.AfternoonGveRankItemList[i]:SetItem(self.ItemDataList[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
		end

	end

    local bi = AfternoonGveRankItem:New(tr.gameObject);

	bi.gameObject:SetActive(true);
	bi:SetItem(self.ItemDataList[idx+1]);--idx是从0开始，如果ItemDataList是lua的table需要idx+1

    table.insert(self.AfternoonGveRankItemList, bi);

end




function AfternoonGveRankRewardWidget:Refesh()

    if self.ItemDataList==nil then
        self.ItemDataList ={}
    else
        for k,_ in pairs(self.ItemDataList) do self.ItemDataList[k] = nil end
    end


	--开服天数
	local openServerDay=30;
	if GameManager.Instance.GloInfoSrv~=nil then
		openServerDay=GameManager.Instance.GloInfoSrv.GRisk.OffDay;
	end

	--自己的排名
	local selfData=nil;
	local selfRank=0;

	if self.idxNow==1 then
		--个人排行奖励
		--这里要判断下数组越界
		local cfg = ConfigDataAccessor.GetGNActivRank(openServerDay);

		if self.PersonalData~= nil then
			selfRank=self.PersonalData.Rank;
		end

		for i=1,20 do
			local rank=cfg["Rank"..i];
			if rank==nil then
				break;
			end
	
			local data={};
			data.rank=rank;
			data.rankTex=""..rank;
	
			local nextRank=cfg["Rank"..i+1];
			if nextRank==nil then
				data.rankTex=""..rank.."~";
			else
				if nextRank- rank >1 then
					data.rankTex=""..rank.."~"..(nextRank-1);
				end
			end
	
	
			data.reward=cfg["Reward"..i];
			data.num=cfg["Reward"..i.."Num"];
	
	
			if selfRank>0 then
	
				if nextRank==nil then
					if selfRank>= rank then
						selfData=data;
					end
				else
					if selfRank>= rank and selfRank<nextRank then
						selfData=data;
					end
				end
			end
	
			table.insert( self.ItemDataList, data );
		end
	

	elseif self.idxNow==2 then
		--冒险团排行奖励
		--这里要判断下数组越界
		local cfg = ConfigDataAccessor.GetGNActivGuildRank(openServerDay);


		if self.GuildData~= nil then
			selfRank=self.GuildData.Rank;
		end

		for i=1,20 do
			local rank=cfg["Rank"..i];
			if rank==nil then
				break;
			end
	
			local data={};
			data.rank=rank;
			data.rankTex=""..rank;
	
			local nextRank=cfg["Rank"..i+1];
			if nextRank==nil then
				data.rankTex=""..rank.."~";
			else
				if nextRank- rank >1 then
					data.rankTex=""..rank.."~"..(nextRank-1);
				end
			end
	
	
			data.reward=cfg["Reward"..i.."Show"];

	
	
			if selfRank>0 then
	
				if nextRank==nil then
					if selfRank>= rank then
						selfData=data;
					end
				else
					if selfRank>= rank and selfRank<nextRank then
						selfData=data;
					end
				end
			end
	
			table.insert( self.ItemDataList, data );
		end
	
	end






	--self:SortItem(self.ItemDataList);
	self.EventRewardPersonalScorllView:ContinueScroll(#self.ItemDataList);

	if selfRank>0 and selfData~=nil then
		--刷新自己的
		self.MyRankItem:SetItem(selfData);
	else
		self.MyRankItem:SetNull();
	end
end


function AfternoonGveRankRewardWidget:SortItem(tbl)

	if #tbl > 1 then

		table.sort(tbl, function(a, b)
			--升序
			if a.rank < b.rank then
				return true
			end
			return false
		end)

	end

end

end

end

function AfternoonGveRankRewardWidget:OnDisable()
	
end

function AfternoonGveRankRewardWidget:OnDestroy()

end

end

function AfternoonGveRankRewardWidget:OnOpen(arg)


	self.GuildData=arg.GuildData;
	self.PersonalData=arg.PersonalData;

	self.idxNow=-1;
	self:OnMenuBtnClick(arg.OpenIdx);

end

end

	self.gameObject:SetActive(false);
	UIManager.Instance:CloseWidget("Event/AfternoonGveRankRewardWidget",nil,false);

end

function FirstPayWidget:OnSubShopTagClick(idx)

    self.SelectedIdx = idx;

    for i = 1, #self.subTagsPool.m_ObjPool do
        self.subTagsPool.m_ObjPool[i]:MakeSelected(false);
    end
    self.subTagsPool.m_ObjPool[idx]:MakeSelected(true);

    local cfg = self.subTagsPool.m_ObjPool[idx].cfg; -- 配置表 SignIn

    self.go_BtnGo:SetActive(false);
    self.go_BtnGet:SetActive(false);
    self.go_Get:SetActive(false);
    self.go_NextDay:SetActive(false);

    if self.FirstPaySignInfo ~= nil then
        local HaveFirstPayAct = self.FirstPaySignInfo.HaveFirstPayAct;
        local GetFirstPayAwardNum = self.FirstPaySignInfo.GetFirstPayAwardNum;
        local LastGetFirstPayAwardTime = self.FirstPaySignInfo.LastGetFirstPayAwardTime;

        local diff = idx - GetFirstPayAwardNum; -- 天数相差
        local canGet = false;
        local canJump = false;

        -- 领完了
        if GetFirstPayAwardNum < self.MaxDay then

            -- 一次都没领
            if GetFirstPayAwardNum == 0 then

                -- 判断是否有充值信息，有就领取，没有就跳转
                if self.FinishedProductIds ~= nil and self.FinishedProductIds.Count > 0 then
                    if idx == 1 then
                        canGet = true;
                        -- self.go_BtnGet:SetActive(true);
                    end
                else
                    canJump = true;
                    -- self.go_BtnGo:SetActive(true);
                end

            else
                -- 判断是不是同一天
                local serverNowTime = math.floor(GameManager.Instance:GetServerTimeStamp() / 1000) -- 凌晨五点刷新（扣除5小时）
                local day1 = GetDay(serverNowTime - 5 * 3600);
                local day2 = 0;
                if LastGetFirstPayAwardTime ~= 0 then
                    day2 = GetDay(LastGetFirstPayAwardTime - 5 * 3600); -- 凌晨五点刷新（扣除5小时）
                end

                -- 不是同一天
                if day1 ~= day2 then
                    -- 可以领取
                    if diff == 1 then
                        canGet = true;
                    end
                end

            end
        end

        if canJump then
            self.go_BtnGo:SetActive(true);
        elseif canGet then
            self.go_BtnGet:SetActive(true);
        else
            if diff <= 0 then
                self.go_Get:SetActive(true);
            elseif diff == 1 then
                self.go_NextDay:SetActive(true);
            end
        end

    else
        self.go_BtnGo:SetActive(true);
    end

    self.rewardItemPool:RecycleAll();

    if cfg ~= nil then
        for i = 1, #cfg.Item do
            local d = {}
            d.ItemID = cfg.Item[i];
            d.Title = cfg.Title[i];
            d.ItemNum = cfg.Num[i];

            local item = self.rewardItemPool:Spawn();
            item:SetItem(d);

        end
    end

end

function FirstPayWidget:Refesh()
    self.FirstPaySignInfo = GameManager.Instance.FirstPaySignInfo;
    self.subTagsPool:RecycleAll();

    local idx = 0;
    local SignInCfgs = ConfigDataAccessor.SignIn;

    local ii = 0;
    for k, v in pairs(SignInCfgs) do
        ii = ii + 1;
    end
    self.MaxDay = ii;
    for k, v in pairs(SignInCfgs) do
        if v.Group == 1 and v.SignInType == 1 then
            idx = idx + 1
            local subTag = self.subTagsPool:Spawn();
            subTag:InitItem(idx, nil);
            subTag.cfg = v;
            -- subTag:SetText("第"..idx.."天");
            subTag:SetText(StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_AccumulateItem_day2"), {idx}));

            -- 判断是否有红点
            local go_redPoint = subTag.transform:Find("RedPoint").gameObject;
            local canGet = false;
            if self.FirstPaySignInfo ~= nil then
                local HaveFirstPayAct = self.FirstPaySignInfo.HaveFirstPayAct;
                local GetFirstPayAwardNum = self.FirstPaySignInfo.GetFirstPayAwardNum;
                local LastGetFirstPayAwardTime = self.FirstPaySignInfo.LastGetFirstPayAwardTime;

                local diff = idx - GetFirstPayAwardNum; -- 天数相差

                -- 领完了
                if GetFirstPayAwardNum < self.MaxDay then

                    -- 一次都没领
                    if GetFirstPayAwardNum == 0 then

                        -- 判断是否有充值信息，有就领取
                        if self.FinishedProductIds ~= nil and self.FinishedProductIds.Count > 0 then
                            if idx == 1 then
                                canGet = true;
                            end
                        end

                    else
                        -- 判断是不是同一天
                        local serverNowTime = math.floor(GameManager.Instance:GetServerTimeStamp() / 1000) -- 凌晨五点刷新（扣除5小时）
                        local day1 = GetDay(serverNowTime - 5 * 3600);
                        local day2 = 0;
                        if LastGetFirstPayAwardTime ~= 0 then
                            day2 = GetDay(LastGetFirstPayAwardTime - 5 * 3600); -- 凌晨五点刷新（扣除5小时）
                        end

                        -- 不是同一天
                        if day1 ~= day2 then
                            -- 可以领取
                            if diff == 1 then
                                canGet = true;
                            end
                        end

                    end
                end
            end

            go_redPoint:SetActive(canGet);

        end
    end

    self.FinishedProductIds = nil;

    local RechargeInfo = GameManager.Instance.RechargeInfo;
    if RechargeInfo ~= nil then
        self.FinishedProductIds = RechargeInfo.FinishedProductIds;
    end

    local aimIdx = 1;

    if self.FirstPaySignInfo ~= nil then
        local HaveFirstPayAct = self.FirstPaySignInfo.HaveFirstPayAct;
        local GetFirstPayAwardNum = self.FirstPaySignInfo.GetFirstPayAwardNum;
        local LastGetFirstPayAwardTime = self.FirstPaySignInfo.LastGetFirstPayAwardTime;

        -- 没完了
        if GetFirstPayAwardNum < self.MaxDay then

            -- 一次都没领
            if GetFirstPayAwardNum == 0 then
                aimIdx = 1;
            else
                -- 判断是不是同一天
                local serverNowTime = math.floor(GameManager.Instance:GetServerTimeStamp() / 1000)
                local day1 = GetDay(serverNowTime - 5 * 3600);
                local day2 = 0;
                if LastGetFirstPayAwardTime ~= 0 then
                    day2 = GetDay(LastGetFirstPayAwardTime - 5 * 3600); -- 凌晨五点刷新（扣除5小时）
                end

                -- 不是同一天
                if day1 ~= day2 then
                    aimIdx = GetFirstPayAwardNum + 1;
                else
                    aimIdx = GetFirstPayAwardNum + 1;
                end

            end
        else
            -- 领完了(最后一天)
            aimIdx = GetFirstPayAwardNum;
        end

    end

    self:OnSubShopTagClick(aimIdx);

end

function FirstPayWidget:OnOpen(arg)

    --[[
	if type(arg)=="table" then
		--Lua调用
		self.GNGPersonRewardNtf =arg[1];
		self.GNGGuildRewardNtf =arg[2];
	else
		--C#调用
		self.GNGPersonRewardNtf =arg[0];
		self.GNGGuildRewardNtf =arg[1];
	end
	]]

    -- 监听请求签到
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.FirstPaySignInfoNtfID, function(data)
        self:OnFirstPaySignInfoNtf(data)
    end, nil);

    self:Refesh();

end

function FirstPayWidget:OnFirstPaySignInfoNtf(data)

    local ret = data.data

    DOVirtual.DelayedCall(0.02, function()
        self:Refesh();
    end)

end

function FirstPayWidget:OnPlayBack()

    self.gameObject:SetActive(false);
    UIManager.Instance:CloseWidget("Shop/FirstPayWidget", nil, false);

    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.FirstPaySignInfoNtfID, function(data)
        self:OnFirstPaySignInfoNtf(data)
    end, nil);
end

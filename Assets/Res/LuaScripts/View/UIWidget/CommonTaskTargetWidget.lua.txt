---@class CommonTaskTargetWidget
CommonTaskTargetWidget = class("CommonTaskTargetWidget", UIPanel)
local EctypeTargetItem = require("View/Item/EctypeTargetItem")
CommonTaskTargetWidget.m_RootTween = nil;
CommonTaskTargetWidget.m_RootBtn = nil;
CommonTaskTargetWidget.m_TaskName = nil;
CommonTaskTargetWidget.m_ScrollViewEx = nil;
CommonTaskTargetWidget.Data = nil;
CommonTaskTargetWidget.Tables = nil;
CommonTaskTargetWidget.ScrollViewEx = nil;
CommonTaskTargetWidget.m_ItemGrid = nil;
CommonTaskTargetWidget.Items = {}
CommonTaskTargetWidget.BG = nil;
CommonTaskTargetWidget.List = nil;
CommonTaskTargetWidget.Select = nil;

local OnActiveSelfFunc = nil;

function CommonTaskTargetWidget:Awake()
    self.onCloseDestroy(true)
    self.m_RootTween = self:Find("Root"):GetComponent(typeof(EasyTweenPosition));
    self.m_TaskName = self:Find("Root/TaskName"):GetComponent(typeof(Text));
    self.ScrollViewEx = self:Find("Root/ScrollView"):GetComponent(typeof(SuperScrollViewController));
    self.m_ItemGrid = self:Find("Root/TargetItem").gameObject;
    self.BG = self:Find("Root/BG"):GetComponent(typeof(RectTransform));
    self.List = self:Find("Root/ScrollView/Mask/List"):GetComponent(typeof(RectTransform));
    self.ScrollViewEx:NewScroll(self.m_ItemGrid, 0, function(tr, idx)
        self:OnItemRefesh(tr, idx);
    end);

    OnActiveSelfFunc = function(active)
        self:OnActiveSelf(active);
    end
    self:OnActiveSelf(false);
    self.OnEctypeTargetDataChangeFunc = function(data)
        self:OnEctypeTargetDataChange(data);
    end
    local PersonTowerTargetChangedEvent=LuaModuleDispatcher:Event(ModuleDef.PersonalTowerModule,"PersonTowerTargetChanged");
    PersonTowerTargetChangedEvent:AddListener(self.OnEctypeTargetDataChangeFunc)
end

function CommonTaskTargetWidget:OnActiveSelf(active)
    --self.m_RootTween.gameObject:SetActive(active);

    --self.ScrollViewEx:ContinueScroll(0);
	--	self.m_TaskName.text =""

end

function CommonTaskTargetWidget:OnItemRefesh(tr, idx)
    if self.Items == nil then
        return;
    end
    for i = 1, #self.Items do
        if self.Items[i].transform == tr then
            self.Items[i]:UpdateContent(self.Tables[idx+1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end
    end

    local bi = EctypeTargetItem.new(tr.gameObject, self.AtlasPath);
    bi.prefab:SetActive(true);
    bi:UpdateContent(self.Tables[idx+1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.Items, bi);
end

function CommonTaskTargetWidget:Start()

end

function CommonTaskTargetWidget:itemCountFunc()
    if self.Tables == nil then
        return 0
    end
    return self.Tables.Count;
end

function CommonTaskTargetWidget:OnEctypeTargetDataChange(data)
    self.Items = {};
    self.Data = data;
    self.Tables = self.Data.Contents;
    self.m_TaskName.text = self.Data.TileName
    if self.Tables then
        self.ScrollViewEx:ContinueScroll(#self.Tables);
    end
	CS.UnityEngine.UI.LayoutRebuilder.ForceRebuildLayoutImmediate(self.List);
    self.BG:SetHeight(self.List.sizeDelta.y+20)
    if self.Data.IsNew then
        self.m_RootTween:Play()
    end
end

function CommonTaskTargetWidget:OnEnable()

end

function CommonTaskTargetWidget:OnDisable()

end

function CommonTaskTargetWidget:OnDestroy()
end

function CommonTaskTargetWidget:OnOpen(arg)
    if arg and arg ~= {} then
        self:OnEctypeTargetDataChange(arg)
    end
end

function CommonTaskTargetWidget:OnClose(arg)
    local PersonTowerTargetChangedEvent=LuaModuleDispatcher:Event(ModuleDef.PersonalTowerModule,"PersonTowerTargetChanged");
    PersonTowerTargetChangedEvent:RemoveListener(self.OnEctypeTargetDataChangeFunc)
end

return CommonTaskTargetWidget

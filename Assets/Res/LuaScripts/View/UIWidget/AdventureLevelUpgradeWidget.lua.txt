AdventureLevelUpgradeWidget=class("AdventureLevelUpgradeWidget",UIPanel)


function AdventureLevelUpgradeWidget:Awake()
	self.transform=self.prefab.transform;
	self.gameObject=self.transform.gameObject;

    self.LvName=self.transform:Find("Level"):GetComponent(typeof(Text));
    self.Icon=self.transform:Find("Icon"):GetComponent(typeof(Image));

	self.transform:Find("BtnClose"):GetComponent(typeof(JButton)).OnClick=(function()
		self:OnPlayBack();
	end);
	
end

function AdventureLevelUpgradeWidget:Start()

end

function AdventureLevelUpgradeWidget:OnEnable()
	
end

function AdventureLevelUpgradeWidget:OnDisable()
	
end

function AdventureLevelUpgradeWidget:OnDestroy()

end

function AdventureLevelUpgradeWidget:OnOpen(arg)

	local riskLevel = tonumber(arg);

	local cfg=ConfigDataAccessor.AdvGradeExp[riskLevel];

	if cfg==nil then
		return;
	end

	self.Cfg=ConfigDataAccessor.AdvGradeExp[riskLevel-1];--拿前一级的奖励

	if self.Cfg==nil then
		return;
	end

	local order=cfg.AdvOrder;
	

	local oldOrder=self.Cfg.AdvOrder;
    ResourceHelper:LoadIconByAtlasAsync(
        self.Icon,
        ResourceHelper.AtlasPathAdventure,
        "adventure_"..oldOrder,
        false
    )

	self.LvName.text=self.Cfg.AdvClass;

	DOVirtual.DelayedCall(1,function ()
		ResourceHelper:LoadIconByAtlasAsync(
			self.Icon,
			ResourceHelper.AtlasPathAdventure,
			"adventure_"..order,
			false
		)

		self.LvName.text=cfg.AdvClass;
	end)

	--ResourceHelper:LoadIconByAtlasAsync(self.Icon, ResourceHelper.AtlasPathAdventure, "adventure_"..order, false)



	--local riskLevel=1;
	--UIManager.Instance:OpenWidget("AdventureLevel/AdventureLevelUpgradeWidget", false, riskLevel, nil, CS.StarProjectDef.MainPageCommond.HideNone,true,false);

end


function AdventureLevelUpgradeWidget:OnClose(arg)

end

function AdventureLevelUpgradeWidget:OnPlayBack()
	self.gameObject:SetActive(false);

	self:AfterCloseEvent();

end


--弹奖励
function AdventureLevelUpgradeWidget:AfterCloseEvent()
	


	local rewardsList={};
	if self.Cfg.IfBreak then
		for i=1,#self.Cfg.BreakAward do
			local rw={};
			rw.BaseID=self.Cfg.BreakAward[i];
			rw.Num=self.Cfg.BreakAwardNum[i];
			rw.EntityID=0;
			table.insert( rewardsList,rw);
		end
	else
		for i=1,#self.Cfg.AwardItem do
			local rw={};
			rw.BaseID=self.Cfg.AwardItem[i];
			rw.Num=self.Cfg.AwardNum[i];
			rw.EntityID=0;
			table.insert( rewardsList,rw);
		end
	end

	--弹窗逻辑（先塞进去之后在外面调用）
	LuaModuleDispatcher:SendMessage(ModuleDef.RewardsPopModule, "OnOpenRewardsPop",{rewardsList});


end

return  AdventureLevelUpgradeWidget

local CashItem = require("View/Item/CashItem");

function BuyCashWidget:Awake()
    self.transform = self.prefab.transform;
    self.gameObject = self.transform.gameObject;

    self.transform:Find("Cut4Cam90/AssemblyRatio2/BtnClose"):GetComponent(typeof(JButton)).OnClick = (function()
        self:OnPlayBack();

    end);

    self.go_ChargeBtn = self.transform:Find("Cut4Cam90/AssemblyRatio2/ChargeBtn").gameObject;
    self.go_ChargeBtn:GetComponent(typeof(JButton)).OnClick = (function()

    end);
    self.go_ChargeBtn:SetActive(false);

    self.ShopItemList = {};
    -- 无限列表
    self.go_ShopItem = self.transform:Find("Cut4Cam90/AssemblyRatio2/CashItem").gameObject;

    self.tr_ItemRoot = self.transform:Find("Cut4Cam90/AssemblyRatio2/Scroll View/Mask/List");
    self.go_ShopItem:SetActive(false);
    self.ShopScorllView = self.transform:Find("Cut4Cam90/AssemblyRatio2/Scroll View"):GetComponent(typeof(
        SuperScrollViewController));
    self.ShopScorllView:NewScroll(self.go_ShopItem, 0, function(tr, idx)
        self:OnShopItemRefesh(tr, idx);
    end);

end

function BuyCashWidget:BindModule(module)
    self.module = module;
end

function BuyCashWidget:OnShopItemRefesh(tr, idx)

    for i = 1, #self.ShopItemList do
        if self.ShopItemList[i].transform == tr then
            self.ShopItemList[i]:SetItem(self.ItemDataList[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
            return
        end

    end

    local bi = CashItem:New(tr.gameObject);
    bi:InitItem(function(ritem)
        self:OnItemClick(ritem);
    end)
    bi.gameObject:SetActive(true);
    bi:SetItem(self.ItemDataList[idx + 1]); -- idx是从0开始，如果ItemDataList是lua的table需要idx+1
    table.insert(self.ShopItemList, bi);

end

function BuyCashWidget:OnItemClick(ritem)

    local getRechargeOrderReq = CS.ProtoMsg.GetRechargeOrderReq();
    getRechargeOrderReq.ProductID = ritem.GoodsId;
    getRechargeOrderReq.OS = 1;
    getRechargeOrderReq.ChannelId = 1;
    getRechargeOrderReq.Extension = "";
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby,
        getRechargeOrderReq, false);

end

function BuyCashWidget:ClearItemData()

    if self.ItemDataList == nil then
        self.ItemDataList = {}
    else
        for k, _ in pairs(self.ItemDataList) do
            self.ItemDataList[k] = nil
        end
    end
end

function BuyCashWidget:RefeshItems()

    local RechargeInfo = GameManager.Instance.RechargeInfo;
    local FinishedProductIds = nil;
    if RechargeInfo ~= nil then
        FinishedProductIds = RechargeInfo.FinishedProductIds;
    end

    -- 获取List
    self:ClearItemData();

    for k, v in pairs(ConfigDataAccessor.ExternalPay) do
        local good = v;
        if good.PayType == 1 then
            local shopData = {};
            shopData.cfg = good;

            shopData.isFirst = true;

            if FinishedProductIds ~= nil then
                for k2, v2 in pairs(FinishedProductIds) do
                    if k2 == v.ID and v2 > 0 then
                        shopData.isFirst = false;
                    end
                end
            end

            -- 限购相关
            table.insert(self.ItemDataList, shopData);
        end
    end

    -- 排序
    self:SortItem(self.ItemDataList);

    self.ShopScorllView:ContinueScroll(#self.ItemDataList, false);

end

function BuyCashWidget:SortItem(tbl)

    if #tbl > 1 then

        table.sort(tbl, function(a, b)
            return a.cfg.Price < b.cfg.Price
        end)

    end

end

function BuyCashWidget:Refesh()

    self:RefeshItems();

end

end

end

end

end

function BuyCashWidget:OnOpen(arg)

    self.OpenFrom = tonumber(arg);

    --[[
	if type(arg)=="table" then
		--Lua调用
		self.GNGPersonRewardNtf =arg[1];
		self.GNGGuildRewardNtf =arg[2];
	else
		--C#调用
		self.GNGPersonRewardNtf =arg[0];
		self.GNGGuildRewardNtf =arg[1];
	end
	]]

    -- 监听充值订单回调
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.GetRechargeOrderRetID, function(data)
        self:OnGetRechargeOrderRet(data)
    end, nil);

    -- Local_Str_Auto_Tips_86并发货通知
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.PaySuccessNtfID, function(data)
        self:OnPaySuccessNtf(data)
    end, nil);

    self:Refesh();

end

function BuyCashWidget:OnGetRechargeOrderRet(data)

    local ret = data.data

    print("请求充值订单回调")

    local OrderId = ret.OrderId;

end

function BuyCashWidget:OnPaySuccessNtf(data)

    local ret = data.data

    Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_86"));

    print("充值成功回调")

    local ProductID = ret.ProductID;

    self:Refesh();

end

function BuyCashWidget:OnClose(arg)

end

function BuyCashWidget:OnPlayBack()

    self.gameObject:SetActive(false);
    UIManager.Instance:CloseWidget("Shop/BuyCashWidget", nil, false);

    -- 请求充值订单
    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.GetRechargeOrderRetID, function(data)
        self:OnGetRechargeOrderRet(data)
    end, nil);

    -- 充值成功并发货通知
    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.PaySuccessNtfID, function(data)
        self:OnPaySuccessNtf(data)
    end, nil);

    if self.OpenFrom == 1 then
        LuaModuleDispatcher:SendMessage(ModuleDef.CommercializationModule, "OnOpenCommercialization", {1})
        -- UIManager.Instance:OpenWidget("Shop/FirstPayWidget", false, nil, nil, CS.StarProjectDef.MainPageCommond.HideNone,false,false);	
    end

end

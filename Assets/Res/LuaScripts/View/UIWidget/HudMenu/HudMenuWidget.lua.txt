function HudMenuWidget:RefeshShopBtn()
    if not (GameManager.Instance:IsUnLockPayFunction()) then
        -- 没有开启支付系统不显示首充
        self.go_JBtnFirstCharge1.transform.parent.gameObject:SetActive(false)
        return
    end

    local FirstPaySignInfo = GameManager.Instance.FirstPaySignInfo

    if FirstPaySignInfo == nil then
        -- 临时关闭钻石商店
        -- self.go_JBtnFirstCharge1:SetActive(false);
        -- self.go_JBtnShop:SetActive(true);
        self.go_JBtnFirstCharge1.transform.parent.gameObject:SetActive(false)

        return
    end

    local isFinishFirstCharge = FirstPaySignInfo.HaveFirstPayAct
    local isGetAllFirstChargeItem = FirstPaySignInfo.GetFirstPayAwardNum > 3
    local LastGetFirstPayAwardTime = FirstPaySignInfo.LastGetFirstPayAwardTime

    -- 判断是不是同一天
    local serverNowTime = math.floor(GameManager.Instance:GetServerTimeStamp() / 1000)
    local day1 = GetDay(serverNowTime - 5 * 3600)
    local day2 = 0
    if LastGetFirstPayAwardTime ~= 0 then
        day2 = GetDay(LastGetFirstPayAwardTime - 5 * 3600) -- 凌晨五点刷新（扣除5小时）
    end

    if isGetAllFirstChargeItem then
        -- 临时关闭钻石商店
        -- 领完了(最后一天)
        -- self.go_JBtnFirstCharge1:SetActive(false);
        -- self.go_JBtnShop:SetActive(true);
        self.go_JBtnFirstCharge1.transform.parent.gameObject:SetActive(false)
    else
        if isFinishFirstCharge then
            -- 临时关闭钻石商店
            -- 首充过了
            -- self.go_JBtnFirstCharge1:SetActive(false);
            -- self.go_JBtnShop:SetActive(true);
            self.go_JBtnFirstCharge1.transform.parent.gameObject:SetActive(false)
        else
            -- 临时关闭钻石商店
            -- 没首充
            self.go_JBtnFirstCharge1:SetActive(true)
            self.go_JBtnShop:SetActive(false)
            self.go_JBtnFirstCharge1.transform.parent.gameObject:SetActive(true)
        end
    end
end

function HudMenuWidget:InitData()
    self.M_EntityBase = GameManager.Instance.M_MainPlayerCtrlBase
    if self.M_EntityBase == nil then
        return
    end
    -- 等级
    self.m_LevelCur = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,
        CS.StarProjectDef.AOIAttrDefine.PlayerLevel)
    self:SetLevel(self.m_LevelCur)
    -- 经验
    self.m_EXPSum = self:GetCurLevelExp()
    self.m_EXPCur = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,
        CS.StarProjectDef.AOIAttrDefine.PlayerExp)
    self:UpdatePlayerExp(self.m_EXPCur)
    -- 渠道名
    self.AreaName = ""
    -- 玩家角色名
    self.PlayerName.text = self.M_EntityBase.M_Curr.M_Name

    -- 冒险等级
    local RiskInfo = GameManager.Instance.M_MainPlayerCtrlBase.Data.Attrs:GetProtoValue(
        CS.StarProjectDef.AOIAttrDefine.RiskInfo)
    if RiskInfo ~= nil then
        local curRiskLevel = RiskInfo.CurRiskLevel
        local cfg = ConfigDataAccessor.AdvGradeExp[curRiskLevel]
        if cfg ~= nil then
            self.RiskText.text = cfg.AdvClass .. GetRomaNum(cfg.AdvOrder)
        end
    end

    -- 冒险团
    local guidName = BusinessManager.Instance:GetGuildName()
    if guidName == "" then
        guidName = LanguageHelp:GetLanguageByKey("Local_Str_guidName")
    end
    self.GuildName.text = guidName
    -- 职业头像
    local jobId = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Uint,
        CS.StarProjectDef.AOIAttrDefine.Job)
    local jobCfg = ConfigDataAccessor.Job[jobId]
    if jobCfg ~= nil then
        local avatarDataCell = ConfigDataAccessor.Avatar[jobCfg.AvatarID]
        if avatarDataCell ~= nil then
            local headCfg = ConfigDataAccessor.ModelHead[avatarDataCell.HeadID]
            if headCfg ~= nil then
                self:GetSpriteByAtlas(headCfg.HeadAtlasName, headCfg.MiddleHead, function(sp)
                    self.JobHeadImg.sprite = sp
                end)
            end
        end
    end
    -- 职业标签图标
    local jobshi = math.floor(jobId / 10)
    local jobLabelIconPath = self.BigIconPath .. "IconRoleLabel_" .. jobshi
    ResourceHelper:LoadIconAsync(self.JobLabelIcon, "", jobLabelIconPath, false)

    -- 职业类型icon设置
    local jobTypeBgName = "Icon_ArmBg" .. jobshi
    ResourceHelper:LoadIconByAtlasAsync(self.JobTypeBg, ResourceHelper.AtlasPathRole, jobTypeBgName, false)
    local jobTypeName = "Icon_Arm" .. jobshi
    ResourceHelper:LoadIconByAtlasAsync(self.JobTypeIcon, ResourceHelper.AtlasPathRole, jobTypeName, false)
    self:RefeshShopBtn()

    self.BlockAll:SetActive(true)
    self.PlayAnimation("Com_Position_R_Open", function()
        self.BlockAll:SetActive(false)
        self.PanelOffLineData.enabled = true;
        SoundManager.Instance:PlayEventName("UI_Windows_Side_01", self.gameObject, self.gameObject)
    end)
end

function HudMenuWidget:GetCurLevelExp()
    if self.m_LevelCur == 0 then
        return 0
    end
    local cfg = ConfigDataAccessor.LevelExp[self.m_LevelCur]
    if cfg ~= nil then
        return cfg.Exp
    end
    return 0
    -- return ConfigDataAccessor.LevelExp[self.m_LevelCur].Exp
end

function HudMenuWidget:UpdatePlayerExp(value)
    if self.M_EntityBase == nil or self.M_EntityBase.Data == nil then
        return
    end
    self.m_EXPCur = value
    self:SetExpUI()
end

function HudMenuWidget:UpdatePlayerExpSum(value)
    if self.M_EntityBase == nil or self.M_EntityBase.Data == nil then
        return
    end
    self.m_EXPSum = value
    self:SetExpUI()
end

function HudMenuWidget:SetExpUI()
    local ratio = 0
    if self.m_EXPSum > 0 then
        ratio = self.m_EXPCur / self.m_EXPSum
        if ratio > 1 then
            ratio = 1
        end
    end
    self.ExpProgress.fillAmount = ratio
    self.Exp.text = tostring(self.m_EXPCur) .. "/" .. tostring(self.m_EXPSum)
end

function HudMenuWidget:SetLevel(value)
    if self.M_EntityBase == nil or self.M_EntityBase.Data == nil then
        return
    end
    local isUpgrade = self.m_LevelCur ~= tonumber(value)
    self.m_LevelCur = tonumber(value)
    -- self.Level.text = "角色等级 " .. self.m_LevelCur
    self.Level.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Level2"), {self.m_LevelCur})

    if isUpgrade then
        self:UpgradeLevel()
    end
end

function HudMenuWidget:UpgradeLevel()
    local expSum = self:GetCurLevelExp()
    self:UpdatePlayerExpSum(expSum)
end

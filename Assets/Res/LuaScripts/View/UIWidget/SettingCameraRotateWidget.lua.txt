SettingCameraRotateWidget = class("SettingCameraRotateWidget", UIPanel)

function SettingCameraRotateWidget:Awake()
    self:InitItems()
end

function SettingCameraRotateWidget:BindModule(m)
    self.module = m
end

function SettingCameraRotateWidget:InitItems()
    self.transform = self.prefab.transform
    self.gameObject = self.transform.gameObject

    --动画相关
    --self.anim = self.transform:GetComponent(typeof(Animator))

    --确认
    local btnConfirm = self.transform:Find("Cut4Cam90/AssemblyRatio2/Btn"):GetComponent(typeof(JButton))
    btnConfirm.OnClick = (function()
        self:OnConfirmClick()
    end)

    self.m_IsCanRotateCamera = true

    local btnL = self.transform:Find("Cut4Cam90/AssemblyRatio2/1"):GetComponent(typeof(JButton))
    btnL.OnClick = (function()
        self.m_IsCanRotateCamera = true
        self.title1.color = Color(1, 1, 1, 1)
        self.title2.color = Color(1, 1, 1, 0.5)
        self.desc1.color = Color(1, 1, 1, 1)
        self.desc2.color = Color(1, 1, 1, 0.5)
        self.select1:SetActive(true)
        self.select2:SetActive(false)
    end)
    local btnR = self.transform:Find("Cut4Cam90/AssemblyRatio2/2"):GetComponent(typeof(JButton))
    btnR.OnClick = (function()
        self.m_IsCanRotateCamera = false
        self.title1.color = Color(1, 1, 1, 0.5)
        self.title2.color = Color(1, 1, 1, 1)
        self.desc1.color = Color(1, 1, 1, 0.5)
        self.desc2.color = Color(1, 1, 1, 1)
        self.select1:SetActive(false)
        self.select2:SetActive(true)
    end)

    self.title1 = self.transform:Find("Cut4Cam90/AssemblyRatio2/tt1"):GetComponent(typeof(Text))
    self.title2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/tt2"):GetComponent(typeof(Text))
    self.desc1 = self.transform:Find("Cut4Cam90/AssemblyRatio2/tt1_2"):GetComponent(typeof(Text))
    self.desc2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/tt2_2"):GetComponent(typeof(Text))

    self.select1 = self.transform:Find("Cut4Cam90/AssemblyRatio2/sle1").gameObject
    self.select2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/sle2").gameObject
end

function SettingCameraRotateWidget:OnConfirmClick()
    self:OnCloseInfoClick()
end

function SettingCameraRotateWidget:Start()
    print("SettingCameraRotateWidget:Start")
end

function SettingCameraRotateWidget:OnEnable()
end

function SettingCameraRotateWidget:OnDisable()
end

function SettingCameraRotateWidget:OnDestroy()
end

function SettingCameraRotateWidget:OnOpen(arg)
    self.m_IsCanRotateCamera = true
    self.title1.color = Color(1, 1, 1, 1)
    self.title2.color = Color(1, 1, 1, 0.5)
    self.desc1.color = Color(1, 1, 1, 1)
    self.desc2.color = Color(1, 1, 1, 0.5)
    self.select1:SetActive(true)
    self.select2:SetActive(false)
end

function SettingCameraRotateWidget:OnClose(arg)
    self:CloseWidget()
end

function SettingCameraRotateWidget:OnCloseInfoClick()
    --self.anim:Play("Out")
    DOVirtual.DelayedCall(
        0.2,
        function()
            UIManager.Instance:CloseWidget("SettingWindow/Prefab/SettingCameraRotateWidget", nil, true)
        end
    )
    GameManager.Instance:SetRotateCameraState(self.m_IsCanRotateCamera)
    CS.StarProject.GlobalEvent.onSetRotateCamera:Invoke(-1)
end

function SettingCameraRotateWidget:CloseWidget()
end

return SettingCameraRotateWidget

function WorldLvUpWidget:Awake()
    self.transform = self.prefab.transform
    self.gameObject = self.transform.gameObject

    local FrameTransForm = self.transform:Find("Cut4Cam90/AssemblyRatio2").transform;
    self.ServiceDays = FrameTransForm:Find("bg/ServiceDays"):GetComponent(typeof(Text))
    self.LevelUp = FrameTransForm:Find("bg/LevelUp"):GetComponent(typeof(Text))
    self.CurLevel = FrameTransForm:Find("bg/CurLevel"):GetComponent(typeof(Text))
    self.LevelTips = FrameTransForm:Find("bg/LevelTips"):GetComponent(typeof(Text))
    self.JBtnSure = FrameTransForm:Find("bg/JBtnSure"):GetComponent(typeof(JButton))
    self.JBtnSureFun = function()
        self:OnClickSureFunJBtn()
    end
    self.JBtnSure.OnClick = self.JBtnSureFun

end

-- 打开界面
function WorldLvUpWidget:OnOpen(data)
    local SrvOpenDay = data.SrvOpenDay;
    self.ServiceDays.text = StringFormat(LanguageHelp:GetLanguageByKey("WorldLvUpWidget_1"), {SrvOpenDay})
    local LevelLimitId = data.LevelLimitId;

    local cfg = ConfigDataAccessor.LevelLimit[LevelLimitId];
    if cfg == nil then
        return;
    end

    local lastCfg = ConfigDataAccessor.GetAddIndexLevelLimitCfg(LevelLimitId, -1);
    local lastLevelUp = 1;
    if lastCfg ~= nil then
        lastLevelUp = lastCfg.Level;
    end

    self.LevelUp.text = StringFormat(LanguageHelp:GetLanguageByKey("WorldLvUpWidget_2"), {lastLevelUp});
    self.CurLevel.text = tostring(cfg.Level);

    local levelTips = "";
    local MainPlayerAllExp = GameManager.Instance:GetPlayerEXP();
    local mainPlayerLevel = GameManager.Instance:GetPlayerLevel();
    local LevelExpCfg = ConfigDataAccessor.LevelExp[mainPlayerLevel];
    if LevelExpCfg ~= nil then
        if MainPlayerAllExp - LevelExpCfg.Exp > 0 then
            levelTips = StringFormat(LanguageHelp:GetLanguageByKey("WorldLvUpWidget_3"), {MainPlayerAllExp, mainPlayerLevel + 1})
        end
    end

    self.LevelTips.text = levelTips;
end

function WorldLvUpWidget:OnClickSureFunJBtn()
    local battleSocket = NetworkManager.Instance.gameSocket
    local message = CS.ProtoMsg.WorldLevelConfirmReport()
    battleSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, message, false)

    self.Close()
end

-- 关闭界面
function WorldLvUpWidget:OnClose()
    -- 清理资源，关闭界面的逻辑
end

return WorldLvUpWidget

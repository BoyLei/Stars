ShowAreaNameWidget = class("ShowAreaNameWidget", UIPanel)

function ShowAreaNameWidget:Awake()
    self.transform = self.prefab.transform
    self.gameObject = self.prefab

    self.map_AutoAlpha = self:Find("Map"):GetComponent(typeof(AutoAlpha))
    self.area_AutoAlpha = self:Find("Area"):GetComponent(typeof(AutoAlpha))

    self.map_AutoAlpha.gameObject:SetActive(false)
    self.area_AutoAlpha.gameObject:SetActive(false)

    self.go_map = self:Find("Map").gameObject
    self.go_area = self:Find("Area").gameObject

    self.map_name = self:Find("Map/TextTile"):GetComponent(typeof(Text))
    self.map_nameImage = self:Find("Map/LevelImage"):GetComponent(typeof(Image))

    self.area_name = self:Find("Area/TextTile"):GetComponent(typeof(Text))
    self.area_nameImage = self:Find("Area/LevelImage"):GetComponent(typeof(Image))

    self.is_play_map_AutoAlpha = false

    self.module = nil
end

function ShowAreaNameWidget:BindModule(m)
    print(".............")
    self.module = m
end

function ShowAreaNameWidget:Start()
end

function ShowAreaNameWidget:OnEnable()
end

function ShowAreaNameWidget:OnDisable()
end

function ShowAreaNameWidget:OnDestroy()
end

function ShowAreaNameWidget:OnOpen(arg)
    --新手引导期间不显示(新手村，王城)
    if not (TaskHelper.IsTaskFinsh(268500992)) or not (TaskHelper.IsTaskFinsh(268632064)) then
        self:OnPlayBack()
        return
    end

    local name = arg[0]
    local type = arg[1]
    local imageName = arg[2]

    --地图响应
    if type == 1 then
        self.go_map:SetActive(true)
        self.go_area:SetActive(false)

        --[[

		if imageName~="" then
			self.map_name.gameObject:SetActive(false);
			self.map_nameImage.gameObject:SetActive(false);
			ResourceHelper:LoadIconByAtlasAsync(self.map_nameImage, ResourceHelper.AtlasPathTextMap, imageName, true,function ()
				self.map_nameImage.gameObject:SetActive(true);
			end);
		else
			self.map_name.text=name;
			self.map_name.gameObject:SetActive(true);
			self.map_nameImage.gameObject:SetActive(false);
		end
		]]
        self.map_name.text = string.upper(name)
        self.map_name.gameObject:SetActive(true)
        self.map_nameImage.gameObject:SetActive(false)

        self.is_play_map_AutoAlpha = true

        if not self.map_AutoAlpha.activeInHierarchy then
            self.map_AutoAlpha.gameObject:SetActive(true)
        end
        self.map_AutoAlpha:Play(
            function()
                self:OnPlayBack()
                self.is_play_map_AutoAlpha = false
            end
        )
    else
        --区域响应

        --如果正在播放地图响应区域那么就不播区域响应

        if not (self.is_play_map_AutoAlpha) then
            self.go_map:SetActive(false)
            self.go_area:SetActive(true)
            self.area_name.text = name

            --[[

			if imageName~="" then
				self.area_name.gameObject:SetActive(false);
				self.area_nameImage.gameObject:SetActive(true);
				ResourceHelper:LoadIconByAtlasAsync(self.area_nameImage, ResourceHelper.AtlasPathTextMap, imageName, true);
			else
				self.area_name.text=name;
				self.area_name.gameObject:SetActive(true);
				self.area_nameImage.gameObject:SetActive(false);
			end
			]]
            self.area_name.text = string.upper(name)
            self.area_name.gameObject:SetActive(true)
            self.area_nameImage.gameObject:SetActive(false)

            if not self.area_AutoAlpha.activeInHierarchy then
                self.area_AutoAlpha.gameObject:SetActive(true)
            end
            self.area_AutoAlpha:Play(
                function()
                    self:OnPlayBack()
                end
            )
        end
    end
end

function ShowAreaNameWidget:OnClose(arg)
end

function ShowAreaNameWidget:OnPlayBack()
    UIManager.Instance:CloseWidget("StarWorld/ShowAreaNameWidget", nil, true)
    --self.gameObject:SetActive(false);
end

return ShowAreaNameWidget

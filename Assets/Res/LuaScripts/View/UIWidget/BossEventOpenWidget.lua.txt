BossEventOpenWidget = class("BossEventOpenWidget", UIPanel)

function BossEventOpenWidget:Awake()
    self.m_Item = self:Find("Item"):GetComponent(typeof(AutoAlpha))
    self.m_Level = self:Find("Item/Level"):GetComponent(typeof(Text))
    self.m_transform = self.prefab.transform
end

function BossEventOpenWidget:Start()
end

function BossEventOpenWidget:OnEnable()
end

function BossEventOpenWidget:OnDisable()
end

function BossEventOpenWidget:OnDestroy()
end

function BossEventOpenWidget:OnOpen(arg)
    --local num = tostring(arg);
    --self.m_Level.text = num;
    if not self.m_transform.gameObject.activeSelf then
        self.m_transform.gameObject:SetActive(true)
    end
    if not self.m_Item.activeInHierarchy then
        self.m_Item.gameObject:SetActive(true)
    end
    self.m_Item:Play(
        function()
            self:OnPlayBack()
        end
    )
end

function BossEventOpenWidget:OnClose(arg)
end

function BossEventOpenWidget:OnPlayBack()
    self.m_transform.gameObject:SetActive(false)
end

return BossEventOpenWidget

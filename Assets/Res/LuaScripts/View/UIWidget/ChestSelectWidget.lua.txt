ChestSelectWidget = class("ChestSelectWidget", UIPanel)
ChestSelectWidget.IsOpen = false

local ObjectPool = require("Common/Helper/ObjectPool")


function ChestSelectWidget:Awake()
    self:InitItems()
end

function ChestSelectWidget:BindModule(m)
    self.module = m
end

function ChestSelectWidget:InitItems()
    self.transform = self.prefab.transform
    self.gameObject = self.transform.gameObject

    local btnCloseInfo = self.transform:Find("Background"):GetComponent(typeof(JButton))
    btnCloseInfo.OnClick = (function()
        self:OnCloseInfoClick()
    end)

    local btnConfirm = self.transform:Find("Cut4Cam90/AssemblyRatio2/BtnConfirm"):GetComponent(typeof(JButton))
    btnConfirm.OnClick = (function()
        self:OnBtnConfirmClick()
    end)

    local btnCancel = self.transform:Find("Cut4Cam90/AssemblyRatio2/CloseBtn"):GetComponent(typeof(JButton))
    btnCancel.OnClick = (function()
        self:OnBtnCancelClick()
    end)

    self.Title = self.transform:Find("Cut4Cam90/AssemblyRatio2/Title"):GetComponent(typeof(Text))

    self.SelectNum = self.transform:Find("Cut4Cam90/AssemblyRatio2/Num"):GetComponent(typeof(Text))

    self.go_Pool = self.transform:Find("Cut4Cam90/AssemblyRatio2/Grid").gameObject
    --对象池
    local _go = self.transform:Find("Cut4Cam90/AssemblyRatio2/ChestSelectItem").gameObject
    local _parent = self.transform:Find("Cut4Cam90/AssemblyRatio2/Grid/Mask/List")
    self.selectItemPool =
        ObjectPool:New(
        3,
        "View/Item/ChestSelectItem",
        _go,
        _parent,
        --生成回调
        function(ritem)
            ritem:InitItem(
                --点击回调
                function(ritem)
                    self:OnItemClick(ritem)
                end
            )
        end,
        --回收回调
        function(ritem)
            ritem:SetUse(false)
        end
    )

    self.go_Pool2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/Grid2").gameObject
    --对象池(4个)
    local _go2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/ChestSelectItem2").gameObject
    local _parent2 = self.transform:Find("Cut4Cam90/AssemblyRatio2/Grid2")
    self.selectItemPool2 =
        ObjectPool:New(
        3,
        "View/Item/ChestSelectItem",
        _go2,
        _parent2,
        --生成回调
        function(ritem)
            ritem:InitItem(
                --点击回调
                function(ritem)
                    self:OnItemClick(ritem)
                end
            )
        end,
        --回收回调
        function(ritem)
            ritem:SetUse(false)
        end
    )
end

function ChestSelectWidget:OnItemClick(ritem)
    if ritem.IsSelected then
        --取消选中
        for i = #self.selectedIds, 1, -1 do
            if self.selectedIds[i] == ritem.FixAwardID then
                self.selectedNum = self.selectedNum - 1
                table.remove(self.selectedIds, i)
                ritem:MakeSelected(false)
                self:RefeshSelected()
            end
        end
    else
        --选中
        if self.selectedNum < self.canSelectedMaxNum then
            self.selectedNum = self.selectedNum + 1
            table.insert(self.selectedIds, ritem.FixAwardID)
            ritem:MakeSelected(true)
            self:RefeshSelected()
        else
            Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_87"))
        end
    end
end

function ChestSelectWidget:RefeshItems(id)
    --刷新
    self.selectItemPool:RecycleAll()
    self.selectItemPool2:RecycleAll()

    local baseID = BusinessManager.Instance:GetItemByItemEntityId(self.EntityID).BaseID

    local cfg = ConfigDataAccessor.Boxes[baseID]

    local canSelectedIds = {}

    for k, v in pairs(ConfigDataAccessor.FixAward) do
        if v.AwardID == cfg.FixAward then
            table.insert(canSelectedIds, v.ID)
        end
    end

    self.canSelectedMaxNum = cfg.FixChoiceCount

    self.selectedNum = 0
    self.selectedIds = {}

    if #canSelectedIds > 4 then
        self.go_Pool:SetActive(true)
        self.go_Pool2:SetActive(false)
        for i = 1, #canSelectedIds do
            local item = self.selectItemPool:Spawn()
            item:SetItem(canSelectedIds[i])
        end
    else
        self.go_Pool:SetActive(false)
        self.go_Pool2:SetActive(true)
        for i = 1, #canSelectedIds do
            local item = self.selectItemPool2:Spawn()
            item:SetItem(canSelectedIds[i])
        end
    end

    self:RefeshSelected()
end

function ChestSelectWidget:RefeshSelected()
    --self.SelectNum.text = "请从以下奖励中选择<color=#C06666>" .. self.canSelectedMaxNum .. "</color>个(<color=#C06666>" .. self.selectedNum .. "</color>/" .. self.canSelectedMaxNum .. ")"
    self.SelectNum.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SelectNum"), {"<color=#C06666>" .. self.canSelectedMaxNum .. "</color>", "(<color=#C06666>" .. self.selectedNum .. "</color>/" .. self.canSelectedMaxNum .. ")"});
end

function ChestSelectWidget:OnBtnConfirmClick()
    if self.selectedNum < self.canSelectedMaxNum then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_DonateSumbitWidget_Message"))
        return
    end

    --发送选择请求

    local openBoxReq = CS.ProtoMsg.OpenBoxReq()
    openBoxReq.BoxEid = self.EntityID
    for i = 1, #self.selectedIds do
        openBoxReq.CIndex:Add(self.selectedIds[i])
    end
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, openBoxReq, false)

    --PrintTable(self.selectedIds)

    --全部关闭
    UIManager.Instance:CloseWidget("Common/ChestSelectWidget", nil, true)
end

function ChestSelectWidget:OnBtnCancelClick()
    --关闭
    UIManager.Instance:CloseWidget("Common/ChestSelectWidget", nil, true)
end

function ChestSelectWidget:Start()
    print("ChestSelectWidget:Start")
end

function ChestSelectWidget:OnEnable()
end

function ChestSelectWidget:OnDisable()
end

function ChestSelectWidget:OnDestroy()
end

function ChestSelectWidget:OnOpen(arg)
    self.IsOpen = true

    local id = tonumber(arg)
    self.EntityID = id
    self:RefeshItems(id)
end

function ChestSelectWidget:OnClose(arg)
    self:CloseWidget()
end

function ChestSelectWidget:OnCloseInfoClick()
    --关闭道具Tips
    if self.module.IsOpenTips then
        self.module.IsOpenTips = false
        self.module:CloseItemTips()

        self:CleanAllSelected()
    else
        UIManager.Instance:CloseWidget("Common/ChestSelectWidget", nil, true)
    end
end

function ChestSelectWidget:CloseWidget()
    self.IsOpen = false
end

return ChestSelectWidget

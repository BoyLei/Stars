function PlayerInfoWidget:OnOpen(arg)
    self.M_EntityBase = arg;
    if self.M_EntityBase == nil or self.M_EntityBase.Data == nil then
        LogError("PlayerInfoWidget:OnOpen() arg is nil error!!!");
        return;
    end
    MonoHelper.RemoveUpdateListener(self.UpdateFun);

    local jobId = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Uint,
        CS.StarProjectDef.AOIAttrDefine.Job);
    if jobId ~= 0 then
        -- 获取职业量谱
        local jobshi1 = math.floor(jobId / 10)
        local jobSpectralPrefabName = "JobSpectralWidget" .. jobshi1 .. "0";
        -- local jobSpectralPrefabName = "JobSpectralWidget30";
        self.JobSpectralWidgetPath = "StarWorld/Prefab/" .. jobSpectralPrefabName;
        local JobSpectralWidget = UIManager.Instance:OpenWidget(self.JobSpectralWidgetPath, false, self.M_EntityBase,
            self.M_JobSpectralRoot, CS.StarProjectDef.MainPageCommond.HideNone, true, true);
        if JobSpectralWidget ~= nil then
            self.JobSpectralWidget = JobSpectralWidget:GetLuaPanel().ScriptTable;
        end
    end

    -- 职业头像
    local jobCfg = ConfigDataAccessor.Job[jobId]
    if jobCfg ~= nil then
        local avatarDataCell = ConfigDataAccessor.Avatar[jobCfg.AvatarID]
        if avatarDataCell ~= nil then
            local headCfg = ConfigDataAccessor.ModelHead[avatarDataCell.HeadID]
            if headCfg ~= nil then
                ResourceHelper:LoadIconByAtlasAsync(self.JobHeadImg, headCfg.HeadAtlasName, headCfg.MinHead, false)
            end
        end
    end
    -- 职业标签图标
    local jobshi = math.floor(jobId / 10)
    local jobHeadImgBgPath = "Hud_RoleFrame" .. jobshi
    ResourceHelper:LoadIconByAtlasAsync(self.JobHeadImgBg, ResourceHelper.AtlasPathHud, jobHeadImgBgPath, false)
    -- 职业类型icon设置
    local jobTypeBgName = "Icon_ArmBg" .. jobshi
    ResourceHelper:LoadIconByAtlasAsync(self.JobTypeBg, ResourceHelper.AtlasPathRole, jobTypeBgName, false)
    local jobTypeName = "Icon_Arm" .. jobshi
    ResourceHelper:LoadIconByAtlasAsync(self.JobTypeIcon, ResourceHelper.AtlasPathRole, jobTypeName, false)

function PlayerInfoWidget:SetHp(value)
    -- Log("主角血条 当前的=" .. value);
    if self.M_EntityBase == nil or self.M_EntityBase.Data == nil then
        return
    end
    self.m_HPCur = value;
    -- if self.m_HPCur > self.m_HPSum then
    -- self.m_HPSum = self.m_HPCur;
    -- end
    self:SetMixHpShield();
    self.M_HPText.text = tostring(self.m_HPCur) .. "/" .. tostring(self.m_HPSum);
end

function PlayerInfoWidget:UpdateTruthHp(value)
    -- Log("主角血条 最大值的=" .. value);
    if self.M_EntityBase == nil or self.M_EntityBase.Data == nil then
        return
    end
    self.m_HPSum = value;
    -- if self.m_HPCur > self.m_HPSum then
    -- self.m_HPSum = self.m_HPCur;
    -- end
    self:SetMixHpShield();
    self.M_HPText.text = tostring(self.m_HPCur) .. "/" .. tostring(self.m_HPSum);
end

function PlayerInfoWidget:SetMixHpShield()
    local isHaveShield = self.ShieldValSum > 0;
    if isHaveShield then
        local mixHp = self.ShieldValSum + self.m_HPCur;
        local isExceedMaxHp = mixHp > self.m_HPSum;
        if isExceedMaxHp then
            -- 超过实体的最大血量
            local ratio = 0;
            if self.m_HPSum > 0 then
                ratio = self.m_HPCur / mixHp;
                if ratio > 1 then
                    ratio = 1;
                end
            end
            self.M_HPBar.fillAmount = ratio;

            local ratio2 = 0;
            if self.ShieldValSum > 0 then
                ratio2 = self.ShieldValSum / mixHp;
                if ratio2 > 1 then
                    ratio2 = 1;
                end
            end
            self.M_ShieldValBarRight.fillAmount = ratio2;
            self.M_ShieldValBarLeft.fillAmount = 0;
        else
            local ratio = 0;
            if self.m_HPSum > 0 then
                ratio = self.m_HPCur / self.m_HPSum;
                if ratio > 1 then
                    ratio = 1;
                end
            end
            self.M_HPBar.fillAmount = ratio;

            local ratio2 = 0;
            if self.m_HPSum > 0 then
                ratio2 = self.ShieldValSum / self.m_HPSum;
                if ratio2 > 1 then
                    ratio2 = 1;
                end
            end
            self.M_ShieldValBarLeft.fillAmount = ratio + ratio2;
            self.M_ShieldValBarRight.fillAmount = 0;
        end
    else
        local ratio = 0;
        if self.m_HPSum > 0 then
            ratio = self.m_HPCur / self.m_HPSum;
            if ratio > 1 then
                ratio = 1;
            end
        end
        self.M_HPBar.fillAmount = ratio;
        self.M_ShieldValBarLeft.fillAmount = 0;
        self.M_ShieldValBarRight.fillAmount = 0;
    end
end

-- 蓄力进度条
function PlayerInfoWidget:UpdateData()
    if self.m_isShowEnergy and self.m_skillEntity ~= nil then
        self.curEnergyTime = self.M_EntityBase:GetCurEnergyTime(true);
        self.energyedCount = self.M_EntityBase:GetCurEnergyCount(true);

        local time = math.ceil(self.curEnergyTime) / 1000;
        local times = string.format("%.1f", time);
        self.M_SkillTime.text = times .. "s";
        self:UpdatErnergy();
    end
end

    self:SetEenergy(skillEntity);
end

function PlayerInfoWidget:SetEenergy(skillEntity)
    self.m_skillEntity = skillEntity;
    self.energyTime = self.m_skillEntity.EnergyItemTime;
    self.energyMaxNum = self.m_skillEntity.EnergyMaxCount;
    self.energyedCount = self.M_EntityBase:GetCurEnergyCount(true);
    self.curEnergyTime = self.M_EntityBase:GetCurEnergyTime(true);

    local skillName = "";
    if self.m_skillEntity.skillContainer.CurSkillInfo ~= nil then
        skillName = self.m_skillEntity.skillContainer.CurSkillInfo:GetSkillName();
    end

    self.M_SkillName.text = skillName;

    ClearAllChild(self.M_LimitRoot);
    if self.energyMaxNum > 1 then
        for i = 0, self.energyMaxNum - 1, 1 do
            local _go = newObject(self.M_SkillErnergyLimit)
            _go.transform:SetParent(self.M_LimitRoot)
            _go.transform.localPosition = Vector3.zero
            _go.transform.localScale = Vector3.one
            if i == 0 then
                local canvasGroup = _go.transform:GetComponent(typeof(CanvasGroup));
                canvasGroup.alpha = 0;
            end
        end
    end

    self:SetEnergyValShow(true)
end

end

        if ratio > 1 then

    end
    self.M_BarSkill.fillAmount = ratio;
    --self.M_BarSkill:SetWidth(self.M_BarSkillWidth * ratio);

function PlayerInfoWidget:OnActionOnBuffUpdate(skillBuff)
    -- 获取BUFF护盾总数
    local isChanage = self:CheckBuffShieldVal(skillBuff, true);
    if isChanage then
        self:SetShieldValSum();
    end
    -- Log("霸服 PlayerInfoWidget:OnActionOnBuffUpdate 委托通知更新111111 runtimeid=",skillBuff.RuntimeID,"层数=",skillBuff.StackCount);
    local buffItem = self:GetBuffItemBySkillBuff(skillBuff);
    if buffItem ~= nil then
        -- 过滤
        if buffItem.BuffID == skillBuff.BuffID and buffItem.LiveTime == skillBuff.LiveTime and buffItem.StackCount ==
            skillBuff.StackCount then
            return;
        end

        -- Log("霸服 PlayerInfoWidget:OnActionOnBuffUpdate 委托通知更新 runtimeid=",skillBuff.RuntimeID,"层数=",skillBuff.StackCount);
        self:RemoveSkillBuffListByIndex(skillBuff);
        self:AddSkillBuffList(skillBuff);
        buffItem:SetBuffData(skillBuff, true);
        self:RefreshBuffItemIndex();

        self:RefreshBuffList();
    end
end

function PlayerInfoWidget:IsContainsBuff(skillBuff)
    if self.SkillBuffList == nil or self.SkillBuffList == {} or #self.SkillBuffList <= 0 then
        return false;
    end
    for i = 1, #self.SkillBuffList do
        local item = self.SkillBuffList[i];
        if item.RuntimeID == skillBuff.RuntimeID then
            return true;
        end
    end
    return false;
end

function PlayerInfoWidget:AddBuffItem(skillBuff)
    self:AddSkillBuffList(skillBuff);
    local count = #self.SkillBuffList;
    local index = 1;
    -- 判断是否有空闲的item
    local buffItem = self:GetIdleBuffItem();
    if buffItem == nil then
        self:CreateBuffItem(skillBuff, index);
    else
        self:UpdateBuffItem(skillBuff, buffItem, index);
    end
    if count > 5 then
        self:RefreshBuffItemIndex();
    end
end

-- BUFFitem 删除回调
function PlayerInfoWidget:BuffItemReleaseCb(skillBuffItem)
    -- list中删除
    self:RemoveSkillBuffListByIndex(skillBuffItem.BuffEntity);
    -- 刷新在场的item index
    self:RefreshBuffItemIndex();

    self:RefreshBuffList();
end

-- 刷新在场的index
function PlayerInfoWidget:RefreshBuffItemIndex()
    for i = 1, #self.BuffItems do
        local buffItem = self.BuffItems[i];
        if buffItem ~= nil and buffItem.BuffEntity ~= nil then
            local index = self:GetBuffIndexBySkillBuff(buffItem.BuffEntity);
            if index ~= -1 then
                buffItem:SetIndex(index);
            else
                buffItem:SetUse(false);
            end
        end
    end
end

function PlayerInfoWidget:SetBuffSpTimeShowDic(key, isOpen)
    local add = -1;
    if isOpen then
        add = 1;
    end
    for k, v in pairs(self.buffSpTimeShowDic) do
        if k == key then
            v = v + add;
            return;
        end
    end
    self.buffSpTimeShowDic[key] = add;
end

function PlayerInfoWidget:GetBuffSpTimeShowDicVal(key)
    for k, v in pairs(self.buffSpTimeShowDic) do
        if k == key then

        end
    end
    return -1;
end

function PlayerInfoWidget:SetOverheadClock(skillBuff, isOpen)
    if isOpen then
        if skillBuff ~= nil then

                self.JobSpectralWidget:SetCountdown(skillBuff.MaxTime / 1000, skillBuff:GetEndime() / 1000);
                self.JobSpectralWidget:SetOpenCountdown(true);
            end
        end
    else

PlayerInfoWidget.buffShieldDic = {}
PlayerInfoWidget.ShieldValSum = 0

function PlayerInfoWidget:CheckBuffShieldVal(skillBuff, isOpen)
    local isChanage = false;
    if skillBuff == nil or skillBuff.BuffInfo == nil or skillBuff.BuffInfo.Cfg == nil then
        -- LogWarning("霸服 PlayerInfoWidget:CheckBuffShieldVal skillBuff=nil err");
        return isChanage;
    end
    local shieldVal = 0;
    if isOpen then
        shieldVal = skillBuff.ShieldVal
    end

    local lastShield = 0;
    local isHave = false;
    for key, value in pairs(self.buffShieldDic) do
        if key == skillBuff.RuntimeID then
            lastShield = value;
            isHave = true;
            break
        end
    end
    if isHave then
        if shieldVal > 0 then
            if shieldVal ~= lastShield then
                self.buffShieldDic[skillBuff.RuntimeID] = shieldVal;
                isChanage = true;
            end
        else
            -- table.remove(self.buffShieldDic, skillBuff.RuntimeID);
            self.buffShieldDic[skillBuff.RuntimeID] = 0;
            isChanage = true;
        end
    elseif shieldVal > 0 then
        self.buffShieldDic[skillBuff.RuntimeID] = shieldVal;
        isChanage = true;
    end

    return isChanage;
end

function PlayerInfoWidget:SetShieldValSum()
    local sum = 0;
    for key, value in pairs(self.buffShieldDic) do
        sum = sum + value;
    end
    self.ShieldValSum = sum;
    -- 通知血条比例修改
    self:SetMixHpShield();
end

function PlayerInfoWidget:ReleaseInfo()
    if self.M_EntityBase ~= nil and self.M_EntityBase.Data ~= nil then
        -- 解绑属性同步回调
        self:DelAttribute();

        -- 解绑技能回调
        self.M_EntityBase:DelEnergyStartLuaAction(self.AddEnergyStartLuaAction);

        self.M_EntityBase:DelEnergyEndLuaAction(self.AddEnergyEndLuaAction);

        self.M_EntityBase:DelEndSkillStageLuaAction(self.AddEndSkillStageLuaAction);

        -- 解绑BUFF回调------  START
        self.M_EntityBase:DelBuffCreateLuaAction(self.AddBuffCreateLuaAction);

        self.M_EntityBase:DelBuffEndLuaAction(self.AddBuffEndLuaAction);

        self.M_EntityBase:DelBuffUpdateLuaAction(self.AddBuffUpdateLuaAction);
    end

    if self.JobSpectralWidget ~= nil then
        UIManager.Instance:CloseWidget(self.JobSpectralWidgetPath, self.M_JobSpectralRoot, false);
    end
    self.JobSpectralWidget = nil;

    self:ResetBuffItem();
    self:ReleaseBuffSpTimeShowDic();
    self.m_skillEntity = nil;
    self.SkillBuffList = {};
    self.SkillDeBuffList = {};

    if self.m_BuffList ~= nil and self.m_BuffList.M_IsOpen then
        local buffList = {};
        self.m_BuffList:UpdateData(buffList);
    end

    MonoHelper.RemoveUpdateListener(self.UpdateFun);
	
	-- 冒险者档案回调
	--NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.PropPanelRetID, self.PropPanelRetCmd, nil)
end

return PlayerInfoWidget

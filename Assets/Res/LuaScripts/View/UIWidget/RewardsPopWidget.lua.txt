RewardsPopWidget = class("RewardsPopWidget", UIPanel)
RewardsPopWidget.IsOpen = false

local ObjectPool = require("Common/Helper/ObjectPool")

function RewardsPopWidget:Awake()
end

function RewardsPopWidget:InitItems()
    self.transform = self.prefab.transform
    self.IsOpen = false

    --动画相关
    self.anim = self.transform:GetComponent(typeof(Animator))

    self.go_bg1 = self.transform:Find("Bg").gameObject
    self.go_grid1 = self.transform:Find("Grid").gameObject
    self.go_grid2 = self.transform:Find("Grid2").gameObject

    self.rect_grid2 = self.transform:Find("Grid2/Mask/List"):GetComponent(typeof(RectTransform))

    --奖励对象池
    local _go = self.transform:Find("ItemGrid").gameObject
    local _parent = self.transform:Find("Grid")
    self.rewardItemPool =
        ObjectPool:New(
        7,
        "View/Item/ItemGrid",
        _go,
        _parent,
        --生成回调
        function(ritem)
            ritem:InitItem(
                false,
                --不使用底板
                --点击回调
                function(ritem)
                    self:OnItemClick(ritem)
                end
            )
        end,
        --回收回调
        function(ritem)
            ritem:SetUse(false)
        end
    )

    local _go2 = self.transform:Find("ItemGrid_").gameObject
    local _parent2 = self.transform:Find("Grid2/Mask/List")
    self.rewardItemPool2 =
        ObjectPool:New(
        14,
        "View/Item/ItemGrid",
        _go2,
        _parent2,
        --生成回调
        function(ritem)
            ritem:InitItem(
                false,
                --不使用底板
                --点击回调
                function(ritem)
                    self:OnItemClick(ritem)
                end
            )
        end,
        --回收回调
        function(ritem)
            ritem:SetUse(false)
        end
    )

    self.go_grid1:SetActive(false)
    self.go_grid2:SetActive(false)

    local btnCloseInfo = self.transform:Find("Background"):GetComponent(typeof(Button))
    btnCloseInfo.onClick:AddListener(
        function()
            self:OnCloseClick()
        end
    )
end

function RewardsPopWidget:RefeshRewards(items)
    SoundManager.Instance:PlayEventName("UI_TanChuang_JiangLi_01", self.gameObject, self.gameObject)

    self.anim:Play("RewardsPopWidgetAnim_Open")

    self.module.IsOpenTips = false
    self.module:CloseItemTips()

    self.transform:SetAsLastSibling()

    local rewardsList = items
    if #rewardsList > 0 then
        local msgList = {}
        for _, v in ipairs(rewardsList) do
            local chatMsgNotice = CS.ProtoMsg.ChatMsgNotice()
            chatMsgNotice.EID = 0
            chatMsgNotice.PID = 0
            local strLanVal = LanguageHelp:GetLanguageByKey("Local_Str_Tips_GetReward")
			local equipStrSp = ExtraData.GetEquipStrByID(v.BaseID);
            local strContentVal = equipStrSp .. GetQualityColor(ConfigDataAccessor.Item[v.BaseID].Quality) .. ConfigDataAccessor.Item[v.BaseID].Name .. "</color>x" .. v.Num
            chatMsgNotice.Data = StringFormat(strLanVal, {strContentVal})
            chatMsgNotice.Channel = CS.ProtoMsg.ChatChannel.ChannelSystem
            chatMsgNotice.ChatTime = TimeUtils.ServerNowStampMilli
            chatMsgNotice.ChatLevel = 1
            chatMsgNotice.JobID = GameManager.Instance:GetPlayerJob()
            table.insert(msgList, chatMsgNotice)
        end
        LuaModuleDispatcher:SendMessage(ModuleDef.ChatModule, "OnAnalogReceptionChatMsg", {msgList})
    end

    if #items > 7 then
        --背包格子
        self.rewardItemPool2:RecycleAll()
        for i = 1, #items do
            DOVirtual.DelayedCall(
                0.5 + (i - 1) * 0.1,
                function()
                    local item = self.rewardItemPool2:Spawn()
                    item.gameObject.name = "ItemGrid" .. i
                    item:SetItem(items[i])
                    if item.Quality ~= nil then
                        local isHaveEffect = (item.Quality >= 5)
                        local goEff = item.transform:Find("FX_UI_Com_Icon03").gameObject
                        if goEff ~= nil then
                            goEff:SetActive(isHaveEffect)
                        end
                    end
                end
            )
        end

        self.rect_grid2.anchoredPosition = Vector2(self.rect_grid2.anchoredPosition.x, 0)

        self.go_grid1:SetActive(false)
        self.go_grid2:SetActive(true)
    else
        --背包格子
        self.rewardItemPool:RecycleAll()
        for i = 1, #items do
            DOVirtual.DelayedCall(
                0.5 + (i - 1) * 0.1,
                function()
                    local item = self.rewardItemPool:Spawn()
                    item.gameObject.name = "ItemGrid" .. i
                    item:SetItem(items[i])
                    if item.Quality ~= nil then
                        local isHaveEffect = (item.Quality >= 5)
                        local goEff = item.transform:Find("FX_UI_Com_Icon03").gameObject
                        if goEff ~= nil then
                            goEff:SetActive(isHaveEffect)
                        end
                    end
                end
            )
        end
        self.go_grid1:SetActive(true)
        self.go_grid2:SetActive(false)
    end
end

function RewardsPopWidget:OnItemClick(ritem)
    print("RewardItem:OnItemClick")
    self.module.IsOpenTips = true
    self.module:OpenItemTips(ritem.BaseID, ritem.EntityID, ritem.transform)
end

function RewardsPopWidget:BindModule(m)
    print(".............")
    self.module = m
end

function RewardsPopWidget:Start()
end

function RewardsPopWidget:OnEnable()
end

function RewardsPopWidget:OnDisable()
end

function RewardsPopWidget:OnDestroy()
end

function RewardsPopWidget:OnOpen(arg)
    self.IsOpen = true
end

function RewardsPopWidget:OnClose(arg)
    self.IsOpen = false
    self.module.CanPop = true
end

function RewardsPopWidget:OnCloseClick()
    self.anim:Play("RewardsPopWidgetAnim_Close")

    DOVirtual.DelayedCall(
        0.1,
        function()
            --关闭道具Tips
            if self.module.IsOpenTips then
                self.module.IsOpenTips = false
                self.module:CloseItemTips()
            else
                self:CloseWidget()
            end
        end
    )
end

function RewardsPopWidget:CloseWidget()
    print("RewardsPopWidget:CloseWidget")
    UIManager.Instance:CloseWidget("RewardsPop/RewardsPopWidget", nil, true)
    self.module:OnCloseRewardsPop()
end

return RewardsPopWidget

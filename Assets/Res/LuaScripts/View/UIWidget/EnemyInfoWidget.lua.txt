function EnemyInfoWidget:SetEnemyInfo(arg)
	if arg == nil then
		self:CloseWidget();
		return;
	end

	if self.m_EntityBase ~= nil and arg.EntityId == self.m_EntityBase.EntityId then
		return;
	end
	self.m_EntityBase = arg;
	
	if self.m_EntityBase == nil or self.m_EntityBase.Data == nil or self.m_EntityBase.Data.Attrs == nil then
		return;
	end
	-- 血条配置信息
	self:InitEnemyHpConfData();
	-- 敌方头像
	self:InitEnemyHeadImg();
	-- 名字等级
	self:InitLevelAndName();
	-- 当前目标
	self:InitTargetId();
	-- 获取BUFF信息
	self:ReleaseBuffItems();
	self:InitGetBuffs();
	-- 注册委托
	self:InitRegisterAction();

	self.m_NewPlayerPreFlag = false;
	self.m_NewPlayerPreFlagList = {};
	local mapSubType = GameManager.Instance:GetMapSubType()
	if mapSubType ~= nil and mapSubType ~= "" then
		self.m_NewPlayerPreFlag = mapSubType == "PlotSecond";
	end

	if not self.m_transform.gameObject.activeSelf then
		self.m_transform.gameObject:SetActive(true);
	end
end

function EnemyInfoWidget:OnClose(arg)
	if self.m_EntityBase ~= nil then
		self:ReleaseInfo();
	end
end

function EnemyInfoWidget:CloseWidget()
	if self.m_EntityBase ~= nil then
		self:ReleaseInfo();
	end
	self.Close();
	self.m_transform.gameObject:SetActive(false);
end

-- 血条配置信息
function EnemyInfoWidget:InitEnemyHpConfData()
	self.m_HPSum = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Long,
		CS.StarProjectDef.AOIAttrDefine.TruthHp);
	self.m_HPCur = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Long,
		CS.StarProjectDef.AOIAttrDefine.curHp);
	self.m_HPLastCur = self.m_HPCur;

	self.m_HpRowSum = 1;
	if self.m_EntityBase.EntityType == CS.StarProjectDef.E_EntityType.Monster then
		local indexValue = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Uint,
			CS.StarProjectDef.AOIAttrDefine.Index);
		local monsterAttrDataCell = ConfigDataAccessor.Monster[tonumber(indexValue)];
		if monsterAttrDataCell ~= nil then
			--self.m_HpRowSum = monsterAttrDataCell.HealthBarCount;
			MonsterType = monsterAttrDataCell.MonType;
		end
		-- 1是普通怪物 2是精英怪物 3是boss
		if MonsterType == 1 then
			self.m_HpRowSum = 1;
		elseif MonsterType == 2 then
			self.m_HpRowSum = 3;
		elseif MonsterType == 3 then
			self.m_HpRowSum = 5;
		end
	end

	self:RefreshHpData();
end

-- 当前目标
function EnemyInfoWidget:InitTargetId()
	local targetEntityId = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Ulong,
		CS.StarProjectDef.AOIAttrDefine.TargetBattleEntityID);
	local targetId = tonumber(targetEntityId);
	local targetName = "";
	if targetId > 0 then
		targetName = GameManager.Instance:GetEntityNameById(targetId);
	end
	self.m_TargetName.text = targetName;
	self.m_TargetName.gameObject:SetActive(targetName ~= "");
end

-- 名字等级
function EnemyInfoWidget:InitLevelAndName()
	local level = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,
		CS.StarProjectDef.AOIAttrDefine.EntityLevel);
	if level < 1 then
		level = 1;
	end
	local showLevel = self.m_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Int,
		CS.StarProjectDef.AOIAttrDefine.ShowLevel);
	if showLevel < 1 then
		showLevel = 1;
	end
	local show = level;
	if showLevel > 1 then
		show = showLevel
	end
	
	local Name=self.m_EntityBase.M_Name;
	if Name==nil or Name=='' or Name=="" then
		local cfg= ConfigDataAccessor.Monster[self.m_EntityBase.ConfigIndex]
		if cfg~=nil then
			Name=cfg.Name
		end
	end
	
	--levelAndEntityName = "<color=#FFFFEB><size=20>" .. show .. "级</size></color> " .. self.m_EntityBase.M_Name;
	self.m_Name.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_levelAndEntityName"), {show, Name});
end

-- 刷新血量信息
function EnemyInfoWidget:RefreshHpData()
	-- 计算血量平均数，，， 一管血条的数值
	self.m_HpRowAverage = self.m_HPSum / self.m_HpRowSum;
	-- 当前血条的数量
	self.m_CurHpRowCount = math.ceil(self.m_HPCur / self.m_HpRowAverage);
	self.m_WhiteHpRowCount = self.m_CurHpRowCount;
	self.m_HpCount.text = "x" .. self.m_CurHpRowCount;
	-- 设置血条的图片信息
	self:SetHpIcon();
	-- 一关血 目前的比例数值
	self.m_OnceHpResidueNum = self.m_HPCur % self.m_HpRowAverage; -- 剩余一管血的数值
	if (self.m_HPCur % self.m_HpRowAverage) == 0 then
		self.m_OnceHpResidueNum = self.m_HpRowAverage;
	end
	if self.m_OnceHpResidueNum <= 0.0001 then
		self.m_OnceHpResidueNum = self.m_HpRowAverage;
	end
	local ratio = self.m_OnceHpResidueNum / self.m_HpRowAverage;
	if ratio > 1 then
		ratio = 1;
	end

	-- self.m_BarHpRect:SetWidth(self.m_HpWidthCfg * ratio);
	-- self.m_BarWhiteHpRect:SetWidth(self.m_HpWidthCfg * ratio);

	self.m_BarHp.fillAmount = ratio;
	self.m_BarWhiteHp.fillAmount = ratio;
	self:SetBarGlowX(ratio);

	--self.m_HpSumText.text = tostring(self.m_HPCur).."/"..tostring(self.m_HPSum);
	local hpRatio = self.m_HPCur / self.m_HPSum;
	if hpRatio > 1 then
		hpRatio = 1;
	end
	local percent = roundToDecimal(hpRatio * 100, 2);
	self.m_HpSumText.text = percent .. "%";
end

-- 设置血条图标
function EnemyInfoWidget:SetHpIcon()
	-- 计算当前血条颜色的下标
	local m_CurHpIdx = self.m_CurHpRowCount;
	if m_CurHpIdx > 5 then
		local remainder = math.ceil(m_CurHpIdx % 5);
		if remainder == 0 then
			remainder = 5;
		end
		m_CurHpIdx = remainder;
	end

	-- 当前血条的图片
	--local res,color = CS.UnityEngine.ColorUtility.TryParseHtmlString(self.HpColorList[m_CurHpIdx]);
	--self.m_BarHp.color = color;

	self:GetSprite(self.m_HpIconPathList[m_CurHpIdx], function(sp)
		self.m_BarHp.sprite = sp
	end);

	local nextHpIconIdx = m_CurHpIdx - 1;
	if nextHpIconIdx >= 1 then
		--local res,color = CS.UnityEngine.ColorUtility.TryParseHtmlString(self.HpColorList[nextHpIconIdx]);
		--self.m_BarHp2.color = color;
		self:GetSprite(self.m_HpIconPathList[nextHpIconIdx], function(sp)
			self.m_BarHp2.sprite = sp
		end);
		self.m_BarHp2.gameObject:SetActive(true);
	else
		self.m_BarHp2.gameObject:SetActive(false);
	end
end

function EnemyInfoWidget:SetHp(value)
	if self.m_HPLastCur == value then
		return;
	end
	self.m_HPCur = value;
	if self.m_EntityBase == nil or self.m_EntityBase.Data == nil then
		return;
	end
	-- if self.m_HPCur > self.m_HPSum then
	-- 	self.m_HPSum = self.m_HPCur;
	-- end

	local delHp = self.m_HPLastCur - self.m_HPCur;
	if delHp < 0 then
		-- 说明怪物脱战，回满血了
		self:RefreshHpData();
		self.m_HPLastCur = self.m_HPCur;
	else
		local residueNum = self.m_OnceHpResidueNum - delHp;
		if residueNum > 0 then
			self.m_OnceHpResidueNum = residueNum;
		else
			self.m_OnceHpResidueNum = 0;
		end
		if self.m_OnceHpResidueNum == 0 then
			if self.m_HPCur > 0 then
				self.m_OnceHpResidueNum = self.m_HPCur % self.m_HpRowAverage; -- 剩余一管血的数值
			end
			-- 当前血条的数量
			self.m_CurHpRowCount = math.ceil(self.m_HPCur / self.m_HpRowAverage);
		else
			local ratio = self.m_OnceHpResidueNum / self.m_HpRowAverage;
			if ratio > 1 then
				ratio = 1;
			end
			--self.m_BarHpRect:SetWidth(self.m_HpWidthCfg * ratio);

				isFind = true;

				stepID = "11" .. "_80";
				isFind = true;
			end
			if isFind then
				local isHave = self:GetNewPlayerPreFlagIsContainsValue(stepID)

					GameManager.Instance:EventPreNewPlayerEvent(stepID);

function EnemyInfoWidget:UpdateTruthHp(value)
	if self.m_EntityBase == nil or self.m_EntityBase.Data == nil then
		return
	end
	self.m_HPSum = value;
	-- if self.m_HPCur > self.m_HPSum then
	-- 	self.m_HPSum = self.m_HPCur;
	-- end

	self.m_OnceHpResidueNum = self.m_HPCur % self.m_HpRowAverage; -- 剩余一管血的数值
	if (self.m_HPCur % self.m_HpRowAverage) == 0 then
		self.m_OnceHpResidueNum = self.m_HpRowAverage;
	end
	if self.m_OnceHpResidueNum <= 0 then
		self.m_OnceHpResidueNum = self.m_HpRowAverage;
	end
	--self.m_HpSumText.text = tostring(self.m_HPCur).."/"..tostring(self.m_HPSum);
	local hpRatio = self.m_HPCur / self.m_HPSum;
	if hpRatio > 1 then
		hpRatio = 1;
	end
	local percent = roundToDecimal(hpRatio * 100, 2);
	self.m_HpSumText.text = percent .. "%";
end

function EnemyInfoWidget:SetMixHpShield()
	local isHaveShield = self.ShieldValSum > 0;
	if isHaveShield then
		local mixHp = self.ShieldValSum + self.m_HPCur;
		local isExceedMaxHp = mixHp > self.m_HPSum;
		if isExceedMaxHp then
			-- 超过实体的最大血量
			local ratio = 0;
			if self.m_HPSum > 0 then
				ratio = self.m_HPCur / mixHp;
				if ratio > 1 then
					ratio = 1;
				end
			end
			self.M_HPBar.fillAmount = ratio;

			local ratio2 = 0;
			if self.ShieldValSum > 0 then
				ratio2 = self.ShieldValSum / mixHp;
				if ratio2 > 1 then
					ratio2 = 1;
				end
			end
			self.M_ShieldValBarRight.fillAmount = ratio2;
			self.M_ShieldValBarLeft.fillAmount = 0;
		else
			local ratio = 0;
			if self.m_HPSum > 0 then
				ratio = self.m_HPCur / self.m_HPSum;
				if ratio > 1 then
					ratio = 1;
				end
			end
			self.M_HPBar.fillAmount = ratio;

			local ratio2 = 0;
			if self.m_HPSum > 0 then
				ratio2 = self.ShieldValSum / self.m_HPSum;
				if ratio2 > 1 then
					ratio2 = 1;
				end
			end
			self.M_ShieldValBarLeft.fillAmount = ratio + ratio2;
			self.M_ShieldValBarRight.fillAmount = 0;
		end
	else
		local ratio = 0;
		if self.m_HPSum > 0 then
			ratio = self.m_HPCur / self.m_HPSum;
			if ratio > 1 then
				ratio = 1;
			end
		end
		self.M_HPBar.fillAmount = ratio;
		self.M_ShieldValBarLeft.fillAmount = 0;
		self.M_ShieldValBarRight.fillAmount = 0;
	end
end

function EnemyInfoWidget:SetBarGlowX(ratio)
	local BarGlowX = self.m_HpWidthCfg * ratio;
	if BarGlowX > self.m_HpWidthCfg / 2 then
		BarGlowX = BarGlowX - self.m_HpWidthCfg / 2;
		BarGlowX = BarGlowX - 2;
	else
		BarGlowX = self.m_HpWidthCfg / 2 - BarGlowX;
		BarGlowX = -BarGlowX - 2;
	end
	local v3 = Vector3.zero;
	v3.x = BarGlowX;
	self.m_BarGlow.transform.localPosition = v3;
end

function EnemyInfoWidget:InitGetBuffs()
	local buffsList = self.m_EntityBase:GetSkillBuffs();
	if buffsList == nil or buffsList.Count <= 0 then
		return
	end

	-- 区分增益减益BUFF
	for i = 0, buffsList.Count - 1 do
		local buff = buffsList[i];
		if buff ~= nil and buff.BuffInfo ~= nil and buff.BuffInfo.Cfg ~= nil then
			local isHave = self:GetBuffIndexById(buff.BuffInfo.Cfg.IsDebuff, buff.RuntimeID);
			local isHaveCfg = buff.BuffInfo:GetHaveBUFFUIShowState(E_BUFFUIShowPosEnum.HUDHpUp);
			if isHaveCfg then
				if isHave == -1 then
					self:AddBuffList(buff);
					local item = self:OnBuffCreate(buff.BuffInfo.Cfg.IsDebuff);
					if item ~= nil then
						local count = self:GetBuffListCount(buff.BuffInfo.Cfg.IsDebuff);
						item:SetBuffCout(count, i);
						item:SetCallBack(self.BuffItemClickFun);
						item:SetItem(buff);
					end
				end
			end
			if isHave == -1 then
				-- 获取BUFF护盾总数
				self:CheckBuffShieldVal(buff, true);
			end
		end
	end
end

function EnemyInfoWidget:CreateBuff(isDeBuff)
	local _go;
	if isDeBuff then
		if #self.DeBuffItems < 3 then
			_go = newObject(self.m_DeBuffPrefab);
			_go.transform:SetParent(self.m_DeBuffContainer);
		end
	else
		if #self.BuffItems < 3 then
			_go = newObject(self.m_BuffPrefab);
			_go.transform:SetParent(self.m_BuffContainer);
		end
	end
	if _go ~= nil then
		_go.transform.localPosition = Vector3.zero;
		_go.transform.localScale = Vector3.one;
		local item = ItemBuff:New(_go, self.OnItemBuffClickFun);
		if isDeBuff then
			table.insert(self.DeBuffItems, item);
		else
			table.insert(self.BuffItems, item);
		end
		return item;
	end
	return nil;
end

	if skillItem ~= nil then
		local isDeBuff = false;

			self:DelBuffList(skillItem.BuffEntity);
			isDeBuff = skillItem.BuffEntity.BuffInfo.Cfg.IsDebuff;
			----删除BUFFUI时，如果buff总数超过了5个
			---- 需要通知最后一个更新BUFF总数
			--if #SkillDeBuffList >= 5 or #SkillBuffList >= 5 then

			--if lastBuffItem ~= nil then
			--local count = self:GetBuffListCount(skillBuff.BuffInfo.Cfg.IsDebuff);
			--lastBuffItem:SetBuffCout(count);
			--end
			--end
		else
			self:DelBuffItemInList(skillItem.IsDeBuff, skillItem.RunTimeId);
			isDeBuff = skillItem.IsDeBuff;
		end
		-- 重新刷新其他Item的信息
		local count = self:GetBuffListCount(isDeBuff);
		if isDeBuff then
			for i = 1, #self.DeBuffItems do

				if #self.SkillDeBuffList >= i then

					buffItem:SetItem(item);
				else
					buffItem:SetUse(false);
				end
			end
		else
			for i = 1, #self.BuffItems do
				local buffItem = self.BuffItems[i];
				if #self.SkillBuffList >= i then
					local item = self.SkillBuffList[i];
					buffItem:SetBuffCout(count, i);
					buffItem:SetItem(item);
				else
					buffItem:SetUse(false);
				end
			end
		end
	end

function EnemyInfoWidget:GetLastBuffItem(isDeBuff)
	local item = nil;
	local max = -1;
	if isDeBuff then
		for i = #self.DeBuffItems, 1, -1 do
			local buffItem = self.DeBuffItems[i];
			if buffItem.IsUsed and buffItem.BuffEntity ~= nil then
				local index = buffItem.Index;
				if index > max then
					max = index;
					item = buffItem;
				end
			end
		end
	else
		for i = #self.BuffItems, 1, -1 do
			local buffItem = self.BuffItems[i];
			if buffItem.IsUsed and buffItem.BuffEntity ~= nil then
				local index = buffItem.Index;
				if index > max then
					max = index;
					item = buffItem;
				end
			end
		end
	end
	return item;
end

function EnemyInfoWidget:AddBuffList(skillBuff)

		table.insert(self.SkillDeBuffList, skillBuff);
	else
		table.insert(self.SkillBuffList, skillBuff);
	end

		local buffList = self:GetAllBuffList(skillBuff.BuffInfo.Cfg.IsDebuff);

			self.m_BuffList:UpdateData(buffList);
		end
	end
end

-- 【防错方法】根据buff类型和运行时ID在buffList里面删除
function EnemyInfoWidget:DelBuffItemInList(isDebuff, runtimeID)
	local index = self:GetBuffIndexById(isDebuff, runtimeID);
	if index == -1 then
		return;
	end
	if isDebuff then
		table.remove(self.SkillDeBuffList, index);
	else
		table.remove(self.SkillBuffList, index);
	end

	if self.m_BuffList ~= nil and self.m_BuffList.M_IsOpen then
		local buffList = self:GetAllBuffList(isDebuff);
		if buffList ~= nil then
			self.m_BuffList:UpdateData(buffList);
		end
	end
end

-- 【防错方法】根据buff类型和运行时ID在buffList里面找到相同运行时ID的下标
function EnemyInfoWidget:GetBuffIndexById(isDebuff, runtimeID)
	if isDebuff then
		for i = 1, #self.SkillDeBuffList do
			local item = self.SkillDeBuffList[i];
			if item.RuntimeID == runtimeID then
				return i;
			end
		end
	else
		for i = 1, #self.SkillBuffList do
			local item = self.SkillBuffList[i];
			if item.RuntimeID == runtimeID then
				return i;
			end
		end
	end
	return -1;
end

function EnemyInfoWidget:DelBuffList(skillBuff)
	local index = self:GetBuffIndex(skillBuff);
	if index == -1 then
		return;
	end
	if skillBuff.BuffInfo.Cfg.IsDebuff then
		table.remove(self.SkillDeBuffList, index);
	else
		table.remove(self.SkillBuffList, index);
	end

	if self.m_BuffList ~= nil and self.m_BuffList.M_IsOpen then
		local buffList = self:GetAllBuffList(skillBuff.BuffInfo.Cfg.IsDebuff);
		if buffList ~= nil then
			self.m_BuffList:UpdateData(buffList);
		end
	end
end

function EnemyInfoWidget:GetBuffIndex(skillBuff)
	if skillBuff.BuffInfo.Cfg.IsDebuff then
		for i = 1, #self.SkillDeBuffList do
			local item = self.SkillDeBuffList[i];
			if item.RuntimeID == skillBuff.RuntimeID then
				return i;
			end
		end
	else
		for i = 1, #self.SkillBuffList do
			local item = self.SkillBuffList[i];
			if item.RuntimeID == skillBuff.RuntimeID then
				return i;
			end
		end
	end
	return -1;
end

function EnemyInfoWidget:OnBuffCreate(isDeBuff)
	local item = nil;
	if isDeBuff then
		item = self:GetIdleDeBuffItem();
	else
		item = self:GetIdleBuffItem();
	end
	if item == nil then
		item = self:CreateBuff(isDeBuff);
	end
	return item;
end

function EnemyInfoWidget:OnActionOnBuffCreate(skillBuff, buffCreateRet)
	--Log("霸服 EnemyInfoWidget:OnActionOnBuffEnd 委托通知创建 runtimeid=",skillBuff.RuntimeID);
	if skillBuff ~= nil then
		if skillBuff.BuffInfo ~= nil and skillBuff.BuffInfo.Cfg ~= nil then
			-- 获取BUFF护盾总数
			local isChanage = self:CheckBuffShieldVal(skillBuff, true);
			if isChanage then
				self:SetShieldValSum();
			end
			local isHaveCfg = skillBuff.BuffInfo:GetHaveBUFFUIShowState(E_BUFFUIShowPosEnum.HUDHpUp);
			if not isHaveCfg then
				return;
			end
			local isHave = self:GetBuffIndexById(skillBuff.BuffInfo.Cfg.IsDebuff, skillBuff.RuntimeID);
			if isHave == -1 then
				self:AddBuffList(skillBuff);
				local item = self:OnBuffCreate(skillBuff.BuffInfo.Cfg.IsDebuff);
				if item ~= nil then
					local count = self:GetBuffListCount(skillBuff.BuffInfo.Cfg.IsDebuff);
					local index = #self.SkillBuffList;
					if skillBuff.BuffInfo.Cfg.IsDebuff then
						index = #self.SkillDeBuffList;
					end
					item:SetBuffCout(count, index);
					item:SetCallBack(self.BuffItemClickFun);
					item:SetItem(skillBuff);
				else
					--添加BUFFUI时，如果没创建出来
					-- 那就是超过5个了，需要通知最后一个增加BUFF总数
					if #self.SkillDeBuffList >= 3 or #self.SkillBuffList >= 3 then
						-- 更新最后一个
						local lastBuffItem = self:GetLastBuffItem(skillBuff.BuffInfo.Cfg.IsDebuff);
						if lastBuffItem ~= nil then
							local count = self:GetBuffListCount(skillBuff.BuffInfo.Cfg.IsDebuff);
							local index = #self.SkillBuffList;
							if skillBuff.BuffInfo.Cfg.IsDebuff then
								index = #self.SkillDeBuffList;
							end
							lastBuffItem:SetBuffCout(count, index);
						end
					end
				end
			end
		end
	end
end

function EnemyInfoWidget:OnActionOnBuffEnd(skillBuff, buffEndRet)
	if skillBuff == nil then
		Log("霸服 EnemyInfoWidget:OnActionOnBuffEnd 委托通知关闭 skillBuff=nil err");
		return;
	end
	--Log("霸服 EnemyInfoWidget:OnActionOnBuffEnd 委托通知关闭 runtimeid=",skillBuff.RuntimeID);
	local buffItem = self:GetBuff(skillBuff);
	if buffItem ~= nil then
		buffItem:ExecuteCallBack();
		----删除BUFFUI时，如果buff总数超过了5个
		---- 需要通知最后一个更新BUFF总数
		--if #SkillDeBuffList >= 5 or #SkillBuffList >= 5 then
		---- 更新最后一个
		--local lastBuffItem = self:GetLastBuffItem(skillBuff.BuffInfo.Cfg.IsDebuff);
		--if lastBuffItem ~= nil then
		--local count = self:GetBuffListCount(skillBuff.BuffInfo.Cfg.IsDebuff);
		--lastBuffItem:SetBuffCout(count);
		--end
		--end
	else
		self:DelBuffList(skillBuff);
	end

	local isChanage = self:CheckBuffShieldVal(skillBuff, false);
	if isChanage then

	end
end

function EnemyInfoWidget:OnActionOnBuffUpdate(skillBuff)

	if buffItem ~= nil then
		buffItem:BuffUpdate(skillBuff);
	end
	-- 获取BUFF护盾总数
	local isChanage = self:CheckBuffShieldVal(skillBuff, true);
	if isChanage then
		self:SetShieldValSum();
	end
end

EnemyInfoWidget.buffShieldDic = {}
EnemyInfoWidget.ShieldValSum = 0

function EnemyInfoWidget:CheckBuffShieldVal(skillBuff, isOpen)
	local isChanage = false;
	if skillBuff == nil or skillBuff.BuffInfo == nil or skillBuff.BuffInfo.Cfg == nil then
		LogWarning("霸服 EnemyInfoWidget:CheckBuffShieldVal skillBuff=nil err");
		return isChanage;
	end
	local shieldVal = 0;
	if isOpen then
		shieldVal = skillBuff.ShieldVal
	end

	local lastShield = 0;
	local isHave = false;
	for key, value in pairs(self.buffShieldDic) do
		if key == skillBuff.RuntimeID then
			lastShield = value;
			isHave = true;
			break;
		end
	end
	if isHave then
		if shieldVal > 0 then
			if shieldVal ~= lastShield then
				self.buffShieldDic[skillBuff.RuntimeID] = shieldVal;
				isChanage = true;
			end
		else
			--table.remove(self.buffShieldDic, skillBuff.RuntimeID);
			self.buffShieldDic[skillBuff.RuntimeID] = 0;
			isChanage = true;
		end
	elseif shieldVal > 0 then
		self.buffShieldDic[skillBuff.RuntimeID] = shieldVal;
		isChanage = true;
	end

	return isChanage;
end

function EnemyInfoWidget:SetShieldValSum()
	local sum = 0;
	for key, value in pairs(self.buffShieldDic) do
		sum = sum + value;
	end
	self.ShieldValSum = sum;
	-- 通知血条比例修改
	--self:SetMixHpShield();
end

CurrencyMenuWidget = class("CurrencyMenuWidget", UIPanel)
CurrencyMenuWidget.IsOpen = false

local ObjectPool = require("Common/Helper/ObjectPool")

function CurrencyMenuWidget:Awake()
    self.transform = self.prefab.transform

    local btnCloseInfo = self.transform:Find("Block"):GetComponent(typeof(JButton))
    btnCloseInfo.OnClick = (function()
        self:CloseWidget()
    end)

    --中间tag页
    self.go_info = self.transform:Find("AnimRoot/Info").gameObject
    local _go = self.transform:Find("AnimRoot/Info/CurrencyJumpItem").gameObject
    self.currencyMenuJumpItemsPool =
        ObjectPool:New(
        4,
        "View/Item/CurrencyMenuJumpItem",
        _go,
        _go.transform.parent,
        function(ritem)
            ritem:InitItem(
                function(ritem)
                    --点击事件
                    self:OnCurrencyMenuJumpItemClick(ritem)
                end
            )
        end,
        function(ritem)
            ritem:SetUse(false)
        end
    )

    --中间tag2页
    self.go_info2 = self.transform:Find("AnimRoot/Info2").gameObject
    local _go2 = self.transform:Find("AnimRoot/Info2/Scroll View/Viewport/Content/CurrencyJumpItem").gameObject
    self.currencyMenuJumpItemsPool2 =
        ObjectPool:New(
        4,
        "View/Item/CurrencyMenuJumpItem",
        _go2,
        _go2.transform.parent,
        function(ritem)
            ritem:InitItem(
                function(ritem)
                    --点击事件
                    self:OnCurrencyMenuJumpItemClick(ritem)
                end
            )
        end,
        function(ritem)
            ritem:SetUse(false)
        end
    )

    --动画相关
    --self.anim=self.transform:GetComponent(typeof(Animator));

    self.module = nil
end

function CurrencyMenuWidget:OnCurrencyMenuJumpItemClick(ritem)
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {ritem.BaseID, false, 0, ritem.tipsTr, 0})
end

function CurrencyMenuWidget:RefeshCurrencyMenu()
    self:ClearAllAnim()
    --self.anim:Play("CurrencyMenuWidgetAnim_Open");

    self.transform:SetAsLastSibling()

    local isShowBg = false
    if #self.Coins <= 8 then
        self.go_info:SetActive(true)
        self.go_info2:SetActive(false)
        self.currencyMenuJumpItemsPool:RecycleAll()
        for i = 1, #self.Coins do
            local cur = self.currencyMenuJumpItemsPool:Spawn()
            cur:SetItem(self.Coins[i].ID, isShowBg)
            isShowBg = not (isShowBg)
        end
    else
        self.go_info:SetActive(false)
        self.go_info2:SetActive(true)

        self.currencyMenuJumpItemsPool2:RecycleAll()
        for i = 1, #self.Coins do
            local cur = self.currencyMenuJumpItemsPool2:Spawn()
            cur:SetItem(self.Coins[i].ID, isShowBg)
            isShowBg = not (isShowBg)
        end
    end
end

function CurrencyMenuWidget:Start()
end

function CurrencyMenuWidget:OnEnable()
end

function CurrencyMenuWidget:OnDisable()
end

function CurrencyMenuWidget:OnDestroy()
end

function CurrencyMenuWidget:OnOpen(arg)
    if self.Coins == nil then
        self.Coins = {}
    else
        --self.ItemDataList ={}
        for k, _ in pairs(self.Coins) do
            self.Coins[k] = nil
        end
    end

    for k, v in pairs(ConfigDataAccessor.CurrencyCfg) do
        local id = k

        if v.IsShowAllCurrency then
            local isSysOpen = true
            if v.SysOpen > 0 then
                --判断是否系统开启
                isSysOpen = CheckPlayerGuidanceOpen(v.SysOpen, "", false)
            end

            if isSysOpen then
                local isIn = false

                for j = 1, #arg do
                    if arg[j] == id then
                        isIn = true
                        break
                    end
                end

                if not (isIn) then
                    local it = {}
                    it.ID = id
                    it.Sort = v.AllCurrencySort
                    table.insert(self.Coins, it)
                end
            end
        end
    end

    self:SortItem(self.Coins)

    self:RefeshCurrencyMenu()
    self.IsOpen = true
end

function CurrencyMenuWidget:SortItem(tbl)
    if #tbl > 1 then
        table.sort(
            tbl,
            function(a, b)
                if a.Sort < b.Sort then
                    -- 按Quality升序
                    return true
                end
                return false
            end
        )
    end
end

function CurrencyMenuWidget:OnClose(arg)
    self.IsOpen = false
end

function CurrencyMenuWidget:CloseWidget()
    self.IsOpen = false
    --self.anim:Play("CurrencyMenuWidgetAnim_Close");
    self:ClearAllAnim()
    self.tw_close =
        DOVirtual.DelayedCall(
        0.01,
        function()
            UIManager.Instance:CloseWidget("Bag/CurrencyMenuWidget", nil, true)
        end
    )
end

function CurrencyMenuWidget:ClearAllAnim()
    if self.tw_close ~= nil then
        self.tw_close:Kill()
        self.tw_close = nil
    end
end

return CurrencyMenuWidget

BossRankRewardsWidget=class("BossRankRewardsWidget",UIPanel)
local ItemGrid=require("View/Item/ItemGrid");


function BossRankRewardsWidget:Awake()
	self.transform=self.prefab.transform;
	self.gameObject=self.transform.gameObject;

	self.transform:Find("Cover"):GetComponent(typeof(JButton)).OnClick=(function()
		self:OnPlayBack();
	end);
	self.transform:Find("AnimRoot/Title/CloseBtn"):GetComponent(typeof(JButton)).OnClick=(function()
		self:OnPlayBack();
	end);

	self.Icon=self.transform:Find("AnimRoot/Title/bossIcon"):GetComponent(typeof(Image));

	self.BossName=self.transform:Find("AnimRoot/Title/name"):GetComponent(typeof(Text));

	--道具
	self.items={}
	for i=1,3 do
		self.items[i]={}

		for j=1,5 do
			local it = ItemGrid:New(self.transform:Find("AnimRoot/Info/BossRankItem"..i.."/Items/ItemGrid"..j).gameObject);
			it:InitItem(false,function(ritem)  
				self:OnItemClick(ritem);
			end);
			table.insert( self.items[i],it);
		end
	end

end



function BossRankRewardsWidget:Start()

end

function BossRankRewardsWidget:OnEnable()
	
end

function BossRankRewardsWidget:OnDisable()
	
end

function BossRankRewardsWidget:OnDestroy()

end


function BossRankRewardsWidget:LoadBossIcon(bossBaseID)
	local bossType=1;
	local cfgs=ConfigDataAccessor.GVEBossSelect;
	for k,v in pairs(cfgs) do
		if v.UseBoss==bossBaseID then
			bossType=v.BossType
		end
	end

	local idx=1;
	if bossType==1 then
		idx=2;
	elseif bossType==2 then
		idx=3;
	end

	ResourceHelper:LoadIconByAtlasAsync(self.Icon, ResourceHelper.AtlasPathHud, "Map_icon_0"..idx, false);
end

function BossRankRewardsWidget:OnOpen(arg)


	local bossCfgId = arg;

	self:LoadBossIcon(bossCfgId);
	self.BossName.text=ConfigDataAccessor.Monster[bossCfgId].Name;
	local gBRCfg=ConfigDataAccessor.GVEBossReward;


	self.EffectLv=0;
    local GVEStartI2S = GameManager.Instance.GVEStartI2S;
    if GVEStartI2S~=nil and GVEStartI2S.PlayID==self.PlayMode then
        --生效等级
        self.EffectLv=GVEStartI2S.EffectLv;
    end


    if self.EffectLv==0 then
        local openServerDay=0;
        if GameManager.Instance.GloInfoSrv~=nil then
            openServerDay=GameManager.Instance.GloInfoSrv.GRisk.OffDay;
        end
        self.EffectLv=ConfigDataAccessor.GetGVEEffectLevel(openServerDay);
    end


    --消耗道具
	for i=1,3 do
        for j=1,#self.items do
            self.items[i][j].gameObject:SetActive(false);
        end
	end


	--奖励等级
	for i=1,3 do
		local idx=0;
		for k2,v2 in pairs(gBRCfg) do
			if v2.Rank==i and v2.Boss==bossCfgId and self.EffectLv == v2.EffectLevel then;

				for j=1,#v2.Reward do

					local Id=v2.Reward[j];
					idx=idx+1;
					self.items[i][idx].gameObject:SetActive(true);
					self.items[i][idx]:SetItemNum(Id,0);
				end

			end

		end

	end

end

function BossRankRewardsWidget:OnItemClick(ritem)
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {ritem.BaseID, false, 0, ritem.transform, 0})
end

function BossRankRewardsWidget:OnClose(arg)

end

function BossRankRewardsWidget:OnPlayBack()

	self.gameObject:SetActive(false);
	UIManager.Instance:CloseWidget("Event/BossRankRewardsWidget",nil,false);

end



return  BossRankRewardsWidget

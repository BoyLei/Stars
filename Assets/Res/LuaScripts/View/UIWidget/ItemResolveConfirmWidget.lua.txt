ItemResolveConfirmWidget = class("ItemResolveConfirmWidget", UIPanel)
ItemResolveConfirmWidget.IsOpen = false

local ObjectPool = require("Common/Helper/ObjectPool")

function ItemResolveConfirmWidget:Awake()
end

function ItemResolveConfirmWidget:InitItems()
    self.transform = self.prefab.transform
    self.IsOpen = false

    --动画相关
    --self.anim=self.transform:GetComponent(typeof(Animator));

    self.go_grid1 = self.transform:Find("Root/Grid").gameObject
    self.go_grid2 = self.transform:Find("Root/Grid2").gameObject

    self.rect_grid2 = self.transform:Find("Root/Grid2/Mask/List"):GetComponent(typeof(RectTransform))

    --奖励对象池
    local _go = self.transform:Find("Root/ItemGrid").gameObject
    local _parent = self.transform:Find("Root/Grid")
    self.rewardItemPool =
        ObjectPool:New(
        8,
        "View/Item/ItemGrid",
        _go,
        _parent,
        --生成回调
        function(ritem)
            ritem:InitItem(
                false,
                --不使用底板
                --点击回调
                function(ritem)
                    self:OnItemClick(ritem)
                end
            )
        end,
        --回收回调
        function(ritem)
            ritem:SetUse(false)
        end
    )

    local _go2 = self.transform:Find("Root/ItemGrid2").gameObject
    local _parent2 = self.transform:Find("Root/Grid2/Mask/List")
    self.rewardItemPool2 =
        ObjectPool:New(
        6,
        "View/Item/ItemGrid",
        _go2,
        _parent2,
        --生成回调
        function(ritem)
            ritem:InitItem(
                false,
                --不使用底板
                --点击回调
                function(ritem)
                    self:OnItemClick(ritem)
                end
            )
        end,
        --回收回调
        function(ritem)
            ritem:SetUse(false)
        end
    )

    self.go_grid1:SetActive(false)
    self.go_grid2:SetActive(false)

    local btnCloseInfo = self.transform:Find("BackGround"):GetComponent(typeof(JButton))
    btnCloseInfo.OnClick = function()
        self:OnCloseClick()
    end

    local btnCancel = self.transform:Find("Root/BtnCancel"):GetComponent(typeof(JButton))
    btnCancel.OnClick = function()
        self:OnCloseClick()
    end

    local btnConfirm = self.transform:Find("Root/BtnConfirm"):GetComponent(typeof(JButton))
    btnConfirm.OnClick = function()
        --点击确认
        self:OnCloseClick()
        LuaModuleDispatcher:SendMessage(ModuleDef.ItemResolveModule, "OnItemResolveConfirm", {})
    end

    self.go_check = self.transform:Find("Root/CheckBox/selected").gameObject
    self.isCheck = false
    self.go_check:SetActive(self.isCheck)
    local btnCheckBox = self.transform:Find("Root/CheckBox"):GetComponent(typeof(JButton))
    btnCheckBox.OnClick = function()
        self.isCheck = not (self.isCheck)
        self.go_check:SetActive(self.isCheck)
    end
end

function ItemResolveConfirmWidget:RefeshRewards(items)
    SoundManager.Instance:PlayEventName("UI_TanChuang_JiangLi_01", self.gameObject, self.gameObject)

    --self.anim:Play("ItemResolveConfirmWidgetAnim_Open");

    self.module.IsOpenTips = false
    self.module:CloseItemTips()

    self.transform:SetAsLastSibling()

    if items == nil then
        do
            return
        end
    end

    if #items > 7 then
        self.go_grid1:SetActive(false)
        self.go_grid2:SetActive(true)

        --背包格子
        self.rewardItemPool2:RecycleAll()
        for i = 1, #items do
            local item = self.rewardItemPool2:Spawn()
            item:SetItem(items[i])
            local infoLb = item.transform:Find("Have/info"):GetComponent(typeof(Text))
            if items[i].PolishNum > 0 then
                local maxNum = ConfigDataAccessor.GetMaxPolishNum(items[i].Quality) --铭刻次数的上限
                local str = "" .. items[i].PolishNum .. "/" .. maxNum
                infoLb.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ItemSalvage_8"), {str})
            elseif items[i].Refined > 0 then
                local str = "" .. items[i].Refined
                infoLb.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ItemSalvage_7"), {str}) .. "%"
            else
                infoLb.text = ""
            end
        end

        self.rect_grid2.anchoredPosition = Vector2(self.rect_grid2.anchoredPosition.x, 0)
    else
        self.go_grid1:SetActive(true)
        self.go_grid2:SetActive(false)

        --背包格子
        self.rewardItemPool:RecycleAll()
        for i = 1, #items do
            local item = self.rewardItemPool:Spawn()
            item:SetItem(items[i])
            local infoLb = item.transform:Find("Have/info"):GetComponent(typeof(Text))
            if items[i].PolishNum > 0 then
                local maxNum = ConfigDataAccessor.GetMaxPolishNum(items[i].Quality) --铭刻次数的上限
                local str = "" .. items[i].PolishNum .. "/" .. maxNum
                infoLb.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ItemSalvage_8"), {str})
            elseif items[i].Refined > 0 then
                local str = "" .. items[i].Refined
                infoLb.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_ItemSalvage_7"), {str}) .. "%"
            else
                infoLb.text = ""
            end
        end
    end
end

function ItemResolveConfirmWidget:OnItemClick(ritem)
    print("RewardItem:OnItemClick")
    self.module.IsOpenTips = true
    self.module:OpenItemTips(ritem.BaseID, ritem.EntityID, ritem.transform)
end

function ItemResolveConfirmWidget:BindModule(m)
    print(".............")
    self.module = m
end

function ItemResolveConfirmWidget:Start()
end

function ItemResolveConfirmWidget:OnEnable()
end

function ItemResolveConfirmWidget:OnDisable()
end

function ItemResolveConfirmWidget:OnDestroy()
end

function ItemResolveConfirmWidget:OnOpen(arg)
    self.IsOpen = true
end

function ItemResolveConfirmWidget:OnClose(arg)
    self.IsOpen = false
end

function ItemResolveConfirmWidget:OnCloseClick()
    --self.anim:Play("ItemResolveConfirmWidgetAnim_Close");

    --保存下当日不在显示
    if self.go_check.activeSelf then
        LocalCacheManager.Instance:SetDayValue("ItemResolveNoConfirmToDay", true)
    end

    DOVirtual.DelayedCall(
        0.1,
        function()
            --关闭道具Tips
            if self.module.IsOpenTips then
                self.module.IsOpenTips = false
                self.module:CloseItemTips()
            else
                self:CloseWidget()
            end
        end
    )
end

function ItemResolveConfirmWidget:CloseWidget()
    print("ItemResolveConfirmWidget:CloseWidget")
    UIManager.Instance:CloseWidget("ItemResolve/ItemResolveConfirmWidget", nil, true)
end

return ItemResolveConfirmWidget

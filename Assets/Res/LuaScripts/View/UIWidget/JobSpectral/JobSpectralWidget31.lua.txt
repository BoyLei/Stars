function JobSpectralWidget31:UpdateData()
	if self.m_IsCountdown then
		if self.m_Countdown > 0 then
			local fixedDeltaTime = Time.deltaTime
			self.m_Countdown = self.m_Countdown - fixedDeltaTime
			local dif = self.m_MaxTime - self.m_Countdown;
			local num = math.floor(dif / self.m_EventItemPer)
			if self.m_LastIndex ~= num then
				self:SetUISpectral1Item(num, true);
				self.m_LastIndex = num;
			end
			if self.m_Countdown <= 0 then
				self.m_IsCountdown = false;
			end
		end
	end
end

function JobSpectralWidget31:SetUISpectral11()
	local ratio = 0;
	if self.m_TruthSpectral1 > 0 then
		ratio = self.m_CurSpectral1 / self.m_TruthSpectral1;
		if ratio > 1 then
			ratio = 1;
		end
	end
	--self.Progress.fillAmount = ratio;
	self.Progress:DOFillAmount(ratio, 0.2);
	if self.m_lastRatio ~= -1 and ratio > self.m_lastRatio and ratio ~= 1 and self.AddAnim ~= nil then
		if self.tw_fg1 ~= nil then
			self.tw_fg1:Kill();
		end
		self:SetAddAnimShow(false);
		self:SetAddAnimShow(true);
		self.tw_fg1 = self.AddAnim:DOAnchorPosX(150 * (1 - ratio), 100):SetSpeedBased():SetEase(Ease.Linear):OnComplete(function ()
			self:SetAddAnimShow(false);
		end);
	end
	self:SetMaxAnimShow(false);
	if ratio == 1 then
		self:SetMaxAnimShow(true);
		self.m_lastRatio = 0;
		self.AddAnim.anchoredPosition = self.AddAnimInitPos;
	else
		self.m_lastRatio = ratio;
	end
end

function JobSpectralWidget31:SetUISpectral1Item(idx, isShow)
	--local isShow = self.m_CurSpectral1 >= idx;
	if self.CommonStateSpectral2List == nil or self.CommonStateSpectral2List == {} then
		return;
	end
	if idx < 1 then
		return;
	end
	if idx > #self.CommonStateSpectral2List then
		return;
	end
	self.CommonStateSpectral2List[idx]:SetActive(isShow);
	--self.CommonStateSpectral2List[idx].transform:GetChild(0).gameObject:SetActive(isShow);
end

function JobSpectralWidget31:SetUISpectral1ItemFun(idx)

	self.CommonStateSpectral2AnimList[idx]:SetActive(isShow);
end

function JobSpectralWidget31:SetMaxAnimShow(isShow)
	if self.MaxAnim == nil then
		return
	end
	self.MaxAnim:SetActive(isShow);
end

function JobSpectralWidget31:SetAddAnimShow(isShow)
	if self.AddAnim == nil then
		return
	end
	self.AddAnim.gameObject:SetActive(isShow);
end

function JobSpectralWidget31:SetCountdown(countdown, startCountdown)
	self.m_MaxTime = countdown;
	self.m_Countdown = startCountdown;
	self.m_EventItemPer = self.m_MaxTime / MaxLength;
end

function JobSpectralWidget31:SetOpenCountdown(isOpen)
	self.m_IsCountdown = isOpen;
	self.m_LastIndex = 0;
	if self.m_IsCountdown == false then
		self:SetUISpectral1(false);
	end

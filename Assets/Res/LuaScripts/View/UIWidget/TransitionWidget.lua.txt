TransitionWidget = class("TransitionWidget", UIPanel)

function TransitionWidget:Awake()
    self.transform = self.prefab.transform
    self.gameObject = self.prefab

    self.bg = self:Find("bg"):GetComponent(typeof(Image))
    --动画相关
    self.anim = self.transform:GetComponent(typeof(Animator))
end

function TransitionWidget:BindModule(m)
    print(".............")
    self.module = m
end

function TransitionWidget:Start()
end

function TransitionWidget:OnEnable()
end

function TransitionWidget:OnDisable()
end

function TransitionWidget:OnDestroy()
end

function TransitionWidget:OnOpen(arg)
    local Type = arg[1]
    --0黑 1白
    local EffectID = arg[2]
    self.EffectID = EffectID

    local time = 1.5

    if self.tw ~= nil then
        self.tw:Kill()
    end

    if Type == 0 then
        self.anim:Play("Fade_In_Out")
        --0黑

        self.bg.color = Color(0, 0, 0, 1)

        self.tw =
            DOVirtual.DelayedCall(
            time,
            function()
                self:OnPlayBack()
            end
        )
    elseif Type == 2 then
        self.anim:Play("Fade_Out")
        self.bg.color = Color(0, 0, 0, 1)

        self.tw =
            DOVirtual.DelayedCall(
            1.5,
            function()
                self:OnPlayBack()
            end
        )
    elseif Type == 3 then
        self.anim:Play("Enter")
        self.bg.color = Color(0, 0, 0, 1)
    else
        --1白
        self.anim:Play("Fade_In_Out")
        self.bg.color = Color(1, 1, 1, 1)

        self.tw =
            DOVirtual.DelayedCall(
            time,
            function()
                self:OnPlayBack()
            end
        )
    end
end

function TransitionWidget:OnClose(arg)
end

function TransitionWidget:OnPlayBack()
    CS.StarProject.GlobalEvent.OnEffectEnd:Invoke(self.EffectID)
    UIManager.Instance:CloseWidget("Avg/TransitionWidget", nil, true)
end

return TransitionWidget

local SignItem = {}

SignItem.__index = SignItem

local ItemGrid=require("View/Item/ItemGrid");


function SignItem:New(_go) 
    local self = {};    
    setmetatable(self, SignItem); 
    
    self.gameObject=_go;
    self.transform=_go.transform;


    self.Name=self.transform:Find("Name"):GetComponent(typeof(Text));
    self.Title=self.transform:Find("Title"):GetComponent(typeof(Text));
   
    self.ItemGrid=ItemGrid:New(self.transform:Find("ItemGrid").gameObject);--第一个不填会默认创建一个
	self.ItemGrid:InitItem(false,function(it) 
		self:OnItemClick(it);
	end)

    
    self.IsUsed=false;
    self.gameObject:SetActive(false);

    return self;    --返回自身
end

function SignItem:SetUse(b)
    self.IsUsed=b;
    self.gameObject:SetActive(b);
end

function SignItem:InitItem()

    self:SetUse(false);
end


function SignItem:OnItemClick(ritem)
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {ritem.BaseID, false, 0, ritem.transform, 0})
end

function SignItem:SetItem(item)
    self:SetUse(true);


    --商品id
    local itemID=item.ItemID;

    --底板
    local itemCfg = ConfigDataAccessor.Item[itemID];
    self.Name.text=itemCfg.Name;
    
    self.Title.text=LanguageHelp:GetLanguageByKey(item.Title);
    

    local it={};
    it.BaseID=itemID;
    it.Num=item.ItemNum;
    self.ItemGrid:SetItem(it);




end




return SignItem;

local AvgMindOneLine = {}

AvgMindOneLine.__index = AvgMindOneLine

function AvgMindOneLine:New(_go) 
    local self = {};    
    setmetatable(self, AvgMindOneLine); 
    
    self.gameObject=_go;
    self.transform=_go.transform;

	self.m_TalkBoxContent=self.transform:Find("TalkBoxContent"):GetComponent(typeof(Text));

	self.tw_content=nil;
	self.talkConetntNow=nil;
	self.text_speed=40;


    return self;    --返回自身
end

function AvgMindOneLine:InitItem(onFinishedTalk,onFinishedWaitting,onUpdatePos)
	self.OnFinishedTalk=onFinishedTalk;
	self.OnFinishedWaitting=onFinishedWaitting;
	self.OnUpdatePos=onUpdatePos;
    self:SetUse(false);
end

function AvgMindOneLine:SetUse(b)
    self.IsUsed=b;
    self.gameObject:SetActive(b);
end

function AvgMindOneLine:SetItem(info)

    self:SetUse(true);

	self.talkConetntNow = info;
	self:RefeshTalkInfo();
end




function AvgMindOneLine:RefeshTalkInfo()

	self.transform:SetAsLastSibling();

	if self.tw_content~= nil then
		self.tw_content:Kill();
	end
	if self.tw_delay~= nil then
		self.tw_delay:Kill();
	end

	DOVirtual.DelayedCall(0.05,function ()
		self.OnUpdatePos();
	end)

	self.m_TalkBoxContent.text="";


	--延迟一段时间后调用
	if self.talkConetntNow.Send_delay>0  then
		--等待时间
		local waitTime=self.talkConetntNow.Send_delay/1000;
		self.tw_delay=DOVirtual.DelayedCall(waitTime,function ()
			self.OnFinishedWaitting();
			self:ShowContent();
		end)
	else
		self:ShowContent();
	end




end


function AvgMindOneLine:ShowContent()
		--[[

	--显示内容
	self.tw_content=self.m_TalkBoxContent:DOText(self.talkConetntNow.Dialog_desc2,self.text_speed):SetSpeedBased():SetEase(Ease.Linear):OnComplete(function ()
		
		if self.OnFinishedTalk~=nil then
			self.OnFinishedTalk();
		end
		
	end):OnUpdate(function ()
		--计算文本框大小
		self.OnUpdatePos();
	end);;

	]]

	self.m_TalkBoxContent.text=self.talkConetntNow.Dialog_desc2;
	DOVirtual.DelayedCall(0.05,function ()
		self.OnFinishedTalk();
		self.OnUpdatePos();
	end)


	



	--播放语音
	if self.talkConetntNow.Voice_id ~= 0 then
		--Util.PlayAvgVoice("Weapon_sword__pickup");
	end
end



function AvgMindOneLine:FinishTalkNow()

	if self.tw_content~= nil then
		self.tw_content:Kill();
	end

	self.m_TalkBoxContent.text=self.talkConetntNow.Dialog_desc2;
	self.OnFinishedTalk();
end





return AvgMindOneLine;

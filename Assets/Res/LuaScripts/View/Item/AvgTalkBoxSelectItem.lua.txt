local AvgTalkBoxSelectItem = {}

AvgTalkBoxSelectItem.__index = AvgTalkBoxSelectItem

function AvgTalkBoxSelectItem:New(_go, onItemCkick)
    local self = {}
    setmetatable(self, AvgTalkBoxSelectItem)

    self.gameObject = _go
    self.transform = _go.transform

    self.m_content = self.transform:Find("Anim/content"):GetComponent(typeof(Text))

    self.Idx = 0

    --点击事件
    local btnClick = self.transform:GetComponent(typeof(Button))
    btnClick.onClick:AddListener(
        function()
            if onItemCkick ~= nil then
                onItemCkick(self.Idx, self.contendID)
            end
        end
    )

    return self --返回自身
end

function AvgTalkBoxSelectItem:SetItem(contentText, idx, cID)
    self.m_content.text = contentText
    self.m_content.color = Color(1, 1, 1)
    self.Idx = idx
    self.contendID = cID
end

function AvgTalkBoxSelectItem:SetDark()
    self.m_content.color = Color(0.57, 0.55, 0.53)
end

return AvgTalkBoxSelectItem

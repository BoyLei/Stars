EctypeTargetItem=class("EctypeTargetItem",UIPanel)
EctypeTargetItem.Text=nil;
EctypeTargetItem.Finished=nil;
EctypeTargetItem.Runing=nil;

function EctypeTargetItem:ctor(go,atlas)
	self.prefab=go;
	self.AtlasPath=atlas;
	self.Text=self:Find("Text"):GetComponent(typeof(Text));
	self.Finished=self:Find("Finished"):GetComponent(typeof(Image));
	self.Runing=self:Find("Runing"):GetComponent(typeof(Image));
end



function EctypeTargetItem:UpdateContent(data)
	self.Text.text=data.Content
	self.Finished.gameObject:SetActive(data.Finished);
	self.Runing.gameObject:SetActive(not data.Finished);
end

return  EctypeTargetItem

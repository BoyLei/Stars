local EquipAttrItem = {}

EquipAttrItem.__index = EquipAttrItem

function EquipAttrItem:New(_go)
    local self = {}
    setmetatable(self, EquipAttrItem)

    self.gameObject = _go
    self.transform = _go.transform

    self.Name = self.transform:Find("name"):GetComponent(typeof(Text))
    self.Add = self.transform:Find("add"):GetComponent(typeof(Text))

    return self --返回自身
end

--设置Item
function EquipAttrItem:RefeshAttrProb(data, isEquiped, slot_id, idx)
    local config = ConfigDataAccessor.GetAttrByID(data.ID)
    local attrName = ""
    if config ~= nil then
        attrName = config.Name
    end
    local name = attrName

    local add = data.Value

    --决定颜色
    --local quality = data.Quality;
    --self.Name.text=GetQualityColor(quality)..name.."</color>";
    --self.Add.text=GetQualityColor(quality).."+"..add.."</color>";
    self.Name.text = name
    self.Add.text = "" .. add

    --装备中的要显示槽位加成
    if isEquiped then
        local equipSlotData = GameManager.Instance:GetEquipSlotData(slot_id)
        local slotAdd = 0
        if equipSlotData == nil or equipSlotData.UpLv == 0 then
            slotAdd = 0
        else
            local intensify = ConfigDataAccessor.GetEquipSlotIntensify(slot_id, equipSlotData.UpLv, GameManager.Instance:GetPlayJobBaseID())
            local addVal1 = intensify.Value[idx]

            slotAdd = addVal1
        end
        if slotAdd > 0 then
            self.Add.text = "" .. (add + slotAdd) .. " (<color=#67D16A>+" .. add .. "</color><color=#CAB27D> +" .. slotAdd .. "</color>)"
        end
    end
end

--设置Item
function EquipAttrItem:RefeshCfgAttrProb(ID)
    local config = ConfigDataAccessor.GetAttrByID(ID)
    local attrName = ""
    if config ~= nil then
        attrName = config.Name
    end
    local name = attrName

    self.Name.text = name
    self.Add.text = "+???"
end

return EquipAttrItem

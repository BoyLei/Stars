local EquipItemSlot = {}

EquipItemSlot.__index = EquipItemSlot

function EquipItemSlot:New(_go)
	local self = {}
	setmetatable(self, EquipItemSlot)

	self.gameObject = _go
	self.transform = _go.transform

	self.Level = self.transform:Find("Have/Num"):GetComponent(typeof(Text)) --等级

	--暂时先隐藏
	--self.transform:Find("Have/Num").gameObject:SetActive(false);
	--self.transform:Find("Have/NumBg").gameObject:SetActive(false);

	self.Icon = self.transform:Find("Have/mask/Icon"):GetComponent(typeof(Image))
	self.Rare = self.transform:Find("Have/BgRare"):GetComponent(typeof(Image))

	self.go_Selected = self.transform:Find("Selected").gameObject
	self.go_Selected:SetActive(false)

	self.go_Lock = self.transform:Find("Have/Lock").gameObject
	self.go_Lock:SetActive(false)

	if self.transform:Find("Have/Equip") ~= nil then
		self.go_Equip = self.transform:Find("Have/Equip").gameObject
		self.go_Equip:SetActive(false)
	end

	self.go_Bar = self.transform:Find("bar").gameObject
	self.go_Bar:SetActive(false)
	self.perfectBar = self.transform:Find("bar/Image"):GetComponent(typeof(Image))
	self.perfectNum = self.transform:Find("bar/barNum"):GetComponent(typeof(Text))

	self.go_New = self.transform:Find("Have/New").gameObject
	self.go_New:SetActive(false)

	self.go_Have = self.transform:Find("Have").gameObject
	self.go_Have:SetActive(false)

	self.go_Null = self.transform:Find("Null").gameObject
	self.go_Null:SetActive(true)

	if self.transform:Find("Null/redpoint")~=nil then
		self.redpoint=self.transform:Find("Null/redpoint").gameObject
		self.redpoint:SetActive(false)
	end

	self.IsUsed = false

	--点击事件
	self.btnClick = self.transform:GetComponent(typeof(JButton))
	self.btnClick.OnClick = (function()
			if self.OnItemCkick ~= nil then
				if self.IsUsed then
					self.OnItemCkick(self, self.Idx)
				else
					self.OnItemCkick(nil, self.Idx)
				end
			end
		end)

	--长按事件
	self.btnClick.OnLongPressDown = (function()
			if self.OnItemLongPress ~= nil then
				if self.IsUsed then
					self.OnItemLongPress(self)
				end
			end
		end)

	return self --返回自身
end

--初始化Item
function EquipItemSlot:InitItem(isHaveBoard, onItemCkick, onItemLongPress)
	self.isHaveBoard = isHaveBoard
	self.OnItemLongPress = onItemLongPress
	self.OnItemCkick = onItemCkick
	self:SetUse(false)
end

function EquipItemSlot:SetUse(b)
	self.IsUsed = b
	self.go_Have:SetActive(b)
	self.go_Null:SetActive(not (b))

	--如果没有有底板就隐藏
	if not self.isHaveBoard then
		self.gameObject:SetActive(b)
	end
end

--设置Item
function EquipItemSlot:SetItem(item, idx, isShowPerfect)
	if self.go_Equip ~= nil then
		self.go_Equip:SetActive(false)
	end

	self.Idx = idx

	local red=false;

	if  item.BaseID==0 then
		local items = BusinessManager.Instance:GetItemsByItemType(4, self.Idx);
		red=items~=nil and items.Count>0
	end
	
	if self.redpoint~=nil then
		self.redpoint:SetActive(red)
	end

	--背包特判
	if item.BaseID == 0 then
		self:SetUse(false)
		self:MakeSelected(false)
		self.go_Bar:SetActive(false)
	else
		self:SetUse(true)

		self.EntityID = item.EntityID
		self.BaseID = item.BaseID

		--稀有度底板
		local cfg = ConfigDataAccessor.Item[self.BaseID]
		local quality = cfg.Quality
		if quality==99 then
            quality=1;
        end
		--self.Rare.sprite= AtlasManager.Instance:GetSprite("ui/common/atlas/commonitem","Bag_Grid_sq"..quality);
		ResourceHelper:LoadIconByAtlasAsync(
			self.Rare,
			ResourceHelper.AtlasPathCommonItem,
			"Bag_Grid_sq" .. quality,
			false
		)

		--LoadIcon(self.Icon,cfg.Icon);
		ResourceHelper:LoadIconAsync(self.Icon, "", cfg.Icon, false)

		--是否是新物品
		if item.IsNew ~= nil then
			--self.go_New:SetActive(item.IsNew);
		end

		--是否显示完美度
		if isShowPerfect == 1 then
			--计算完美度
			local perfectAllValue = 0
			local eCfg = ConfigDataAccessor.Equip[self.BaseID]
			--主属性
			local itemData = BusinessManager.Instance:GetItemByItemEntityId(self.EntityID)
			local MainProps = itemData.EquipProps.MainPropList
			for i = 0, MainProps.Count - 1 do
				local maxAttr = eCfg.NumMax[i + 1]
				perfectAllValue = perfectAllValue + (MainProps[i].Value / maxAttr)
			end

			--这里要计算平均完美度（不完美的话上面就变成false）
			local Perfection = math.floor(perfectAllValue / MainProps.Count * 100)
			self.Level.text = "" .. Perfection

			self.transform:Find("Have/Num").gameObject:SetActive(true)
			if self.transform:Find("Have/NumBg") ~= nil then
				self.transform:Find("Have/NumBg").gameObject:SetActive(true)
			end

			self.perfectBar.fillAmount = Perfection / 100
			self.Level.text = ""
			self.perfectNum.text = "" .. Perfection
			self.go_Bar:SetActive(true)
		else
			--计算装备槽强度
			local equipSlotData = GameManager.Instance:GetEquipSlotData(self.Idx)
			if equipSlotData == nil or equipSlotData.UpLv == 0 then
				self.transform:Find("Have/Num").gameObject:SetActive(false)
				if self.transform:Find("Have/NumBg") ~= nil then
					self.transform:Find("Have/NumBg").gameObject:SetActive(false)
				end
			else
				self.transform:Find("Have/Num").gameObject:SetActive(true)
				if self.transform:Find("Have/NumBg") ~= nil then
					self.transform:Find("Have/NumBg").gameObject:SetActive(true)
				end
				self.Level.text = "+" .. equipSlotData.UpLv
			end

			self.go_Bar:SetActive(false)
		end

		--是否显示已装备
		if self.EntityID ~= nil and cfg.ItemType == 4 and item.Heroid > 0 then
			if self.go_Equip ~= nil then
				self.go_Equip:SetActive(true)
			end
		end
	end
	--self:MakeSelected(false);
end

--是否被选中
function EquipItemSlot:MakeSelected(b)
	self.go_Selected:SetActive(b)
end

return EquipItemSlot

local LootItemAdv = {}

LootItemAdv.__index = LootItemAdv

function LootItemAdv:New(_go)
    local self = {}
    setmetatable(self, LootItemAdv)

    self.gameObject = _go
    self.transform = _go.transform

    self.tr_MeshRoot = self.transform:Find("MeshRoot")
    self.go_perfab = nil
    self.perfabName = ""

    self.name = self.transform:Find("ItemName/Name"):GetComponent(typeof(TextMeshPro))
    --self.name.gameObject:SetActive(false)

    self.go_effectRoot = self.transform:Find("DropEff").gameObject

    self.ModelName = ""

    self.go_model = nil

    self.isShowEff = false
    self.CanCheckEff = false

    self.quality = 0
    self.IsUsed = false

    self.Enable = true

    MonoHelper.AddFixedUpdateListener(
        function()
            self:UpdateLoot()
        end
    )

    --参数

    self.mBounceHeight = 0
    self.mInitDelay = 0
    self.mIdleRotationSpeed = 0
    self.mState = 3
    self.mTimeSinceBorn = 0

    self.mBounceTime = 0
    self.mFallTime = 0

    self.mTimeSinceBounceStart = 0
    self.mTimeSinceFallStart = 0

    self.mHorizontalVelocity = Vector3.zero

    self.mHighPosY = 0
    self.mInitPosY = 0
    self.mBounceRotateSpeed = 0

    return self --返回自身
end

function LootItemAdv:InitItem(onCreateEff)
    self.OnCreateEff = onCreateEff --创建特效回调
    self.IsUsed = false
    self.gameObject:SetActive(false)
end

function LootItemAdv:SetItem(info)
    --参数
    self.mState = 0
    local brustCenter = info.pos

    local bounceEaseType = CS.TweenEaseType.easeOutQuad
    local fallEaseType = CS.TweenEaseType.easeInCubic
    local ScatterRadius = 3 * 1.5
    local MinDelay = 0.1
    local MaxDelay = 0.2
    local MaxBounceTime = 0.7
    local MinBounceTime = 0.4
    local MaxFallTime = 0.5
    local MinFallTime = 0.2
    local MinBounceRotation = 180
    local MaxBounceRotation = 360
    local MaxBounceHeight = 4 * 1.5
    local MinBounceHeight = 2 * 1.5
    local IdleRotationSpeed = 2

    local randomInCircle = CS.UnityEngine.Random.insideUnitCircle
    local xz = randomInCircle * ScatterRadius
    local dropHorizontalDir = ScatterRadius * randomInCircle.magnitude
    local y = BusinessManager.Instance:GetGroundHeight(brustCenter.x + xz.x, brustCenter.z + xz.y)
    --local targetPos = Vector3(brustCenter.x + xz.x, y, brustCenter.z + xz.y)

    local bounceHeight = CS.UnityEngine.Random.Range(MinBounceHeight, MaxBounceHeight)
    local bounceTime = CS.UnityEngine.Random.Range(MinBounceTime, MaxBounceTime)
    local fallTime = CS.UnityEngine.Random.Range(MinFallTime, MaxFallTime)
    local bounceRotateSpeed = CS.UnityEngine.Random.Range(MinBounceRotation, MaxBounceRotation)

    local dropHorizontalVelocity = dropHorizontalDir / (bounceTime + fallTime) * randomInCircle.normalized
    local delay = CS.UnityEngine.Random.Range(MinDelay, MaxDelay)

    --参数
    self.mBounceHeight = bounceHeight
    self.mInitPosY = y
    self.mHighPosY = y + bounceHeight
    self.mInitDelay = delay
    self.mIdleRotationSpeed = IdleRotationSpeed
    self.mHorizontalVelocity = Vector3(dropHorizontalVelocity.x, 0, dropHorizontalVelocity.y)
    self.mBounceEaseType = bounceEaseType
    self.mFallEaseType = fallEaseType
    self.mBounceTime = bounceTime
    self.mFallTime = fallTime
    self.mBounceRotateSpeed = bounceRotateSpeed

    self.isShowEff = false
    self.CanCheckEff = false
    self.Enable = true

    local itemId = info.baseID
    local num = info.num

    local cfg = ConfigDataAccessor.Item[itemId]
    --local cfg=LocalDataManager.Instance:GetItemDataCell(itemId);
    local quality = cfg.Quality

    self.id = itemId
    self.num = num
    self.enyityID = info.entityID

    self.quality = quality
    self.transform.position = brustCenter
    self.transform:Rotate(Vector3.up, CS.UnityEngine.Random.Range(0, 360))

    self.name.text = cfg.Name
    self.name.color = GetLootQualityColorRGB(self.quality)
    self.name.gameObject:SetActive(false)

    --加载模型
    local avatarDataCell = ConfigDataAccessor.Avatar[cfg.AvatarID]
    if avatarDataCell ~= nil then
        local modelDataCell = ConfigDataAccessor.Model[avatarDataCell.ModelId]
        if modelDataCell ~= nil then
            self:LoadModel(modelDataCell.ModelsPath)
        end
    end

    self.go_effectRoot:SetActive(false)

    self.IsUsed = true
    self.gameObject:SetActive(true)
end

function LootItemAdv:SetUse(b)
    self.IsUsed = b
    self.gameObject:SetActive(b)
end

--加载模型
function LootItemAdv:LoadModel(path)
    --加载成功之后
    if path ~= "" and self.perfabName ~= path then
        self.perfabName = path

        if self.go_perfab ~= nil then
            GameObject.Destroy(self.go_perfab)
        end

        self.go_perfab = ResourceManager.Instance:LoadGameObject(path)

        self.go_perfab.transform:SetParent(self.tr_MeshRoot)
        self.go_perfab.transform.localScale = Vector3(0.7, 0.7, 0.7)
        self.go_perfab.transform.localPosition = Vector3(0, 0, 0)
    end
end

function LootItemAdv:UpdateLoot()
    if not (self.IsUsed) then
        return
    end

    local timeDelta = Time.fixedDeltaTime
    if self.mState == 0 then
        self.mTimeSinceBorn = self.mTimeSinceBorn + timeDelta
        if self.mTimeSinceBorn > self.mInitDelay then
            self.mState = 1
            self.mTimeSinceBounceStart = 0
        end
    elseif self.mState == 1 then
        local moveOffset = self.mHorizontalVelocity * timeDelta
        self.transform.position = self.transform.position + moveOffset

        self.mTimeSinceBounceStart = self.mTimeSinceBounceStart + timeDelta

        self.mBounceTimeVal = self.mTimeSinceBounceStart / self.mBounceTime
        local posY = CS.SGame_TweenUtils.getEasedVal(self.mInitPosY, self.mHighPosY, self.mBounceEaseType, self.mBounceTimeVal)
        self.transform.position = Vector3(self.transform.position.x, posY, self.transform.position.z)
        self.transform:Rotate(Vector3.up, self.mBounceRotateSpeed * timeDelta)

        if self.mTimeSinceBounceStart > self.mBounceTime then
            self.mState = 2
            self.mTimeSinceFallStart = 0
        end
    elseif self.mState == 2 then
        local moveOffset = self.mHorizontalVelocity * timeDelta
        self.transform.position = self.transform.position + moveOffset

        self.mTimeSinceFallStart = self.mTimeSinceFallStart + timeDelta

        self.mFallTimeVal = self.mTimeSinceFallStart / self.mFallTime
        local posY = CS.SGame_TweenUtils.getEasedVal(self.mHighPosY, self.mInitPosY, self.mFallEaseType, self.mFallTimeVal)
        if posY < self.mInitPosY then
            posY = self.mInitPosY
        end
        self.transform.position = Vector3(self.transform.position.x, posY, self.transform.position.z)

        if self.mTimeSinceFallStart > self.mFallTime then
            self.mState = 3

            --光柱延迟
            DOVirtual.DelayedCall(
                0.15,
                function()
                    self.go_effectRoot:SetActive(true)
                    self.name.gameObject:SetActive(true)
                end
            )

            local time = 1.8
            if self.id == 121 or self.id == 122 then
                time = 0.5
            end
            DOVirtual.DelayedCall(
                time,
                function()
                    self.gameObject:SetActive(false)
                    self.IsUsed = false

                    --产生特效
                    self.OnCreateEff(self.transform.position, self.quality, self.id, self.num, self.enyityID)
                end
            )
        end
    elseif self.mState == 3 then
        self.transform:Rotate(Vector3.up, self.mIdleRotationSpeed * timeDelta)
    end

    --[[
        if self.Enable then
        self.rb.position = self.rb.position + self.velocity * Time.deltaTime

        if self.velocity.y < -4 then
            self.velocity.y = -4
        else
            self.velocity = self.velocity - Vector3.up * self.gravity * Time.deltaTime
        end

        if math.abs(self.rb.position.y - self.startPosition.y) < 0.25 and self.velocity.y < 0 then
            self.rb.useGravity = true
            self.rb.isKinematic = false
            self.rb.velocity = self.velocity + Vector3(0, self.addSpeed, 0)
            self.CanCheckEff = true
            self.Enable = false
        end
    end


    if self.CanCheckEff and not (self.isShowEff) and math.abs(self.rb.velocity.y) < 0.01 then
        self.isShowEff = true

        self.go_effectRoot:SetActive(true)
        --self.name.gameObject:SetActive(true);

        local time = 1.8
        if self.id == 121 or self.id == 122 then
            time = 0.5
        end
        DOVirtual.DelayedCall(
            time,
            function()
                self.gameObject:SetActive(false)
                self.IsUsed = false

                --产生特效
                self.OnCreateEff(self.transform.position, self.quality, self.id, self.num, self.enyityID)
            end
        )
    end
        ]]
end

return LootItemAdv

local AmuletAttr = {}

AmuletAttr.__index = AmuletAttr

function AmuletAttr:New(_go)
    local self = {}
    setmetatable(self, AmuletAttr)

    self.gameObject = _go
    self.transform = _go.transform

    self.name = self.transform:Find("name"):GetComponent(typeof(Text))

    self.point = self.transform:Find("point"):GetComponent(typeof(Text))

    self.go_light = self.transform:Find("star/li").gameObject

    return self --返回自身
end

function AmuletAttr:Refesh(maxnum, cs)
    if cs ~= nil then
        self.Cs = cs

        --显示图标
        local ID = cs.ID --技能id
        local Points = cs.Points --点数分布，二进制文件

        local pList = self:_byte2bin(Points)

        --点的数量
        local num = 0

        for i = 1, maxnum do
            if pList[33 - i] > 0 then
                num = num + 1
            end
        end

        --计算技能等级
        local lv = 1

        local pd = ConfigDataAccessor.GetPassiveDesc(ID, lv)

        self.name.text = pd.SkillName
        if num > 0 then
            self.name.color = Color(1, 0.54, 0.16)
        else
            self.name.color = Color(0.47, 0.47, 0.47)
        end
        self.go_light:SetActive(num > 0)
        --self.point.text = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_point"), {num, maxnum})
        self.point.text = "" .. num .. "/" .. maxnum
    end
end

function AmuletAttr:_byte2bin(n)
    --print("_byte2bin"..n);

    local t = {}
    for i = 32, 0, -1 do
        t[#t + 1] = math.floor(n / 2 ^ i)
        n = n % 2 ^ i
    end

    --print(table.concat(t));

    return t
    --return table.concat(t)
end

return AmuletAttr

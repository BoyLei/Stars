function SecondToDate(sec)
    local date = {}
    date.day = math.floor(sec / 86400)
    date.hours = math.floor(math.fmod(sec, 86400) / 3600)
    date.minutes = math.floor(math.fmod(sec, 3600) / 60)
    date.seconds = math.floor(math.fmod(sec, 60))
    return date
end

function SecondToHMS(sec)
    local date = {}
    date.hours = math.floor(sec / 3600)
    date.minutes = math.floor(math.fmod(sec, 3600) / 60)
    date.seconds = math.floor(math.fmod(sec, 60))
    return date
end

function SecondToFormat(sec)
    local date = SecondToHMS(sec)
    if date.hours > 0 then
        -- return string.format("%2d时", date.hours)
        return string.format("%2d", date.hours) .. LanguageHelp:GetLanguageByKey("Local_Str_Time_hour");
    elseif date.minutes > 0 then
        -- return string.format("%2d分", date.minutes)
        return string.format("%2d", date.minutes) .. LanguageHelp:GetLanguageByKey("Local_Str_Time_minute");
    else
        -- return string.format("%2d秒", date.seconds)
        return string.format("%2d", date.seconds) .. LanguageHelp:GetLanguageByKey("Local_Str_Time_second");
    end
end

-- 把秒数转化为“XX”形式，如果时针分针为0也保持为00
-- 例子
-- SecToStringColon2(3600)  return 1时
-- SecToStringColon2(1600)  return 27分
-- SecToStringColon2(600)  return 10分
-- SecToStringColon3(8)  return 1分
function SecToStringColon4(sec)
    local hasLast = false;
    local days = math.floor(sec / 86400);
    if days > 0 then
        -- return string.format("%.1d天", days);
        return string.format("%.1d", days) .. LanguageHelp:GetLanguageByKey("Local_Str_Time_day");
    end

    local hours = math.floor(sec / 3600);
    if hours > 0 then
        -- return string.format("%.1d时", hours);
        return string.format("%.1d", hours) .. LanguageHelp:GetLanguageByKey("Local_Str_Time_hour");
    end

    local minutes = math.floor(sec / 60);
    if minutes > 0 then
        -- return string.format("%.1d分", minutes);
        return string.format("%.1d", minutes) .. LanguageHelp:GetLanguageByKey("Local_Str_Time_minute");
    end

    local seconds = sec;
    if seconds > 0 then
        -- return string.format("%.1d秒", seconds);
        -- return "1分"
        return "1" .. LanguageHelp:GetLanguageByKey("Local_Str_Time_minute");
    end
end

-- 把秒数转化为“XX:XX:XX”形式，如果时针分针为0也保持为00
-- 例子
-- SecToStringColon2(86401)  return 1天
-- SecToStringColon2(3600)  return 1:00:00
-- SecToStringColon2(1600)  return 26:40
-- SecToStringColon2(600)  return 10:00
-- SecToStringColon2(8)  return 00:08
function SecToStringColon5(sec)
    local tb = {};
    local hasLast = false;
    local days = math.floor(sec / 86400);
    local hours = math.floor(math.fmod(sec, 86400) / 3600);
    local minutes = math.floor(math.fmod(sec, 3600) / 60);
    local seconds = math.floor(math.fmod(sec, 60));
    if days > 1 then
        -- return string.format("%.1d天", days);
        return string.format("%.1d", days) .. LanguageHelp:GetLanguageByKey("Local_Str_Time_day");
    else
        if hours > 0 then
            table.insert(tb, string.format("%.2d:", hours));
        end
        -- if minutes > 0 or hours > 0 then

        -- end
        if seconds > 0 or minutes > 0 or hours > 0 then
            if minutes > 0 then
                table.insert(tb, string.format("%.2d:", minutes));
            end
            table.insert(tb, string.format("%.2d", seconds));
        end
    end

    return table.concat(tb);
end

-- 取相隔最近的星期几的几点的剩余时间（秒）,weekDay表示星期几,time指的是刷新是几点
function CalculationNextFreshTime(weekDay, time)
    if weekDay > 8 then
        return
    end
    local nowTime = TimeManager:GetServerRealTime()
    local nowYear = tonumber(os.date("%Y", nowTime))
    local nowMonth = tonumber(os.date("%m", nowTime))
    local nowDay = tonumber(os.date("%d", nowTime))
    local nowHour = tonumber(os.date("%H", nowTime))
    local resumeDay = 0
    local todayWeekDay = os.date("*t", nowTime).wday - 1
    if weekDay == 8 then
        local refreshDate = os.time({
            day = nowDay,
            month = nowMonth,
            year = nowYear,
            hour = time,
            minute = 0,
            second = 0
        })
        if refreshDate > nowTime then
            return refreshDate - nowTime
        else
            return refreshDate - nowTime + 86400
        end
        return refreshDate
    else
        if todayWeekDay < weekDay then
            resumeDay = weekDay - todayWeekDay
            local refreshDate = os.time({
                day = nowDay + resumeDay,
                month = nowMonth,
                year = nowYear,
                hour = time,
                minute = 0,
                second = 0
            })
            return refreshDate - nowTime
        elseif todayWeekDay == weekDay then
            local refreshDate = os.time({
                day = nowDay,
                month = nowMonth,
                year = nowYear,
                hour = time,
                minute = 0,
                second = 0
            })
            if refreshDate >= nowTime then
                return refreshDate - nowTime
            else
                -- 加上下个星期当天的刷新时间
                return refreshDate - nowTime + 604800
            end
        else
            resumeDay = (weekDay + 7) - todayWeekDay
            local refreshDate = os.time({
                day = nowDay + resumeDay,
                month = nowMonth,
                year = nowYear,
                hour = time,
                minute = 0,
                second = 0
            })
            return refreshDate - nowTime
        end
    end
end

-- 获取次日凌晨时间戳
function GetWeeHourTime(curTimeSec)
    local year = os.date("%Y", curTimeSec)
    local month = os.date("%m", curTimeSec)
    local day = os.date("%d", curTimeSec)
    local time = os.time({
        year = year,
        month = month,
        day = day,
        hour = 23,
        min = 59,
        sec = 59
    })
    return time + 1
end

-- 获取指定天数后的凌晨时间戳
function GetAfterDaysWeeHourTime(curTimeSec, afterDays)
    if afterDays <= 0 then
        return
    end

    local year = os.date("%Y", curTimeSec)
    local month = os.date("%m", curTimeSec)
    local day = os.date("%d", curTimeSec)
    local time = os.time({
        year = year,
        month = month,
        day = day,
        hour = 23,
        min = 59,
        sec = 59
    })
    local addSecond = (afterDays - 1) * 86400 + 1
    return time + addSecond
end

-- 获取指定参数后的时间戳
function GetFutureTimeSec(curTimeSec, afterDays, afterHours, afterMins, afterSecs)
    local addSecond = (afterDays * 86400) + (afterHours * 3600) + (afterMins * 60) + afterSecs
    return curTimeSec + addSecond
end

-- 获得服务器刷新天数的剩余时间
function GetLeftTimeSecond(serverTimestap, addSecond)
    local stamp = math.floor(serverTimestap / 1000);
    local nextDayStamp = GetWeeHourTime(stamp);

    local leftTimeToNextDay = nextDayStamp - stamp + addSecond;

    return leftTimeToNextDay;
end

-- 获得服务器刷新天数的剩余时间
function GetLeftTimeToRefesh(serverTimestap, addSecond)
    local leftSecond = GetLeftTimeSecond(serverTimestap, addSecond);

    return SecToStringColon2(leftSecond);
end

-- 流失的时间转换秒
function LossTimeSecToString(sec)
    if sec < 60 then
        -- return math.floor(sec) .. "秒前"
        return math.floor(sec) .. LanguageHelp:GetLanguageByKey("Local_Str_Time_second") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");
    end

    if sec >= 60 and sec < 3600 then
        local minutes = math.floor(sec / 60)
        -- return minutes .. "分钟前"
        return minutes .. LanguageHelp:GetLanguageByKey("Local_Str_Time_minute") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");

    end

    if sec >= 86400 then
        local date = math.floor(sec / 86400)
        -- return date .. "天前"
        return date .. LanguageHelp:GetLanguageByKey("Local_Str_Time_day") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");

    end

    if sec >= 3600 and sec < 86400 then
        local hour = math.floor(sec / 3600)
        -- return hour .. "小时前"
        return hour .. LanguageHelp:GetLanguageByKey("Local_Str_Time_hour") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");

    end
end

function LossTimeSecToString2(diffValue)
    if diffValue < 0 then
        return "";
    end
    local minute = 60;
    local hour = minute * 60;
    local day = hour * 24;
    local month = day * 30;
    local year = day * 12;

    local minC = math.ceil(diffValue / minute); -- 分
    local hourC = math.floor(diffValue / hour); -- 时
    local dayC = math.floor(diffValue / day); -- 天
    local weekC = math.floor(diffValue / (7 * day)); -- 周
    local monthC = math.floor(diffValue / month); -- 月
    local yearC = math.floor(diffValue / year); -- 年

    if yearC >= 1 then
        -- return yearC .. "年前";
        return yearC .. LanguageHelp:GetLanguageByKey("Local_Str_Time_year") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");

    elseif monthC >= 1 then
        -- return monthC .. "月前";
        return monthC .. LanguageHelp:GetLanguageByKey("Local_Str_Time_month") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");

    elseif weekC >= 1 then
        -- return weekC .. "周前";
        return weekC .. LanguageHelp:GetLanguageByKey("Local_Str_Time_week") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");

    elseif dayC >= 1 then
        -- return dayC .. "天前";
        return dayC .. LanguageHelp:GetLanguageByKey("Local_Str_Time_day") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");

    elseif hourC >= 1 then
        -- return hourC .. "小时前";
        return hourC .. LanguageHelp:GetLanguageByKey("Local_Str_Time_hour") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");

    elseif minC >= 1 then
        -- return minC .. "分钟前";
        return minC .. LanguageHelp:GetLanguageByKey("Local_Str_Time_minute") ..
                   LanguageHelp:GetLanguageByKey("Local_Str_Time_ago");
    else
        return "";
    end
end

    end

    tb.year = tonumber(os.date("%Y", unixTime))
    tb.month = tonumber(os.date("%m", unixTime))
    tb.day = tonumber(os.date("%d", unixTime))
    tb.hour = tonumber(os.date("%H", unixTime))
    tb.minute = tonumber(os.date("%M", unixTime))
    tb.second = tonumber(os.date("%S", unixTime))
    return tb
end

        return LanguageHelp:GetLanguageByKey("Local_Str_Sunday");
    elseif weekday == 1 then
        return LanguageHelp:GetLanguageByKey("Local_Str_Monday");
    elseif weekday == 2 then
        return LanguageHelp:GetLanguageByKey("Local_Str_Tuesday");
    elseif weekday == 3 then
        return LanguageHelp:GetLanguageByKey("Local_Str_Wednesday");
    elseif weekday == 4 then
        return LanguageHelp:GetLanguageByKey("Local_Str_Thursday");
    elseif weekday == 5 then
        return LanguageHelp:GetLanguageByKey("Local_Str_Friday");

function MT(KeyMap, table)
    local base = {
        __index = function(table, key)
            local keyIndex = KeyMap[key]
            if not keyIndex then
                print("Key not found: ", key)
                return nil
            end
            return table[keyIndex]
        end
        --[[
        __newindex = function()
            local keyIndex = KeyMap[key]
            if not keyIndex then
                print("Key not found: ", key)
            end
            table[keyIndex] = val
            error(""Attempt to modify read-only table"")
        end
]]
        --
    }
    for k, v in pairs(table) do
        setmetatable(v, base)
    end
    base.__metatable = false
    return base
end

function print_r(root)
    local cache = {
        [root] = "."
    }
    local function _dump(t, space, name)
        local temp = {}
        for k, v in pairs(t) do
            local key = tostring(k)
            if cache[v] then
                tinsert(temp, "+" .. key .. " {" .. cache[v] .. "}")
            elseif type(v) == "table" then
                local new_key = name .. "." .. key
                cache[v] = new_key
                tinsert(temp, "+" .. key .. _dump(v, space .. (next(t, k) and "|" or " ") .. srep(" ", #key), new_key))
            else
                tinsert(temp, "+" .. key .. " [" .. tostring(v) .. "]")
            end
        end
        return tconcat(temp, "\n" .. space)
    end
    print(_dump(root, "", ""))
end

function logError(...)
    local t = {}

        table.insert(t, "\t")

    table.insert(t, debug.traceback(""))
    CS.UnityEngine.Debug.LogError("error" .. table.concat(t))
end

--[[
解析 ”a,b; c,d;e,f"格式的字符串, 结果是
{
{[1], [2], [3], ...},
{[1], [2], [3], ...},
...
}
]]
function SplitString(str, paramCount)
    paramCount = paramCount or 2
    local result = {}
    local arr = string.split(str, ";")
    for _, v in pairs(arr) do
        local tmp = string.split(v, ",")
        if tmp then
            local info = {}
            for i = 1, paramCount do
                if nil ~= tmp[i] then
                    table.insert(info, tmp[i])
                end
            end
            table.insert(result, info)
        end
    end
    return result
end

function string.split(str, delimiter)
    str = tostring(str)
    delimiter = tostring(delimiter)
    if (delimiter == "") then
        return false
    end
    local pos, arr = 0, {}
    -- for each divider found
    for st, sp in function()
        return string.find(str, delimiter, pos, true)
    end do
        table.insert(arr, string.sub(str, pos, st - 1))
        pos = sp + 1
    end
    table.insert(arr, string.sub(str, pos))
    return arr
end

function io.exists(path)
    local file = io.open(path, "r")
    if file then
        io.close(file)
        return true
    end
    return false
end

function io.readfile(path)
    local file = io.open(path, "r")
    if file then
        local content = file:read("*a")
        io.close(file)
        return content
    end
    return nil
end

function io.writefile(path, content, mode)
    mode = mode or "w+b"
    local file = io.open(path, mode)
    if file then
        if file:write(content) == nil then
            return false
        end
        io.close(file)
        return true
    else
        return false
    end
end

function io.pathinfo(path)
    local pos = string.len(path)
    local extpos = pos + 1
    while pos > 0 do
        local b = string.byte(path, pos)
        if b == 46 then -- 46 = char "."
            extpos = pos
        elseif b == 47 then -- 47 = char "/"
            break
        end
        pos = pos - 1
    end

    local dirname = string.sub(path, 1, pos)
    local filename = string.sub(path, pos + 1)
    extpos = extpos - pos
    local basename = string.sub(filename, 1, extpos - 1)
    local extname = string.sub(filename, extpos)
    return {
        dirname = dirname,
        filename = filename,
        basename = basename,
        extname = extname
    }
end

function io.filesize(path)
    local size = false
    local file = io.open(path, "r")
    if file then
        local current = file:seek()
        size = file:seek("end")
        file:seek("set", current)
        io.close(file)
    end
    return size
end

local function urlencodechar(char)
    return "%" .. string.format("%02X", string.byte(char))
end
function string.urlencode(input)
    -- convert line endings
    input = string.gsub(tostring(input), "\n", "\r\n")
    -- escape all characters but alphanumeric, '.' and '-'
    input = string.gsub(input, "([^%w%.%- ])", urlencodechar)
    -- convert spaces to "+" symbols
    return string.gsub(input, " ", "+")
end

function string.urldecode(input)
    input = string.gsub(input, "+", " ")
    input =
        string.gsub(
        input,
        "%%(%x%x)",
        function(h)
            return string.char(CheckNumber(h, 16))
        end
    )
    input = string.gsub(input, "\r\n", "\n")
    return input
end

-- 计算 UTF8 字符串的长度，每一个中文算一个字符
function string.utf8len(input)
    local len = string.len(input)
    local left = len
    local cnt = 0
    local arr = {0, 0xc0, 0xe0, 0xf0, 0xf8, 0xfc}
    while left ~= 0 do
        local tmp = string.byte(input, -left)
        local i = #arr
        while arr[i] do
            if tmp >= arr[i] then
                left = left - i
                break
            end
            i = i - 1
        end
        cnt = cnt + 1
    end
    return cnt
end

function string.formatnumberthousands(num)
    local formatted = tostring(CheckNumber(num))
    local k
    while true do
        formatted, k = string.gsub(formatted, "^(-?%d+)(%d%d%d)", "%1,%2")
        if k == 0 then
            break
        end
    end
    return formatted
end

function ConvertHexToColor(hex)
    hex = string.gsub(hex, "#", "")
    local color = {}
    for i = 1, 3 do
        local num
        local tmp = string.sub(hex, i * 2 - 1, i * 2)
        num = tonumber(tmp, 16)
        table.insert(color, num)
    end
    return color
end

function ConvertHexToColorWithAlpha(hex, alpha)
    hex = string.gsub(hex, "#", "")
    local color = {}
    for i = 1, 3 do
        local num
        local tmp = string.sub(hex, i * 2 - 1, i * 2)
        num = tonumber(tmp, 16)
        table.insert(color, num)
    end
    return Color(color[1] / 255, color[2] / 255, color[3] / 255, alpha)
end

    for i = 1, #list2 do

end

function GetInt64High2Low8(value)
    -- 取 最高的 8位
    local v = ((value >> 40) & 0xff)
    if v > 0 then
        return v
    end
    -- 取 最高的 8位
    local v = ((value >> 32) & 0xff)
    if v > 0 then
        return v
    end
    -- 取 最高的 8位
    local v = ((value >> 24) & 0xff)
    if v > 0 then
        return v
    end
    -- 取 不到 最高 8位, 那就取高 8-16 位
    v = ((value >> 16) & 0xff)
    if v > 0 then
        return v
    end
    -- 取 低8-16 位
    v = ((value >> 8) & 0xff)
    if v > 0 then
        return v
    end
    -- 取低 8 位
    v = (value & 0xff)
    return v
end

-- 计算 UTF8 字符串的长度，每一个中文算一个字符
function utf8len(input)
    local len = string.len(input)
    local left = len
    local cnt = 0
    local arr = {0, 0xc0, 0xe0, 0xf0, 0xf8, 0xfc}
    while left ~= 0 do
        local tmp = string.byte(input, -left)
        local i = #arr
        while arr[i] do
            if tmp >= arr[i] then
                left = left - i
                break
            end
            i = i - 1
        end
        cnt = cnt + 1
    end
    return cnt
end

-- 保留N位小数点
function roundToDecimal(number, decimalPlaces)
    local decimalOffset = 10 ^ decimalPlaces
    return math.floor(number * decimalOffset) / decimalOffset
end

function roundToDecimal2(number, decimalPlaces)
    local decimalOffset = 10 ^ decimalPlaces
    local num = math.floor(number * decimalOffset) / decimalOffset
    return FormatNum(num)
end

---如果小数位数为0，则只保留整数
function FormatNum(num)
    if num <= 0 then
        return 0
    else
        local t1, t2 = math.modf(num)
        ---小数如果为0，则去掉
        if t2 > 0 then
            return num
        else
            return t1
        end
    end
end

function AddMapData(mapData, key, value)
    if mapData[key] ~= nil then
        UpdateMapData(mapData, key, value)
        return
    end

    mapData[key] = value
    mapData.Count = mapData.Count + 1
end

function UpdateMapData(mapData, key, value)
    -- 如果 value 为nil, 那就删除这个元素
    if value == nil then
        RemoveMapDataKey(mapData, key)
        return
    end

    -- 不为空, 就去判断 是否存在这个 key, 如果都不存在,那就是 添加这个属性值
    if mapData[key] == nil then
        AddMapData(mapData, key, value)
        return
    end

    -- 如果 属性key 存在, 就去更新 它的值
    mapData[key] = value
end

function RemoveMapDataKey(mapData, key)
    -- 如果本来 就不存在这个 属性 值, 那就直接 return
    if mapData[key] == nil then
        return
    end

    -- 删除这个 元素的 值, 但是需要注意, 由于 key 的不确定性，所以 只能通过 设置nil 来约定 nil 为元素删除
    -- table.remove 必须 传入元素的 整数位置
    mapData[key] = nil
    mapData.Count = mapData.Count - 1
    mapData.Count = mapData.Count < 0 and 0 or mapData.Count
end

function GetCharStrLen(str)
    local bytes = {str:byte(1, #str)}
    local length, begin = 0, false
    for _, byte in ipairs(bytes) do
        if byte < 128 or byte >= 192 then
            begin = false
            length = length + 1
        elseif not begin then
            begin = true
            length = length + 1
        end
    end
    return length

local strLocked = ""
function CheckPlayerGuidanceOpen(playerGuidance, playName, needMsg, needLockedStr)
    local cfg = ConfigDataAccessor.GuidSysOpen[playerGuidance]

    if cfg ~= nil then
        -- 玩家等级
        local playerLevel = GameManager.Instance:GetPlayerLevel()
        if cfg.NeedLv > playerLevel then
            if needMsg or needLockedStr then
                if playName == "" then
                    strLocked = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Tips64"), {cfg.NeedLv})
                else
                    strLocked = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Tips65"), {cfg.NeedLv, playName})
                end
                if needMsg then
                    Util.ShowMessage(strLocked)
                end
            end
            return false, strLocked
        end

        -- 冒险等级
        local mainPlayerBase = GameManager.Instance.M_MainPlayerCtrlBase
        if mainPlayerBase ~= nil then
            local RiskInfo = GameManager.Instance.M_MainPlayerCtrlBase.Data.Attrs:GetProtoValue(CS.StarProjectDef.AOIAttrDefine.RiskInfo)
            if RiskInfo ~= nil then
                local curRiskLevel = RiskInfo.CurRiskLevel
                if cfg.NeedAdvLv > curRiskLevel then
                    if needMsg or needLockedStr then
                        local cfg = ConfigDataAccessor.AdvGradeExp[cfg.NeedAdvLv]

                        if cfg == nil then
                            return false, ""
                        end

                        if playName == "" then
                            -- Util.ShowMessage("冒险等级达到"..cfg.AdvClass..GetRomaNum(cfg.AdvOrder).."开放");
                            strLocked = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Tips66"), {cfg.AdvClass .. GetRomaNum(cfg.AdvOrder)})
                        else
                            -- Util.ShowMessage("冒险等级达到"..cfg.AdvClass..GetRomaNum(cfg.AdvOrder).."后开放"..playName.."玩法");
                            strLocked = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Tips67"), {cfg.AdvClass .. GetRomaNum(cfg.AdvOrder), playName})
                        end

                        if needMsg then
                            Util.ShowMessage(strLocked)
                        end
                    end
                    return false, strLocked
                end
            end
        end

        -- 任务完成情况
        if cfg.NeedQuestCom > 0 then
            if not TaskHelper.IsTaskFinsh(cfg.NeedQuestCom) then
                if needMsg or needLockedStr then
                    local task = TaskHelper.GetTaskConfig(cfg.NeedQuestCom)
                    if task ~= nil then
                        local taskName = task.Base.TaskName
                        local taskType = TaskHelper.TaskTypeToName(task.Base.TaskType)
                        -- Util.ShowMessage("完成"..taskType.."任务“"..taskName.."”后开放");
                        strLocked = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Tips68"), {taskType, taskName})

                        if needMsg then
                            Util.ShowMessage(strLocked)
                        end
                    end
                end
                return false, strLocked
            end
        end
    end

    return true, strLocked
end

-- 通过跳转表枚举来跳转
function JumpByJumpType(itemGetID, itemBaseID, fromModule, fromArgs, tr)
    local cfg = ConfigDataAccessor.ItemGetJump[itemGetID]

    if cfg == nil then
        return
    end

    -- 判断是否系统开启
    local sysOpen = cfg.SysOpen
    if sysOpen > 0 then
        if not (CheckPlayerGuidanceOpen(sysOpen, "", true)) then
            -- 不满足开放条件
            return
        end
    end

    -- 跳转枚举
    local enumName = cfg.WayString

    if enumName == "Buy_Trade" then -- 1001
        -- "系统-交易行购买"

        JumpBySystemJumpEnum(enumName, fromModule, fromArgs, itemBaseID)
    elseif enumName == "Sys_Activity_PartnerTarget" then -- 1014
        -- "伙伴目标"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_Bag_All" then -- 1014
        -- "背包全部"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_Power" then -- 1020
        -- 前往战力
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_Bag_Item" then -- 1015
        -- "背包道具"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_Bag_Equip" then -- 1016
        -- "背包装备"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_EquipRecast" then
        -- "装备重构"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_Bag_Partner" then -- 1017
        -- "背包伙伴"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_Bag_Amulet" then -- 1018
        -- "背包纹章"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_Bag_Other" then -- 1019
        -- "背包其他"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_Gacha" then -- 1002
        -- "系统-启梦获取"
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
        LuaModuleDispatcher:SendMessage(ModuleDef.DrawCardModule, "CloseTenTimesWindow", {})
        MutiScenesMergeManager.Instance:UnloadScene()
    elseif enumName == "Sys_Item_Resolve" then
        -- 分解
        JumpBySystemJumpEnum(enumName, fromModule, fromArgs)
    elseif enumName == "Sys_WorldLine" then

-- 判断玩法今天是否开启
function CheckRecyclePlayIsOpen(playID)
    --[[
	1.搜索出当天的
	2.按照时间排序
	3.要开的放在左
	4.最后面补三个空的
	]]
    local tt = GameManager.Instance:GetServerTimeStamp()
    local serverNowTime = math.floor(tt / 1000)

    local v = ConfigDataAccessor.RecyclePlay[playID]

    local DateList = v.DateList -- 开放日期，暂时不知道干啥(暂时不管)

    local LoopType = v.LoopType -- 开放循环类型 0 不循环(暂时不管) 1 日循环 2 周循环 3 半月循环

    local LoopParam = v.LoopParam -- 循环时间参数  0/1 不填； 2 填对应的周几；周日需填0；3 填半月对应的第几天；

    -- 周循环
    if LoopType == 2 then
        -- 半月循环
        -- 获得当前是周几
        local weekDay = GetWeekDay(serverNowTime)
        for i = 1, #LoopParam do
            if weekDay == LoopParam[i] then
                return true
            end
        end
    elseif LoopType == 3 then
        -- 日循环
        -- 获得当前是几号
        local day = GetDay(serverNowTime)

        for i = 1, #LoopParam do
            if day == LoopParam[i] or day + 15 == LoopParam[i] then
                return true
            end
        end
    elseif LoopType == 1 then
        return true
    end

    return false
end

--判断是否是今天
function CheckIsToday(timestamp)
    local System_DailyReset = ConfigDataAccessor.SystemList[63].Value --0点之后的多少秒

    local tt = GameManager.Instance:GetServerTimeStamp()
    local serverNowTime = math.floor(tt / 1000) - System_DailyReset
    local serverDay = GetDay(serverNowTime)
    local serverWeekDay = GetWeekDay(serverNowTime)

    local timeDay = GetDay(timestamp - System_DailyReset)
    local timeWeekDay = GetWeekDay(timestamp)

    if serverDay == timeDay and serverWeekDay == timeWeekDay then
        return true
    end

    return false
end

-- 获取几号
function GetDay(timestamp)
    local day = tonumber(os.date("%d", timestamp))

    return day
end

-- 获取周几
function GetWeekDay(timestamp)
    local weekday = tonumber(os.date("%w", timestamp))

    if weekday == 0 then
        weekday = 7
    end

    return weekday
end

-- 创建单个本地实体
-- realDropData 奖励数据 type(DropNotify) 必传
-- boxType 宝箱类型 E_LocalEntityType.XXX 必传
-- boxTypeSource 宝箱创建来源 E_LocalEntitySource.XXX 必传
-- pos 宝箱创建位置，默认为玩家当前位置
-- boxId 宝箱id, 默认为9999991
-- isAllOpen 宝箱是否全部打开，默认为true
-- canOpen 宝箱是否可以被打开，默认为可打开
-- openCallBackFunc 点击开启宝箱交互后的执行的callBackFunc<bool> 无需求可不传
function CreateSingleLocalEntity(realDropData, boxType, boxTypeSource, pos, boxId, isAllOpen, canOpen, openCallBackFunc)
    if realDropData == nil or realDropData.Infos == nil or realDropData.Infos.Count <= 0 then
        return
    end
    if boxType == nil then
        return
    end
    if boxTypeSource == nil then
        return
    end

    if pos == nil then
        pos = GameManager.Instance:GetEntityPosById(GameManager.Instance.mainPlayerId)
    end
    if canOpen == nil then
        canOpen = true
    end
    if isAllOpen == nil then
        isAllOpen = true
    end
    if boxId == nil then
        if boxTypeSource == E_LocalEntitySource.WildBoss then
            boxId = 9999998 --宝箱高
        else
            boxId = 9999991
        end
    end

    -- 构建奖励数据
    local localDropData = LocalDropList()
    for i = 0, realDropData.Infos.Count - 1 do
        local data = realDropData.Infos[i]
        localDropData:AddData(data.Itemid, data.ItemNum, data.EntityID)
    end
    localDropData.CanOpen = canOpen
    localDropData.OpenCallBackFunc = openCallBackFunc

    pos = BusinessManager.Instance:GetRandomPos(pos, 2.5)
    CS.StarProject.GlobalEvent.onCreateLocalEntity:Invoke(boxType, boxId, boxTypeSource, pos, localDropData, isAllOpen, -1)
end

-- 通过道具组ID筛选道具
function GetItemEntityIDsByItemSet(itemSetId)
    local allEntityIds = {}

    local itemSetCfg = ConfigDataAccessor.DonateItemSet[itemSetId]

    for i = 1, #itemSetCfg.ItemType do
        local itemType = itemSetCfg.ItemType[i]

        local datas = nil

        if #itemSetCfg.SubType == 0 then
            datas = BusinessManager.Instance:GetItemsByItemType(itemType, 0)
        else
            for j = 1, #itemSetCfg.SubType do
                local itemSubType = itemSetCfg.SubType[j]
                if datas == nil then
                    datas = BusinessManager.Instance:GetItemsByItemType(itemType, itemSubType)
                else
                    local dd = BusinessManager.Instance:GetItemsByItemType(itemType, itemSubType)
                    for p = 0, dd.Count - 1 do
                        datas:Add(dd[p])
                    end
                end
            end
        end

        for j = datas.Count - 1, 0, -1 do
            local itemMD = datas[j]
            local itemCfg = ConfigDataAccessor.Item[itemMD.BaseID]

            local isRemoved = false

            -- 品质
            local isPassQuality = false
            if #itemSetCfg.Quality == 0 or itemSetCfg.Quality[1] == 0 then
                isPassQuality = true
            else
                for k = 1, #itemSetCfg.Quality do
                    if itemCfg.Quality == itemSetCfg.Quality[k] then
                        isPassQuality = true
                        break
                    end
                end
            end

            if not (isPassQuality) then
                datas:RemoveAt(j)
                isRemoved = true
            end

            -- 其他继续需要筛选的
        end

        if itemType == 2 then
            -- 玩家装备

            for j = datas.Count - 1, 0, -1 do
                local itemMD = datas[j]
                local equipCfg = ConfigDataAccessor.Equip[itemMD.BaseID]

                local isRemoved = false

                -- 装备等级
                local isPassLevel = false
                if #itemSetCfg.Level == 0 or itemSetCfg.Level[1] == 0 then
                    isPassLevel = true
                else
                    for k = 1, #itemSetCfg.Level do
                        if equipCfg.Equip_Level == itemSetCfg.Level[k] then
                            isPassLevel = true
                            break
                        end
                    end
                end

                if not (isPassLevel) then
                    datas:RemoveAt(j)
                    isRemoved = true
                end

                -- 穿戴职业
                if not (isRemoved) then
                    local aimJob = equipCfg.JobBaseType

                    local isPassJobType = false
                    if #itemSetCfg.JobType == 0 or aimJob == 0 or itemSetCfg.JobType[1] == 0 then
                        isPassJobType = true
                    else
                        for k = 1, #itemSetCfg.JobType do
                            if aimJob == itemSetCfg.JobType[k] then
                                isPassJobType = true
                                break
                            end
                        end
                    end

                    if not (isPassJobType) then
                        datas:RemoveAt(j)
                        isRemoved = true
                    end
                end
            end
        elseif itemType == 5 then
            -- 纹章

            for j = datas.Count - 1, 0, -1 do
                local itemMD = datas[j]
                local amuletEquipCfg = ConfigDataAccessor.AmuletEquip[itemMD.BaseID]

                local isRemoved = false

                -- 装备等级
                local isPassLevel = false
                if #itemSetCfg.Level == 0 or itemSetCfg.Level[1] == 0 then
                    isPassLevel = true
                else
                    for k = 1, #itemSetCfg.Level do
                        if amuletEquipCfg.Level == itemSetCfg.Level[k] then
                            isPassLevel = true
                            break
                        end
                    end
                end

                if not (isPassLevel) then
                    datas:RemoveAt(j)
                    isRemoved = true
                end

                -- 纹章阶数
                if not (isRemoved) then
                    local isPassEquipLevel = false
                    if #itemSetCfg.EquipLevel == 0 or itemSetCfg.EquipLevel[1] == 0 then
                        isPassEquipLevel = true
                    else
                        for k = 1, #itemSetCfg.EquipLevel do
                            if amuletEquipCfg.EquipLevel == itemSetCfg.EquipLevel[k] then
                                isPassEquipLevel = true
                                break
                            end
                        end
                    end

                    if not (isPassEquipLevel) then
                        datas:RemoveAt(j)
                        isRemoved = true
                    end
                end
            end
        end

        for j = datas.Count - 1, 0, -1 do
            -- 排除玩家身上的
            if datas[j].SpaceId ~= 1 and datas[j].SpaceId ~= 13 and datas[j].SpaceId ~= 20 then
                table.insert(allEntityIds, datas[j].EntityID)
            end
        end
    end

    return allEntityIds
end

-- 刷新gameObject对象数量
function RefreshGoCachePool(prefabGo, parentRoot, showCount)
    local curCount = parentRoot.childCount
    -- 需要创建
    if (showCount > curCount) then
        local needAddCount = showCount - curCount
        for i = 1, needAddCount do
            local item = GameObject.Instantiate(prefabGo, parentRoot)
            item:SetActive(true)
        end
    else
        -- 不需要显示的时候，防止prefab不在root子节点导致显隐异常
        if showCount == 0 then
            prefabGo:SetActive(false)
        end
        -- 所需的显示 多余的隐藏
        for i = 0, curCount - 1 do
            local item = parentRoot.transform:GetChild(i)
            if (i >= showCount) then
                item.gameObject:SetActive(false)
            else
                item.gameObject:SetActive(true)
            end
        end
    end
end

function GetTaskState(taskID)
    local state = 0

    -- 任务状态
    if TaskHelper.ContainTask(taskID) then
        -- 正在进行
        local TaskInfo = TaskHelper.GetRunningTask(taskID)
        if TaskInfo ~= nil then
            if TaskInfo.TaskState == CS.ProtoMsg.TaskStateEnum.TaskStageRunning then
                -- 任务进行中
                state = 1
            elseif TaskInfo.TaskState == CS.ProtoMsg.TaskStateEnum.TaskStageFinished then
                -- 任务已完成未提交
                state = 2
            end
        end
    elseif TaskHelper.IsTaskFinsh(taskID) then
        -- 已经完成
        state = 2
    else
        -- 未接取
        state = 0
    end

    return state
end

function GetShowItemIDByJob(RewardMarks)
    if RewardMarks == nil then
        return 0
    end

    if #RewardMarks <= 1 then
        return 0
    end

    -- 如果是全职业就不用读取其他职业的
    if RewardMarks[1] == 0 then
        return RewardMarks[2]
    end

    local baseJob = GameManager.Instance:GetPlayJobBaseID()

    local idx = 1
    local tempJob = RewardMarks[idx]

    -- 如果找到职业的就返回职业的
    while tempJob ~= nil do
        if tempJob == baseJob then
            if RewardMarks[idx + 1] == nil then
                return 0
            end
            return RewardMarks[idx + 1]
        end
        idx = idx + 2
        tempJob = RewardMarks[idx]
    end

    -- 什么都没找到就不显示
    return 0
end

function DeepCopyTable(orig)
    local copy
    if type(orig) == "table" then
        copy = {}
        for orig_key, orig_value in next, orig, nil do
            copy[DeepCopyTable(orig_key)] = DeepCopyTable(orig_value)
        end
        setmetatable(copy, DeepCopyTable(getmetatable(orig)))
    else -- 值类型直接复制
        copy = orig
    end
    return copy
end

    end
    SlotResonateLvMax = minLv
    -- 当前最大平均等级
    if SlotResonateLv > 0 then
        -- SlotResonateLvMax = ConfigDataAccessor.EquipSlotUpLimit[SlotResonateLv].Max;
    else
        --SlotResonateLvMax = 0
    end

    -- 当前最大等级
    local SlotMaxLv = ConfigDataAccessor.EquipSlotUpLimit[1].Max
    for i = 2, #ConfigDataAccessor.EquipSlotUpLimit do
        local equipSlotUpLimit = ConfigDataAccessor.EquipSlotUpLimit[i]
        if equipSlotUpLimit.Condition > 0 then
            --[[
            --冒险等级
            local RiskInfo = GameManager.Instance.M_MainPlayerCtrlBase.Data.Attrs:GetProtoValue(CS.StarProjectDef.AOIAttrDefine.RiskInfo)
            if RiskInfo ~= nil then
                local curRiskLevel = RiskInfo.CurRiskLevel
                if equipSlotUpLimit.Condition > curRiskLevel then
                    -- 不满足
                    break
                end
            end
            ]]
            --玩家等级
            local playerLv = GameManager.Instance:GetPlayerLevel()
            if equipSlotUpLimit.Condition > playerLv then
                -- 不满足
                break
            end
        end
        if equipSlotUpLimit.ConditionNum > SlotResonateLvMax then
            -- 不满足
            break
        end
        SlotMaxLv = equipSlotUpLimit.Max
    end

    local isNeedCalculate = true

    if SlotCfgMaxLv == UpLv then
        -- 已经不需要强化了
        isNeedCalculate = false
    end
    if SlotMaxLv == UpLv then
        -- 已经不需要强化了
        isNeedCalculate = false
    end

    local IsMoneyEnough = true
    if isNeedCalculate then
        -- 强化需要的资源
        local expend = ConfigDataAccessor.GetEquipSlotUpExpend(SlotID, UpLv + 1)

        -- 道具消耗（货币）暂时不用，但是先不删
        local cfgM = ConfigDataAccessor.Item[expend.Money[1]]
        local haveMoneyNum = BusinessManager.Instance:GetCurrencyNum(expend.Money[1])
        if haveMoneyNum < expend.Money[2] then
            IsMoneyEnough = false
        end

        -- 消耗道具
        for i = 1, #expend.Item_id do
            local haveMoneyNum2 = BusinessManager.Instance:GetCurrencyNum(expend.Item_id[i])
            if haveMoneyNum2 < expend.Item_num[i] then
                IsMoneyEnough = false
                break
            end
        end
    else
        -- 显示强化到最高的UI
        IsMoneyEnough = false
    end

    return IsMoneyEnough
end

-- 检查装备洗练红点
function CheckEquipRefineRedPoint(SlotID)
    local itemData = BusinessManager.Instance:GetHeroEquipedEquip(SlotID)

    local isAllPerfect = true -- 是否完美度全部满了？

    local perfectAllValue = 0

    local eCfg = ConfigDataAccessor.Equip[itemData.BaseID]
    -- 主属性
    local MainProps = itemData.EquipProps.MainPropList

    for i = 0, MainProps.Count - 1 do
        local maxAttr = eCfg.NumMax[i + 1]
        perfectAllValue = perfectAllValue + (MainProps[i].Value / maxAttr)
    end

    -- 这里要计算平均完美度（不完美的话上面就变成false）
    local Perfection = math.floor(perfectAllValue / MainProps.Count * 100)

    isAllPerfect = (Perfection == 100)

    local IsMoneyEnough = true

    if not (isAllPerfect) then
        -- 强化需要的资源
        local expend = ConfigDataAccessor.GetEquipResetExpend(SlotID, eCfg.Equip_Level)

        -- 道具消耗（货币）暂时不用，但是先不删
        local cfgM = ConfigDataAccessor.Item[expend.Money[1]]
        local haveMoneyNum = BusinessManager.Instance:GetCurrencyNum(expend.Money[1])
        if haveMoneyNum < (expend.Money[2]) then
            IsMoneyEnough = false
        end

        -- 消耗道具
        for i = 1, #expend.Item_id do
            local haveMoneyNum2 = BusinessManager.Instance:GetCurrencyNum(expend.Item_id[i])
            if haveMoneyNum2 < expend.Item_num[i] then
                IsMoneyEnough = false
                break
            end
        end
    else
        IsMoneyEnough = false
    end

    return IsMoneyEnough
end

function FormatShowTime(timeInSec)
    local day = 0
    local hour = 0
    local min = 0
    if (timeInSec / 86400) > 0 then
        day = math.floor(timeInSec / 86400)
    --_text = showTime.ToString() + "天";
    end

    timeInSec = timeInSec - 86400 * day
    if (timeInSec / 3600) > 0 then
        hour = math.floor(timeInSec / 3600)
    --_text = showTime.ToString() + "小时";
    end

    timeInSec = timeInSec - 3600 * hour

    if (timeInSec / 60) > 0 then
        min = math.ceil(timeInSec / 60)
    --_text = showTime.ToString() + "分钟";
    end

    --return string.format("%d天%d小时%d分钟", day, hour, min);
    return StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Time"), {day, hour, min})
end

LanguageHelp = class("LanguageHelp")

function LanguageHelp:ctor()
    self.LanguageDic = {};
end

function LanguageHelp:Clear()
    self.LanguageDic = {};
end

function LanguageHelp:GetLanguageByKey(key)
    if key == "" or key == nil or type(key) ~= "string" then
        return "";
    end
    local str = self.LanguageDic[key];
    if str == nil then
        str = LanguageManager.Instance:GetLanguageByKey(key);
        if str ~= "" then
            self.LanguageDic[key] = str;
        end
    end
    return str;
end

return LanguageHelp

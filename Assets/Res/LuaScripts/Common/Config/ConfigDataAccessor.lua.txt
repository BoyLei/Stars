function GetTaskReward(taskID)
    local taskReward = ConfigDataAccessor.TaskReward[taskID]

    local Awards = {}

    for i = 1, #taskReward.Reward, 2 do
        local id = taskReward.Reward[i]
        local num = taskReward.Reward[i + 1]
        if num ~= nil then
            if Awards[id] == nil then
                Awards[id] = num
            else
                Awards[id] = Awards[id] + num
            end
        end
    end

    for i = 1, #taskReward.RewardHid, 2 do
        local id = taskReward.RewardHid[i]
        local num = taskReward.RewardHid[i + 1]
        if num ~= nil then
            if Awards[id] == nil then
                Awards[id] = num
            else
                Awards[id] = Awards[id] + num
            end
        end
    end

    return Awards
end

function InitNpcDialogGroup()
    local data = require "Common.Config.NpcDialogGroup"

    -- NPC随机开场白
    ConfigDataAccessor.NpcDialogGroup = {}

    for k, v in pairs(data) do
        if ConfigDataAccessor.NpcDialogGroup[v.GroupID] == nil then
            ConfigDataAccessor.NpcDialogGroup[v.GroupID] = {}
        end
        table.insert(ConfigDataAccessor.NpcDialogGroup[v.GroupID], v)
    end
end

function GetNpcDialogIdByGroupID(groupID)
    local group = ConfigDataAccessor.NpcDialogGroup[groupID]

    if group == nil then
        return 0
    end

    local leftDatas = {}
    for i = 1, #group do
        local data = group[i]
        if data.DelCondition > 0 then
            local isMet = GameManager.Instance:IsConditionMete(data.DelCondition) -- 判断是否满足
            -- 不满足删除的才加进去
            if not (isMet) then
                table.insert(leftDatas, data)
            end
        else
            if data.GroupID > 0 then
                local isMet = GameManager.Instance:IsConditionMete(data.OpenCondition) -- 判断是否满足
                -- 满足删除的才加进去
                if isMet then
                    table.insert(leftDatas, data)
                end
            end
        end
    end

    if #leftDatas > 0 then
        -- 按照权重随机
        local talkIdx = math.random(1, #leftDatas)
        return leftDatas[talkIdx].DialogID
    end

    return 0
end

function GetMaxAdvGradeExpLv()
    if ConfigDataAccessor.MaxAdvGradeExpLv ~= nil then
        return ConfigDataAccessor.MaxAdvGradeExpLv
    end

    ConfigDataAccessor.MaxAdvGradeExpLv = 1
    for k, v in pairs(ConfigDataAccessor.AdvGradeExp) do
        if v.ID > ConfigDataAccessor.MaxAdvGradeExpLv then
            ConfigDataAccessor.MaxAdvGradeExpLv = v.ID
        end
    end
    return ConfigDataAccessor.MaxAdvGradeExpLv
end

--获取每次的奖励【玩家等级版本】
function GetDonateOrderRewardEach(playerLv)
    local minLv = 0
    local aim_id = 0
    for k, v in pairs(ConfigDataAccessor.DonateOrderReward) do
        if v.LvLimit <= playerLv and minLv < v.LvLimit then
            minLv = v.LvLimit
            aim_id = k
        end
    end

    -- 找个最大的
    if aim_id == 0 then
        for k, v in pairs(ConfigDataAccessor.DonateOrderReward) do
            if v.LvLimit >= minLv then
                minLv = v.LvLimit
                aim_id = k
            end
        end
    end

    -- 还是没找到就找个最小的
    if aim_id == 0 then
        minLv = 9999999
        for k, v in pairs(ConfigDataAccessor.DonateOrderReward) do
            if v.LvLimit <= minLv then
                minLv = v.LvLimit
                aim_id = k
            end
        end
    end

    local rews = nil
    for k, v in pairs(ConfigDataAccessor.DonateOrderReward) do
        if v.LvLimit == minLv and v.RewardType == 1 then
            rews = v
        end
    end

    return rews
end

function InitWorldLine()
    ConfigDataAccessor.WorldLineDic = {}

    local worldLines = ConfigDataAccessor.WorldLine

    for k, v in pairs(worldLines) do
        if v.NodeType == 1 or v.NodeType == 2 then
            ConfigDataAccessor.WorldLineDic[k] = {}
        end
    end

    for k, v in pairs(ConfigDataAccessor.WorldLineDic) do
        for k2, v2 in pairs(worldLines) do
            if (v2.NodeType ~= 1 and v2.NodeType ~= 2) and v2.PreNode == k then
                ConfigDataAccessor.WorldLineDic[k][k2] = {}
                table.insert(ConfigDataAccessor.WorldLineDic[k][k2], k2)
            end
        end
    end

    for k, v in pairs(ConfigDataAccessor.WorldLineDic) do
        for k2, v2 in pairs(v) do
            local id = FindNextWorldLineItem(k2)
            while id > 0 do
                table.insert(v2, id)
                id = FindNextWorldLineItem(id)
            end
        end
    end
end

function FindNextWorldLineItem(k)
    local id = 0
    local worldLines = ConfigDataAccessor.WorldLine
    for k2, v2 in pairs(worldLines) do
        if v2.NodeType ~= 1 and v2.PreNode == k then
            id = k2
        end
    end

    return id
end

-- 初始化 job 职业相关的 数据
function InitJobDatas()
    -- 职业对应的转职信息
    -- {10 = [10,11,12]  , 11 = [10,11,12] , 12 = [10,11,12] }
    ConfigDataAccessor.JobTransferLists = {}

    for k, v in pairs(ConfigDataAccessor.Job) do
        -- 如果 有职业转换 列表并且数量大于0
        if v.TransferJob ~= nil and #v.TransferJob > 0 then
            local jobs = {}
            table.insert(jobs, 1, v.ID)
            for i = 1, #v.TransferJob, 1 do
                table.insert(jobs, #jobs + 1, v.TransferJob[i])
            end

            -- 将 转职数据 存入 JobTransferLists 中, 格式为 {10 = [10,11,12]  , 11 = [10,11,12] , 12 = [10,11,12] }
            for i = 1, #jobs, 1 do
                ConfigDataAccessor.JobTransferLists[jobs[i]] = jobs
            end
        end
    end
end

-- LuaModule 创建
function AdventureLevelModule:OnModuleCreate(arg)
    LuaModuleDispatcher:Register(ModuleDef.AdventureLevelModule, self)

    -- 冒险等级升级监听
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.RiskLevelUpRetID, function(data)
        self:OnRiskLevelUpRet(data)
    end, nil);

end

-- 冒险等级升级监听
function AdventureLevelModule:OnRiskLevelUpRet(data)
    if data == nil then
        return;
    end

    -- 获取伙伴数据
    local prevRiskLevel = data.data.NowRiskLevel - 1;

    local cfg = ConfigDataAccessor.AdvGradeExp[data.data.NowRiskLevel];
    local prewCfg = ConfigDataAccessor.AdvGradeExp[prevRiskLevel];

    if cfg == nil then
        return;
    end
    if prewCfg == nil then
        return;
    end

    if prewCfg.IfBreak then
        -- UIManager.Instance:OpenWidget(AdventureLevelUpgradeWidgetPath, false, data.data.NowRiskLevel, nil,
        --     CS.StarProjectDef.MainPageCommond.HideNone, false, false);
        UIManager.Instance:OpenWidgetAsync(AdventureLevelUpgradeWidgetPath, nil, false, data.data.NowRiskLevel, nil,
            CS.StarProjectDef.MainPageCommond.HideNone, false, false);
    else

        if prewCfg.AdvOrder ~= cfg.AdvOrder then
            -- 飘字升级成功
            Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_1"));
            LuaModuleDispatcher:SendMessage(ModuleDef.PersonalTowerModule, "RefreshLevelRedPoint",
                {data.data.NowRiskLevel});
        end

        local rewardsList = {};
        -- 直接显示奖励
        for i = 1, #prewCfg.AwardItem do
            local rw = {};
            rw.BaseID = prewCfg.AwardItem[i];
            rw.Num = prewCfg.AwardNum[i];
            rw.EntityID = 0;
            table.insert(rewardsList, rw);
        end
        -- 弹窗逻辑（先塞进去之后在外面调用）
        LuaModuleDispatcher:SendMessage(ModuleDef.RewardsPopModule, "OnOpenRewardsPop", {rewardsList});

    end

    -- 发送 冒险等级升级 的 全局事件
    CS.StarProject.GlobalEvent.OnPlayerDataChange:Invoke(CS.StarProjectDef.AOIAttrDefine.RiskInfo, nil);

end

function AdventureLevelModule:OnListener()
    -- 属性同步回调
    if self.M_EntityBase == nil then
        self.M_EntityBase = GameManager.Instance.M_MainPlayerCtrlBase;
        self.attrLuaKey = "RiskInfo_" .. GameManager.Instance.mainPlayerId;

        self.M_EntityBase.Data:AddAttributeChangeLuaAction(CS.StarProjectDef.AOIAttrDefine.RiskInfo, self.attrLuaKey,
            function(key, value)
                print("属性同步开始")

                if self.AdventureLevelWindowView ~= nil then
                    -- 刷新
                    self.AdventureLevelWindowView:Refesh();
                end
            end)
    end

end

function AdventureLevelModule:OffListener()
    -- 解绑属性同步回调
    if self.M_EntityBase ~= nil then
        self.M_EntityBase.Data:DelAttributeChangeLuaAction(CS.StarProjectDef.AOIAttrDefine.RiskInfo, self.attrLuaKey);
        self.M_EntityBase = nil;
    end
end

end

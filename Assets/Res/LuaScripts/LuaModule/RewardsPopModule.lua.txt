function RewardsPopModule:OpenItemTips(baseID, entityID, tr)

    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {baseID, false, entityID, tr, 0})
end

    print("RewardsPopModule:CloseItemTips")
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnCloseTips", {})
end

function RewardsPopModule:GetItemsByItemSpaceId()
    return BusinessManager.Instance:GetItemsByItemSpaceId(self.SpaceID)
end

--关闭回调
function RewardsPopModule:OnCloseRewardsPopEvt(arg)
    self.evtOnCloseRewardsPop:Invoke()
end

--关闭
function RewardsPopModule:OnCloseRewardsPop()
    self:OnCloseRewardsPopEvt()
    self.CanPop = true
    --所有的New置空
    --BusinessManager.Instance:UnNewAllItems();
end

--进队列
function RewardsPopModule:PushBack(it)
    Queue.pushBack(self.DropQueue, it)
end

--出队列
function RewardsPopModule:PopFront()
    local it = nil

    if self.DropQueue.first <= self.DropQueue.last then
        it = Queue.popFront(self.DropQueue)
    end

    return it
end

--LuaModule 创建
function RewardsPopModule:OnModuleCreate(arg)
    LuaModuleDispatcher:Register(ModuleDef.RewardsPopModule, self)

    self.CanPop = true

    MonoHelper.AddUpdateListener(
        function()
            self:PopCheck()
        end
    )

    self.DropQueue = Queue.new() --队列

    self.OnBackLoginFunc = function(againLoginType)
        -- 清除队列
        for k, v in pairs(self.DropQueue) do
            self.DropQueue[k] = nil
        end
        self.DropQueue.first = 0
        self.DropQueue.last = -1
        self.CanPop = true
    end
    CS.StarProject.GlobalEvent.OnBackLogin:AddListener(self.OnBackLoginFunc)
end

function RewardsPopModule:PopCheck()
    --[[
        if Input.GetKey(CS.UnityEngine.KeyCode.W) then
        --奖励弹窗测试
        local rewardsList = {}
        for i = 1, 17 do
            local rw = {}
            rw.BaseID = 103001
            rw.Num = 1
            table.insert(rewardsList, rw)
        end
        LuaModuleDispatcher:SendMessage(ModuleDef.RewardsPopModule, "OnOpenRewardsPop", {rewardsList})
    end
    ]]
    --找到池子里没有使用的
    if self.CanPop then
        local data = self:PopFront()
        if data ~= nil then
            self.AsyncLoadRewardsPopWidgetCb = function(ui)
                if ui ~= nil then
                    if self.rewardsPopWidgetView == nil then
                        self.rewardsPopWidgetView = ui:GetLuaPanel().ScriptTable
                        self.rewardsPopWidgetView:BindModule(self)
                        self.rewardsPopWidgetView:InitItems()
                        self.evtOnCloseRewardsPop = LuaModuleDispatcher:Event(ModuleDef.RewardsPopModule, "OnCloseRewardsPopEvt")
                    end
                    self.IsOpenTips = false
                    --显示
                    self.rewardsPopWidgetView:RefeshRewards(data)
                end
                self.AsyncLoadRewardsPopWidgetCb = nil
            end
            UIManager.Instance:OpenWidgetAsync(RewardsPopWidgetPath, self.AsyncLoadRewardsPopWidgetCb, false, nil, nil, CS.StarProjectDef.MainPageCommond.HideBoth, false, true)

            self.CanPop = false
        end
    end
end

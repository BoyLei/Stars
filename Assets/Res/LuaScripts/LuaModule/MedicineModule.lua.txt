    -- 打开伙伴出战
    ["OnOpenMedicine"] = function(...)

-- 监听打开出战界面事件
function MedicineModule:OnOpenMedicine(...)
    local tb = ...

    local itemBaseID = 0
    local entityBase = nil
    if type(tb) == "table" then
        -- Lua调用
        if #tb > 0 then
            entityBase = tb[1]
        end
        if #tb > 1 then
            itemBaseID = tb[2]
        end
    else
        -- C#调用
        if tb.Length > 0 then
            entityBase = tb[0]
        end
        if tb.Length > 1 then
            itemBaseID = tb[1]
        end
    end

    if entityBase == nil then
        entityBase = GameManager.Instance.M_MainPlayerCtrlBase
    end

    self.M_EntityBase = entityBase

    -- local medicineWindowCS = UIManager.Instance:OpenWindow(MedicinePath, nil,
    --     CS.StarProjectDef.MainPageCommond.HideBoth);
    -- if self.medicineWindowView == nil then
    --     self.medicineWindowView = medicineWindowCS:GetLuaPanel().ScriptTable;
    --     self.medicineWindowView:BindModule(self);
    --     self.medicineWindowView:Init(entityBase);
    -- end
    -- self.medicineWindowView:Refresh();

    self.AsyncLoadMedicineCb = function(ui)
        if ui ~= nil then
            if self.medicineWindowView == nil then
                self.medicineWindowView = ui:GetLuaPanel().ScriptTable
                self.medicineWindowView:BindModule(self)
                self.medicineWindowView:Init(entityBase)
            end
            self.medicineWindowView:InitEntityBase(entityBase)
            self.medicineWindowView:Refresh()

            if itemBaseID > 0 then
                -- 打开对应的二级界面
                self.medicineWindowView:OpenMedicineSecondWindow(itemBaseID)
            end
        end
        self.AsyncLoadMedicineCb = nil
    end
    UIManager.Instance:OpenWindowAsync(MedicinePath, self.AsyncLoadMedicineCb, nil, CS.StarProjectDef.MainPageCommond.HideBoth)
end

-- LuaModule 创建
function MedicineModule:OnModuleCreate(...)
    LuaModuleDispatcher:Register(ModuleDef.MedicineModule, self)

    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.UserSundryNtfID,
        function(data)
            self:OnUserSundryNtf(data)
        end,
        nil
    )

    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.ItemUseSrvID,
        function(data)
            self:OnItemUseAck(data)
        end,
        nil
    )

    FixMessageManager.Instance:OnMessage(
        FixUpdateDef.UserSundry,
        function(data)
            self:OnUserSundry(data)
        end,
        nil
    )
end

function MedicineModule:OnRegisterAttribute(...)
    local tb = ...

    local entityBase = 0
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        entityBase = tb[1]
    else
        -- C#调用
        entityBase = tb[0]
    end
    self.M_EntityBase = entityBase

    self.storageHP = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Long, CS.StarProjectDef.AOIAttrDefine.StorageHP)
    self.storageMP = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Long, CS.StarProjectDef.AOIAttrDefine.StorageMP)

    -- 绑定属性同步回调  START
    self.attrLuaKey = "MedicineModule_" .. GameManager.Instance.mainPlayerId
    self:InitAttribute()
    -- 绑定属性同步回调----- END

    self.storageDrugs = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Proto, CS.StarProjectDef.AOIAttrDefine.StorageDrugs)

    self:RefreshRedPoint()
end

function MedicineModule:InitAttribute()
    self.M_EntityBase.Data:AddAttributeChangeLuaAction(
        CS.StarProjectDef.AOIAttrDefine.StorageHP,
        self.attrLuaKey,
        function(key, value)
            if self.storageHP == 0 then
                self.storageHP = value
                return
            end
            if self.storageHP > value then
                -- 正在回血
                local del = self.storageHP - value
                GameManager.Instance:OnCureFloatingText(del, self.M_EntityBase.M_Curr.EntityId)
                self.storageHP = value
            end
        end
    )

    self.M_EntityBase.Data:AddAttributeChangeLuaAction(
        CS.StarProjectDef.AOIAttrDefine.StorageMP,
        self.attrLuaKey,
        function(key, value)
            if self.storageMP == 0 then
                self.storageMP = value
                return
            end
            if self.storageMP > value then
                -- 正在回蓝
                local del = self.storageMP - value
                GameManager.Instance:OnCureFloatingText(del, self.M_EntityBase.M_Curr.EntityId)
                self.storageMP = value
            end
        end
    )
end

function MedicineModule:OnBagChanged()
    if self.medicineWindowView ~= nil then
        self.medicineWindowView:OnBagChanged()
    end

    BusinessManager.Instance:RefreshMedicineBtn()
end

function MedicineModule:DelAttribute()
    if self.M_EntityBase ~= nil and self.M_EntityBase.Data ~= nil then
        self.M_EntityBase.Data:DelAttributeChangeLuaAction(CS.StarProjectDef.AOIAttrDefine.StorageHP, self.attrLuaKey)

        self.M_EntityBase.Data:DelAttributeChangeLuaAction(CS.StarProjectDef.AOIAttrDefine.StorageMP, self.attrLuaKey)
    end
end

function MedicineModule:OnUseMedicine(...)
    local tb = ...

    local itemID = 0
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        itemID = tb[1]
    else
        -- C#调用
        itemID = tb[0]
    end

    local item = BusinessManager.Instance:GetItemByItemEntityId(itemID)

    local itemUseReq = CS.ProtoMsg.ItemUseReq()
    itemUseReq.ItemEntityID = item.EntityID
    itemUseReq.ItemSpaceId = item.SpaceId
    itemUseReq.Count = 1
    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, itemUseReq, false)
end

-- LuaModule 消息处理
function MedicineModule:OnModuleMessage(msg, ...)
    self.ModuleMessages[msg](...)
end

function MedicineModule:OnUserSundryNtf(data)
    if data == nil or data.data == nil then
        return
    end

    self.userSundry = data.data.UserStting
    if self.medicineWindowView ~= nil then
        self.medicineWindowView:RefreshSundry()
    end
end

function MedicineModule:OnUserSundry(data)
    local pid = GameManager.Instance.mainPlayerId
    local sundryMD = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.UserSundry):GetUserSundryMDByPid(pid)

    if sundryMD == nil then
        sundryMD = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.UserSundry):GetUserSundryMD()
    end

    if sundryMD ~= nil then
        self.userSundry = sundryMD.UserSetting
        if self.medicineWindowView ~= nil then
            self.medicineWindowView:RefreshSundry()
        end
    end
end

function MedicineModule:OnItemUseAck(data)
    if data == nil or data.data == nil then
        return
    end

    print("MedicineModule:OnItemUseAck")
end

-- LuaModule显示
function MedicineModule:OnModuleShow(arg)
    self:RefreshRedPoint()
end

-- LuaModule 释放
function MedicineModule:OnModuleRelease()
    self:DelAttribute()

    LuaModuleDispatcher:UnRegister(ModuleDef.MedicineModule)
end

-- 是否能使用血量药品
function MedicineModule:CanUseHPMedicine()
    local totalHP = ConfigDataAccessor.SystemList[49].Value
    local curHP = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Long, CS.StarProjectDef.AOIAttrDefine.StorageHP)
    return totalHP > curHP
end

-- 是否能使用法力药品
function MedicineModule:CanUseMPMedicne()
    local totalMP = ConfigDataAccessor.SystemList[50].Value
    local curMP = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Long, CS.StarProjectDef.AOIAttrDefine.StorageMP)
    return totalMP > curMP
end

-- 是否能使用战斗药品
function MedicineModule:CanUseBattleMedicine()
    local hp = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Long, CS.StarProjectDef.AOIAttrDefine.curHp)
    local sum = self.M_EntityBase.Data.Attrs:GetLuaAoiValue(CS.StarProjectDef.EnumAOIType.Long, CS.StarProjectDef.AOIAttrDefine.TruthHp)
    return hp < sum
end

function MedicineModule:RefreshRedPoint()
    if self.storageHP == nil or self.storageMP == nil then
        return
    end

    local ids = self.storageDrugs.Eitemid

    local sundry = self.userSundry
    local l = BusinessManager.Instance:GetUserSundryData(sundry)

    local act1 = l[1]

    local isNoShowToDay = false --今日不在显示
    if LocalCacheManager.Instance:GetDayValue("Medicine2and3") ~= nil then
        isNoShowToDay = LocalCacheManager.Instance:GetDayValue("Medicine2and3")
    end
    RedPointManager.Instance:TriggerConditionTypeBoolValue(RedPointConditionType.Medicine2, self.storageHP <= 0 and act1 and not (isNoShowToDay))
    RedPointManager.Instance:TriggerConditionTypeBoolValue(RedPointConditionType.Medicine3, self.storageMP <= 0 and act1 and not (isNoShowToDay))

    local items = BusinessManager.Instance:GetItemsByItemType(3, 3)
    RedPointManager.Instance:TriggerConditionTypeBoolValue(RedPointConditionType.Medicine1, (ids == nil or ids.Count == 0) and items.Count > 0)
end

return MedicineModule

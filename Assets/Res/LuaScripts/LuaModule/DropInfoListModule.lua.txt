DropInfoListModule = class("DropInfoListModule", BaseLuaModule)
DropInfoListModule.DropQueue = Queue.new() --队列

local DropInfoListWidgetPath = "DropInfoList/DropInfoListWidget"

--LuaModule 消息模块
DropInfoListModule.ModuleMessages = {
    --打开掉落
    --ModuleManager.Instance.SendMessage(ModuleDef.DropInfoListModule, "OnOpenDrop", null);
    ["OnOpenDrop"] = function(...)
        DropInfoListModule:OnOpenDrop(...)
    end,
    --关闭掉落
    --ModuleManager.Instance.SendMessage(ModuleDef.DropInfoListModule, "OnCloseDrop", null);
    ["OnCloseDrop"] = function(...)
        DropInfoListModule:OnCloseDrop(...)
    end,
    --添加掉落
    --ModuleManager.Instance.SendMessage(ModuleDef.DropInfoListModule, "OnAddDrop", new object[] { it }); //只要传一个掉落信息
    ["OnAddDrop"] = function(...)
        DropInfoListModule:OnAddDrop(...)
    end
}

--打开掉落监听UI
function DropInfoListModule:OnOpenDrop(...)
    print("DropInfoListModule:OnOpenDrop")

    --local tb=...
    -- local dropInfoListWidgetCs =UIManager.Instance:OpenWidget(DropInfoListWidgetPath,false,nil,nil,CS.StarProjectDef.MainPageCommond.HideNone,true)
    -- if self.dropInfoListWidgetView == nil then
    -- 	self.dropInfoListWidgetView=dropInfoListWidgetCs:GetLuaPanel().ScriptTable;
    -- 	self.dropInfoListWidgetView:BindModule(self);
    -- 	self.DropQueue=Queue.new();--队列
    -- else
    -- 	self.dropInfoListWidgetView.gameObject:SetActive(true);
    -- end

    self.AsyncLoadDropInfoListWidgetCb = function(ui)
        if ui ~= nil then
            if self.dropInfoListWidgetView == nil then
                self.dropInfoListWidgetView = ui:GetLuaPanel().ScriptTable
                self.dropInfoListWidgetView:BindModule(self)
                self.DropQueue = Queue.new() --队列
            else
                self.dropInfoListWidgetView.gameObject:SetActive(true)
            end
        end
        self.AsyncLoadDropInfoListWidgetCb = nil
    end
    UIManager.Instance:OpenWidgetAsync(DropInfoListWidgetPath, self.AsyncLoadDropInfoListWidgetCb, false, nil, nil, CS.StarProjectDef.MainPageCommond.HideNone, true, false, true)
end

--关闭掉落监听UI
function DropInfoListModule:OnCloseDrop(...)
    self.dropInfoListWidgetView.gameObject:SetActive(false)
end

--添加掉落事件
function DropInfoListModule:OnAddDrop(...)
    local tb = ...
    --传入道具ID
    if type(tb) == "table" then
        --Lua调用
        local dropinfo = tb[1]
        self:PushBack(dropinfo)
    else
        --C#调用
        local dropinfo = {}
        dropinfo.Itemid = tb[0].BaseID
        dropinfo.ItemNum = tb[0].Num
        --dropinfo.AwardType = 1
        self:PushBack(dropinfo)
    end
end

--进队列
function DropInfoListModule:PushBack(it)
    --筛选出快捷穿戴的
    --[[
	local cfg = ConfigDataAccessor.Item[it.Itemid];
	if cfg.ItemType==2 and it.EntityID~=nil and it.EntityID~=0 then
		LuaModuleDispatcher:SendMessage(ModuleDef.QuickEquipModule, "OnAddQuickEquip",{it} );
	end
	]]
    --UI显示
    -- 展示道具名UI
    -- 过滤秘境的进度条道具
    if it ~= nil then
        if it.Itemid == 121 or it.Itemid == 122 then
            return
        end
    end
    Queue.pushBack(self.DropQueue, it)
end

--出队列
function DropInfoListModule:PopFront()
    local it = nil

    if self.DropQueue.first <= self.DropQueue.last then
        it = Queue.popFront(self.DropQueue)
    end

    return it
end

--LuaModule 创建
function DropInfoListModule:OnModuleCreate(arg)
    LuaModuleDispatcher:Register(ModuleDef.DropInfoListModule, self)

    self.OnBackLoginFunc = function(againLoginType)
        if self.dropInfoListWidgetView ~= nil and self.dropInfoListWidgetView.gameObject ~= nil then
            -- 清除队列
            for k, v in pairs(self.DropQueue) do
                self.DropQueue[k] = nil
            end
            self.DropQueue.first = 0
            self.DropQueue.last = -1
            self.dropInfoListWidgetView:CleanAll()
        end
    end
    CS.StarProject.GlobalEvent.OnBackLogin:AddListener(self.OnBackLoginFunc)
end

--LuaModule 消息处理
function DropInfoListModule:OnModuleMessage(msg, ...)
    self.ModuleMessages[msg](...)
end

--LuaModule显示
function DropInfoListModule:OnModuleShow(arg)
end

--LuaModule 释放
function DropInfoListModule:OnModuleRelease()
    LuaModuleDispatcher:UnRegister(ModuleDef.DropInfoListModule)
end

return DropInfoListModule

function MindRepairModule:OnOpenMindRepairWindow(...)
	local tb = ...
    local cfgId = 0
    --列表
    if type(tb) == "table" then
        --Lua调用
        cfgId = tb[1]
    else
        --C#调用
        cfgId = tb[0]
    end

	local Cb = function(ui)
		if ui ~= nil then
			self.MindRepairWindowView = ui:GetLuaPanel().ScriptTable;	
			if self.MindRepairWindowView ~= nil then
				self.MindRepairWindowView.InstPanel = ui
				self.MindRepairWindowView:BindModule(self)
				self.MindRepairWindowView:RenderAllView(true, cfgId)
			end
		end
	end

	UIManager.Instance:OpenWindowAsync("MiniGame/Prefab/MindRepairWindow", Cb, cfgId)
end

--LuaModule 创建
function MindRepairModule:OnModuleCreate(arg)
	LuaModuleDispatcher:Register(ModuleDef.MindRepairModule,self)
	-- self:BuildAllSkillData()
end

--LuaModule 消息处理

end

function MindRepairModule:OnSkillOpenWidgetDestroy(arg)

end

--LuaModule 释放
function MindRepairModule:OnModuleRelease()
	LuaModuleDispatcher:UnRegister(ModuleDef.MindRepairModule)
	self.MindRepairWindowView = nil
end

return MindRepairModule

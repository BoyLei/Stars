-- LuaModule 创建
function MailModule:OnModuleCreate(arg)
    LuaModuleDispatcher:Register(ModuleDef.MailModule, self)

    self.MailListNtf = function(data)
        self:OnMailListNtf(data)
    end
    ---邮件ID列表推送
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.MailListNtfID, self.MailListNtf, self);

    self.MailGetInfosRet = function(data)
        self:OnMailGetInfosRet(data)
    end
    ---回复邮件详细信息 、收到新邮件
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.MailGetInfosRetID, self.MailGetInfosRet, self);

    self.MailGetRewardRet = function(data)
        self:OnMailGetRewardRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.MailGetRewardRetID, self.MailGetRewardRet, self);

    self.MailDeleteRet = function(data)
        self:OnMailDeleteRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.MailDeleteRetID, self.MailDeleteRet, self);

    self.MailsMarkReadedRet = function(data)
        self:OnMailsMarkReadedRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.MailsMarkReadedRetID, self.MailsMarkReadedRet, self);

    self.GetMailAttachedContentNtf = function(data)
        self:OnGetMailAttachedContentNtf(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.GetMailAttachedContentNtfID, self.GetMailAttachedContentNtf, self);
end

-- LuaModule 释放
function MailModule:OnModuleRelease()
    ---邮件ID列表推送
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.MailListNtfID, self.MailListNtf, self);

    ---回复邮件详细信息 、收到新邮件
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.MailGetInfosRetID, self.MailGetInfosRet, self);

    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.MailGetRewardRetID, self.MailGetRewardRet, self);

    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.MailDeleteRetID, self.MailDeleteRet, self);

    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.MailsMarkReadedRetID, self.MailsMarkReadedRet, self);

    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.GetMailAttachedContentNtfID, self.GetMailAttachedContentNtf,
        self);

    self.MailListNtf = nil;
    self.MailGetInfosRet = nil;
    self.MailGetRewardRet = nil;
    self.MailDeleteRet = nil;
    self.MailsMarkReadedRet = nil;
    self.GetMailAttachedContentNtf = nil;
    LuaModuleDispatcher:UnRegister(ModuleDef.MailModule)
end

function MailModule:IsNewMail(uid)
    for i = 1, #self.MailKeys do
        if self.MailKeys[i] == uid then
            return true;
        end
    end
    return false;
end

function MailModule:OnMailGetInfosRet(data)
    local msg = data.data
    local change = false;
    for i = 0, msg.DataList.Count - 1 do
        local mail = msg.DataList[i]
        if self:IsNewMail(mail.ID) == false then
            -- 新推送邮件
            table.insert(self.MailKeys, mail.ID)
            self.MailStatusMap[mail.ID] = {
                IsRead = false,
                Reward = self:HasAttachment(mail)
            }
            self:SortMailKey();
        end
        self.MailMap[mail.ID] = mail;
        change = true
    end
    if change then
        CS.StarProject.GlobalEvent.OnFreshMail:Invoke(self:GetMails());
    end
    self:UpdateRedPoint();
end

function MailModule:OnMailGetRewardRet(data)
    local msg = data.data;
    local change = false;
    for i = 0, msg.ListIDs.Count - 1 do
        if self.MailMap[msg.ListIDs[i]] ~= nil then
            self.MailMap[msg.ListIDs[i]].IsRead = true
            self.MailMap[msg.ListIDs[i]].IsGetReward = true
            change = true;
        end

        if self.MailStatusMap[msg.ListIDs[i]] ~= nil then
            self.MailStatusMap[msg.ListIDs[i]].IsRead = true
            self.MailStatusMap[msg.ListIDs[i]].Reward = true
        end
    end

    if change then
        CS.StarProject.GlobalEvent.OnFreshMail:Invoke(self:GetMails());
    end
    self:UpdateRedPoint();
end
function MailModule:GetMailIndex(uid)
    for i, v in ipairs(self.MailMap) do
        if v == uid then
            return i;
        end
    end
    return -1;
end

function MailModule:OnMailDeleteRet(data)
    local msg = data.data;
    local change = false;
    for i = 0, msg.ListIDs.Count - 1 do
        self.MailMap[msg.ListIDs[i]] = nil;
        self.MailStatusMap[msg.ListIDs[i]] = nil;
        local index = self:GetMailIndex(msg.ListIDs[i]);
        if index > -1 then
            table.remove(self.MailKeys, index);
        end
        CS.StarProject.GlobalEvent.OnDeleteMail:Invoke(msg.ListIDs[i]);
        change = true;
    end

    if change then
        self:SortMailKey();
        CS.StarProject.GlobalEvent.OnFreshMail:Invoke(self:GetMails());
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_24"))
    end
    self:UpdateRedPoint();
end

function MailModule:SortMailKey()
    table.sort(self.MailKeys, function(m1, m2)
        return m1 > m2;
    end)
end

function MailModule:OnMailListNtf(data)
    local msg = data.data;
    self.MailMap = {}
    self.MailKeys = {}
    self.MailStatusMap = {}

    for i = 0, msg.ListIDs.Count - 1 do
        table.insert(self.MailKeys, msg.ListIDs[i]);
        self.MailStatusMap[msg.ListIDs[i]] = {
            IsRead = msg.ReadList[i],
            Reward = msg.GetList[i]
        }
    end
    self:UpdateRedPoint();
    self:SortMailKey();
    self:MailGetInfosReq();
end

function MailModule:HasAttachment(mail)

    if mail.Addbc ~= nil and mail.Addbc.Data ~= nil and mail.Addbc.Data.Count > 0 then
        return false;
    end

    if mail.ItemList ~= nil and mail.ItemList.ItemList ~= nil and mail.ItemList.ItemList.Count > 0 then
        return false;
    end

    return true;
end

function MailModule:UpdateRedPoint()
    -- 未读个数
    local hasNew = 0;
    -- 未领取 
    local Reward = false;
    if self.MailStatusMap ~= nil then
        for k, v in pairs(self.MailStatusMap) do
            if not v.IsRead then
                hasNew = hasNew + 1;
            end

            if not v.Reward then
                Reward = true;
            end
        end
    end

    RedPointManager.Instance:RefreshRedPointCount(RedPointType.Mail_New, hasNew);

    RedPointManager.Instance:TriggerConditionTypeBoolValue(RedPointConditionType.Mail_Red, (hasNew == 0 and Reward));

end

function MailModule:OnMailsMarkReadedRet(data)
    local msg = data.data;
    local change = false;
    for i = 0, msg.ListIDs.Count - 1 do
        if self.MailMap[msg.ListIDs[i]] ~= nil then
            self.MailMap[msg.ListIDs[i]].IsRead = true;
        end

        if self.MailStatusMap[msg.ListIDs[i]] ~= nil then
            self.MailStatusMap[msg.ListIDs[i]].IsRead = true;
        end

        change = true
    end
    if change then
        CS.StarProject.GlobalEvent.OnFreshMail:Invoke(self:GetMails());
    end
    self:UpdateRedPoint();
end

function MailModule:OnGetMailAttachedContentNtf(data)
    local msg = data.data;
    local rewardsList = {}
    for i = 0, msg.List.Count - 1 do
        local rw = {};
        rw.BaseID = msg.List[i].Index;
        rw.Num = msg.List[i].Num;
        rw.EntityID = msg.List[i].EID;
        table.insert(rewardsList, rw);
    end
    LuaModuleDispatcher:SendMessage(ModuleDef.RewardsPopModule, "OnOpenRewardsPop", {rewardsList});

end

function MailModule:GetMails()
    local list = {}
    for k, v in pairs(self.MailMap) do
        if v ~= nil then
            table.insert(list, v)
        end
    end

    table.sort(list, function(m1, m2)
        return m1.CreateTime > m2.CreateTime;
    end)

    return list;
end

function MailModule:MailGetInfosReq()
    local msg = CS.ProtoMsg.MailGetInfosReq();

    for i = 1, #self.MailKeys do
        if self.MailMap[self.MailKeys[i]] == nil then
            msg.ListIDs:Add(self.MailKeys[i])
        end
        if msg.ListIDs.Count >= self.MaxQueryCount then
            break
        end
    end
    if msg.ListIDs.Count == 0 then
        return
    end
    local gameSocket = NetworkManager.Instance.gameSocket;
    gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, msg, false, 0, false);
end

function MailModule:OnMailMarkReadedReq(...)
    local tb = ...
    local mailID = 0
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        mailID = tb[1]
    else
        -- C#调用
        mailID = tb[0]
    end
    if self.MailMap[mailID] ~= nil then
        if not self.MailMap[mailID].IsRead then
            local msg = CS.ProtoMsg.MailMarkReadedReq();
            msg.UID = mailID;
            local gameSocket = NetworkManager.Instance.gameSocket;
            gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, msg, false, 0, false);
        end
    end
end

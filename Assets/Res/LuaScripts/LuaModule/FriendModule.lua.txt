-- LuaModule 创建
function FriendModule:OnModuleCreate(arg)
    --------------------------------  【监听协议】 -------------------------------------
    self.OnClient2ThirdRetCmd = function(msgName, data)
        self:OnClient2ThirdRet(msgName, data)
    end
    Client2ThirdMsgManager.Instance:OnMessageEnum(CS.MsgIDEnum.Client2ThirdRetID, self.OnClient2ThirdRetCmd, self);

    --------------------------------  【注册模块的事件】 -------------------------------------
    LuaModuleDispatcher:Register(ModuleDef.FriendModule, self);

    --------------------------------  【注册模块的事件】 -------------------------------------
    -- update事件
    self.UpdateFun = function()
        self:UpdateData();
    end

    ------------------------------- 【初始化模块的本地变量】----------------------
    self.Client2ThirdReqList = {};

    --------------- 好友申请
    self.m_Countdown = self.m_CfgInterval
    self.m_ApplyFriends = {};
    self.m_ApplyFriendsRead = {}
    self.FriendList = {}
    self.BlackList = {}
    self.ApplyFinishPIDList = {}
    self.selectPlayerId = 0;
    -- 全量数据回调
    self.OnFullFriendsApplyToRespFun = function(data)
        self:OnFullFriendsApplyToResp(data);
    end
    -- 差量数据回调
    self.OnDispersionFriendsApplyToRespFun = function(data)
        self:OnDispersionFriendsApplyToResp(data);
    end

    -- 获取黑名单列表的回调
    self.OnGetPageBlacksRespFun = function(data)
        self:OnGetPageBlacksResp(data);
    end
    self.OnGetPageBlacksResp2Fun = function(data)
        self:OnGetPageBlacksResp2(data);
    end
    --------------- 好友申请
    	-- 申请加好友 事件
	self.OnApplyToAddFriendRespFun = function(data)
		self:OnApplyToAddFriendResp(data);
	end
end

-- LuaModule 释放

-- 第三方回复客户端
function FriendModule:OnClient2ThirdRet(msgName, data)
    -- 判断有无监听
    local isHave = self:GetIsHaveCacheListenByMsgName(msgName);
    if not isHave then
        Log("FriendModule:OnClient2ThirdRet() msg=" .. msgName .. "<-这个协议名，我居然没有监听 err!!!");
        return;
    end
    -- Log("收到 第三方回复客户端 协议 =》" .. msgName);
    -- 执行回调
    for i, v in ipairs(self.Client2ThirdReqList) do
        -- if v ~= nil and v.MsgName == msgName and v.Count > 0 then
        -- v.CB(data);	-- 执行回调
        -- v.Count = v.Count - 1;
        -- end
        if v ~= nil and v.MsgName == msgName and v.CB ~= nil then
            v.CB(data); -- 执行回调
            v.CB = nil;
        end
    end
end

-- 发送协议
--- 协议名、协议参数信息、协议的回复key、协议回复要监听的协议名、协议的回复执行回调
function FriendModule:OnSendClient2ThirdReq(...)
    local tb = ...

    local msgName = nil; -- 协议名
    local data = nil; -- 协议参数信息
    local key = nil; -- 协议的回复key
    local revertMsgName = nil; -- 协议回复要监听的协议名
    local cb = nil; -- 协议的回复执行回调

    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        msgName = tb[1]; -- 协议名
        data = tb[2]; -- 协议参数信息
        key = tb[3]; -- 协议的回复key
        revertMsgName = tb[4]; -- 协议回复要监听的协议名
        cb = tb[5]; -- 协议的回复执行回调
    else
        -- C#调用
        msgName = tb[0]; -- 协议名
        data = tb[1]; -- 协议参数信息
        key = tb[2]; -- 协议的回复key
        revertMsgName = tb[3]; -- 协议回复要监听的协议名
        cb = tb[4]; -- 协议的回复执行回调
    end

    self:SendClient2ThirdReq(msgName, data, key, revertMsgName, cb);
end

-- 发送 对好友申请响应（同意或拒绝）
function FriendModule:OnSendRespondFriendApply(...)
    local tb = ...
    local pids = {};
    local res = -2;
    local key = "";
    local cb = nil;
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        pids = tb[1];
        res = tb[2];
        key = tb[3];
        cb = tb[4];
    else
        -- C#调用
        pids = tb[0];
        res = tb[1];
        key = tb[2];
        cb = tb[3];
    end

    local message = CS.ProtoMsg.RespondFriendApplyReq();
    message.ToUserPID = self.mainPlayerPID;
    for i = 1, #pids do
        local pid = pids[i];
        message.FromUserPID:Add(pid);
    end
    message.HandleResult = res;

    local msgName = "RespondFriendApplyReq";
    local data = ExtraData.ToByteString(message);
    local revertMsgName = "RespondFriendApplyResp";

    self:SendClient2ThirdReq(msgName, data, key, revertMsgName, cb);
end

function FriendModule:OnSyncMsgData(...)
    local tb = ...
    local msgDatas = nil;
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        msgDatas = tb[1];
    else
        -- C#调用
        msgDatas = tb[0];
    end
    local curChatList = {};
    if msgDatas ~= nil and msgDatas ~= {} and #msgDatas > 0 then
        local isSavePid = false;
        if self.m_CacheRecentlyPIDList == nil then
            self:GetCacheRecentlyPIDList();
        end
        for k, v in pairs(msgDatas) do
            if v ~= nil then
                local isContains = self.m_CacheRecentlyPIDList:Contains(v.PID);
                if isContains == false then
                    self:SetRecentlyPid(v.PID, false, false);
					self:SetRecentlyPid(v.AcceptPID, false, false);
                    isSavePid = true;
                end

                self:CacheFriendChatRecords(v);
                self:CacheFriendChatRedPointRecords(v);
                
                local pid = v.AcceptPID;
                if pid == self.mainPlayerPID then
                    pid = v.PID;
                end
                if curChatList[pid] == nil then
                    curChatList[pid] = {};
                end
                table.insert(curChatList[pid], v);
            end
        end
        if isSavePid then
            self:CacheRecentlyPIDList();
        end
    end
    if self.FriendWindow ~= nil then
        self.FriendWindow:OnSyncMsgData(curChatList);
    end
    RedPointManager.Instance:RefreshRedPointCount(RedPointType.FriendNotice, self:GetAllFriendChatRedPointCount());
end

function FriendModule:CacheFriendChatRecords(msgData)
    if self.m_CacheAllPrivateChatMsgs == nil then
        self.m_CacheAllPrivateChatMsgs = {};
    end
    local pid = msgData.AcceptPID;
    if pid == self.mainPlayerPID then
        pid = msgData.PID;
    end
    if self.m_CacheAllPrivateChatMsgs[pid] == nil then
        self.m_CacheAllPrivateChatMsgs[pid] = {};
    end
    table.insert(self.m_CacheAllPrivateChatMsgs[pid], msgData);
end

function FriendModule:CacheFriendChatRedPointRecords(msgData)
    if self.m_UnreadFriendChatRedPoint == nil then
        self.m_UnreadFriendChatRedPoint = {};
    end
    if msgData.PID == self.mainPlayerPID then
        return;
    end
	local pid = msgData.AcceptPID;
	if pid == self.mainPlayerPID then
		pid = msgData.PID;
	end
    if self.m_UnreadFriendChatRedPoint[pid] == nil then
        self.m_UnreadFriendChatRedPoint[pid] = 1;
    else
        local count = self.m_UnreadFriendChatRedPoint[pid];
        self.m_UnreadFriendChatRedPoint[pid] = count + 1;
    end
end

function FriendModule:GetAllFriendChatRedPointCount()
    if self.m_UnreadFriendChatRedPoint == nil or self.m_UnreadFriendChatRedPoint == {} then
        return 0;
    end
    local count = 0;
    for k, v in pairs(self.m_UnreadFriendChatRedPoint) do
        count = count + v;
    end
    return count;
end

function FriendModule:GetFriendChatRedPointCountByPid(pid)
    if self.m_UnreadFriendChatRedPoint == nil then
        return 0;
    end
    if self.m_UnreadFriendChatRedPoint[pid] == nil then
        return 0;
    else
        local count = self.m_UnreadFriendChatRedPoint[pid];
        return count;
    end
end

function FriendModule:ClearFriendChatRedPointCountByPid(pid)
    if self.m_UnreadFriendChatRedPoint == nil then
        return;
    end
	self.m_UnreadFriendChatRedPoint[pid] = 0;
    RedPointManager.Instance:RefreshRedPointCount(RedPointType.FriendNotice, self:GetAllFriendChatRedPointCount());
end

function FriendModule:ClearAllFriendChatRedPoint()
    self.m_UnreadFriendChatRedPoint = {};
    RedPointManager.Instance:RefreshRedPointCount(RedPointType.FriendNotice, self:GetAllFriendChatRedPointCount());
end

    if isDel then
        if self.m_CacheRecentlyPIDList ~= nil and self.m_CacheRecentlyPIDList.Count > 0 then
            local isRemove = self.m_CacheRecentlyPIDList:Remove(pid);
            if isRemove and isSave then
                self:CacheRecentlyPIDList();
            end
        end
    else
        if pid ~= self.mainPlayerPID then
            local isContains = self.m_CacheRecentlyPIDList:Contains(pid);
            if isContains == false then
                self.m_CacheRecentlyPIDList:Add(pid);
                if isSave then
                    self:CacheRecentlyPIDList();
                end
            end
        end
    end
end

-- 判断与该玩家PID的关系
------ 判断是否好友关系和在不在双方黑名单中
-------- 0没有任何关系 1只有好友关系 2在我的黑名单中
function FriendModule:GetFriendRelation(PID)
    local isFriend = false;
    local isBlack = false;
    local res = 0;
    -- 判断是否好友
    if self.FriendList ~= nil and self.FriendList ~= {} and #self.FriendList > 0 then
        for i = 1, #self.FriendList do
            local friendInfo = self.FriendList[i];
            if friendInfo ~= nil and friendInfo.FriendUser.PID == PID then
                isFriend = true;
                break
            end
        end
    end

    -- 判断是否在黑名单中
    if self.BlackList ~= nil and self.BlackList ~= {} and #self.BlackList > 0 then
        for i = 1, #self.BlackList do
            local blackInfo = self.BlackList[i];
            if blackInfo ~= nil and blackInfo.BlackUserInfo.UserPID == PID then
                isBlack = true;
                break
            end
        end
    end

    if isBlack then
        res = 2;
    elseif isFriend and isBlack == false then
        res = 1;
    else
        res = 0;
    end
    return res;
end

-- 执行全量申请数据
function FriendModule:OnFullFriendsApplyToResp(data)
    if data == nil then
        return;
    end
    self.m_ApplyFriends = {}
    for i = 0, data.FriendRequests.Count - 1 do
        table.insert(self.m_ApplyFriends, data.FriendRequests[i]);
    end
    -- 排序
    if self.m_ApplyFriends ~= nil and self.m_ApplyFriends ~= {} and #self.m_ApplyFriends > 1 then
        -- 定义比较函数，按照id从大到小排序
        local function compare(a, b)
            return a.CreateTime > b.CreateTime
        end
        -- 对values进行排序
        table.sort(self.m_ApplyFriends, compare);
    end
    -- 通知界面
    self:SendFriendApplyListWidget()
end

-- 执行差量申请数据
function FriendModule:OnDispersionFriendsApplyToResp(data)
    if data == nil then
        return;
    end
    -- 差量更新全量的数据
    local isHaveNew = false
    for i = 0, data.FriendRequests.Count - 1 do
        local child = data.FriendRequests[i];
        if child ~= nil then
            local isHave = false;
            for j = 1, #self.m_ApplyFriends do
                local item = self.m_ApplyFriends[j];
                if item ~= nil and item.FriendInfo ~= nil then
                    if child.FromUserPID == item.FromUserPID then
                        self.m_ApplyFriends[j] = child;
                        isHave = true;
                    end
                end
            end
            if isHave == false then
                table.insert(self.m_ApplyFriends, child);
                isHaveNew = true;
            end
        end
    end
    -- 排序
    if self.m_ApplyFriends ~= nil and self.m_ApplyFriends ~= {} and #self.m_ApplyFriends > 1 then
        -- 定义比较函数，按照id从大到小排序
        local function compare(a, b)
            return a.CreateTime > b.CreateTime
        end
        -- 对values进行排序
        table.sort(self.m_ApplyFriends, compare);
    end
    -- 通知界面
    if isHaveNew == true then
        self:SendFriendApplyListWidget()
    end
end

function FriendModule:SendClient2ThirdReq(msgName, data, key, revertMsgName, cb)
    -- Log("发送 第三方回复客户端 协议 =》" .. msgName);
    local battleSocket = NetworkManager.Instance.gameSocket;
    if battleSocket == nil then
        Log("FriendModule:OnClient2ThirdReq() battleSocket=nil err!!!!");
        return;
    end
    local message = CS.ProtoMsg.Client2ThirdReq();
    message.MsgName = msgName;
    message.MsgData = data;
    -- battleSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, message, false);
    battleSocket:SendEnumPbByteMsg(CS.MsgIDEnum.Client2ThirdReqID, message, false);
    -- 缓存监听协议名的回调
    self:CacheListenMsgCB(key, revertMsgName, cb);
end

-- 缓存监听协议名的回调
function FriendModule:CacheListenMsgCB(key, msgName, cb)
    if cb == nil then
        return;
    end
    -- 查找如果有相同的key,相同的msgName,那就监听次数+1
    for i, v in ipairs(self.Client2ThirdReqList) do
        if v ~= nil and v.Key == key and v.MsgName == msgName then
            -- v.Count = v.Count + 1;
            v.CB = cb;
            return;
        end
    end
    -- 如果没有就新加一个
    -------------- key:监听的回调key
    -------------- MsgName:监听的协议名
    -------------- CB:回调方法
    -------------- Count:监听次数
    local data = {
        Key = key,
        MsgName = msgName,
        CB = cb,
        Count = 1
    };
    table.insert(self.Client2ThirdReqList, data);
end

-- 获取是否有该协议名的监听的缓存
function FriendModule:GetIsHaveCacheListenByMsgName(msgName)
    for i, v in ipairs(self.Client2ThirdReqList) do
        -- if v ~= nil and v.MsgName == msgName and v.Count > 0 then
        if v ~= nil and v.MsgName == msgName then
            return true;
        end
    end

    return false;
end

ObjectInteractiveModule = class("ObjectInteractiveModule", BaseLuaModule)

local ObjectInteractiveWidgetPath = "ObjectInteractive/ObjectInteractiveWidget"

ObjectInteractiveModule.IsOpen = false
--LuaModule 消息模块
ObjectInteractiveModule.ModuleMessages = {
    --打开物件交互
    --ModuleManager.Instance.SendMessage(ModuleDef.ObjectInteractiveModule, "OnOpenObjectInteractive", null);
    ["OnOpenObjectInteractive"] = function(...)
        ObjectInteractiveModule:OnOpenObjectInteractive(...)
    end,
    --关闭物件交互
    --ModuleManager.Instance.SendMessage(ModuleDef.ObjectInteractiveModule, "OnCloseObjectInteractive", null);
    ["OnCloseObjectInteractive"] = function(...)
        ObjectInteractiveModule:OnCloseObjectInteractive(...)
    end,
    --刷新物件交互
    --ModuleManager.Instance.SendMessage(ModuleDef.ObjectInteractiveModule, "OnRefeshObjectInteractive", new object[] { it }); //只要传一个掉落信息
    ["OnRefeshObjectInteractive"] = function(...)
        ObjectInteractiveModule:OnRefeshObjectInteractive(...)
    end,
    --刷新本地物件交互
    --ModuleManager.Instance.SendMessage(ModuleDef.ObjectInteractiveModule, "OnRefeshLocalObjectInteractive", new object[] { it }); //只要传一个掉落信息
    ["OnRefeshLocalObjectInteractive"] = function(...)
        ObjectInteractiveModule:OnRefeshLocalObjectInteractive(...)
    end,
    --打开物件交互
    --ModuleManager.Instance.SendMessage(ModuleDef.ObjectInteractiveModule, "OnOpenObjectInteractive", null);
    ["OnUpdateService"] = function(...)
        ObjectInteractiveModule:OnUpdateService(...)
    end,
    --显隐物件交互
    --ModuleManager.Instance.SendMessage(ModuleDef.ObjectInteractiveModule, "OnShowHideObjectInteractive", {true});
    ["OnShowHideObjectInteractive"] = function(...)
        ObjectInteractiveModule:OnShowHideObjectInteractive(...)
    end
}

function ObjectInteractiveModule:OnUpdateService(...)
    --local tb = ...
    --local isShow = false
    ----传入道具ID
    --if type(tb) == "table" then
    ----Lua调用
    --isShow = tb[1]
    --else
    ----C#调用
    --isShow = tb[0]
    --end
    if self.objectInteractiveWidgetView ~= nil then
        self:RefeshInteractiveList()
    end
end

function ObjectInteractiveModule:OnShowHideObjectInteractive(...)
    local tb = ...
    local isShow = false
    --传入道具ID
    if type(tb) == "table" then
        --Lua调用
        isShow = tb[1]
    else
        --C#调用
        isShow = tb[0]
    end

    if self.objectInteractiveWidgetView ~= nil then
        self.objectInteractiveWidgetView.gameObject:SetActive(isShow)
    end
end

--打开掉落监听UI
function ObjectInteractiveModule:OnOpenObjectInteractive(...)
    print("ObjectInteractiveModule:OnOpenObjectInteractive")

    if self.objectInteractiveWidgetView == nil then
        local objectInteractiveWidgetCs = UIManager.Instance:OpenWidget(ObjectInteractiveWidgetPath, false, nil, nil, CS.StarProjectDef.MainPageCommond.HideNone, true, false, true)
        self.objectInteractiveWidgetView = objectInteractiveWidgetCs:GetLuaPanel().ScriptTable
        self.objectInteractiveWidgetView:BindModule(self)
    else
        self.objectInteractiveWidgetView.gameObject:SetActive(true)
    end

    self.IsOpen = true

    --[[
        MonoHelper.AddUpdateListener(
        function()
            self:RefeshInteractiveList()
        end
    )
    ]]
end

function ObjectInteractiveModule:RefeshInteractiveList()
    if not self.IsOpen then
        return
    end

    --判断所有实体到自己的距离
    --local ls=GameManager.Instance:GetAllInteractiveList();
    --self.objectInteractiveWidgetView:RefeshInteractive(ls);

    self.objectInteractiveWidgetView:ClearAllItem() -- 先清空
    -- 优先判断交互物件实体
    local aim_entityID = GameManager.Instance:GetCloestInteractive()
    --print(aim_entityID)
    if aim_entityID ~= 0 then
        self.objectInteractiveWidgetView:RefeshAimInteractive(aim_entityID)
    else
        -- 后判断客户端模拟的交互物件实体
        local aim_localEntity = GameManager.Instance:GetCloestLocalEntity()
        if aim_localEntity ~= nil then
            local Type = aim_localEntity:GetTypeInt()
            local ID = aim_localEntity.ID
            local TriggerKey = aim_localEntity.TriggerKey
            local EntityKey = aim_localEntity.EntityKey
            self.objectInteractiveWidgetView:RefeshLocalInteractive(Type, ID, TriggerKey, EntityKey)
        end
    end
end

--关闭掉落监听UI
function ObjectInteractiveModule:OnCloseObjectInteractive(...)
    self.objectInteractiveWidgetView.gameObject:SetActive(false)
    self.IsOpen = false
end

function ObjectInteractiveModule:OnRefeshObjectInteractive(...)
    local tb = ...

    local interactiveList = nil
    --传入道具ID
    if type(tb) == "table" then
        --Lua调用
        interactiveList = tb[1]
    else
        --tb.Length
        --C#调用
        interactiveList = tb[0]
    end

    --如果列表里的东西一样就不去更新了，做一个优化

    --显示
    self.objectInteractiveWidgetView:RefeshInteractive(interactiveList)
end

function ObjectInteractiveModule:OnRefeshLocalObjectInteractive(...)
    local tb = ...
    local Type = nil -- 本地实体类型
    local ID = nil -- 交互物件ID
    local TriggerKey = nil -- 交互回调key
    local EntityKey = nil -- 特殊的实体key
    --列表
    if type(tb) == "table" then
        --Lua调用
        Type = tb[1]
        ID = tb[2]
        TriggerKey = tb[3]
        EntityKey = tb[4]
    else
        --C#调用
        Type = tb[0]
        ID = tb[1]
        TriggerKey = tb[2]
        EntityKey = tb[3]
    end
    self.objectInteractiveWidgetView:ClearAllItem() -- 先清空
    self.objectInteractiveWidgetView:RefeshLocalInteractive(Type, ID, TriggerKey, EntityKey)
end

--LuaModule 创建
function ObjectInteractiveModule:OnModuleCreate(arg)
    LuaModuleDispatcher:Register(ModuleDef.ObjectInteractiveModule, self)

    self.OnBackLoginFunc = function(againLoginType)
        if self.objectInteractiveWidgetView ~= nil and self.objectInteractiveWidgetView.gameObject ~= nil then
            DOVirtual.DelayedCall(
                1,
                function()
                    -- 清除队列
                    self.objectInteractiveWidgetView:ClearAllItem() -- 先清空
                end
            )
        end
    end
    CS.StarProject.GlobalEvent.OnBackLogin:AddListener(self.OnBackLoginFunc)
end

--LuaModule 消息处理
function ObjectInteractiveModule:OnModuleMessage(msg, ...)
    self.ModuleMessages[msg](...)
end

--LuaModule显示
function ObjectInteractiveModule:OnModuleShow(arg)
end

--LuaModule 释放
function ObjectInteractiveModule:OnModuleRelease()
    LuaModuleDispatcher:UnRegister(ModuleDef.ObjectInteractiveModule)
end

return ObjectInteractiveModule

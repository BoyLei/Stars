function ItemResolveModule:OnItemResolveConfirm(...)
    if self.itemResolveWidgetView ~= nil then
        self.itemResolveWidgetView:ItemResolveReq()
    end
end

function ItemResolveModule:OnOpenItemResolve(...)
    -- local tb=...
    -- local itemResolveWidgetCs =UIManager.Instance:OpenWidget(ItemResolveWidgetPath,false,nil,nil,CS.StarProjectDef.MainPageCommond.HideBoth,false,true);
    -- if self.itemResolveWidgetView == nil then
    -- 	self.itemResolveWidgetView=itemResolveWidgetCs:GetLuaPanel().ScriptTable;
    -- 	self.itemResolveWidgetView:BindModule(self);
    -- 	self.itemResolveWidgetView:InitItems();
    -- 	self.evtOnCloseItemResolve = LuaModuleDispatcher:Event(ModuleDef.ItemResolveModule, "OnCloseItemResolveEvt")
    -- end

    -- self.IsOpenTips=false
    -- --显示
    -- self.itemResolveWidgetView:RefeshItemResolve();

    self.AsyncLoadItemResolveWidgetCb = function(ui)
        if ui ~= nil then
            if self.itemResolveWidgetView == nil then
                self.itemResolveWidgetView = ui:GetLuaPanel().ScriptTable
                self.itemResolveWidgetView:BindModule(self)
                self.itemResolveWidgetView:InitItems()
                self.evtOnCloseItemResolve = LuaModuleDispatcher:Event(ModuleDef.ItemResolveModule, "OnCloseItemResolveEvt")
            end

            self.IsOpenTips = false
            --显示
            self.itemResolveWidgetView:RefeshItemResolve()
        end
        self.AsyncLoadItemResolveWidgetCb = nil
    end
    UIManager.Instance:OpenWidgetAsync(ItemResolveWidgetPath, self.AsyncLoadItemResolveWidgetCb, false, nil, nil, CS.StarProjectDef.MainPageCommond.HideBoth, false, false)
end

function ItemResolveModule:OnOpenItemResolveConfirmWidget(...)
    local tb = ...

    local data = nil
    --列表
    local parentWind = nil
    if type(tb) == "table" then
        --Lua调用
        data = tb[1]
    else
        --C#调用
        data = tb[0]
    end

    self.AsyncLoadItemResolveConfirmWidgetCb = function(ui)
        if ui ~= nil then
            if self.ItemResolveConfirmWidgetView == nil then
                self.ItemResolveConfirmWidgetView = ui:GetLuaPanel().ScriptTable
                self.ItemResolveConfirmWidgetView:BindModule(self)
                self.ItemResolveConfirmWidgetView:InitItems()
            end

            self.IsOpenTips = false
            --显示
            self.ItemResolveConfirmWidgetView:RefeshRewards(data)
        end
        self.AsyncLoadItemResolveConfirmWidgetCb = nil
    end
    UIManager.Instance:OpenWidgetAsync(ItemResolveConfirmWidgetPath, self.AsyncLoadItemResolveConfirmWidgetCb, false, nil, nil, CS.StarProjectDef.MainPageCommond.HideBoth, false, true)
end

function ItemResolveModule:OpenItemTips(baseID, entityID, tr)
    --print("ItemResolveModule:OpenItemTips:"..baseID);
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnOpenTips", {baseID, false, entityID, tr, 0})
end

function ItemResolveModule:CloseItemTips()
    print("ItemResolveModule:CloseItemTips")
    LuaModuleDispatcher:SendMessage(ModuleDef.ItemTipsModule, "OnCloseTips", {})
end

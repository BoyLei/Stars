        nil
    )

        function(isOpen)
            if self.ectypeBagWindow ~= nil then

            end
        end

    end

end

function EctypeBagModule:CreateLocalBox(CommonDrop, CommonChestNum, ExtraChestNum, pos)
    local rewards = self:FormattingReward(CommonDrop) -- 格式化奖励
    local rewardSum = #rewards -- 奖励总数量
    local ownerId = 0
    local boxCount = CommonChestNum + ExtraChestNum -- 宝箱总数量
    local averageBoxNum = math.floor((rewardSum / boxCount)) -- 平均奖励数量
    local cha = rewardSum - (averageBoxNum * boxCount)
    local lastNum = averageBoxNum + cha -- 最后一个的数量
    self.BirthPos = Vector3.zero -- 怪物死亡位置
    self.BirthPos.x = pos.x
    self.BirthPos.y = pos.Y
    self.BirthPos.z = pos.Z

    local pos = GameManager.Instance:GetEntityPosById(GameManager.Instance.mainPlayerId)

    local mapPox = GameManager.Instance:GetMapBossPos()
    if mapPox ~= Vector3.zero then
        pos = mapPox
    end

    --随机下位置
    local pos2 = BusinessManager.Instance:GetRandomPos(pos, 2.5)

    for i = 0, 0 do
        local startIndex = 1 + (i * averageBoxNum)
        local num = averageBoxNum
        local InteractId = 9999997 -- 宝箱种
        local rewardList = self:GetRewards(rewards, startIndex, num)
        CS.StarProject.GlobalEvent.onCreateLocalEntity:Invoke(E_LocalEntityType.TreasureBox, InteractId, E_LocalEntitySource.DailyEctype, pos2, rewardList, true, -1)
    end
end

-- 格式化奖励 转换成一维数组的奖励数据结构
function EctypeBagModule:FormattingReward(CommonDrops)
    local rewards = {}
    local infos = CommonDrops.Infos
    for j = 0, infos.Count - 1 do
        local data = infos[j]
        table.insert(rewards, data)
    end
    return rewards
end

-- 拿取奖励 [开始下标、数量]
function EctypeBagModule:GetRewards(reward, startIndex, num)
    local drops = LocalDropList()
    for i = 1, #reward do
        if i >= startIndex and i < startIndex + num then
            local dropinfo = reward[i]
            drops:AddData(dropinfo.Itemid, dropinfo.ItemNum, dropinfo.EntityID)
        end
    end
    return drops
end

function EctypeBagModule:OnDropNotify(data, pos)
    local infos = data.Infos
    pos.y = pos.y + 1.0

    local dropIdx = 0

    for i = 0, infos.Count - 1 do
        local dropinfo = infos[i]

        local EntityID = dropinfo.EntityID
        local Itemid = dropinfo.Itemid
        local ItemNum = dropinfo.ItemNum

        -- 排除角色经验,冒险经验
        if Itemid ~= 111 and Itemid ~= 101 then
            dropIdx = dropIdx + 1

            -- 生成掉落表现
            DOVirtual.DelayedCall(
                0.15 * dropIdx,
                function()
                    -- 生成的掉落物品
                    LuaModuleDispatcher:SendMessage(ModuleDef.LootControllerModule, "OnAddLoot", {Itemid, pos, ItemNum, EntityID})
                end
            )
        end
    end
end

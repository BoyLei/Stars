-- LuaModule 创建
function SecretAreaModule:OnModuleCreate(arg)
    LuaModuleDispatcher:Register(ModuleDef.SecretAreaModule, self)

    -- 申请进入个人秘境的回复
    self.EnterPersonSercetRetCmd = function(data)
        self:OnEnterPersonSercetRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.EnterPersonSercetRetID, self.EnterPersonSercetRetCmd, nil)
    -- 秘境当前击杀进度值
    self.PersonSecretLevelDataRetCmd = function(data)
        self:OnPersonSecretLevelData(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.PersonSecretLevelDataRetID, self.PersonSecretLevelDataRetCmd, nil)
    -- 个人秘境结算消息
    self.GameEndPersonSecretRetCmd = function(data)
        self:OnGameEndPersonSecret(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.GameEndPersonSecretRetID, self.GameEndPersonSecretRetCmd, nil)
    -- 个人秘境 s->c 结算信息
    self.PersonSecretReviveRetCmd = function(data)
        self:OnPersonSecretReviveRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.PersonSecretReviveRetID, self.PersonSecretReviveRetCmd, nil)
    -- 领取个人秘境昨日奖励
    self.PersonSecretDayAwardRetCmd = function(data)
        self:OnPersonSecretDayAwardRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.PersonSecretDayAwardRetID, self.PersonSecretDayAwardRetCmd, nil)
    ---请求领取个人秘境成就奖励
    self.PersonSecretAchieveAwardRetCmd = function(data)
        self:OnPersonSecretAchieveAwardRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.PersonSecretAchieveAwardRetID, self.PersonSecretAchieveAwardRetCmd, nil)
    ---请求领取个人秘境每日任务奖励
    self.PSGetDailyRewardRetCmd = function(data)
        self:OnPSGetDailyRewardRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.PSGetDailyRewardRetID, self.PSGetDailyRewardRetCmd, nil)
    -- 个人秘境数据(全量/差量同步)
    self.OnFixMessagePersonSecretCmd = function(data)
        self:OnFixMessagePersonSecret(data)
    end
    FixMessageManager.Instance:OnMessage(FixUpdateDef.PersonSecret, self.OnFixMessagePersonSecretCmd)
    -- 上个赛季排名领奖 返回
    self.PersonSecretLastRewardRetCmd = function(data)
        self:OnPersonSecretLastRewardRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.PersonSecretLastRewardRetID, self.PersonSecretLastRewardRetCmd, nil)
    -- 本地实体点击回调
    self.OnClickLocalEntityFun = function(Type, ID, TriggerKey, Key)
        self:OnClickLocalEntity(Type, ID, TriggerKey, Key)
    end
    CS.StarProject.GlobalEvent.onClickLocalEntity:AddListener(self.OnClickLocalEntityFun)
    -- 主角创建完成（用于断线重连）
    self.OnMainPageLoadedCompleteFun = function(res)
        self:OnMainPageLoadedCompleteCB(res)
    end
    CS.StarProject.GlobalEvent.OnMainPageLoadedComplete:AddListener(self.OnMainPageLoadedCompleteFun)

    ---PersonSecretMissions
    self.GotAchieveList = {}
    self.AchieveList = {}
    self.MissionDatas = {}
    local list = ConfigDataAccessor.SeasonAchieve
    if list ~= nil then
        for k, v in pairs(list) do
            if v ~= nil then
                self.GotAchieveList[v.AchieveID] = false
                self.AchieveList[v.SubType] = 0
                -- 格式化秘境成就
                local val = {
                    State = 0,
                    Config = v
                }
                self.MissionDatas[v.AchieveID] = val -- 0:未完成 1：可领取 2：已领取
            end
        end
    end
end

-- LuaModule 释放
function SecretAreaModule:OnModuleRelease()
    -- 申请进入个人秘境的回复
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.EnterPersonSercetRetID, self.EnterPersonSercetRetCmd, nil)
    -- 秘境当前击杀进度值
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.PersonSecretLevelDataRetID, self.PersonSecretLevelDataRetCmd, nil)
    -- 个人秘境结算消息
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.GameEndPersonSecretRetID, self.GameEndPersonSecretRetCmd, nil)
    -- 个人秘境 s->c 结算信息
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.PersonSecretReviveRetID, self.PersonSecretReviveRetCmd, nil)
    -- 领取个人秘境昨日奖励
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.PersonSecretDayAwardRetID, self.PersonSecretDayAwardRetCmd, nil)
    ---请求领取个人秘境成就奖励
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.PersonSecretAchieveAwardRetID, self.PersonSecretAchieveAwardRetCmd, nil)
    -- 差量同步
    FixMessageManager.Instance:OffMessage(FixUpdateDef.PersonSecret, self.OnFixMessagePersonSecretCmd)
    -- 上个赛季排名领奖 返回
    NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.PersonSecretLastRewardRetID, self.PersonSecretLastRewardRetCmd, nil)
    CS.StarProject.GlobalEvent.onClickLocalEntity:RemoveListener(self.OnClickLocalEntityFun)

    CS.StarProject.GlobalEvent.OnMainPageLoadedComplete:RemoveListener(self.OnMainPageLoadedCompleteFun)

    self.EnterFloor = -1
    LuaModuleDispatcher:UnRegister(ModuleDef.SecretAreaModule)
end

            if v ~= nil then
                self.GotAchieveList[v.AchieveID] = false
                self.AchieveList[v.SubType] = 0
                -- 格式化秘境成就
                local val = {
                    State = 0,
                    Config = v
                }
                self.MissionDatas[v.AchieveID] = val -- 0:未完成 1：可领取 2：已领取
            end
        end
    end

    m_RedPointAchieveFloorCount = 0
    m_RedPointAchieveActiveCount = 0
    m_RedPointAchieveChallengeCount = 0
    m_RedPoinDailyRewardCount = 0
    m_RedPointDailyTaskCount = 0

    RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaAchieveActive, m_RedPointAchieveActiveCount)
    RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaAchieveChallenge, m_RedPointAchieveChallengeCount)
    RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaDailyReward, m_RedPoinDailyRewardCount)
    RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaDailyTask, m_RedPointDailyTaskCount)
end

function SecretAreaModule:OnCustomLevelEctype(...)

    self:CloseSecretAreaTaskWidget()
    self:CloseSecretAreaSueecssWindow()
    self:CloseSecretAreaFailWindow()

    self:CloseSecretAreaSuccessTipsWidget()

	if self.SeasonPassFloor > SystemConstConfigs.PersonSecret_ExitWithoutMainHud and self.EnterFloor > SystemConstConfigs.PersonSecret_ExitWithoutMainHud then
		-- 从个人秘境副本退出后，回到个人秘境玩法界面
		self:OpenSecretAreaWindow()
	end
end

---comment 打开赛季排名领奖结算界面
---@param LastSeasonID any 赛季ID
---@param LastSeasonRank any 赛季排名
---@param LastSeasonEndTime any 赛季结算时间
function SecretAreaModule:OpenRankRewardPartWidgetPath(LastSeasonID, LastSeasonRank, LastSeasonEndTime)
    if LastSeasonID <= 0 or LastSeasonRank <= 0 then
        return
    end
    local data = {
        CostItems = {},
        CutOffTime = LastSeasonEndTime,
        Rank = LastSeasonRank
    }
    -- 奖励
    local rewardCfg = ConfigDataAccessor.GetSecretRankClientBySeasonIDAndRank(LastSeasonID, LastSeasonRank)
    if rewardCfg ~= nil then
        for i = 1, #rewardCfg.RewardID do
            local itemID = rewardCfg.RewardID[i]
            local rewardNum = rewardCfg.RewardNum[i]
            if itemID ~= nil and rewardNum ~= nil then
                table.insert(
                    data.CostItems,
                    {
                        ItemID = itemID,
                        ItemNum = rewardNum
                    }
                )
            end
        end
    end

    -- UIManager.Instance:OpenWidget(
    --     RankRewardPartWidgetPath,
    --     true,
    --     data,
    --     nil,
    --     CS.StarProjectDef.MainPageCommond.HideNone,
    --     false,
    --     true
    -- )
    UIManager.Instance:OpenWidgetAsync(RankRewardPartWidgetPath, nil, true, data, nil, CS.StarProjectDef.MainPageCommond.HideNone)
end

-- 打开秘境主界面
function SecretAreaModule:OpenSecretAreaWindow()
    local data = {
        PassFloor = self.PassFloor,
        Rank = self.Rank,
        UnlockFloor = self.UnlockFloor,
        CurSeasonID = self.CurSeasonID
    }
    -- local secretAreaWindow = UIManager.Instance:OpenWindow(
    --     SecretAreaWindowPath,
    --     data,
    --     CS.StarProjectDef.MainPageCommond.HideBoth,
    --     true
    -- )
    -- if self.SecretAreaWindow == nil or self.SecretAreaWindow.IsOpen == false then
    --     self.SecretAreaWindow = secretAreaWindow:GetLuaPanel().ScriptTable
    -- end
    self.AsyncLoadSecretAreaWindowCb = function(ui)
        if ui ~= nil then
            if self.SecretAreaWindow == nil or self.SecretAreaWindow.IsOpen == false then
                self.SecretAreaWindow = ui:GetLuaPanel().ScriptTable
                self.SecretAreaWindow:BindModule(self)
                self.SecretAreaWindow:SetRankTime()
            end
        end
        self.AsyncLoadSecretAreaWindowCb = nil
    end
    UIManager.Instance:OpenWindowAsync(SecretAreaWindowPath, self.AsyncLoadSecretAreaWindowCb, data, CS.StarProjectDef.MainPageCommond.HideBoth)

    self:SendPersonSecretRankReq()
    self:SendPersonSecretLastRankReq()

    if self.LastSeasonRank > 0 then
        -- 上个赛季排名 如果存在ID说明没有领奖
        -- 自动弹出ID
        -- 发送给服务器领取奖励
        self:SendPersonSecretLastRewardReq()
    end
end

-- 关闭秘境主界面
function SecretAreaModule:CloseSecretAreaWindow()
    if self.SecretAreaWindow ~= nil and self.SecretAreaWindow.IsOpen then
        self.SecretAreaWindow.Close()
    else
        UIManager.Instance:CloseWindow(SecretAreaWindowPath);
    end
    self.SecretAreaWindow = nil
end

-- 打开左边秘境任务挂件
function SecretAreaModule:OpenSecretAreaTaskWidget(action)
    if self.TaskRoot == nil then
        return
    end
    if self.TaskWidget ~= nil and self.TaskWidget.IsOpen then
        if self.CacheLevelData ~= nil then
            self.TaskWidget:UpdateTaskData(self.CacheLevelData)
        end
        if action ~= nil then
            action()
        else
            self.TaskWidget:InitData(self.EnterFloor)
        end
    else
        self.AsyncLoadTaskWidgetCb = function(ui)
            if ui ~= nil then
                if self.TaskWidget == nil then
                    self.TaskWidget = ui:GetLuaPanel().ScriptTable
                    self.TaskWidget:BindModule(self)
                end
                if self.CacheLevelData ~= nil then
                    self.TaskWidget:UpdateTaskData(self.CacheLevelData)
                end
                if action ~= nil then
                    action()
                end
            end
            self.AsyncLoadTaskWidgetCb = nil
        end
        UIManager.Instance:OpenWidgetAsync(TaskWidgetPath, self.AsyncLoadTaskWidgetCb, true, self.EnterFloor, self.TaskRoot.transform, CS.StarProjectDef.MainPageCommond.HideNone)
    end

    -- 关闭秘境主界面
    self:CloseSecretAreaWindow()
end

-- 关闭左边秘境任务挂件
function SecretAreaModule:CloseSecretAreaTaskWidget()
    if self.TaskWidget ~= nil and self.TaskWidget.IsOpen then
        -- self.TaskWidget.Close();
        UIManager.Instance:CloseWidget(TaskWidgetPath, self.TaskRoot.transform, false)
    end
    IsMainPageLoadedCompleteOpenWidget = false
end

-- 打开秘境复活界面
function SecretAreaModule:OpenSecretAreaReviveWidget(...)
    local tb = ...
    local num = 0
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        num = tb[1]
    else
        -- C#调用
        num = tb[0]
    end
    -- 是否复活，已经复活的次数，当前层数
    local data = {
        IsRevive = true,
        ReviveCount = num,
        Floor = self.EnterFloor
    }
    -- 1.没有打开过失败界面
    if self.FailWindow == nil then
        self:OpenSecretAreaFailWindow(data)
    elseif self.FailWindow ~= nil and self.FailWindow.IsOpen == false then
        -- 2.有界面并且当前没打开
        self:OpenSecretAreaFailWindow(data)
    end
end

-- 个人秘境数据(全量/差量同步)
function SecretAreaModule:OnFixMessagePersonSecret(data)
    local IsPartial = true -- 是否差量
    if data.AllDatas ~= nil and data.AllDatas.Count > 0 then
        IsPartial = false
    end

    if IsAlreadySyncAllSecretData == false and IsPartial == true then
        -- 如果没有收到全量同步、先收到了差量，就过滤
        return
    end

    local pid = GameManager.Instance.mainPlayerId
    local md = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.PersonSecret):GetMDByPid(pid)
    if md == nil then
        md = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.PersonSecret):GetMD()
    end

    if md ~= nil then
        self.PassFloor = md.PassFloor --- 今日通过层数
        self.LastPassFloor = md.LastPassFloor --- 昨日通过层数
        self.SeasonPassFloor = md.SeasonPassFloor --- 赛季历史通过层数
        self.UnlockFloor = md.UnlockFloor --- 已解锁的最大层数
        self.Rank = md.SeasonRank --- 当前赛季排名
        self.CurSeasonID = md.SeasonID --- 当前赛季ID
        --Log("当前赛季 服务器全量同步 =" .. self.CurSeasonID);
        self.LastSeasonRank = md.LastSeasonRank --- 上个赛季排名 如果存在ID说明没有领奖
        self.SeasonEndTime = md.RefrshTime ---本赛季结算时间
        self.DailyChallange = md.DailyChallange
        self.DailyRewardRecords = md.DailyRewardRecords
		self.EnterFloor = md.InFloorID
        --- 成就进度数据
        for k, v in pairs(md.ProgressList.Data) do
            self.AchieveList[k] = v
        end
        --- 已领奖成就列表
        for k, v in pairs(md.GotList.Data) do
            self.GotAchieveList[k] = v
        end
        -- 全量就同步后，初始化一次
        if IsPartial == false then
            -- 成就奖励状态格式化成客户端
            IsAlreadySyncAllSecretData = true
            self:InitMissionDatas()
            -- 每日任务格式化成客户端
            self:InitDailyTaskData()
        else
            -- 成就差量更新
            CS.StarProject.GlobalEvent.OnPersonSecretAchieveChange:Invoke(
                {
                    AchieveList = self.AchieveList,
                    GotAchieveList = self.GotAchieveList
                }
            )
            -- 有刚可以领取的弹提示框
            self:SendAchievementMessage()

            self:OnRedPointDailyTaskCount()
        end
        -- 更新界面全量数据
        if self.SecretAreaWindow ~= nil and self.SecretAreaWindow.IsOpen then
            self.SecretAreaWindow:UpdataAllData(self.PassFloor, self.Rank, self.UnlockFloor, self.CurSeasonID)
        end
        -- 刷新赛季界面信息
        self:UpdateSecretAreaRankWidget()
        ---获取赛季最大层数
        self:GetSeasonMaxFloor()
        -- 缓存主角的系统玩法数据
        GameManager.Instance.SeasonPassFloor = self.SeasonPassFloor

        local RedPoinDailyRewardCount = 0
        if self.LastPassFloor > 0 then
            RedPoinDailyRewardCount = 1
        end
        if m_RedPoinDailyRewardCount ~= RedPoinDailyRewardCount then
            m_RedPoinDailyRewardCount = RedPoinDailyRewardCount
            RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaDailyReward, m_RedPoinDailyRewardCount)
        end
    end
end

-- 成就奖励状态格式化成客户端
function SecretAreaModule:InitMissionDatas()
    local RedPointAchieveFloorCount = 0
    local RedPointAchieveActiveCount = 0
    local RedPointAchieveChallengeCount = 0
    for k, v in pairs(self.MissionDatas) do
        if v ~= nil then
            local config = v.Config
            local value = 0
            if self.AchieveList ~= nil then
                if self.AchieveList[config.SubType] ~= nil then
                    value = self.AchieveList[config.SubType]
                end
            end
            local state = 0 -- 0:未完成 1：可领取 2：已领取
            if self.GotAchieveList ~= nil then
                if self.GotAchieveList[k] == true then
                    state = 2
                end
            end
            if state == 0 and value >= config.TargetNum then
                state = 1
            end
            v.State = state

            if state == 1 then
                if config.AchieveType == 1 then
                    RedPointAchieveFloorCount = RedPointAchieveFloorCount + 1
                elseif config.AchieveType == 2 then
                    RedPointAchieveActiveCount = RedPointAchieveActiveCount + 1
                elseif config.AchieveType == 3 then
                    RedPointAchieveChallengeCount = RedPointAchieveChallengeCount + 1
                end
            end
        end
    end
    if RedPointAchieveFloorCount ~= m_RedPointAchieveFloorCount then
        m_RedPointAchieveFloorCount = RedPointAchieveFloorCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaAchieveFloor, m_RedPointAchieveFloorCount)
    end
    if RedPointAchieveActiveCount ~= m_RedPointAchieveActiveCount then
        m_RedPointAchieveActiveCount = RedPointAchieveActiveCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaAchieveActive, m_RedPointAchieveActiveCount)
    end
    if RedPointAchieveChallengeCount ~= m_RedPointAchieveChallengeCount then
        m_RedPointAchieveChallengeCount = RedPointAchieveChallengeCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaAchieveChallenge, m_RedPointAchieveChallengeCount)
    end
end

function SecretAreaModule:InitDailyTaskData()
    self.DailyChallangeCfg = {}
    local list = ConfigDataAccessor.SercretDailyTask
    if self.DailyChallange == nil then
        m_RedPointDailyTaskCount = RedPointDailyTaskCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaDailyTask, m_RedPointDailyTaskCount)
        return
    end
    local DailyChallange = self.DailyChallange.List

    for k, v in pairs(DailyChallange) do
        local item = {
            TaskID = k,
            Cfg = nil,
            CurProgress = v,
            TargetNum = 1,
            IsGet = false
        }
        item.Cfg = list[k]
        item.IsGet = self:GetDailyRewardRecordsState(k)
        item.TargetNum = tonumber(item.Cfg.Paramas[#item.Cfg.Paramas])
        self.DailyChallangeCfg[k] = item
        -- table.insert(self.DailyChallangeCfg, item)
    end

    local RedPointDailyTaskCount = 0
    for k, v in pairs(self.self.DailyChallangeCfg) do
        if v.IsGet then
        else
            if v.CurProgress >= v.TargetNum then
                RedPointDailyTaskCount = RedPointDailyTaskCount + 1
            end
        end
    end
    if RedPointDailyTaskCount ~= m_RedPointDailyTaskCount then
        m_RedPointDailyTaskCount = RedPointDailyTaskCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaDailyTask, m_RedPointDailyTaskCount)
    end
end

function SecretAreaModule:GetDailyRewardRecordsState(taskID)
    if self.DailyRewardRecords == nil then
        return false
    end
    local DailyRewardRecords = self.DailyRewardRecords.List
    for k, v in pairs(DailyRewardRecords) do
        if k == taskID then
            return v
        end
    end
    return false
end

        -- 门票道具不足
        str = LanguageHelp:GetLanguageByKey("ShowSystemMessage_New_34")
        self.EnterFloor = -1

        self.EnterFloor = -1
    end

    if str ~= "" then
        CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(str)
        if self.SecretAreaWindow ~= nil and self.SecretAreaWindow.IsOpen then
            self.SecretAreaWindow:BackWindowAnim()
        end
    end

-- 秘境结算消息回调
function SecretAreaModule:OnGameEndPersonSecret(data)
    local res = data.data
    if res.IsSuccess == 1 then
        self:CreateLocalBox(res.CommonDrop, res.FirstReward, res.CommonChestNum, res.ExtraChestNum, res.BossDeadPos)
        local unlockFloor = self.EnterFloor + res.Unlock
        if unlockFloor > self.m_MaxFloors then
            unlockFloor = self.m_MaxFloors
        end
        self.SuccessData = {
            ResData = res,
            UnlockFloor = unlockFloor,
            CurFloor = self.EnterFloor,
            LastUnlockFloor = self.UnlockFloor,
            MaxFloors = self.m_MaxFloors
        }
        self:UpdatePersonSecretTask2State(true)
        -- self:OpenSecretAreaSueecssWindow(data);
        self:OpenSecretAreaSuccessTipsWidget(self.SuccessData)
        self:CloseSecretAreaBuffTipsWidget()
    else
        -- 秘境失败结算
        if res.NeedShow then
            local data = {
                IsRevive = false,
                ReviveCount = 1,
                Floor = self.EnterFloor
            }
            self:OpenSecretAreaFailWindow(data)
        end
        self:CloseSecretAreaBuffTipsWidget()
    end
end

-- 创建本地宝箱 [掉落道具、普通宝箱数量、额外宝箱数量、]
function SecretAreaModule:CreateLocalBox(CommonDrop, FirstReward, CommonChestNum, ExtraChestNum, pos)
    local rewards = self:FormattingReward(CommonDrop) -- 格式化奖励
    --rewards = self:FormattingFirstReward(rewards, FirstReward);
    local rewardSum = #rewards -- 奖励总数量
    local ownerId = 0
    local boxCount = CommonChestNum -- 宝箱总数量
    --local boxCount = CommonChestNum + ExtraChestNum -- 宝箱总数量
    local averageBoxNum = math.floor((rewardSum / boxCount)) -- 平均奖励数量
    local cha = rewardSum - (averageBoxNum * boxCount)
    local lastNum = averageBoxNum + cha -- 最后一个的数量
    self.BirthPos = Vector3.zero -- 怪物死亡位置
    self.BirthPos.x = pos.X
    self.BirthPos.y = pos.Y
    self.BirthPos.z = pos.Z

    --[[
        local mapPox = GameManager.Instance:GetMapBossPos()
    if mapPox ~= Vector3.zero then
        self.BirthPos = mapPox
    end
    ]]
    for i = 1, boxCount do
        local startIndex = 1
        if i > 1 then
            startIndex = 1 + ((i - 1) * averageBoxNum)
        end
        local num = averageBoxNum
        local InteractId = CommonChestID -- 普通宝箱id
        -- if i == boxCount then
        --     --num = lastNum
        --     --if ExtraChestNum > 0 then
        --     InteractId = ExtraChestID -- 每日宝箱id
        -- --end
        -- end
        local rewardList = self:GetRewards(rewards, startIndex, num)

        local pos = Vector3.zero
        if boxCount == 3 then
            if i == 1 then
                pos.x = self.BirthPos.x - 2.5
                pos.y = self.BirthPos.y
                pos.z = self.BirthPos.z
            elseif i == 2 then
                pos.x = self.BirthPos.x + 2.5
                pos.y = self.BirthPos.y
                pos.z = self.BirthPos.z
            else
                pos.x = self.BirthPos.x
                pos.y = self.BirthPos.y
                pos.z = self.BirthPos.z + 1
            end
        else
            pos.x = self.BirthPos.x + i * 2.5
            pos.y = self.BirthPos.y
            pos.z = self.BirthPos.z
        end
        pos.z = pos.z + 2.5
        CS.StarProject.GlobalEvent.onCreateLocalEntity:Invoke(E_LocalEntityType.TreasureBox, InteractId, E_LocalEntitySource.SecretArea, pos, rewardList, true, -1)
    end
end

-- 格式化奖励 转换成一维数组的奖励数据结构
function SecretAreaModule:FormattingReward(CommonDrops)
    local rewards = {}
    for i = 0, CommonDrops.Count - 1 do
        local infos = CommonDrops[i].Infos
        for j = 0, infos.Count - 1 do
            local data = infos[j]
            table.insert(rewards, data)
        end
    end
    return rewards
end

function SecretAreaModule:FormattingFirstReward(list, FirstReward)
    if FirstReward ~= nil and FirstReward.Data ~= nil then
        local entityID = GameManager.Instance.mainPlayerId
        for k, v in pairs(FirstReward.Data) do
            local dropinfo = {Itemid = k, ItemNum = v, EntityID = entityID};
            table.insert(list, dropinfo)
        end
    end
    return list;
end

-- 拿取奖励 [开始下标、数量]
function SecretAreaModule:GetRewards(reward, startIndex, num)
    local drops = LocalDropList()
    for i = 1, #reward do
        if i >= startIndex and i < startIndex + num then
            local dropinfo = reward[i]
            drops:AddData(dropinfo.Itemid, dropinfo.ItemNum, dropinfo.EntityID)
        end
    end
    return drops
end

-- 显示成就提示
function SecretAreaModule:SendAchievementMessage()
    if self.AchieveList == nil then
        return
    end
    if self.GotAchieveList == nil then
        return
    end

    local RedPointAchieveFloorCount = 0
    local RedPointAchieveActiveCount = 0
    local RedPointAchieveChallengeCount = 0

    for k, v in pairs(self.MissionDatas) do
        if v ~= nil then
            local config = v.Config
            local oldState = v.State

            local value = 0
            if self.AchieveList ~= nil then
                if self.AchieveList[config.SubType] ~= nil then
                    value = self.AchieveList[config.SubType]
                end
            end

            local newState = 0 -- 0:未完成 1：可领取 2：已领取
            if self.GotAchieveList ~= nil then
                if self.GotAchieveList[k] == true then
                    -- newState = self.GotAchieveList[k]
                    newState = 2
                end
            end

            if newState == 0 and value >= config.TargetNum then
                newState = 1
            end

            if oldState == 0 and newState == 1 then
                -- local message = "达成 " .. config.AchieveDec
                local message = StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_SecretAreaModule_AchievementMessage"), {config.AchieveDec})
                CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:AddAchievementMessage(message)
                v.State = newState
            end

            if newState == 1 then
                if config.AchieveType == 1 then
                    RedPointAchieveFloorCount = RedPointAchieveFloorCount + 1
                elseif config.AchieveType == 2 then
                    RedPointAchieveActiveCount = RedPointAchieveActiveCount + 1
                elseif config.AchieveType == 3 then
                    RedPointAchieveChallengeCount = RedPointAchieveChallengeCount + 1
                end
            end
        end
    end
    if RedPointAchieveFloorCount ~= m_RedPointAchieveFloorCount then
        m_RedPointAchieveFloorCount = RedPointAchieveFloorCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaAchieveFloor, m_RedPointAchieveFloorCount)
    end
    if RedPointAchieveActiveCount ~= m_RedPointAchieveActiveCount then
        m_RedPointAchieveActiveCount = RedPointAchieveActiveCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaAchieveActive, m_RedPointAchieveActiveCount)
    end
    if RedPointAchieveChallengeCount ~= m_RedPointAchieveChallengeCount then
        m_RedPointAchieveChallengeCount = RedPointAchieveChallengeCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaAchieveChallenge, m_RedPointAchieveChallengeCount)
    end
end

function SecretAreaModule:OnRedPointDailyTaskCount()
    self.DailyChallangeCfg = {}
    local list = ConfigDataAccessor.SercretDailyTask

    local DailyChallange = self.DailyChallange.List
    for k, v in pairs(DailyChallange) do
        local item = {
            TaskID = k,
            Cfg = nil,
            CurProgress = v,
            TargetNum = 1,
            IsGet = false
        }
        item.Cfg = list[k]
        item.IsGet = self:GetDailyRewardRecordsState(k)
        item.TargetNum = tonumber(item.Cfg.Paramas[#item.Cfg.Paramas])
        self.DailyChallangeCfg[k] = item
        -- table.insert(self.DailyChallangeCfg, item)
    end

    local RedPointDailyTaskCount = 0
    for k, v in pairs(self.self.DailyChallangeCfg) do
        if v.IsGet then
        else
            if v.CurProgress >= v.TargetNum then
                RedPointDailyTaskCount = RedPointDailyTaskCount + 1
            end
        end
    end
    if RedPointDailyTaskCount ~= m_RedPointDailyTaskCount then
        m_RedPointDailyTaskCount = RedPointDailyTaskCount
        RedPointManager.Instance:RefreshRedPointCount(RedPointType.SecretareaDailyTask, m_RedPointDailyTaskCount)
    end
    -- 通知界面刷新
end

-- 重置父节点到hud
function ChatModule:OnResetRoot(...)
    local tb = ...

    if self.chatHudWidget ~= nil then
        self.chatHudWidget.Transform:SetParent(self.OldParent)
        self.chatHudWidget.RectTr.anchoredPosition = Vector2(0, 110)
    end
end

-- 改变父节点
function ChatModule:OnChangeRoot(...)
    local tb = ...

    local rectPos = nil
    local root = nil
    -- 传入道具ID
    if type(tb) == "table" then
        -- Lua调用
        root = tb[1]
        rectPos = tb[2]
    else
        -- C#调用
        root = tb[0]
        rectPos = tb[1]
    end

    if self.chatHudWidget ~= nil then
        self.chatHudWidget.Transform:SetParent(root)
        self.chatHudWidget.RectTr.anchoredPosition = rectPos
    end
end

function ChatModule:OnOpenChatInput(...)
    local tb = ...
    local ip = nil
    local sources = 0

    local textFromChannelWindow = ""
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        ip = tb[1]
        sources = tb[2]
        if #tb > 2 then
            textFromChannelWindow = tb[3]
        end
    else
        -- C#调用
        ip = tb[0]
        sources = tb[1]
        if tb.Length > 2 then
            textFromChannelWindow = tb[2]
        end
    end

    self.AsyncLoadChatInputCb = function(ui)
        if ui ~= nil then
            if self.chatInputWindow == nil then
                self.chatInputWindow = ui:GetLuaPanel().ScriptTable
                self.chatInputWindow:BindModule(self)
                self.chatInputWindow:Init()
            end
            self.chatInputWindow:SyncInput(ip, sources, textFromChannelWindow)
        end
        self.AsyncLoadChatInputCb = nil
    end
    UIManager.Instance:OpenWindowAsync(ChatInputPath, self.AsyncLoadChatInputCb, nil, CS.StarProjectDef.MainPageCommond.HideNone)
end

function ChatModule:MsgFilter(chatMsgNotice)
    local isFilter = false
    if chatMsgNotice.Channel == CS.ProtoMsg.ChatChannel.ChannelPrivate then
        isFilter = true
    end
    return isFilter
end

function ChatModule:OnChatMsgNoticeS(data)
    if data == nil or data.data == nil or data.data.CMNotices == nil then
        return
    end

    Log("聊天信息长度=》" .. data.data.CMNotices.Count)
    local channelPrivateList = {}
    if data.data.CMNotices.Count > 0 then
        for i = 0, data.data.CMNotices.Count - 1 do
            -- 黑名单的就过滤
            local item = data.data.CMNotices[i]
            local isMyBlackList = self:GetChatSendPidIsMyBlackList(item)
            if isMyBlackList == false then
                -- 如果是私聊就不在任何一个频道显示
                if self.chatChannelWindowCS ~= nil then
                    local res = self:MsgFilter(item)
                    if res then
                        table.insert(channelPrivateList, item)
                    else
                        self.chatChannelWindowCS:AddMsgItem(item)
                    end
                end
                -- 通知 -> [弹幕挂件]
                if self.chatBarrageWidget ~= nil and self.chatBarrageWidget.IsShow then
                    self.chatBarrageWidget:AddMsgItem(item)
                end
                -- 通知HUD上的聊天挂件
                if self.chatHudWidget ~= nil and self.chatHudWidget.IsOpen then
                    self.chatHudWidget:AddMsgItem(item)
                end
            end
        end
    end
    if channelPrivateList ~= {} and #channelPrivateList > 0 then
        LuaModuleDispatcher:SendMessage(ModuleDef.FriendModule, "OnSyncMsgData", {channelPrivateList})
    end
end

-- 获取聊天发送的PID是否是我的黑名单
function ChatModule:GetChatSendPidIsMyBlackList(CMNotices)
    if FriendModule == nil then
        FriendModule = LuaModuleDispatcher:GetModule(ModuleDef.FriendModule)
    end
    if CMNotices == nil then
        return false
    end
    local Channel = CMNotices.Channel
    if Channel == CS.ProtoMsg.ChatChannel.ChannelSystem then
        return false
    end

    local friendRelation = FriendModule:GetFriendRelation(CMNotices.PID)

    return friendRelation == 2
end

-- 发送聊天信息
------ 文本信息 包含【标签-道具】
------ 发送频道
------ 字节流 包含【每个类型的特殊信息】
------ PID
function ChatModule:OnSendChatMsg(...)
    local tb = ...
    local chatMsg = ""
    local chatChannel = CS.ProtoMsg.ChatChannel.ChannelNull
    local extrainfos = nil
    local PID = 0
    if type(tb) == "table" then
        -- Lua调用
        chatMsg = tb[1]
        chatChannel = tb[2]
        extrainfos = tb[3]
        PID = tb[4]
    else
        -- C#调用
        chatMsg = tb[0]
        chatChannel = tb[1]
        extrainfos = tb[2]
        PID = tb[3]
    end

    local sendChatMsgReq = CS.ProtoMsg.SendChatMsgReq()
    sendChatMsgReq.ChatMsg = chatMsg
    sendChatMsgReq.Channel = chatChannel
    sendChatMsgReq.ChatTime = TimeUtils.ServerNowStampMilli
    if extrainfos ~= nil then
        sendChatMsgReq.SendData:Add(ExtraData.ToByteString(extrainfos))
    end
    sendChatMsgReq.PID = PID
    NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, sendChatMsgReq, false)
end

-- 模拟接收聊天信息
function ChatModule:OnAnalogReceptionChatMsg(...)
    local tb = ...
    local ChatMsgNotices = nil

    local msg = CS.ProtoMsg.ChatMsgNoticeS()
    local info = {
        data = msg
    }

    if type(tb) == "table" then
        -- Lua调用
        ChatMsgNotices = tb[1]
        if ChatMsgNotices == nil or ChatMsgNotices == {} or #ChatMsgNotices <= 0 then
            return
        end
        for i = 1, #ChatMsgNotices, 1 do
            local item = ChatMsgNotices[i]
            msg.CMNotices:Add(item)
        end
    else
        -- C#调用
        ChatMsgNotices = tb[0]
        for i = 0, ChatMsgNotices.Count - 1, 1 do
            local item = ChatMsgNotices[i]
            msg.CMNotices:Add(item)
        end
    end

    self:OnChatMsgNoticeS(info)
end
function ChatModule:ShowRewardTips(...)
    local tb = ...
    local id = 0
    local num = 0
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        id = tb[1]
        num = tb[2]
    else
        -- C#调用
        id = tb[0]
        num = tb[1]
    end
    local msgList = {}
    local jobID = GameManager.Instance:GetPlayerJob()
    local strLanVal = LanguageHelp:GetLanguageByKey("Local_Str_Tips_GetReward")
    local itemCfg = ConfigDataAccessor.Item[id]
    local equipStrSp = ExtraData.GetEquipStrByID(id);
    local quality = 1
    local itemName = ""
    if itemCfg ~= nil then
        quality = itemCfg.Quality
        itemName = itemCfg.Name
    end

    local chatMsgNotice = CS.ProtoMsg.ChatMsgNotice()
    chatMsgNotice.EID = 0
    chatMsgNotice.PID = 0
    local strContentVal = equipStrSp .. GetQualityColor(quality) .. itemName .. "</color>x" .. num
    chatMsgNotice.Data = StringFormat(strLanVal, {strContentVal})
    chatMsgNotice.Channel = CS.ProtoMsg.ChatChannel.ChannelSystem
    chatMsgNotice.ChatTime = TimeUtils.ServerNowStampMilli
    chatMsgNotice.ChatLevel = 1
    chatMsgNotice.JobID = jobID
    table.insert(msgList, chatMsgNotice)
    LuaModuleDispatcher:SendMessage(ModuleDef.ChatModule, "OnAnalogReceptionChatMsg", {msgList})
end

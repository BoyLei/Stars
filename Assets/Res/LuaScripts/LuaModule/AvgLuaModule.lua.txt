function AvgLuaModule:OnTransition(...)
    local tb = ...

    local id = 0
    local effectID = 0

    --传入道具ID
    if type(tb) == "table" then
        --Lua调用
        id = tb[1]
        effectID = tb[2]
    else
        --C#调用
        id = tb[0]
        effectID = tb[1]
    end

    UIManager.Instance:OpenWidgetAsync("Avg/TransitionWidget", nil, false, {id, effectID}, nil, CS.StarProjectDef.MainPageCommond.HideNone, false, true)
end

function AvgLuaModule:OnOpenAvgFromServer(...)
    local tb = ...

    local dialogID = 0
    local entityID = 0

    --传入道具ID
    if type(tb) == "table" then
        --Lua调用
        dialogID = tb[1]
        entityID = tb[2]
    else
        --C#调用
        dialogID = tb[0]
        entityID = tb[1]
    end

    local dialogItem = ConfigDataAccessor.CommonDialog[dialogID]
    if dialogItem == nil then
        --SGF.Debuger.LogWarning($"OnDialogNotifyMsg() dialogID={dialogID},找不到!!!");
        return
    end
    if dialogItem.Dialog_type == 1 or dialogItem.Dialog_type == 9 then
        self:OnOpenAvgTalkBox({dialogID, nil, entityID})
    elseif dialogItem.Dialog_type == 2 then
        self:OnOpenAvgPopBox({dialogID, 0, entityID})
    elseif dialogItem.Dialog_type == 3 then
        self:OnOpenAvgZoneBox({dialogID})
    elseif dialogItem.Dialog_type == 6 then
        self:OnOpenAvgAsideBox({dialogID})
    elseif dialogItem.Dialog_type == 7 then
        self:OnOpenAvgBookBox({dialogID})
    else
        self:OnOpenAvgZoneBox({dialogID})
    end
    Log("打开AVG:" .. dialogID)
end

function AvgLuaModule:OnShowHideAvgPopBox(...)
    local tb = ...
    local isShow = false
    --传入道具ID
    if type(tb) == "table" then
        --Lua调用
        isShow = tb[1]
    else
        --C#调用
        isShow = tb[0]
    end

    if self.avgPopBoxView ~= nil then
        self.avgPopBoxView.gameObject:SetActive(isShow)
    end
end

--打开AVG对话框
function AvgLuaModule:OnOpenAvg(...)
    local tb = ...

    local scriptIdx = 0
    --传入道具ID
    if type(tb) == "table" then
        --Lua调用
        scriptIdx = tb[1]
    else
        --C#调用
        scriptIdx = tb[0]
    end

    --预热avg
    if self.IsPreLoadAvgTalkbox == nil then
        self:PreLoadAvgTalkbox()
        self.IsPreLoadAvgTalkbox = true
    end

    local data2 = ConfigDataAccessor.TeleDialog[scriptIdx]
    Log("打开AVG:" .. scriptIdx)
    --优先找心灵对话
    if data2 ~= nil then
        self:OnOpenAvgMindBox(tb)
    else
        local data = ConfigDataAccessor.CommonDialog[scriptIdx]
        --local data=LocalDataManager.Instance:GetCommonDialogDataCell(scriptIdx);

        if data == nil then
            print("对话加载失败 id=", scriptIdx)
            return
        end

        if data.Dialog_type == 1 or data.Dialog_type == 9 then
            self:OnOpenAvgTalkBox(tb)
        elseif data.Dialog_type == 2 then
            self:OnOpenAvgPopBox(tb)
        elseif data.Dialog_type == 3 then
            self:OnOpenAvgZoneBox(tb)
        elseif data.Dialog_type == 6 then
            self:OnOpenAvgAsideBox(tb)
        elseif data.Dialog_type == 7 then
            self:OnOpenAvgBookBox(tb)
        end
    end
end
--打开AVG对话框
function AvgLuaModule:OnOpenAvgTalkBoxForTimeLine(...)
    local tb = ...

    --UIManager.Instance:CloseAllLoadedWindowNWidget()

    self.AsyncLoadAvgTalkBoxCb = function(ui)
        if ui ~= nil then
            if self.avgTalkBoxView == nil then
                self.avgTalkBoxView = ui:GetLuaPanel().ScriptTable
                self.avgTalkBoxView:BindModule(self)
                --别人可以这么调用
                self.evtOnFinishAvgTalkBox = LuaModuleDispatcher:Event(ModuleDef.AvgLuaModule, "OnFinishAvgTalkBox")
            --自己可以这么调用
            --self.evtOnFinishAvgTalkBox = self.LuaEventTable:GetEvent("OnFinishAvgTalkBox")
            end

            self.SelectedIdx = 0
            --self.EffectID_TalkBox = nil
            --self.NPCEnityID = nil
            --self.NeedCheckService = false --是否需要检查基础服务
            --self.BaseServiceSelectedIdx = 1 --从外面选择的基础服务的下标
            local scriptIdx = 0
            --传入道具ID
            if type(tb) == "table" then
                --Lua调用
                scriptIdx = tb[1]
                if #tb > 1 then
                    self.EffectID_TalkBox = tb[2]
                end
                if #tb > 2 then
                    self.NPCEnityID = tb[3]
                end
                if #tb > 3 then
                    self.NeedCheckService = tb[4]
                end
                if #tb > 4 then
                    self.BaseServiceSelectedIdx = tb[5]
                end
            else
                --C#调用
                scriptIdx = tb[0]
                if tb.Length > 1 then
                    self.EffectID_TalkBox = tb[1]
                end
                if tb.Length > 2 then
                    self.NPCEnityID = tb[2]
                end
                if tb.Length > 3 then
                    self.NeedCheckService = tb[3]
                end
                if tb.Length > 4 then
                    self.BaseServiceSelectedIdx = tb[4]
                end
            end

            if self.NPCEnityID ~= nil then
                self.BaseID = GameManager.Instance:GetEntityCfgID(self.NPCEnityID)
            else
                self.BaseID = nil
            end

            self.avgTalkBoxView:StartTalkBox(scriptIdx)

            GameManager.Instance.IsOpenTalkBox = true
        end
        self.AsyncLoadAvgTalkBoxCb = nil
    end
    UIManager.Instance:OpenWidgetAsync(AvgTalkBoxPath, self.AsyncLoadAvgTalkBoxCb, true, nil, nil, CS.StarProjectDef.MainPageCommond.HideBoth, false, true)
end

            self.NPCEnityID = tb[3]
        end
        if #tb > 3 then
            self.NeedCheckService = tb[4]
        end
        if #tb > 4 then
            self.BaseServiceSelectedIdx = tb[5]
        end
    else
        --C#调用
        scriptIdx = tb[0]
        if tb.Length > 1 then
            self.EffectID_TalkBox = tb[1]
        end
        if tb.Length > 2 then
            self.NPCEnityID = tb[2]

    self.AsyncLoadAvgTalkBoxCb = function(ui)
        if ui ~= nil then

--完成对话的通用事件
function AvgLuaModule:OnFinishAvgAction()
    if self.NeedCheckService and self.BaseServiceSelectedIdx > 0 then
        local lobbyServiceReq = CS.ProtoMsg.LobbyServiceReq()
        lobbyServiceReq.MapID = GameManager.Instance:GetCurMapId()
        lobbyServiceReq.UID = self.NPCEnityID
        lobbyServiceReq.ObjID = self.BaseID
        lobbyServiceReq.Index = self.BaseServiceSelectedIdx --这里要看外边点的是哪一个
        lobbyServiceReq.ObjType = CS.ProtoMsg.ServiceObjEnum.ObjNpc
        CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, lobbyServiceReq, false)
    end

    self.EffectID = nil
end

    --回调效果

end

--打开AVG气泡对话框
function AvgLuaModule:OnOpenAvgPopBox(...)
    local tb = ...

    if self.avgPopBoxView == nil then
        local avgPopBoxCs = UIManager.Instance:OpenWidget("Avg/AvgPopBox", false, nil, nil, CS.StarProjectDef.MainPageCommond.HideNone, false, false, true)
        self.avgPopBoxView = avgPopBoxCs:GetLuaPanel().ScriptTable
        self.avgPopBoxView:BindModule(self)
        self.evtOnFinishAvgPopBox = LuaModuleDispatcher:Event(ModuleDef.AvgLuaModule, "OnFinishAvgPopBox")
    end

    local scriptIdx = 0
    self.PopEntityID = 0
    self.EffectID_AvgPop = nil
    --传入道具ID
    if type(tb) == "table" then
        --Lua调用
        scriptIdx = tb[1]
        if #tb > 1 then
            self.EffectID_AvgPop = tb[2] --弹框绑定的实体
        end
        if #tb > 2 then
            self.PopEntityID = tb[3] --弹框绑定的实体
        end
    else
        --C#调用
        scriptIdx = tb[0]
        if tb.Length > 1 then
            self.EffectID_AvgPop = tb[1] --弹框绑定的实体
        end
        if tb.Length > 2 then
            self.PopEntityID = tb[2] --弹框绑定的实体
        end
    end

    self.avgPopBoxView:GetTalkContentById(scriptIdx, self.PopEntityID)
end

--获取配置Item
function AvgLuaModule:CommonDialogDataCellById(idx)
    local data = ConfigDataAccessor.CommonDialog[idx]
    --local data=LocalDataManager.Instance:GetCommonDialogDataCell(idx);

    if data == nil then
        print("对话加载失败 id=", idx)
    end
    if data.Dialog_type == 1 then
        data.Dialog_desc2 = BusinessManager.Instance:ReplacePlayerName(data.Dialog_desc)
    else
        data.Dialog_desc2 = BusinessManager.Instance:ReplacePlayerName(data.Dialog_desc, "")
    end
    data.Name2 = BusinessManager.Instance:ReplacePlayerName(data.Name, "")
    --[[
        local targetID = GameManager.Instance.mainPlayerId
    if targetID ~= nil and targetID ~= 0 then
        local playerName = GameManager.Instance:GetEntityNameById(targetID)
        if playerName ~= nil and playerName ~= "" then
            local pname = "<color=#f7cf46>" .. playerName .. "</color>"
            data.Name2 = string.gsub(data.Name, "@player", playerName)
            if data.Dialog_type == 1 then
                data.Dialog_desc2 = string.gsub(data.Dialog_desc, "@player", pname)
            else
                data.Dialog_desc2 = string.gsub(data.Dialog_desc, "@player", playerName)
            end
        end
    end
    ]]
    return data
    --self:RefeshTalkInfo();
end

--完成AVG普通对话框通知
function AvgLuaModule:OnFinishAvgTalkBox(arg)
    --选项回调效果
    if self.EffectID_TalkBox ~= nil then
        if self.SelectedIdx == 0 or self.SelectedIdx == nil then
            CS.StarProject.GlobalEvent.OnSelectEffect:Invoke(self.EffectID_TalkBox, 0)
        else
            CS.StarProject.GlobalEvent.OnSelectEffect:Invoke(self.EffectID_TalkBox, self.SelectedIdx - 1)
        end
    end
    self:OnFinishAvgAction()
    self.evtOnFinishAvgTalkBox:Invoke()
    self.NPCEnityID = nil
    BusinessManager.Instance:ResumeLast(self.BaseID)
    GameManager.Instance.IsOpenTalkBox = false
end

end

    GameManager.Instance.IsOpenTalkBox = false
end

--完成AVG书页对话框通知
function AvgLuaModule:OnFinishAvgBookBox(arg)
    --选项回调效果
    if self.EffectID_AvgBook ~= nil then
        if self.SelectedIdx == 0 or self.SelectedIdx == nil then
            CS.StarProject.GlobalEvent.OnSelectEffect:Invoke(self.EffectID_AvgBook, 0)
        else
            CS.StarProject.GlobalEvent.OnSelectEffect:Invoke(self.EffectID_AvgBook, self.SelectedIdx - 1)
        end
        self.EffectID_AvgBook = nil
    end

    self:OnFinishAvgAction()
    self.evtOnFinishAvgBookBox:Invoke()
    GameManager.Instance.IsOpenTalkBox = false
end

    --print("on lua module show................")

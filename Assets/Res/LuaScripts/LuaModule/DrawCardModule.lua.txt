function DrawCardModule:OnDrawCardInfos(data)
    local gachaMD = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.Gacha):GetGachaMD();

    self.gacheLimit = {};
    if gachaMD ~= nil then
        if gachaMD.Forever ~= nil then
            self:SetSsrLeftTimes(gachaMD.Forever);
        end
        if gachaMD.Activity ~= nil then
            self:SetSsrLeftTimes(gachaMD.Activity);
        end
        if gachaMD.Newbie ~= nil then
            self:SetSsrLeftTimes(gachaMD.Newbie);
        end
        -- if gachaMD.Friend ~= nil then
        --     self:SetSsrLeftTimes(gachaMD.Forever.PoolType, gachaMD.Forever.NoSSR);
        -- end

        self:RefreshSsrLeftTimes(gachaMD);
    end
end

function DrawCardModule:SetSsrLeftTimes(gachaMD)
    local t = self:ConvertEnumToNum(gachaMD.PoolType);
    local ssrTimes = gachaMD.NoSSR;

    if gachaMD ~= nil then
        self.timesLimit = gachaMD.DataList;
    end

    local cardPool = ConfigDataAccessor.CardPool;
    for _, v in pairs(cardPool) do
        if v.GachaTimesLimits > 0 then
            if gachaMD.DataList.Count == 0 then
                if v.CardPoolType == t then
                    if self.drawCardWindow ~= nil then
                        self.drawCardWindow:RefreshLimitTimes(v.ID, 0, v.GachaTimesLimits);
                    end
                end
            else
                for i = 0, gachaMD.DataList.Count - 1 do
                    if gachaMD.DataList[i].PoolID == v.ID then
                        if self.drawCardWindow ~= nil then
                            self.drawCardWindow:RefreshLimitTimes(gachaMD.DataList[i].PoolID,
                                gachaMD.DataList[i].History, v.GachaTimesLimits);

                            if gachaMD.DataList[i].History >= v.GachaTimesLimits then
                                table.insert(self.gacheLimit, gachaMD.DataList[i].PoolID);
                            end
                        end
                    end
                end
            end
        end
    end

    if self.tenTimesWindow ~= nil then
        self.tenTimesWindow:SetTenTimesBtn();
    end
end

function DrawCardModule:RefreshSsrLeftTimes(gachaMD)
    local tb = {};
    table.insert(tb, gachaMD.Forever);
    table.insert(tb, gachaMD.Activity);
    table.insert(tb, gachaMD.Newbie);

    local cardPool = ConfigDataAccessor.CardPool;
    for _, item in pairs(tb) do
        local t = self:ConvertEnumToNum(item.PoolType);
        local ssrTimes = item.NoSSR;
        for _, v in pairs(cardPool) do
            if v.CardPoolType == t then
                local leftTimes = ConfigDataAccessor.Probability[v.UseProbGroup].ForceSSRTimes - ssrTimes;
                if self.drawCardWindow ~= nil then
                    self.drawCardWindow:RefreshSsrTimes(v.ID, leftTimes,
                        ConfigDataAccessor.Probability[v.UseProbGroup].ForceSSRTimes);
                end
            end
        end
    end
end

function DrawCardModule:ConvertEnumToNum(poolType)
    if poolType == CS.ProtoMsg.CardPool.CpForever then
        return 1;
    elseif poolType == CS.ProtoMsg.CardPool.CpActivity then
        return 2;
    elseif poolType == CS.ProtoMsg.CardPool.CpNewbie then
        return 3;
    elseif poolType == CS.ProtoMsg.CardPool.CpFriend then
        return 4;
    end

    return 1;
end

function DrawCardModule:OnGachaRet(data)
    if data == nil or data.data == nil then
        return;
    end

    local gachas = data.data.GachaResult;
    local msgList = {}
    for i = 1,gachas.Count do 
        local curData = gachas[i-1];

        local originDrop = curData.OriginDrop;
        
        local finalDrop = originDrop;
        if curData.FinalDrop.Count > 0 then
            finalDrop = curData.FinalDrop[0];
        end
        local itemData = ConfigDataAccessor.Item[originDrop.Index];
        if itemData == nil then
            return;
        end
        --提交列表
        local str = itemData.Name;
        local nameStr =str:sub(1, 3)..str:sub(4, string.len(str))
        local partnerData = ConfigDataAccessor.Partner[originDrop.Index];
        local lv = partnerData.Quality;

        local chatMsgNotice = CS.ProtoMsg.ChatMsgNotice()
        chatMsgNotice.EID = 0
        chatMsgNotice.PID = 0
        local strLanVal = LanguageHelp:GetLanguageByKey("Local_Str_Tips_GetReward")
        local strContentVal = GetQualityColor(lv) .. nameStr .. "</color>x" .. 1
        chatMsgNotice.Data = StringFormat(strLanVal, {strContentVal})
        chatMsgNotice.Channel = CS.ProtoMsg.ChatChannel.ChannelSystem
        chatMsgNotice.ChatTime = TimeUtils.ServerNowStampMilli
        chatMsgNotice.ChatLevel = 1
        chatMsgNotice.JobID = GameManager.Instance:GetPlayerJob()
        table.insert(msgList, chatMsgNotice)
    end
    LuaModuleDispatcher:SendMessage(ModuleDef.ChatModule, "OnAnalogReceptionChatMsg", {msgList})
    -- 刷新货币
    if self.drawCardWindow ~= nil then
        self.drawCardWindow:RefreshCoin();
    end

    self:OnOpenDrawCardDetail(data.data, 0);

    local info = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.ComCount):GetData(19);
    if info ~= nil then
        -- Util.ShowMessage(string.format(ConfigDataAccessor.CRetMsgList[56].Note, info.RemainCount, info.AddCount));

        -- Util.ShowMessage(string.format("启梦%d次，今日剩余启梦次数%d次", info.AddCount-info.RemainCount, info.RemainCount));
        Util.ShowMessage(StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_235"),
            {info.AddCount - info.RemainCount, info.RemainCount}))
    end

    -- 显示toast “启梦X次，今日剩余启梦次数XX次”

    -- else
    --     -- 十连抽
    -- 	self:OnOpenDrawCardDetail(data.data, 0);

    -- end

end

function DrawCardModule:IsTimesLimit(id)
    if self.timesLimit == nil then
        return false;
    end

    local data = self.timesLimit;
    local v = ConfigDataAccessor.CardPool[id];
    if v.GachaTimesLimits > 0 then
        if data.Count == 0 then
            return false;
        else
            for i = 0, data.Count - 1 do
                if data[i].PoolID == id then
                    return data[i].History >= v.GachaTimesLimits;
                end
            end
        end
    end
    return false;
end

function DrawCardModule:OnCurrencyExchangeRet(data)
    if data == nil then
        return;
    end

    local useCoin = nil;
    for _, v in pairs(ConfigDataAccessor.CardPool) do
        if useCoin == nil then
            useCoin = v.UseToken;
            break
        end
    end

    -- 货币兑换成功，直接购买
    local goodId = ConfigDataAccessor.SystemList[215].Value;
    local goodData = ConfigDataAccessor.Goods[goodId];
    local id = goodData.Price[1]; -- 绑钻
    local changeRate = goodData.Price[2]; -- 汇率
    self:SendShopBuyReq(goodData.Sub_shopid, goodId, data.data.Num / changeRate);
end

function DrawCardModule:SendShopBuyReq(shopID, goodID, goodNum)
    local shopBuyReq = CS.ProtoMsg.ShopBuyReq();
    shopBuyReq.ShopID = shopID; -- 商店ID
    shopBuyReq.GoodID = goodID; -- 商品ID
    shopBuyReq.GoodNum = goodNum; -- 商品数量

    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, shopBuyReq,
        false);
end

function DrawCardModule:OnShopBuyRet(data)
    -- 检测抽卡界面是否打开
    if self.drawCardWindow ~= nil and self.drawCardWindow.IsOpen then
        if self.dcType == 10 then
            self.drawCardWindow:TenDC();
        else
            self.drawCardWindow:SingleDC();
        end
    elseif self.drawCardWindow ~= nil then 
        self.drawCardWindow:RefreshBtns()
    end
end

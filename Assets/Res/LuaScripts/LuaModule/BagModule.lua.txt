BagModule = class("BagModule",BaseLuaModule)

local BagWindowPath = "Bag/BagWindow"


--LuaModule 消息模块
BagModule.ModuleMessages=
{
	--打开背包
	--ModuleManager.Instance.SendMessage(ModuleDef.BagModule, "OnOpenBag", new object[] { 1 }); //只要传一个对话ID
	["OnOpenBag"]=function(...)
		BagModule:OnOpenBag(...);
	end,


	--刷新背包
	--ModuleManager.Instance.SendMessage(ModuleDef.BagModule, "OnRefeshBag", new object[] {  }); 
	["OnRefeshBag"]=function(...)
		BagModule:OnRefeshBag(...);
	end,

	--点击到固定Tag页面
	--ModuleManager.Instance.SendMessage(ModuleDef.BagModule, "OnOpenBagTag", new object[] { 2,1 }); 
	["OnOpenBagTag"]=function(...)
		BagModule:OnOpenBagTagMsg(...);
	end,

}


function BagModule:OnOpenBagTagMsg(...)


	local tb=...;

	local tag=1;
	local subTag=0;
	--列表
	if type(tb)=="table" then
		--Lua调用
		tag=tb[1];
		subTag=tb[2];
	else
		--C#调用
		tag=tb[0];
		subTag=tb[1];
	end

	if self.bagWindowView~=nil and self.bagWindowView.IsOpen then
		self.bagWindowView.EquipMenuIdx=subTag;
		self.bagWindowView:OnMenuBtnClick(tag,true);
	end

end

--监听刷新背包事件
function BagModule:OnRefeshBag(...)

	if self.bagWindowView~=nil and self.bagWindowView.IsOpen then
		local items=self:GetItemsByItemSpaceId()
		self.bagWindowView:RefeshBag(items);
	end
	
end

--监听打开背包事件
function BagModule:OnOpenBag(...)

	
	local tb=...
	
	--local bagWindowCs =UIManager.Instance:OpenWindow("Bag/BagWindow",nil,CS.StarProjectDef.MainPageCommond.HideBoth,false,true);
	local bagWindowCs =UIManager.Instance:OpenWindow(BagWindowPath,nil,CS.StarProjectDef.MainPageCommond.HideBoth,false,false);
	if self.bagWindowView == nil then
		self.bagWindowView=bagWindowCs:GetLuaPanel().ScriptTable;
		self.bagWindowView:BindModule(self);
		self.bagWindowView:InitItems();
		self.evtOnCloseBox = LuaModuleDispatcher:Event(ModuleDef.BagModule, "OnCloseBoxEvt")
	end
	

	--背包ID 0为全部
	self.SpaceID=0;

	--刷新背包
	self:OnRefeshBag();

	--self.bagWindowView.currencyBar:OnListener();


	--[[
	for i=1,10 do

		local it={};
		it.Itemid=300002;
		it.ItemNum=10+i;
		LuaModuleDispatcher:SendMessage(ModuleDef.DropInfoListModule, "OnAddDrop",{it});

	end

	]]


end

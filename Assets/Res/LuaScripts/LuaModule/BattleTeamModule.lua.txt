function BattleTeamModule:OnOpenAssistList(...)
    if self.Partners == nil or #self.Partners < 1 then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_5"))
        return
    end
    local tb = ...
    local id = 0 ---选中的ID
    local seat = 0 ---被选的槽位
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        id = tb[1]
        seat = tb[2]
    else
        -- C#调用
        id = tb[0]
        seat = tb[1]
    end
    local datas = {}
    local IsEmpty = false
    local playerJob = GameManager.Instance:GetPlayerJob()
    local slotInfo = ConfigDataAccessor.PartnerAssistSlot[playerJob][seat]
    local slotpartner = nil
    local select = id
    if slotInfo ~= nil then
        for k, v in pairs(self.Partners) do
            local data = {
                SumAtr = 0,
                Select = 0,
                data = v
            }
            data.SumAtr = 0
            data.Select = 0
            if id ~= 0 and id == v.Index then
                data.Select = 1
                select = v.Index
                slotpartner = v
            end
            for i = 1, #slotInfo.Qualifications do
                data.SumAtr = data.SumAtr + v.AMD.Aptitudes[slotInfo.Qualifications[i]].BaseValue
            end
            table.insert(datas, data)
        end
    end

    if id == 0 then
        IsEmpty = true
        datas[1].Select = 1
        select = datas[1].data.Index
    end

    table.sort(
        datas,
        function(d1, d2)
            if d1.Select ~= d2.Select then
                return d1.Select > d2.Select
            end

            if d1.SumAtr ~= d2.SumAtr then
                return d1.SumAtr > d2.SumAtr
            end

            return false
        end
    )

    self:CloseLastWidget()
    local info =
        self:AddWidget(
        AssistsWidgetPath,
        false,
        {
            datas = datas,
            seat = seat,
            IsEmpty = IsEmpty,
            sid = select,
            Slotpartner = slotpartner
        },
        self.PartnerRoot
    )
    UIManager.Instance:OpenWidgetAsync(info)
end

function BattleTeamModule:OnOpenAssistWidget(...)
    local assites = {}
    for i = 1, 6 do
        assites[i - 1] = {
            Pos = i - 1,
            data = nil
        }
    end

    if self.Partners ~= nil then
        for k, v in pairs(self.Partners) do
            if v.State == CS.ProtoMsg.PartnerState.Assist and v.CurSeat ~= -1 and assites[v.CurSeat] ~= nil then
                assites[v.CurSeat].data = v
            end
        end
    end

    self:CloseLastWidget()
    local info =
        self:AddWidget(
        PartnerAssistWidgetPath,
        false,
        function()
            return self:FreshAssistList()
        end,
        self.PartnerRoot
    )
    UIManager.Instance:OpenWidgetAsync(info)
end

function BattleTeamModule:FreshAssistList()
    local assites = {}
    for i = 1, 6 do
        assites[i - 1] = {
            Pos = i - 1,
            data = nil
        }
    end

    if self.Partners ~= nil then
        for k, v in pairs(self.Partners) do
            if v.State == CS.ProtoMsg.PartnerState.Assist and v.CurSeat ~= -1 and assites[v.CurSeat] ~= nil then
                assites[v.CurSeat].data = v
            end
        end
    end

    return assites
end

function BattleTeamModule:OnOpenDetailWidget(...)

        return
    end
    local tb = ...

    local index = 0 -- 伙伴ID，配置表ID
    local subTag = 0
    -- 传入道具ID
    if type(tb) == "table" then

        index = tb[1]
        subTag = tb[2]
    else
        -- C#调用
        index = tb[0]
        subTag = tb[1]
    end

    end

    end
    self:CloseLastWidget()

    end
    local info =
        self:AddWidget(
        PartnerDetailWidgetPath,
        false,
        {
            Partners = self.Partners,
            SelectIndex = index,
            Module = self,
            SubTag = subTag
        },
        self.PartnerRoot,
        CS.StarProjectDef.MainPageCommond.HideNone,
        false,
        false,
        self.AsyncLoadPartnerDetailWidgetCb
    )
    -- local partnerDetailWidgetCS = UIManager.Instance:OpenWidget(info)
    -- self.partnerDetailWidget = partnerDetailWidgetCS:GetLuaPanel().ScriptTable
    UIManager.Instance:OpenWidgetAsync(info)

function BattleTeamModule:OnSelectBattlePartner(...)
    if self.Partners == nil or GetTableLen(self.Partners) < 1 then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_5"))
        return
    end
    local tb = ...
    local index = 0
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        index = tb[1]
    else
        -- C#调用
        index = tb[0]
    end
    local select = 0
    local isEmpty = true

    -- 根据伙伴 槽位和出战状态筛选
    if self.Partners ~= nil then
        for k, v in pairs(self.Partners) do
            if v.State == CS.ProtoMsg.PartnerState.InBattle or v.State == CS.ProtoMsg.PartnerState.Concretization then
                if v.CurSeat == index then
                    isEmpty = false
                    select = v.Index
                end
            end
        end
    end

    self:CloseLastWidget()
    -- if select==0 then
    -- for k, v in pairs(self.Partners) do
    -- print(k)
    -- select=v.Index;
    -- break;
    -- end
    -- end
    local info =
        self:AddWidget(
        PartnersWidgetPath,
        false,
        {
            Partners = self.Partners,
            SelectIndex = select,
            IsEmpty = isEmpty,
            Index = index
        },
        self.PartnerRoot
    )
    UIManager.Instance:OpenWidgetAsync(info)
end

function BattleTeamModule:ErrorHandler(data)
    local RetMsgID = data.MsgRet.RetMsgID -- 对应的消息号(这个可以判断是哪个协议)

    local RetCode = data.MsgRet.RetCode
    if RetCode ~= 0 then
        if RetMsgID == CS.MsgIDEnum.PartnerBattleReqID then
            CS.StarProject.GlobalEvent.OnPartnerBattleRet:Invoke(false)
        elseif RetMsgID == CS.MsgIDEnum.PartnerAssistReqID then
            CS.StarProject.GlobalEvent.OnPartnerAssistRet:Invoke(false)
        elseif RetMsgID == CS.MsgIDEnum.PartnerFallReqID then
            CS.StarProject.GlobalEvent.OnPartnerFallRet:Invoke(false)
        elseif RetMsgID == CS.MsgIDEnum.PartnerLimitBreakReqID then
            CS.StarProject.GlobalEvent.OnPartnerTopped:Invoke(false)
        elseif RetMsgID == CS.MsgIDEnum.PartnerUpStarReqID then
            CS.StarProject.GlobalEvent.OnPartnerUpStar:Invoke(false)
        end
    end
end

-- 伙伴新的差量同步
function BattleTeamModule:OnPartnerRefesh(data)
    if data == nil then
        return
    end
    -- 修改的
    for i = 0, data.Changes.Count - 1 do
        local item = data.Changes[i]
        local index = self:GetPartnerIndex(item)
        if index == -1 then
            ---新增
            table.insert(self.Partners, item)
        else
            self.Partners[index] = item
        end
        ---数据刷新[这个要问一下]
        CS.StarProject.GlobalEvent.OnPartnerFresh:Invoke(item.Index)
    end
    -- 删
    for i = 0, data.Delets.Count - 1 do
        local item = data.Delets[i]
        local index = self:GetPartnerIndex(item)
        if index ~= -1 then
            table.remove(self.Partners, index)
        end
    end
end

-- 接收全量伙伴列表
function BattleTeamModule:OnAllPartnerList(data)
    if data == nil then
        return
    end

    -- 获取伙伴数据
    self.Partners = {}

    local allPartnerList = data.data
    local partners = allPartnerList.Pars
    -- 伙伴数据
    for i = 0, partners.Count - 1 do
        local partner = partners[i]
        table.insert(self.Partners, partner)
    end
end

-- 替换|出战返回
function BattleTeamModule:OnPartnerBattleRet(data)
    if data == nil then
        return
    end

    -- if data.ReplaceSeat ~= -1 then
    ---- 替换
    -- else
    ---- 上阵
    -- end
    if data.data.Ret == CS.ProtoMsg.PartnerOperate.OpOk then
        CS.StarProject.GlobalEvent.OnPartnerBattleRet:Invoke(true)
    end

    if self.battleTeamWidgetView ~= nil and self.battleTeamWidgetView.IsOpen then
        self.battleTeamWidgetView:OnReplacePartner(self.Partners, data.data)
    end
end

function BattleTeamModule:GetPartnerIndex(dirtyPartner)
    if self.Partners == nil or self.Partners == {} then
        return 0
    end

    for _k, _v in pairs(self.Partners) do
        if _v.Index == dirtyPartner.Index then
            return _k
        end
    end
    return -1
end
function BattleTeamModule:GetPartner(index)
    if self.Partners == nil then
        return nil
    end

    for _k, _v in pairs(self.Partners) do
        if _v.Index == index then
            return _v
        end
    end
    return nil
end

-- 伙伴脏数据
function BattleTeamModule:OnNoticePartnerList(data)
    if data == nil or data.data == nil then
        return
    end

    local dirtyPartnerList = data.data.Pars
    for i = 0, dirtyPartnerList.Count - 1 do
        local dirtyPartner = dirtyPartnerList[i]
        local index = self:GetPartnerIndex(dirtyPartner)
		local config=ConfigDataAccessor.Partner[dirtyPartner.Index]
		if config~=nil and config.TemporaryPar==0 then
			
	        if index == -1 then
	            ---新增
	            table.insert(self.Partners, dirtyPartner)
	            LocalCache.GetPartnerRedPointData:AddPartner(dirtyPartner.Index)
	        else
	            self.Partners[index] = dirtyPartner
	        end
	        ---数据刷新
	        CS.StarProject.GlobalEvent.OnPartnerFresh:Invoke(dirtyPartner.Index)
		end
    end
end

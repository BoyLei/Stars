-- 监听关闭商店事件
function ShopModule:OnRefeshDropMsg(...)
    local tb = ...

    local Item_id = 1;
    local goodNum = 1;
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        Item_id = tb[1];
        goodNum = tb[2];
    else
        -- C#调用
        Item_id = tb[0];
        goodNum = tb[1]
    end

    self:SetBuyDrop(Item_id, goodNum);

end

-- 监听刷新商店事件
function ShopModule:OnRefeshShopMsg(...)

    if self.shopWindowView ~= nil and self.shopWindowView.IsOpen then
        self.shopWindowView:RefeshShop();
    end

    if self.npcShopWindowView ~= nil and self.npcShopWindowView.IsOpen then
        self.npcShopWindowView:RefeshShop();
    end
end

-- 关闭商店回调
function ShopModule:OnCloseShopEvt(arg)
    if self.evtOnCloseShop ~= nil then
        self.evtOnCloseShop:Invoke()
    end
    self.shopWindowView = nil;
end

-- 关闭Npc商店回调
function ShopModule:OnCloseNpcShopEvt(arg)
    if self.evtOnCloseNpcShop ~= nil then
        self.evtOnCloseNpcShop:Invoke()
    end
    self.npcShopWindowView = nil;
end

-- 关闭商店
function ShopModule:OnCloseShop()
    self.shopWindowView.currencyBar:OffListener();

    self:OnCloseShopEvt();
    -- 所有的New置空
end

-- 关闭Npc商店
function ShopModule:OnCloseNpcShop()
    self.npcShopWindowView.currencyBar:OffListener();

    self:OnCloseNpcShopEvt()
    -- 所有的New置空
end

function ShopModule:SetBuyDrop(baseID, num)

    self.BuyNum = num;

end

function ShopModule:OpenBuyMenu(ritem)

    local info = {};
    info.BuyCount = ritem.buyCount; -- 已购数量
    info.GoodsId = ritem.GoodsId; -- 商品ID

    LuaModuleDispatcher:SendMessage(ModuleDef.ItemBuyModule, "OnOpenItemBuy", {info});

end

function ShopModule:SendShopBuyReq(shopID, goodID, goodNum)
    local shopBuyReq = CS.ProtoMsg.ShopBuyReq();
    shopBuyReq.ShopID = shopID; -- 商店ID
    shopBuyReq.GoodID = goodID; -- 商品ID
    shopBuyReq.GoodNum = goodNum; -- 商品数量

    local cfg = ConfigDataAccessor.Goods[goodID];
    self:SetBuyDrop(cfg.Item_id, goodNum);

    CS.SGF.Network.NetworkManager.Instance.gameSocket:SendRPCMsg(CS.SGF.Network.ServerType.ServerTypeLobby, shopBuyReq,
        false);
end

-- LuaModule 创建
function ShopModule:OnModuleCreate(arg)
    LuaModuleDispatcher:Register(ModuleDef.ShopModule, self)
    -- 限购信息
    self.ShopBuyLimit = {};

    -- 是否打开了Tips
    self.IsOpenTips = false;

    -- 请求所有商店购买信息回复
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.AllLimInfoRetID, function(data)
        self:OnAllLimInfoRet(data)
    end, nil);

    -- 购买物品回复
    CS.SGF.Network.NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.ShopBuyRetID, function(data)
        self:OnShopBuyRet(data)
    end, nil);

end

function ShopModule:OnAllLimInfoRet(data)

    print("OnAllLimInfoRet");

    local pb = data.data;

    if self.ShopBuyLimit == nil then
        self.ShopBuyLimit = {}
    else
        for k, _ in pairs(self.ShopBuyLimit) do
            self.ShopBuyLimit[k] = nil
        end
        -- self.ItemDataList ={}
    end

    local dayLimit = nil;
    if pb.DayLimit ~= nil then
        dayLimit = pb.DayLimit.AllInfo; -- int,ShopLimMD
    end

    local weekLimit = nil; -- int,ShopLimMD
    if pb.WeekLimit ~= nil then
        weekLimit = pb.WeekLimit.AllInfo; -- int,ShopLimMD
    end

    local forEverLimit = nil;
    if pb.ForEverLimit ~= nil then
        forEverLimit = pb.ForEverLimit.AllInfo; -- int,ShopLimMD
    end

    if dayLimit ~= nil then
        for k, v in pairs(dayLimit) do
            local shopLimMD = v;
            local shopID = shopLimMD.ShopID;
            local buyInfo = shopLimMD.BuyInfo; -- int,int

            for k2, v2 in pairs(buyInfo) do
                print("buyInfo:" .. k2 .. "   " .. v2);
                self.ShopBuyLimit[k2] = v2;
            end
        end
    end

    if weekLimit ~= nil then
        for k, v in pairs(weekLimit) do
            local shopLimMD = v;
            local shopID = shopLimMD.ShopID;
            local buyInfo = shopLimMD.BuyInfo; -- int,int

            for k2, v2 in pairs(buyInfo) do
                print("buyInfo:" .. k2 .. "   " .. v2);
                self.ShopBuyLimit[k2] = v2;
            end
        end
    end

    if forEverLimit ~= nil then
        for k, v in pairs(forEverLimit) do
            local shopLimMD = v;
            local shopID = shopLimMD.ShopID;
            local buyInfo = shopLimMD.BuyInfo; -- int,int

            for k2, v2 in pairs(buyInfo) do
                print("buyInfo:" .. k2 .. "   " .. v2);
                self.ShopBuyLimit[k2] = v2;
            end
        end
    end

    if self.shopWindowView ~= nil and self.shopWindowView.IsOpen then
        self.shopWindowView:RefeshShop();
    end

end

-- 检查购买限制
function ShopModule:CheckBuyLimit(shopData)
    shopData.buyCount = 0;

    for k, v in pairs(self.ShopBuyLimit) do
        if k == shopData.cfg.Id then
            shopData.buyCount = v;
        end
    end

    local cfg = shopData.cfg;

    -- 判断购买条件
    shopData.MeteFlag = 1;
    local condition = cfg.Condition;
    if #condition > 0 then
        for i = 1, #condition do
            if not (GameManager.Instance:IsConditionMete(condition[i])) then
                shopData.MeteFlag = 0;
            end
        end
    end

    -- 判断是否已经售空
    shopData.SoldOutFlag = 0;
    if cfg.Buy_type == 1 then
        if shopData.buyCount >= cfg.Buy_num then
            shopData.SoldOutFlag = 1;
        end
    elseif cfg.Buy_type == 2 then
        if shopData.buyCount >= cfg.Buy_num then
            shopData.SoldOutFlag = 1;
        end
    elseif cfg.Buy_type == 3 then
        if shopData.buyCount >= cfg.Buy_num then
            shopData.SoldOutFlag = 1;
        end
    end

end

-- LuaModule 创建
function WantedModule:OnModuleCreate(arg)
    --------------------------------  【注册模块的事件】 -------------------------------------
    LuaModuleDispatcher:Register(ModuleDef.WantedModule, self)

    ------------------------------- 【初始化模块的本地变量】----------------------
    self.AllTarInfo = {}
    self.TarInfoStartShowTime = {}

    --------------------------------  【 全局事件 】 -------------------------------------
    self.OnChallengeWantedEntityFun = function(ServerID, MapID, PointID, PointType)
        self:OnChallengeWantedEntity(ServerID, MapID, PointID, PointType)
    end
    CS.StarProject.GlobalEvent.onChallengeWantedEntity:AddListener(self.OnChallengeWantedEntityFun)
    --------------------------------  【监听协议】 -------------------------------------
    -- S-->C 返回客户端 接通缉任务
    self.OnWTaskAcceptRetCmd = function(data)
        self:OnWTaskAccept(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.WTaskAcceptRetID, self.OnWTaskAcceptRetCmd, self)
    -- S-->C 返回客户端 队长发起通缉挑战成功
    self.OnWTaskChallRetCmd = function(data)
        self:OnWTaskChallRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.WTaskChallRetID, self.OnWTaskChallRetCmd, self)
    -- S-->C 客户端 队长发起通缉返回 (已经全员同意了)
    self.OnWTaskStartRetCmd = function(data)
        self:OnWTaskStartRet(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.WTaskStartRetID, self.OnWTaskStartRetCmd, self)
    -- > 通缉任务三阶伤害数据通知
    self.OnWTaskLevMonHurtNtfCmd = function(data)
        self:OnWTaskLevMonHurtNtf(data)
    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.WTaskLevMonHurtNtfID, self.OnWTaskLevMonHurtNtfCmd, self)
    -- > 通缉任务三阶结算数据通知
    self.OnWTaskLevMonResNtfCmd = function(data)

    end
    NetworkManager.Instance:OnMessageEnum(CS.MsgIDEnum.WTaskLevMonResNtfID, self.OnWTaskLevMonResNtfCmd, self)
end

    ------------------------------- 【初始化模块的本地变量】----------------------

    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        data = tb[1]

        data = tb[0]
    end
    if data == true then
        self:CloseFailWindow()
    end
end

-- 共享通缉任务
function WantedModule:OnShareWantedTask(...)
    local tb = ...
    local captainerName = ""
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        captainerName = tb[1]
    else
        -- C#调用
        captainerName = tb[0]
    end

    local isReceiveTask = false
    local rewNum = 0
    local info = GameManager.Instance:GetMainPlayerWantTaskMD()
    if info ~= nil then
        isReceiveTask = info.IsAcc
        rewNum = info.RewNum
    end
    -- 次数是否已经达到上限
    local count = SystemConstConfigs.TeamWantedAwardLimits;
    if rewNum >= count then
        return
    end
    if isReceiveTask == false then
        -- 没有接取任务，弹框提示
        UIAPI.ShowMsgBox(42, function(EventName)
            if EventName == "SURE" then
                self:OnSendWTaskAcceptReq()
            end
        end, captainerName)
    end
end

-- 接取通缉任务打开界面
function WantedModule:OnSendWTaskAcceptReqOpen(...)
    local isReceiveTask = false
    local info = GameManager.Instance:GetMainPlayerWantTaskMD()
    if info ~= nil then
        isReceiveTask = info.IsAcc
    end
    if isReceiveTask == true then
        self:OnOpenWantedWindow()
    else
        self.IsTaskAccepAutoWindow = true;
        self:OnSendWTaskAcceptReq();
    end
end

-- > 通缉任务三阶伤害数据通知
function WantedModule:OnWTaskLevMonHurtNtf(data)
    local res = data.data

    self:OpenWantedRankWidget(res.HurtInfos)
    if self.SendWantedPointType ~= 2 then
        self.SendWantedPointType = 2
        CS.StarProject.GlobalEvent.OnEntityEctype:Invoke(true);
    end
end

-- > 通缉任务三阶结算数据通知
function WantedModule:OnWTaskLevMonResNtf(data)
    local res = data.data
    if res.WTaskType > 2 or self.SendWantedPointType > 2 then
        -- 3阶的通缉目标
        if res.IsSuc == true then
            self:OnOpenWantedChallengeSuccessWindow(res.RankInfos, res.DropList)
        else
            local data = {
                IsRevive = false
            }
            self:OpenFailWindow(data);
        end
    else
        -- 1 2阶的通缉目标
        if res.IsSuc == true then
            CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(
                LanguageHelp:GetLanguageByKey("ShowSystemMessage_New_2"));
        else
            CS.StarProject.Service.DisplayProcess.DisplayProcessDispenser.Instance:ShowSystemMessage(
                LanguageHelp:GetLanguageByKey("ShowSystemMessage_New_1"));
        end
    end
end

-- 通知界面更新数据
function WantedModule:InformWantedWindow(...)
    if self.WantedWindow ~= nil and self.WantedWindow.IsOpen then
        local tb = ...
        local informState = -1
        -- 列表
        if type(tb) == "table" then
            -- Lua调用
            informState = tb[1]
        else
            -- C#调用
            informState = tb[0]
        end
        -- 全部刷新
        if informState == 0 then
            self.WantedWindow:InitData()
        elseif informState == 1 then
            -- 设置3档的目标信息
            self.WantedWindow:SetTargetItemList()
        elseif informState == 2 then
            -- 设置今日次数
            self.WantedWindow:SetTimeTips()
        end
    end
end

-- 挑战通缉实体[分线ID,地图ID,点位ID,点位类型]
function WantedModule:OnChallengeWantedEntity(ServerID, MapID, PointID, PointType)
    local isCanChallenge = self:GetIsCanChallenge();
    if isCanChallenge == false then
        return
    end
    self.SendWantedPointID = PointID
    self.SendWantedPointType = PointType
    -- 请求挑战
    if PointType > 2 then
        self:OnSendWTaskChallReq()
    else
        -- 1、2阶位的直接发起挑战
        self:OnSendWTaskStartReq()
    end
end

-- 获取是否可以挑战
function WantedModule:GetIsCanChallenge()
    -- 判断是否接取了通缉任务
    local isReceiveTask = self:GetIsReceiveWantedTask()
    if isReceiveTask == false then
        return false
    end
    ---- 请求挑战
    -- return true
    -- 1 没有队伍 创建队伍
    -- 2、有队伍 且是队长 人数大于3人 则开始挑战
    -- 3、有队伍 不是队长  飘字提示 仅有队长才能发起挑战
    local module = ModuleManager.Instance:GetModule(ModuleDef.TeamModule)
    local luatable = module:GetLuaTable()
    local teamInfo = luatable:GetTeamInfo()
    local hasTeam = false -- 是否有队伍
    local isCaptain = false -- 是否队长
    local teamPeopleNum = 0 -- 队伍人数
    if teamInfo ~= nil and teamInfo.TeamID ~= 0 and teamInfo.UserInfos.Count > 0 then
        hasTeam = true
        for i = 0, teamInfo.UserInfos.Count - 1 do
            local info = teamInfo.UserInfos[i]
            if info ~= nil then
                if info.IsCaptain then
                    if info.Eid == GameManager.Instance.mainPlayerId then
                        isCaptain = true
                    end
                end
            end
            teamPeopleNum = teamPeopleNum + 1
        end
    end
    if hasTeam then
        if isCaptain then
            if teamPeopleNum >= 3 then
                -- 请求挑战
                return true
            else
                Util.ShowMessage(StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_251"), {3}));
                return false
            end
        else
            Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_60"));
            return false
        end
    else
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_61"));
        return false
    end
    return true
end

-- 设置3档的目标信息的次数/总数/刷新时间
function WantedModule:GetAllTarNumInfo()
    local list = {}
    if GameManager.Instance.AllWantedInfo == nil then
        return list
    end
    local oneInfo = self:GetWTaskTypeCount(1)
    list[1] = oneInfo
    local twoInfo = self:GetWTaskTypeCount(2)
    list[2] = twoInfo
    local threeInfo = self:GetWTaskTypeCount(3)
    list[3] = threeInfo
    return list
end

-- 获取点阶位类型的数量和总数
function WantedModule:GetWTaskTypeCount(wTaskType)
    local sum = 0
    local num = 0
    for i, v in pairs(GameManager.Instance.AllWantedInfo) do
        for i1, v1 in pairs(v) do
            if v1 ~= nil then
                for k2, v2 in pairs(v1) do
                    if k2 == wTaskType then
                        for index, value in pairs(v2) do
                            sum = sum + 1
                            if value ~= nil then
                                if value.CurState == 0 then
                                    num = num + 1
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    local refTime = 0
    if GameManager.Instance.TarInfoStartShowTime ~= nil then
        for k, v in pairs(GameManager.Instance.TarInfoStartShowTime) do
            if k == wTaskType then
                refTime = v;
                break
            end
        end
    end
    local data = {
        Sum = sum,
        Num = num,
        RefTime = refTime
    }
    return data
end

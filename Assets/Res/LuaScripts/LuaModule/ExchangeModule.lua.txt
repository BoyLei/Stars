---打开道具品类筛选
function ExchangeModule:OpenItemTypeFilter(...)
    local tb = ...;
    local ItemType = 0; ---道具类型
    local msgName =CS.MsgIDEnum.BankQueryListReqID;
    -- self.M_Searchs[msgName]
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        ItemType = tb[1];
    else
        -- C#调用
        ItemType = tb[0];
    end

    LuaModuleDispatcher:SendMessage(ModuleDef.SelectorModule, "OnOpenSelectorByType", {ItemType, -- 确认事件
    function(data)
        self.M_PageNum = 1;
        local msgName = CS.MsgIDEnum.BankQueryListReqID;
        local message = self.M_Searchs[msgName]
        -- 是否只看有货（bankitemid表有效）
        local InStock = false;
        if data.commonSelector[1] ~= nil then
            InStock = data.commonSelector[1] == 1;
        end
        message.InStock = InStock;

        -- 是否符合玩家等级
        local IsPlayerLv = false;
        if data.commonSelector[2] ~= nil then
            IsPlayerLv = data.commonSelector[2] == 1;
        end
        message.IsPlayerLv = IsPlayerLv;

        -- 是否符合玩家职业
        local IsJobID = false;
        if data.commonSelector[3] ~= nil then
            IsJobID = data.commonSelector[3] == 1;
        end
        message.IsJobID = IsJobID;

        -- 大于等于品质
        local QualityMin = 0;
        if data.selectQuality ~= nil then
            QualityMin = data.selectQuality
        end
        message.QualityMin = QualityMin;
        -- 小于等于品质

        -- message.QualityMax = 11;
        -- 大于等于玩家等级
        local PlayerLv = 0;
        if data.levelRange ~= nil then
            PlayerLv = data.levelRange
        end
        message.PlayerLv = PlayerLv;

        -- 道具类型
        message.SubType:Clear();
        if data.equipType ~= nil and type(data.equipType) == "table" then
            for k, v in pairs(data.equipType) do
                message.SubType:Add(v)
            end
        end

        -- 筛选器没筛选到数据
        if message.SubType.Count == 0 and data.ItemType ~= nil then
            local temp = ConfigDataAccessor.ItemMainType[data.ItemType];
            if temp ~= nil then
                for k, v in pairs(temp.SubTypes) do
                    message.SubType:Add(v.Id)
                end
            end
        end

        -- 词条品质
        message.EntryQuality = 0;
        local quality = {};
        if data.wordsData ~= nil and data.wordsData.quality ~= nil then
            quality = data.wordsData.quality
        end

        for k, v in pairs(quality) do
            if v ~= 0 then
                message.EntryQuality = k;
                break
            end
        end

        -- 大于等于词条数 如果不带词条的话，写个-1
        message.EntryNum = -1;
        local wordsNum = {};
        if data.wordsData ~= nil and data.wordsData.wordsNum ~= nil then
            wordsNum = data.wordsData.wordsNum
        end
        for k, v in pairs(wordsNum) do
            if v ~= 0 then
                message.EntryNum = k;
                break
            end
        end

        -- 包含词条id
        message.Entrys:Clear();
        if data.wordsData ~= nil and data.wordsData.words ~= nil and type(data.wordsData.words) == "table" then
            for k, v in pairs(data.wordsData.words) do
                message.Entrys:Add(v)
            end
        end
        -- 排序列：使用枚举
        message.OrderBy = self.M_CurrentSortBy;
        -- 页码
        message.PageNum = self.M_PageNum;
        if message.PageNum < 0 then
            message.PageNum = 0;
        end

        SendClient2CenterReq(message);

        self.M_CurrentSearchName = msgName;
    end, ---取消事件
    function()

    end});

end

function ExchangeModule:OpenGoodsFilter(...)
    local tb = ...;
    local ItemID = 0; ---道具类型

    local msgName = CS.MsgIDEnum.BankItemListReqID;
    -- self.M_Searchs[msgName]

    -- 列表
    if type(tb) == "table" then
        -- Lua调用

    else
        -- C#调用

    function(data)

        local msgName = CS.MsgIDEnum.BankItemListReqID;
        local message = self.M_Searchs[msgName]
        -- 道具ID
        message.ItemID = ItemID;
        -- 词条品质
        local quality = 0;
        if data.quality ~= nil then
           -- quality = data.quality
           for k,v in pairs(data.quality) do 
                if v ~= 0 then 
                    quality = k
                end
           end

        self.M_CurrentSearchName = msgName;
    end, ---取消事件

---获取排序索引
function ExchangeModule:GetSortIndex(...)
    local tb = ...;
    local fType = 0;
    local order = 0;
    local id = 0;
    local cb = nil;
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        fType = tb[1];
        order = tb[2];
        id = tb[3];
        cb = tb[4];
    else
        -- C#调用
        fType = tb[0];
        order = tb[1];
        id = tb[2];
        cb = tb[3]
    end

    if cb ~= nil then
        cb((fType * 100 + order * 10 + id))
    end
end

---模糊搜索
function ExchangeModule:FuzzySearch(...)
    local tb = ...;
    local pattern = "";
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        pattern = tb[1];
    else
        -- C#调用
        pattern = tb[0];
    end

    if pattern == nil or pattern == "" or pattern == " " or pattern == '' or pattern == ' ' then
        return nil;
    end
    local list = {}

    for k, v in pairs(ConfigDataAccessor.Item) do
        local start, _, _ = string.find(v.Name, pattern)
        if start then
            table.insert(list, v.Id);
        end
    end

    self:BankTypeListReq({list, pattern})
end

---所有道具类型  查 bankitemid 表 C --> center
function ExchangeModule:BankTypeListReq(...)
    local tb = ...;
    local Items = {};
    local pattern = "";
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        Items = tb[1];
        pattern = tb[2];
    else
        -- C#调用
        Items = tb[0];
        pattern = tb[1];
    end

    if #Items > 0 then
        local msgName = CS.MsgIDEnum.BankTypeListReqID;

        self.M_PageNum = 1;
        self.M_Searchs[msgName].PageNum = self.M_PageNum;
        self.M_Searchs[msgName].OrderBy = self.M_CurrentSortBy;
        self.M_Searchs[msgName].ItemIDs:Clear();

        for k, v in pairs(Items) do
            self.M_Searchs[msgName].ItemIDs:Add(v)
        end
        self.M_Searchs[msgName].Node = pattern
        self.M_CurrentSearchName = msgName;

        local message = self.M_Searchs[msgName];
        SendClient2CenterReq(message);

    else
		self.M_CurrentSearchName = 0;
        ---空数据就不发给服务器,客户端直接模拟回复
        local message = CS.ProtoMsg.BankItemIDListRet()
        message.PageNum = self.M_PageNum;
        if message.PageNum < 0 then
            message.PageNum = 0;
        end
        message.Node = pattern;
        if self.BankItemIDListRetEvent ~= nil then
            self.BankItemIDListRetEvent:Invoke(message)
        end
    end
end

-- 设置排序规则
function ExchangeModule:SetOrderRule(...)
    local tb = ...;
    local needQuery = false;
    local pageType = 0;
    local Order = 0;
    local Index = 0;

    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        needQuery = tb[1];
        pageType = tb[2];
        Order = tb[3];
        Index = tb[4];
    else
        -- C#调用
        needQuery = tb[0];
        pageType = tb[1];
        Order = tb[2];
        Index = tb[3];
    end

    self.M_CurrentSortBy = (pageType * 100 + Index * 10 + Order);
    if needQuery then
        self.M_PageNum = 1;
        if self.M_CurrentSearchName == 0 then
            return
        end
        local message = self.M_Searchs[self.M_CurrentSearchName]
        if message == nil then
            return
        end

        message.PageNum = self.M_PageNum;
        if message.PageNum < 1 then
            message.PageNum = 1;
        end
        message.OrderBy = self.M_CurrentSortBy;

        SendClient2CenterReq(message);

    end

end

function AfternoonGveModule:OnCkeckAndOpenAfternoonGvePopWindow(...)
    local GNGPlayStartI2L = GameManager.Instance.GNGPlayStartI2L -- GNG活动属性

    if GNGPlayStartI2L ~= nil then
        self.StartTime = GNGPlayStartI2L.StartTime -- 开始时间
        self.Duration = GNGPlayStartI2L.Duration -- 持续时间
        self.EndTime = GNGPlayStartI2L.EndTime -- 结束时间
        self.BossID = GNGPlayStartI2L.BossID -- bossid

        self:OnOpenAfternoonGvePopWindow()
    else
        -- 没有活动数据
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_2"))
    end
end

function AfternoonGveModule:OnOpenAfternoonGvePopWindow(...)
    -- 打开GNG弹出面板
    -- local afternoonGvePopWindowCS =UIManager.Instance:OpenWindow(AfternoonGvePopWindowPath,nil,CS.StarProjectDef.MainPageCommond.HideBoth);
    -- if self.afternoonGvePopWindow == nil then
    --     self.afternoonGvePopWindow=afternoonGvePopWindowCS:GetLuaPanel().ScriptTable;
    --     self.afternoonGvePopWindow:BindModule(self);
    -- end

    self.AsyncLoadAfternoonGvePopWindowCb = function(ui)
        if ui ~= nil then
            if self.afternoonGvePopWindow == nil then
                self.afternoonGvePopWindow = ui:GetLuaPanel().ScriptTable
                self.afternoonGvePopWindow:BindModule(self)
            end
        end
        self.AsyncLoadAfternoonGvePopWindowCb = nil
    end
    UIManager.Instance:OpenWindowAsync(AfternoonGvePopWindowPath, self.AsyncLoadAfternoonGvePopWindowCb, nil, CS.StarProjectDef.MainPageCommond.HideBoth)
end

function AfternoonGveModule:OnGNGPlayStartI2L(data)
    print("午间GVE:玩法开启通知")
    self.GNGPersonReward = nil
    self.GNGGuildReward = nil
    -- 发个消息清除
    CS.StarProject.GlobalEvent.OnClearGNGBossReward:Invoke(true)
end

function AfternoonGveModule:OnGNGPlayEndNtf(data)
    print("午间GVE:玩法结束通知")
end

function AfternoonGveModule:OnGNGPersonRewardNtf(data)
    print("午间GVE:个人奖励通知")

    self.GNGPersonReward = data.data

    if self.afternoonGvePopWindow ~= nil and self.GNGGuildReward ~= nil then
        -- 如果当前UI是打开的那么直接打开
        if self.afternoonGvePopWindow.gameObject.activeSelf then
            -- 发个消息清除
            CS.StarProject.GlobalEvent.OnClearGNGBossReward:Invoke(true)

            local data = {}
            table.insert(data, self.GNGPersonReward)
            table.insert(data, self.GNGGuildReward)
            -- UIManager.Instance:OpenWidget(AfternoonGveFinishWidgetPath, false, data, nil, CS.StarProjectDef.MainPageCommond.HideNone,false,false);
            UIManager.Instance:OpenWidgetAsync(AfternoonGveFinishWidgetPath, nil, false, data, nil, CS.StarProjectDef.MainPageCommond.HideNone, false, false)
        end
    end
end

function AfternoonGveModule:OnGNGGuildRewardNtf(data)
    print("午间GVE:冒险团奖励通知")

    self.GNGGuildReward = data.data

    if self.afternoonGvePopWindow ~= nil and self.GNGPersonReward ~= nil then

            UIManager.Instance:OpenWidgetAsync(AfternoonGveFinishWidgetPath, nil, false, data, nil, CS.StarProjectDef.MainPageCommond.HideNone, false, false)
        end
    end
end

function AfternoonGveModule:OnGNGStartNtf(data)
    print("午间GVE:开始副本")

    local ret = data.data

    self.GameStartTime = ret.StartTime
    self.GameEndTime = ret.GameEndTime
    self.BossEid = ret.BossEid

    if self.ectypeAfternoonGveWidget ~= nil then
        -- 初始化UI
        self.ectypeAfternoonGveWidget:Reset()
    end

    -- 关闭pop
    if self.afternoonGvePopWindow ~= nil then
        self.afternoonGvePopWindow:OnClickCloseBtn()
    end
end

function AfternoonGveModule:OnGNGRunNtf(data)
    -- print("午间GVE:开始副本倒计时")

    local ret = data.data

    self.TotalScore = ret.TotalScore -- 当前积分
    self.StartTime = ret.StartTime -- boss开打的1分钟倒计时
    self.EndTime = ret.EndTime -- boss开打的结束时间

    if self.ectypeAfternoonGveWidget ~= nil then
        -- 刷新UI
        self.ectypeAfternoonGveWidget:Refesh()
    end
end

    UIManager.Instance:OpenWindowAsync(AfternoonGveSuccessWindowPath, nil, data)

-- LuaModule 消息处理
function AfternoonGveModule:OnModuleMessage(msg, ...)
    self.ModuleMessages[msg](...)
end

function AfternoonGveModule:OnDisable(...)
end

function AfternoonGveModule:OnCustomEnterEctype(...)
    -- 进入午间GVE副本
    if self.ectypeAfternoonGveWidget ~= nil then
        self.ectypeAfternoonGveWidget:OnActiveSelf(true)
    end
end

function AfternoonGveModule:OnCustomLevelEctype(...)
    -- 离开午间GVE副本
    if self.ectypeAfternoonGveWidget ~= nil then
        self.ectypeAfternoonGveWidget:OnActiveSelf(false)
    end
end

-- LuaModule 创建
function AfternoonGveModule:OnModuleCreate(arg)
    LuaModuleDispatcher:Register(ModuleDef.AfternoonGveModule, self)

    ---副本开始通知消息
    NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.GNGStartNtfID,
        function(data)
            self:OnGNGStartNtf(data)
        end,
        nil
    )

    ---副本进行中消息
    NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.GNGRunNtfID,
        function(data)
            self:OnGNGRunNtf(data)
        end,
        nil
    )

    ---副本结束消息
    NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.GNGEndNtfID,
        function(data)
            self:OnGNGEndNtf(data)
        end,
        nil
    )

    ---玩法开启通知
    NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.GNGPlayStartI2LID,
        function(data)
            self:OnGNGPlayStartI2L(data)
        end,
        nil
    )

    ---玩法结束通知
    NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.GNGPlayEndNtfID,
        function(data)
            self:OnGNGPlayEndNtf(data)
        end,
        nil
    )

    ---个人奖励通知
    NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.GNGPersonRewardNtfID,
        function(data)
            self:OnGNGPersonRewardNtf(data)
        end,
        nil
    )

    ---冒险团奖励通知
    NetworkManager.Instance:OnMessageEnum(
        CS.MsgIDEnum.GNGGuildRewardNtfID,
        function(data)
            self:OnGNGGuildRewardNtf(data)
        end,
        nil
    )
end

-- LuaModule 释放

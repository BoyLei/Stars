function TaskModule:PlayPlot(...)
    local tb = ...
    local id = nil
    local EffectID = nil
	local EnQueque=true
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        id = tb[1]
        EffectID = tb[2]
		EnQueque=tb[3]
    else
        -- C#调用
        id = tb[0]
        EffectID = tb[1]
		EnQueque=tb[2]
    end
    local config = ConfigDataAccessor.Plot[id]
    if config ~= nil then
        -- UIManager.Instance:OpenWindow(ChapterWindowPath, { config, EffectID },
        -- 	CS.StarProjectDef.MainPageCommond.HideBoth);

        if config.Type == 1 then

			if EnQueque then
            		UIManager.Instance:OpenWidgetAsync(PlotWidgetPath, nil, false, {id, EffectID}, nil,
                	CS.StarProjectDef.MainPageCommond.HideNone, false, true);
			else
					UIManager.Instance:OpenQueueWidget(PlotWidgetPath, nil, false, {id, EffectID}, nil,
					CS.StarProjectDef.MainPageCommond.HideNone, false, true);
			end
			
        elseif config.Type == 2 or config.Type == 3 then
            LuaModuleDispatcher:SendMessage(ModuleDef.SecretAreaModule, "OpenEctypeWarningWidget",
            {config.Type, config.Title});
        end

    end
end

-- team部分代码start
function TaskModule:OnGetTargetTeamInfoRet(data)
    if data == nil or data.data == nil then
        print("OnGetTargetTeamInfoRet data is nil...");
        return;
    end

    local tInfo = data.data.TInfo;

    -- teamwidget显示界面
    self.TaskWidget:SetTeamWidget(tInfo);

    LuaModuleDispatcher:SendMessage(ModuleDef.TeamModule, "OnSyncTeamData", data.data.TInfo);
end

function TaskModule:UpdateTeamData(...)
    local tb = ...
    local data = nil;
    -- 列表
    if type(tb) == "table" then
        -- Lua调用
        data = tb[1]
    else
        -- C#调用
        data = tb[0]
    end
    -- 刷新队伍数据
    if self.TaskWidget ~= nil then
        if data == nil then
            self.TaskWidget:RefreshTeamWidget(nil);
        else
            self.TaskWidget:RefreshTeamWidget(data);
        end
    end
end

function TaskModule:OnAcceptTaskAck(data)
    local AcceptTaskAck = data.data

    if AcceptTaskAck.Ret == CS.ProtoMsg.AcceptTaskAck.Types.ResultType.Ok then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_35"));
    else
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_36"));
    end
end

--[[
*****@放弃任务
--]]
function TaskModule:OnDropTaskAck(data)
    local DropTaskAck = data.data

    if DropTaskAck.Ret == CS.ProtoMsg.DropTaskAck.ResultType.Ok then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_37"));
    else
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_38"));
    end
end

function TaskModule:OnSubmitTaskAck(data)
    local SubmitTaskAck = data.data

    if SubmitTaskAck.Ret == CS.ProtoMsg.SubmitTaskAck.Types.ResultType.Ok then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_39"));
    else
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_40"));
    end
end

function TaskModule:OnGetTaskRewardAck(data)
    local GetTaskRewardAck = data.data

    if GetTaskRewardAck.Ret == CS.ProtoMsg.GetTaskRewardAck.ResultType.Ok then
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_41"));
    else
        Util.ShowMessage(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_42"));
    end
end

    TaskRoot = arg;

-- LuaModule 释放
function TaskModule:OnModuleRelease()
    Log("TaskModule OnModuleCreate 创建");
    if CS.SGF.Network.NetworkManager.Instance == nil then
        return;
    end
    ---接收任务服务器回调
    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.AcceptTaskAckID, function(data)
        self:OnAcceptTaskAck(data)
    end, nil);

    ---放弃任务服务器回调
    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.DropTaskAckID, function(data)
        self:OnDropTaskAck(data)
    end, nil);

    ---提交任务服务器回调
    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.SubmitTaskAckID, function(data)
        self:OnSubmitTaskAck(data)
    end, nil);

    ---领取任务奖励服务器回调
    CS.SGF.Network.NetworkManager.Instance:OffMessageEnum(CS.MsgIDEnum.GetTaskRewardAckID, function(data)
        self:OnGetTaskRewardAck(data)
    end, nil);
    LuaModuleDispatcher:UnRegister(ModuleDef.TaskModule)
end

return TaskModule

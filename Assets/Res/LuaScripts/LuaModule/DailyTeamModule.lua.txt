function DailyTeamModule:OnTeamDailyLevelDataNtf(data)
	local msg = data.data
	for k, v in pairs(msg.DeadLogs) do
		if k == GameManager.Instance.mainPlayerId then
			--复活时间必须大于0 ,服务会把过去时间传递过来
			local time=math.ceil((v-TimeUtils.ServerNowStampMilli)/1000)
			if time >0 then				
				---玩家自己
				-- UIManager.Instance:OpenWindow(DailyTeamReviveWindowPath, v, CS.StarProjectDef.MainPageCommond.HideBoth);
				UIManager.Instance:OpenWindowAsync(DailyTeamReviveWindowPath, nil, v, CS.StarProjectDef.MainPageCommond.HideBoth)
				break
			end
		end
	end
end

function DailyTeamModule:OnTeamDailyGameEndNtf(data)
	local msg = data.data
	self.GameEndData = msg
	local BossDeadPos = msg.BossDeadPos
	local IsSuccess = msg.IsSuccess
	local CommonList = msg.CommonList
	local ExtraList = msg.ExtraList
	local hasawrard = CommonList.Count > 0
	local hasExtra = ExtraList.Count > 0
	CS.StarProject.GlobalEvent.OnDailyTeamGameEnd:Invoke(IsSuccess > 0)
	if IsSuccess > 0 and (hasawrard or hasExtra) then
		self:CreateLocalBox(msg.CommonList, BossDeadPos, CommonChestID)
		self:CreateLocalBox(msg.ExtraList, BossDeadPos, ExtraChestID)
		local bossName = ""
		local entry = BusinessManager.Instance:GetTDEntryOpenNtf()
		if entry ~= nil then
			local config = ConfigDataAccessor.Monster[entry.BossID]
			if config ~= nil then
				bossName = config.Name
			end
		end
		-- Util.ShowMessage(bossName .. "已被成功讨伐")
		Util.ShowMessage(StringFormat(LanguageHelp:GetLanguageByKey("Local_Str_Auto_Tips_232"), {bossName}))
	else
		-- UIManager.Instance:OpenWindow(DailyTeamAwardWindowPath,self.GameEndData,CS.StarProjectDef.MainPageCommond.HideBoth);
		UIAPI.ShowUniversalSettlement(
			msg.IsSuccess > 0,
			msg.Second,
			msg.CommonList,
			msg.ExtraList,
			function()
				self:OnQuitTreasureScene()
			end,
			nil
		)
	end
end

function DailyTeamModule:OnQuitTreasureScene()
	-- 切换地图得时候关闭复活框
	UIManager.Instance:CloseWindow(DailyTeamReviveWindowPath)
	GameManager.Instance:BackToWorld()
	local info = FixMessageManager.Instance:GetMDMgr(FixUpdateDef.ComCount):GetData(11)
	if info ~= nil then
		if info.RemainCount > 0 then
			DelayInvoker.DelayInvoke(
				self,
				1,
				function()
					UIAPI.ShowMsgBox(
						71,
						function(eventName)
							if eventName == "SURE" then
								if info.RemainCount > 0 then
									local config = ConfigDataAccessor.PlayOpenConditions[2]
									if not (CheckSysOpen(2, true)) then
										return
									end

									local mapID = config.MapID
									local npcID = config.NPCID
									local mete = GameManager.Instance:IsConditionMete(config.IndexChangeCondition)
									if mete then
										mapID = config.MapID2
										npcID = config.NPCID2
									end
									TaskHelper.FindPathByNpc(
										mapID,
										npcID,
										1,
										function(result)
											if result then
												LuaModuleDispatcher:SendMessage(ModuleDef.DailyTeamModule, "OnDailyTeamWindow", {})
											end
										end
									)
								end
							end
						end,
						info.RemainCount
					)
				end
			)
		end
	end
end

-- 有成员同意/加入了
function DailyTeamModule:OnTDMemberJoinNtf(data)
	local msg = data
	if msg == nil then
		UIManager.Instance:CloseWidget(DailyTeamConfirmWidgetPath)
		return
	end

	local teamid = uint64.new(msg.TeamID)
	local calllist = msg.CallList
	local playeType = msg.PlayeType

	local module = ModuleManager.Instance:GetModule(ModuleDef.TeamModule)
	local luatable = module:GetLuaTable()
	local teamInfo = luatable:GetTeamInfo()
	local myTeamID = uint64.new(teamInfo.TeamID)
	if teamInfo ~= nil and myTeamID ~= 0 and teamInfo.UserInfos.Count > 0 then
		if myTeamID.value == teamid.value then
			-- 自己关心的队伍
			-- 刷新
			if msg.IsCancel then
				UIManager.Instance:CloseWidget(DailyTeamConfirmWidgetPath)
			else
				if calllist == nil or calllist.Count <= 0 then
					-- 如果列表空了，那就是有队员拒绝了
					UIManager.Instance:CloseWidget(DailyTeamConfirmWidgetPath)
					return
				end
				if self.dailyTeamConfirmWidget ~= nil and self.dailyTeamConfirmWidget.IsOpen then
					self.dailyTeamConfirmWidget:RenderAllView(
						not self.dailyTeamConfirmWidget.IsOpen,
						{
							TeamInfo = teamInfo,
							List = calllist,
							PlayeType = playeType
						}
					)
				else
					-- local DailyTeamConfirmWidget = UIManager.Instance:OpenWidget(DailyTeamConfirmWidgetPath, true,
					-- 	{ TeamInfo = teamInfo, List = calllist, PlayeType = playeType }, nil,
					-- 	CS.StarProjectDef.MainPageCommond.WidgetHide, true, false);
					-- if self.dailyTeamConfirmWidget == nil then
					-- 	self.dailyTeamConfirmWidget = DailyTeamConfirmWidget:GetLuaPanel().ScriptTable
					-- end

					self.AsyncLoadDailyTeamConfirmWidgetCb = function(ui)
						if ui ~= nil then
							if self.dailyTeamConfirmWidget == nil then
								self.dailyTeamConfirmWidget = ui:GetLuaPanel().ScriptTable
							end
						end
						self.AsyncLoadDailyTeamConfirmWidgetCb = nil
					end
					UIManager.Instance:OpenWidgetAsync(
						DailyTeamConfirmWidgetPath,
						self.AsyncLoadDailyTeamConfirmWidgetCb,
						true,
						{
							TeamInfo = teamInfo,
							List = calllist,
							PlayeType = playeType
						},
						nil,
						CS.StarProjectDef.MainPageCommond.WidgetHide,
						true,
						false
					)
				end
			end
		end
	end
end

-- 创建本地宝箱 [普通宝箱、额外宝箱,位置]
function DailyTeamModule:CreateLocalBox(Drops, Pos, InteractId)
	if Drops == nil or Drops.Count == 0 then
		return
	end
	self.BirthPos = Vector3.zero -- 怪物死亡位置
	self.BirthPos.x = Pos.X
	self.BirthPos.y = Pos.Y
	self.BirthPos.z = Pos.Z

	local mapPox = GameManager.Instance:GetMapBossPos()
	if mapPox ~= Vector3.zero then
		self.BirthPos = mapPox
	end

	--改成了固定位置
	self.BirthPos = Vector3(52.43, 0.06, 38.07)
	--随机下位置
	--pos=BusinessManager.Instance:GetRandomPos(pos,2.5);

	local rewardList = self:GetRewards(Drops)
	CS.StarProject.GlobalEvent.onCreateLocalEntity:Invoke(E_LocalEntityType.TreasureBox, InteractId, E_LocalEntitySource.DailyTeamEctype, self.BirthPos, rewardList, true, -1)
end

-- 拿取奖励 [开始下标、数量]
function DailyTeamModule:GetRewards(reward)
	local drops = LocalDropList()
	for i = 0, reward.Count - 1 do
		local dropinfo = reward[i]
		drops:AddData(dropinfo.Itemid, dropinfo.ItemNum, dropinfo.EntityID)
	end
	return drops
end

function DailyTeamModule:OnMainPageLoadedCompleteCB(res)
	if TaskRoot ~= nil and IsMainPageLoadedCompleteOpenWidget == true then
		self:OpenDailyTeamTaskWidget()
	end
	IsMainPageLoadedCompleteOpenWidget = false
end

function DailyTeamModule:OpenDailyTeamTaskWidget()
	-- self.TaskWidget = UIManager.Instance:OpenWidget(DailyTeamTaskWidgetPath, true, nil, TaskRoot.transform,
	-- CS.StarProjectDef.MainPageCommond.HideNone, true, false);

	self.AsyncLoadDailyTeamTaskWidgetCb = function(ui)
		if ui ~= nil then
			if self.TaskWidget == nil then
				self.TaskWidget = ui:GetLuaPanel().ScriptTable
			end
		end
		self.AsyncLoadDailyTeamTaskWidgetCb = nil
	end
	UIManager.Instance:OpenWidgetAsync(DailyTeamTaskWidgetPath, self.AsyncLoadDailyTeamTaskWidgetCb, true, nil, TaskRoot.transform, CS.StarProjectDef.MainPageCommond.HideNone, true, false)
end

return DailyTeamModule
